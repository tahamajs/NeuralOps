\section*{پیوست الف: کارنامه جامع ۱۳۱ مکانیزم متعارف توجه (Attention Master Registry)}
\addcontentsline{toc}{section}{پیوست الف: کارنامه جامع ۱۳۱ مکانیزم متعارف توجه}

در این پیوست، مشخصات کامل و ساختار ریاضی تمامی ۱۳۱ مکانیزم متعارف توجه که در رجیستری رسمی چارچوب پژوهش (\lr{\texttt{ATTENTION\_REGISTRY}}) پیاده‌سازی و اعتبارسنجی شده‌اند، ارائه می‌گردد. تمامی این ۱۳۱ ماژول در محیط آزمایشگاهی استاندارد بر روی کارت گرافیک \lr{NVIDIA GeForce RTX 4070 Laptop (SM 8.9)} ارزیابی گردیده و موفق به اخذ تاییدیه دروازه کیفیت (\lr{\textbf{PASS}}) شده‌اند.

\paragraph{روایت تحلیلی خانواده‌های معماری توجه:}
مکانیزم‌های توجه ثبت‌شده در جدول \ref{tab:app_all_131_attentions} در ۷ خانواده بنیادین ریاضیاتی طبقه‌بندی می‌شوند:
\begin{enumerate}
    \item \textbf{مکانیزم‌های توجه خلوت و پنجره‌ای (\lr{Sparse \& Window}):} نظیر \lr{\texttt{adaptive\_span\_attention}}، \lr{\texttt{block\_transformer\_hiba}} و \lr{\texttt{nsa\_attention}} که با تعریف ماسک‌های باینری دلخواه $\mathcal{M}_w$ یا پنجره‌های لغزان محلی، پیچیدگی زمانی و حافظه را از $\mathcal{O}(L^2)$ به $\mathcal{O}(L \cdot w)$ تقلیل می‌دهند.
    \item \textbf{فاکتورسازی تانسوری و تجزیه کم‌مرتبه (\lr{Low-Rank Factorization}):} نظیر \lr{\texttt{adaptive\_svd\_attention}}، \lr{\texttt{lora\_attention}}، \lr{\texttt{asvd\_attention}} و \lr{\texttt{mla\_attention}} که با فرض کم‌مرتبه بودن زیرفضای وزن‌ها، تقریب $W \approx U \Sigma V^\top$ را اعمال کرده و پیچیدگی را به $2 d r + r$ محدود می‌سازند.
    \item \textbf{توجه خطی و نگاشت‌های هسته‌ای (\lr{Linear \& Kernelized}):} نظیر \lr{\texttt{cosformer}}، \lr{\texttt{performer\_attention}} و \lr{\texttt{galerkin\_attention}} که با جایگزینی سافت‌مکس با نگاشت‌های ویژگی تفکیک‌پذیر $\phi(Q) (\phi(K)^\top V)$، ضرب ماتریسی را انجمنی ساخته و هزینه را از درجه دو به درجه یک خطی $\mathcal{O}(L)$ کاهش می‌دهند.
    \item \textbf{مدل‌های حالت‌فضا و ساختارهای بازگشتی (\lr{State-Space \& Recurrent Models}):} نظیر \lr{\texttt{mamba\_block}}، \lr{\texttt{gated\_delta\_net}}، \lr{\texttt{gla\_attention}} و \lr{\texttt{hgrn2\_gated\_deltanet}} که با معادلات حالت $h_t = \bar{A}_t h_{t-1} + \bar{B}_t x_t$ امکان استنتاج با ردپای حافظه ثابت $\mathcal{O}(1)$ در زمان استنتاج را فراهم می‌سازند.
    \item \textbf{معماری‌های مخلوط متخصصان (\lr{Mixture of Experts - MoE}):} نظیر \lr{\texttt{expert\_choice\_moe}}، \lr{\texttt{chela\_slay}}، \lr{\texttt{reap\_moe}} و \lr{\texttt{bam\_umoe}} که با هدایت پویا و گیتینگ $Top-k$، ظرفیت پارامتری مدل را بدون افزایش متناظر محاسبات ضرب خطی گسترش می‌دهند.
    \item \textbf{مکانیزم‌های پرسمان گروه‌بندی‌شده (\lr{Grouped-Query Attention - GQA}):} نظیر \lr{\texttt{gqa\_attention}}، \lr{\texttt{flash\_gqa}} و \lr{\texttt{wgqa\_attention}} که با اشتراک‌گذاری سرهای کلید-مقدار میان چندین سر پرسمان ($G = H_q / H_{\text{kv}}$)، حجم اشغال بافر \lr{KV Cache} را تا ۸ برابر فشرده می‌کنند.
    \item \textbf{ساختارهای ترکیبی و چندشاخه‌ای (\lr{Hybrid \& Multi-Branch}):} نظیر \lr{\texttt{aegis\_hybrid}}، \lr{\texttt{granite\_hybrid}} و \lr{\texttt{jamba}} که کانال‌های توجه متراکم محلی را با بلوک‌های بازگشتی سراسری در هم می‌آمیزند.
\end{enumerate}

\section*{پیوست ب: کارنامه جامع ۷۲ موتور متعارف فشرده‌سازی مدل (Compression Master Registry)}
\addcontentsline{toc}{section}{پیوست ب: کارنامه جامع ۷۲ موتور متعارف فشرده‌سازی مدل}

در این پیوست، تمامی ۷۲ موتور فشرده‌سازی مدل‌های ترنسفورمر ثبت‌شده در رجیستری رسمی چارچوب پژوهش (\lr{\texttt{COMPRESSOR\_REGISTRY}}) به همراه اهداف بهینه‌سازی ریاضی و نرخ‌های همگرایی در جدول \ref{tab:app_all_72_compressors} تصریح گردیده‌اند.

\paragraph{روایت تحلیلی پارادایم‌های فشرده‌سازی:}
موتورهای فشرده‌سازی این پژوهش بر پایه ۵ محور الگوریتمی متمایز طراحی شده‌اند:
\begin{enumerate}
    \item \textbf{تجزیه مقادیر منفرد آگاه از فعال‌سازی (\lr{Activation-Aware SVD}):} نظیر \lr{\texttt{a3\_compressor}}، \lr{\texttt{aa\_svd\_compressor}} و \lr{\texttt{svd\_llm\_compressor}} که به جای بهینه‌سازی خطای خام وزن‌ها $\|W - \hat{W}\|_F^2$، خطای فعال‌سازهای لایه بعدی $\|(W - \hat{W}) X\|_F^2$ را از طریق سفیدسازی چولسکی ماتریس کوواریانس ورودی $S = X X^\top = L L^\top$ به حداقل مطلق اکرت-یانگ-میرسکی می‌رسانند.
    \item \textbf{کوانتایزاسیون برداری و غیرخطی (\lr{Vector Quantization}):} نظیر \lr{\texttt{kmeans\_quantize}}، \lr{\texttt{coopq\_quantizer}} و \lr{\texttt{astro\_quantizer}} که با تعریف کتاب‌کد برداری (\lr{Codebook}) و بهینه‌سازی لوید-مکس، وزن‌ها را در کلاسترهای فشرده نگاشت کرده و برابری رتبه ابعادی فضا را در مقادیر کم‌بیت حفظ می‌نمایند.
    \item \textbf{هرس ساختاریافته و تنک‌سازی با جهت متناوب (\lr{ADMM Pruning}):} نظیر \lr{\texttt{admm\_compressor}}، \lr{\texttt{flap}} و \lr{\texttt{sparsegpt}} که با فرمول‌بندی مسئله تنک‌سازی تحت قیدهای رتبه و مقیاس نُرم صفر، بهینگی سراسری را با نرخ $\mathcal{O}(1/\sqrt{T})$ در مدل‌های بزرگ محقق می‌سازند.
    \item \textbf{فاکتورسازی تانسوری کرونکر و قطار تانسوری (\lr{Kronecker \& Tensor Train}):} نظیر \lr{\texttt{kronecker\_compressor}} و \lr{\texttt{tt\_compressor}} که ماتریس‌های بزرگ پروجکشن را به حاصل‌ضرب تانسوری ماتریس‌های فوق سبک تقلیل می‌دهند.
    \item \textbf{فشرده‌سازی و مدیریت بافر کلید-مقدار (\lr{KV Cache Compression}):} نظیر \lr{\texttt{delta\_kv}}، \lr{\texttt{cachenotes\_compressor}} و \lr{\texttt{aatc\_xkv\_compressor}} که مصرف حافظه حافظه استنتاج خودبازگشتی را با حذف هوشمند توکن‌های بی‌استفاده مهار می‌کنند.
\end{enumerate}
