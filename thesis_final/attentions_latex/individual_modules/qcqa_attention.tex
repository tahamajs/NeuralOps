\section{ماژول qcqa\_attention (Quality and Capacity-Aware Grouped Query Attention)}
\label{sec:module_qcqa_attention}

\subsection{معرفی و انگیزه}
این ماژول در چارچوب پژوهش حاضر پیاده‌سازی و ارزیابی شده و فاقد مقاله مستقل جداگانه در ادبیات تحقیق است.

مکانیزم توجه با سرهای پرس‌وجوی گروه‌بندی‌شده (\lr{Grouped-Query Attention - GQA}) به عنوان استانداردی موثر برای کاهش بار حافظه بافر کلید/مقدار (\lr{KV-Cache}) در مدل‌های ترنسفورمر به کار می‌رود. با این وجود، در روش‌های مرسوم \lr{GQA}، نسبت انتساب سرهای پرس‌وجو به سرهای کلید/مقدار به صورت یکنواخت و صلب در تمامی لایه‌ها و موقعیت‌های متنی یکسان در نظر گرفته می‌شود. این رویکرد ایستا بدون توجه به اهمیت محتوایی و انتروپی اطلاعاتی توکن‌ها عمل کرده و ممکن است به افت کیفیت بازنمایی در سرهای بحرانی منجر گردد.

مکانیزم توجه آگاه از کیفیت و ظرفیت (\lr{Quality and Capacity-Aware GQA - QCQA}) با هدف پویاسازی و یادگیری توزیع تخصیص سرها معرفی شده است. ایده محوری ماژول \lr{QCQA}، ترکیب پیوسته و هوشمند سرهای کلید/مقدار فشرده‌شده از طریق یک شبکه مسیریاب هموار (\lr{Continuous Learned Routing Distribution}) با دمای کنترل‌شده است. بدین ترتیب، برای هر سر پرس‌وجو، یک بازنمایی مجازی سنتز‌شده از کلید و مقدار بر مبنای ترکیبی خطی و وزن‌دار از سرهای فشرده موجود ساخته می‌شود که بدون نیاز به تکثیر صلب داده‌ها، انعطاف بیان‌پذیری مدل را به حداکثر می‌رساند.

\subsection{مبانی نظری و فرمول‌بندی ریاضی}
فرض کنید $\mathbf{X} \in \mathbb{R}^{B \times L \times D}$ ماتریس حالت‌های پنهان ورودی باشد که در آن $B$ اندازه دسته، $L$ طول دنباله و $D$ بعد پنهان مدل است. پروجکشن‌های اولیه پرس‌وجو، کلید و مقدار با ابعاد سرهای نامتقارن تولید می‌شوند:
\begin{equation}
\mathbf{Q} \in \mathbb{R}^{B \times L \times H_q \times d}, \quad \mathbf{K}_{\text{raw}}, \mathbf{V}_{\text{raw}} \in \mathbb{R}^{B \times N_{\text{kv}} \times H_{\text{kv}} \times d}
\end{equation}
که در آن $H_q$ تعداد سرهای پرس‌وجو، $H_{\text{kv}} \le H_q$ تعداد سرهای فشرده کلید/مقدار و $d = D / H_q$ بعد هر سر توجه است.

به منظور سنتز بازنمایی ویژه هر سر پرس‌وجو، شبکه مسیریاب خطی $\mathbf{W}_{\text{route}} \in \mathbb{R}^{d \times H_{\text{kv}}}$ ماتریس لوجیت‌های مسیر را به ازای هر سر پرس‌وجو تولید می‌کند:
\begin{equation}
\mathbf{R}_{\text{logits}} = \frac{\mathbf{Q} \mathbf{W}_{\text{route}}}{\max(\tau_{\text{router}}, 10^{-4})} \in \mathbb{R}^{B \times L \times H_q \times H_{\text{kv}}}
\end{equation}
سپس توزیع وزن‌های پیوسته مسیریابی با تابع بیشینه هموار نرمال‌سازی می‌شود:
\begin{equation}
\mathbf{W}_{\text{route}} = \operatorname{softmax}(\mathbf{R}_{\text{logits}}, \text{dim}=-1)
\end{equation}
برای هر توکن پرس‌وجو در موقعیت زمانی $t$ و سر پرس‌وجوی $h \in \{1, \dots, H_q\}$، بازنمایی کلید و مقدار سنتز‌شده از ترکیب محدب سرهای فشرده کلید/مقدار به دست می‌آید:
\begin{equation}
\tilde{\mathbf{K}}_{b, t, h} = \sum_{k=1}^{H_{\text{kv}}} W_{\text{route}, b, t, h, k} \cdot \mathbf{K}_{\text{raw}, b, :, k}, \quad \tilde{\mathbf{V}}_{b, t, h} = \sum_{k=1}^{H_{\text{kv}}} W_{\text{route}, b, t, h, k} \cdot \mathbf{V}_{\text{raw}, b, :, k}
\end{equation}
سپس امتیازهای توجه و بافتار نهایی با اعمال ضرب داخلی، ماسک‌های علّی/پدینگ و نرم‌سازی بیشینه هموار استاندارد محاسبه می‌گردد:
\begin{equation}
\mathbf{S}_h = \frac{\mathbf{Q}_h \tilde{\mathbf{K}}_h^\top}{\sqrt{d}}, \quad \mathbf{A}_h = \operatorname{softmax}(\mathbf{S}_h + \mathbf{M}), \quad \mathbf{O}_h = \mathbf{A}_h \tilde{\mathbf{V}}_h
\end{equation}
در نهایت، تانسورهای خروجی کلیه سرها ادغام شده و از پروجکشن خروجی عبور داده می‌شوند:
\begin{equation}
\mathbf{O} = \operatorname{Concat}(\mathbf{O}_1, \dots, \mathbf{O}_{H_q}) \mathbf{W}_O
\end{equation}

\subsection{معماری و پیاده‌سازی}
کلاس پایتونی اصلی این ماژول تحت عنوان \lr{QCQAAttention} در فایل \lr{Codes/framework/attentions/qcqa\_attention.py} پیاده‌سازی شده و از کلاس پایه \lr{BaseAttention} ارث‌بری می‌نماید. اهم ویژگی‌های ساختاری و مهندسی این کد عبارتند از:
\begin{itemize}
    \item \textbf{شبکه مسیریاب پویا (\lr{Continuous Soft Router}):} شبکه سبک‌وزن \lr{self.router} وظیفه نگاشت بردار بعد سر به تعداد سرهای کلید/مقدار فشرده را بر عهده دارد و دمای \lr{router\_temp} امکان تنظیم غلظت یا نرمی توزیع وزن‌ها را فراهم می‌سازد.
    \item \textbf{پشتیبانی کامل از کش بافر کلید/مقدار (\lr{KV-Cache}):} سازگار با هر دو ساختار چندتایی و \lr{DynamicCache} پلتفرم \lr{Hugging Face} به همراه اعمال صحیح انحراف زمانی \lr{past\_len} در محاسبات مکانی \lr{RoPE}.
    \item \textbf{انعطاف در انواع ماسک‌های ورودی:} پردازش ماسک‌های دوبعدی پدینگ، ماسک‌های چهاربعدی تلفیقی و ماسک مثلثی علّی برای دیکودرها.
    \item \textbf{مسیر شتاب‌یافته \lr{SDPA}:} در صورت عدم نیاز به بازگرداندن ماتریس‌های توجه و هم‌ترازی سرها، از هسته ادغام‌شده \lr{scaled\_dot\_product\_attention} برای بیشینه‌سازی توان عملیاتی استفاده می‌شود.
\end{itemize}

\subsection{محدودیت‌ها و تحلیل شکست}
تحلیل تفصیلی کارنامه ماژول نمایانگر رفتارهای ساختاری زیر است:
\begin{itemize}
    \item \textbf{افت کارایی در مدل‌های \lr{RoBERTa} (\lr{51.03\%}) و \lr{GPT-2} (\lr{50.92\%}):} نتایج این دو مدل در آستانه \lr{PASS\_LOW} قرار دارد. علت بنیادین این پدیده، تداخل دینامیک وزن‌های مسیریاب با وزن‌های تثبیت‌شده پیش‌آموزش است. از آنجا که مقادیر اولیه مسیریاب به صورت تصادفی با وزن‌های نرمال کوچک مقداردهی می‌شوند، در فازهای نخستین آموزش بدون دوره گرم‌سازی طولانی، بازنمایی کلیدها دچار درهم‌آمیختگی و اغتشاش شده و مدل به جای بازیابی اطلاعات ظریف، به پیش‌بینی میانگین رده‌ها بسنده می‌نماید.
    \item \textbf{چالش مدل علّی \lr{Qwen} (\lr{51.72\%}):} در مدل‌های خودبازگشتی عمیق که از پیش بر پایه فشرده‌سازی صلب \lr{GQA} آموزش دیده‌اند، تغییر دینامیک فضاهای کلید/مقدار در طول دنباله مانع از همگرایی سریع گرادیان‌ها در لایه‌های فوقانی شبکه می‌گردد.
\end{itemize}
