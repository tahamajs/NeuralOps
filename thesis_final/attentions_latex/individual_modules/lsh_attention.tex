\section{ماژول توجه با درهم‌سازی حساس به موقعیت (LSH Attention)}
\label{sec:module_lsh_attention}

\subsection{معرفی و انگیزه}
این مکانیزم بر مبنای مبانی نظری پیشرفته در ادبیات توجه و فشرده‌سازی \cite{kitaev2020reformer} توسعه یافته است.

مکانیزم توجه \lr{Locality-Sensitive Hashing Attention (LSH Attention)} که بر پایه ایده انقلابی مدل \lr{Reformer} بنا شده است، یکی از مطرح‌ترین رهیافت‌ها در جهت کاهش پیچیدگی زمانی و مکانی توجه از $\mathcal{O}(N^2)$ به $\mathcal{O}(N \log N)$ به شمار می‌رود. در ساختار استاندارد ترنسفورمر، سهم عمده‌ای از محاسبات ضرب داخلی ماتریس‌های پرس‌وجو و کلید صرف جفت توکن‌هایی می‌شود که پس از اعمال تابع \lr{Softmax} وزنی بسیار نزدیک به صفر دریافت می‌کنند. محاسبه کامل این ضرب‌ها در طول دنباله‌های بلند منشأ اصلی هدررفت حافظه و زمان پردازش است.

انگیزه اصلی در الگوریتم \lr{LSH Attention}، جایگزینی جستجوی فراگیر با خوشه‌بندی احتمالی توکن‌ها در سطوح زاویه‌ای مشابه است. با ترسیم ابرصفحه‌های تصادفی، توکن‌هایی که ضرب داخلی زاویه‌ای بالاتری دارند با احتمال بسیار بالایی به یک سطل درهم‌سازی (\lr{Hash Bucket}) یکسان نگاشت می‌شوند. در این حالت، عملیات توجه تنها در میان اعضای هر سطل انجام شده و از محاسبات سنگین میان‌سطلی صرف‌نظر می‌گردد. نسخه پیاده‌سازی‌شده در این چارچوب علاوه بر الگوریتم چنددوری (\lr{Multi-round LSH})، از نگاشت‌های فاکتورگیری‌شده \lr{FactorizedLoRALinear} جهت کاهش چشمگیر پارامترهای مدل بهره می‌برد.

\subsection{مبانی نظری و فرمول‌بندی ریاضی}
فرض کنید بردارهای پرس‌وجو و کلید در یک فضای $d$-بعدی به صورت $\mathbf{q}, \mathbf{k} \in \mathbb{R}^d$ قرار داشته باشند. در روش درهم‌سازی مبتنی بر ابرصفحه‌های تصادفی زاویه‌ای، به ازای هر دور $r \in \{1, \dots, R\}$، یک ماتریس تصویر تصادفی $\mathbf{R}_r \in \mathbb{R}^{d \times \frac{b}{2}}$ با توزیع نرمال استاندارد نمونه‌برداری و بهنجار می‌شود:
\begin{equation}
\tilde{\mathbf{R}}_r = \frac{\mathbf{R}_r}{\|\mathbf{R}_r\|_2}
\end{equation}
سپس، شناسه سطل هر بردار با درهم‌سازی متقارن تصادفی تعیین می‌گردد:
\begin{equation}
h_r(\mathbf{x}) = \operatorname{argmax}\left( [\mathbf{x} \tilde{\mathbf{R}}_r ; -\mathbf{x} \tilde{\mathbf{R}}_r] \right) \in \{0, \dots, b - 1\}
\end{equation}
که در آن $b$ تعداد سطل‌های درهم‌سازی است. احتمال قرارگیری دو بردار در یک سطل رابطه‌ای مستقیم با زاویه کسینوسی $\theta$ میان آنها دارد:
\begin{equation}
\Pr[h(\mathbf{q}) = h(\mathbf{k})] = 1 - \frac{\theta(\mathbf{q}, \mathbf{k})}{\pi}
\end{equation}

برای افزایش ضریب اطمینان و جلوگیری از حذف تصادفی توکن‌های مرتبط، الگوریتم به تعداد $R$ دور مستقل اجرا می‌شود. ماتریس امتیازات درون هر سطل به شکل زیر تعریف می‌گردد:
\begin{equation}
S_{i, j}^{(r)} = \begin{cases}
\frac{\mathbf{q}_i \mathbf{k}_j^T}{\sqrt{d}} & \text{if } h_r(\mathbf{q}_i) = h_r(\mathbf{k}_j) \text{ and } m_j = 1 \\
-\infty & \text{otherwise}
\end{cases}
\end{equation}
خروجی نهایی از میانگین‌گیری نتایج تمامی دورهای درهم‌سازی حاصل می‌شود:
\begin{equation}
\mathbf{O} = \frac{1}{R} \sum_{r=1}^R \operatorname{Softmax}\left(\mathbf{S}^{(r)}\right) \mathbf{V}
\end{equation}

\paragraph{تحلیل طیفی و نامساوی‌های پایداری خطا}
برای بررسی میزان دقت تقریب ماتریس وزن یا نگاشت توجه در ماژول \lr{lsh\_attention}، تفاضل ماتریس فاکتورگیری‌شده از نگاشت متراکم اصلی بر اساس نامساوی هافدینگ-فون‌نیومان ارزیابی می‌گردد. فرض کنید $\mathbf{W} \in \mathbb{R}^{d_{\text{in}} \times d_{\text{out}}}$ ماتریس وزن هدف باشد که توسط تقریب کم‌مرتبه یا پراکنده با رتبه موثر $r$ جایگزین شده است. کران بالای خطای بازسازی بر مبنای نرم فروبنیوس به صورت زیر تصریح می‌شود:
\begin{equation}
\label{eq:spectral_bound_lsh_attention}
\| \mathbf{W} - \mathbf{W}_{r} \|_F = \sqrt{\sum_{i=r+1}^{\min(d_{\text{in}}, d_{\text{out}})} \sigma_i^2(\mathbf{W})} \le \epsilon \| \mathbf{W} \|_F
\end{equation}
که در آن $\sigma_i(\mathbf{W})$ بیانگر $i$-امین مقدار منفرد مرتب‌شده ماتریس به صورت نزولی است. نسبت حفظ انرژی طیفی $\eta_r = \frac{\sum_{i=1}^r \sigma_i^2}{\sum_{i=1}^{d} \sigma_i^2}$ تضمین می‌کند که با انتخاب آستانه $\tau \ge 0.90$، بیش از نود درصد از پراکندگی واریانس داده‌ها حفظ شده و افت سیگنال در انتشار گرادیان رو به عقب به حداقل ممکن کاهش می‌یابد.

\paragraph{تحلیل مجانبی و پیچیدگی اشتراک حافظه سخت‌افزاری}
از منظر پیچیدگی محاسباتی و انتقال داده‌ها در سلسومراتب حافظه (حافظه سراسری \lr{HBM} به حافظه محلی \lr{SRAM})، عملیات ضرب ماتریسی در این مکانیزم نیازمند تخصیص بافر موقت با ابعاد فاکتورگیری‌شده است. برای پردازش دنباله‌ای به طول $N$ با بعد مخفی $D$، بار محاسباتی هسته از فرم درجه دو به رابطه زیر تقلیل می‌یابد:
\begin{equation}
\label{eq:complexity_lsh_attention}
\mathcal{C}_{\text{lsh\_attention}} = \mathcal{O}\left( B \cdot N \cdot D \cdot r + B \cdot H \cdot N \cdot d_k \right) \ll \mathcal{O}\left( B \cdot H \cdot N^2 \cdot d_k \right)
\end{equation}
این بهبود مرتبه پیچیدگی، با کاهش چشمگیر حجم تبادل تانسورها در گذرگاه‌های ارتباطی کارت گرافیک، مانع از اشباع پهنای باند حافظه (پدیده \lr{Memory-Bound}) در فرآیند استنتاج طولانی‌مدت می‌گردد.

\subsection{معماری و پیاده‌سازی}
پیاده‌سازی ماژول \lr{LSHAttention} در فایل \lr{Codes/framework/attentions/lsh\_attention.py} با رعایت استانداردهای سخت‌افزاری و برطرف ساختن نواقص شناخته‌شده نسخه‌های پیشین صورت گرفته است:
\begin{itemize}
    \item \textbf{پردازش قطعه‌بندی‌شده بدون ماتریس $N \times N$ (\lr{LSH-09}):} توجه سطل‌بندی‌شده به صورت تکه‌ای (\lr{Chunked}) با اندازه پاره‌های ۶۴ توکنی اجرا می‌گردد تا هیچ‌گاه ماتریس کامل دنباله در حافظه ساخته نشود.
    \item \textbf{نگاشت‌های فاکتورگیری‌شده کم‌مرتبه (\lr{FactorizedLoRALinear}):} جایگزینی لایه‌های خطی استاندارد با ترکیبی از پروجکشن‌های تجزیه‌شده \lr{SVD} با رتبه ۶۴ و آداپتورهای آموزش‌پذیر رتبه ۱۶ برای کنترل ابعاد و بهینه‌سازی پارامترها.
    \item \textbf{انطباق با دیکودرها (\lr{LSH-15}):} در مدل‌های خودبازگشتی، به منظور ممانعت از نقض ماسک علّی در اثر بازآرایی و مرتب‌سازی سطل‌ها، ماژول به صورت ایمن به شاخه توجه نرمال با ماسک پایین‌مثلثی رجوع می‌کند.
\end{itemize}

\subsection{محدودیت‌ها و تحلیل شکست}
بررسی عملکرد ماژول بر روی سایر مدل‌ها حاکی از محدودیت‌های ذاتی تقریب درهم‌سازی است:
\begin{itemize}
    \item \textbf{افت دقت در \lr{RoBERTa} و \lr{GPT-2}:} دقت ۴۹.۵۴٪ در \lr{RoBERTa} و ۵۱.۶۱٪ در \lr{GPT-2} نشان‌دهنده حساسیت بالای این مدل‌ها به خطای برش سطل‌ها (\lr{Bucket Boundary Error}) است. در مدل‌های عمیق، توکن‌هایی که از نظر معنایی به شدت به یکدیگر وابسته‌اند ممکن است به واسطه وجود نویز اندک در فضا به سطل‌های متفاوتی هدایت شوند که به قطع جریان اطلاعات و افت دقت به سطح پایه شانس منجر می‌گردد.
    \item \textbf{ناسازگاری‌های علّی در دیکودرهای مدرن (\lr{Qwen}):} دقت ۵۰.۹۲٪ در مدل \lr{Qwen} حاصل تقابل میان تفکیک ابرصفحه‌ها و نیاز اکید به رعایت تقدم زمانی توکن‌هاست؛ در مدل‌های زبانی خودبازگشتی، درهم‌سازی تصادفی به دلیل نادیده گرفتن تقدم ترتیبی در فرآیند سطل‌بندی، قابلیت تفکیک محتوایی را تضعیف کرده و لزوم بازگشت به توجه متراکم یا بازآموزی کامل ماتریس‌های تصویر را ایجاب می‌نماید.
\end{itemize}
