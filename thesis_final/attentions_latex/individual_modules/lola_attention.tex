\section{ماژول توجه خطی کم‌مرتبه با حافظه تُنُک (LoLA Attention)}
\label{sec:module_lola_attention}

\subsection{معرفی و انگیزه}
این مکانیزم بر مبنای مبانی نظری پیشرفته در ادبیات توجه و فشرده‌سازی \cite{vaswani2017attention} توسعه یافته است.

مکانیزم توجه \lr{LoLA (Low-Rank Linear Attention with Sparse Caching)} به منظور برطرف ساختن دو چالش عمده در استقرار مدل‌های ترنسفورمر مقیاس‌بزرگ طراحی شده است: نخست، حجم عظیم حافظه مصرفی ناشی از ماتریس‌های پروجکشن کامل، و دوم، رشد نامحدود و خطی بافر تانسورهای پیشین (\lr{KV-Cache}) در دیکودرهای خودبازگشتی. در کاربردهای بلادرنگ و سامانه‌های لبه (\lr{Edge Devices})، سربار ناشی از جابه‌جایی داده‌های حافظه پنهان بین حافظه اصلی و پردازنده گرافیکی عامل اصلی افت توان پردازشی به شمار می‌رود.

نوآوری ماژول \lr{LoLA} در ترکیب دو راهبرد مکمل است: از یک سو، استفاده از نگاشت‌های خطی کم‌رتبه (\lr{LowRankLinear}) مبتنی بر تجزیه مقادیر منفرد (\lr{SVD}) به همراه اتصالات باقیمانده یادگیرنده رتبه-پایین (\lr{LoRA}) برای کاهش وزن‌های پایه؛ و از سوی دیگر، طراحی یک سامانه سلسله‌مراتبی کش چندبخشی (\lr{LowRankCache}) که توکن‌های اخیر را به صورت دقیق، توکن‌های تاریخی را با فواصل گزینشی (\lr{Strided/Sparse})، و توکن‌های دوردست را به صورت زیرفضای تصویرشده ارتوگونال با رتبه تقلیل‌یافته (\lr{Orthogonal Low-Rank QR Projection}) نگهداری می‌نماید.

\subsection{مبانی نظری و فرمول‌بندی ریاضی}
در لایه‌های پروجکشن $\mathbf{W} \in \mathbb{R}^{d_{\text{out}} \times d_{\text{in}}}$، ماتریس وزن به حاصل‌ضرب دو ماتریس متعامد با رتبه پایین $r \ll \min(d_{\text{in}}, d_{\text{out}})$ تجزیه شده و توسط جمله باقیمانده \lr{LoRA} بازسازی می‌شود:
\begin{equation}
\mathbf{W} = \mathbf{U} \mathbf{V} + \frac{\alpha}{r_{\text{lora}}} \mathbf{B}_{\text{lora}} \mathbf{A}_{\text{lora}}
\end{equation}
که در آن $\mathbf{U} \in \mathbb{R}^{d_{\text{out}} \times r}$ و $\mathbf{V} \in \mathbb{R}^{r \times d_{\text{in}}}$ با شرایط ارتوگونالیتی اولیه $\mathbf{U}^T \mathbf{U} = \mathbf{I}$ مقداردهی شده و در طی فرآیند تنظیم دقیق تثبیت می‌گردند.

در سمت حافظه پنهان دیکودر، تانسورهای کلید و مقدار $\mathbf{K}$ و $\mathbf{V}$ در سه مخزن زمانی مجزا با ساختار چندسطحی مدیریت می‌شوند:
\begin{enumerate}
    \item \textbf{بافر توکن‌های اخیر (\lr{Recent Window}):} نگهداری دقیق $W_{\text{recent}}$ توکن گذشته برای ثبت وابستگی‌های نحوی بومی:
    \begin{equation}
    \mathbf{K}_{\text{recent}} = \mathbf{K}_{t - W_{\text{recent}} : t}
    \end{equation}
    \item \textbf{بافر نمونه‌برداری تُنُک (\lr{Strided Sparse Window}):} ذخیره توکن‌ها با گام زمانی $s > 1$ برای پوشش پنجره‌های میانی:
    \begin{equation}
    \mathbf{K}_{\text{sparse}} = \{ \mathbf{K}_i \mid i \equiv 0 \pmod s, \; i < t - W_{\text{recent}} \}
    \end{equation}
    \item \textbf{بافر تصویرسازی کم‌رتبه (\lr{Low-Rank Projected Buffer}):} کاهش بعد توکن‌های باستانی با استفاده از تجزیه ارتوگونال \lr{QR} روی ماتریس تصویر تصادفی ثابت $\mathbf{P} \in \mathbb{R}^{d \times k}$:
    \begin{equation}
    \mathbf{Q}_r, \mathbf{R}_r = \operatorname{QR}(\mathbf{K}_{\text{history}} \mathbf{P}), \quad \mathbf{K}_{\text{compressed}} = \mathbf{Q}_r \left( \mathbf{Q}_r^T \mathbf{K}_{\text{history}} \right)
    \end{equation}
\end{enumerate}
در نهایت، امتیازات توجه با الحاق این سه مخزن و با پیچیدگی بسیار کمتر از کش کامل محاسبه می‌گردد.

\subsection{معماری و پیاده‌سازی}
کلاس اصلی \lr{LoLAAttention} در ماژول \lr{Codes/framework/attentions/lola\_attention.py} قرار داشته و اصلاحات مهندسی حساسی به شرح زیر در آن پیاده شده است:
\begin{itemize}
    \item \textbf{مدیریت حافظه در انکودرهای دوطرفه (\lr{LL-FIX-01}):} کش پویا تنها در پردازش‌های دیکودر خودبازگشتی جاری فعال شده و برای مدل‌های انکودر (\lr{DistilBERT} و \lr{RoBERTa}) به طور خودکار غیرفعال می‌گردد تا از آلودگی تانسوری میان دسته‌ها و رشد بی‌رویه مصرف حافظه جلوگیری به عمل آید (کاهش حدود ۷۰ درصدی مصرف حافظه).
    \item \textbf{بازنشانی صریح حافظه کش (\lr{LL-FIX-03}):} متد \lr{cache.reset()} در ابتدای توالی‌های نامرتبط فراخوانی می‌شود تا تانسورهای معلق در حافظه بازیافت گردند.
    \item \textbf{سازگاری کامل با \lr{GQA} و \lr{RoPE}:} پیاده‌سازی توابع انطباق ابعاد سرها و افزودن موقعیت زاویه‌ای با در نظر گرفتن جابه‌جایی تانسورهای فشرده‌شده.
\end{itemize}

\subsection{محدودیت‌ها و تحلیل شکست}
با وجود کارایی بالا در سرعت و حافظه، برخی رفتارهای انحرافی شایان توجه است:
\begin{itemize}
    \item \textbf{اشباع دقت در مدل‌های با رتبه پایه (\lr{PASS\_LOW}):} دقت ۵۰.۹۲٪ در مدل‌های \lr{RoBERTa} و \lr{GPT-2} نشان می‌دهد که تقریب اولیه با رتبه ۳۲ بر روی نگاشت‌های عمیق بدون بازآموزی دقیق لایه‌های بالاتر، سبب از دست رفتن بخش قابل توجهی از اطلاعات رتبه‌های منفرد فرعی ماتریس می‌گردد.
    \item \textbf{تحلیل معماری بر روی \lr{Qwen}:} در مدل \lr{Qwen}، ثبت دقت ۵۲.۰۶٪ حاکی از تثبیت وضعیت در حاشیه پایه است. دلیل این امر، تقابل میان فشرده‌سازی سه‌بخشی کش و نحوه انتشار بردارهای چرخش موقعیتی (\lr{RoPE}) در فرکانس‌های بالا است؛ زیرفضای تصویرشده ارتوگونال در بافر توکن‌های باستانی، تعامد ذاتی فرکانس‌های زاویه‌ای توکن‌های دوردست را تا حدی مغشوش می‌سازد که منجر به تضعیف استدلال‌های حساس به ترتیب کلمات در متون طولانی می‌گردد.
\end{itemize}
