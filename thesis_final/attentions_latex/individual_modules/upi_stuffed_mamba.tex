\section{ماژول مدل فضای حالت گزینشی مامبا (Selective Mamba / UPI-Stuffed Mamba)}
\label{sec:module_upi_stuffed_mamba}

\subsection{معرفی و انگیزه}
این مکانیزم بر مبنای مبانی نظری پیشرفته در ادبیات توجه و فشرده‌سازی \cite{ref_upistuffedmamba_2024} توسعه یافته است.

مکانیزم بلوک فضای حالت گزینشی مامبا \lr{(Selective Mamba SSM Block)} که پیش‌تر با عنوان اولیه \lr{upi\_stuffed\_mamba} نام‌گذاری شده بود، با هدف جایگزینی لایه توجه استاندارد درجه دو $\mathcal{O}(N^2)$ با یک مدل فضای حالت ساختاریافته خطی $\mathcal{O}(N)$ طراحی گردیده است. در معماری مامبا، به جای تشکیل ماتریس توجه تمام‌به‌تمام، تعاملات میان توکن‌ها از طریق یک سیستم دینامیکی پیوسته با پارامترهای وابسته به ورودی (\lr{Selective SSM}) مدل‌سازی می‌شود.

این ماژول در کد مبدا چارچوب تحت نام اصیل \lr{SelectiveMamba} در فایل \lr{Codes/framework/attentions/upi\_stuffed\_mamba.py} بازطراحی و سخت‌سازی شده و با شناسه‌های \lr{selective\_mamba} و شناسه‌های پیشین \lr{upi\_stuffed\_mamba} و \lr{upi\_mamba} ثبت گردیده است. ساختار این ماژول امکان اجرای اسکن بازگشتی علّی برای مدل‌های دیکودر (\lr{GPT-2}) و اسکن متقارن دوطرفه (\lr{Bidirectional Scan}) برای مدل‌های انکودر (\lr{DistilBERT} و \lr{RoBERTa}) را به همراه قابلیت استنتاج بهینه گام‌به‌گام با حافظه $\mathcal{O}(1)$ فراهم می‌سازد.

\subsection{مبانی نظری و فرمول‌بندی ریاضی}
معماری پایه بر مبنای مدل فضای حالت خطی با پارامترهای متغیر با زمان فرمول‌بندی می‌شود:
\begin{align}
\mathbf{h}_t &= \bar{\mathbf{A}}_t \mathbf{h}_{t-1} + \bar{\mathbf{B}}_t \mathbf{x}_t \in \mathbb{R}^{D_{\text{in}} \times d_{\text{state}}}, \\
\mathbf{y}_t &= \mathbf{C}_t \mathbf{h}_t + \mathbf{D} \mathbf{x}_t \in \mathbb{R}^{D_{\text{in}}}
\end{align}
که در آن $d_{\text{state}} = 16$ بعد فضای حالت پنهان است.

\paragraph{۱. گزینش‌پذیری وابسته به ورودی و گسسته‌سازی زمانی:}
ابتدا ورودی لایه $\mathbf{X}$ به دو شاخه تعاملی و دروازه‌بانی تقسیم شده و شاخه تعاملی از یک کانوولوشن یک‌بعدی زمانی با بعد ۴ عبور می‌کند:
\begin{equation}
[\mathbf{X}_{\text{in}}, \mathbf{Z}] = \operatorname{LoRA-Linear}_{\text{in}}(\mathbf{X}), \quad \mathbf{X}_{\text{ssm}} = \operatorname{SiLU}(\operatorname{Conv1D}(\mathbf{X}_{\text{in}}))
\end{equation}
سپس پارامترهای گسسته‌سازی $\Delta_t$ و ماتریس‌های کنترل $\mathbf{B}_t$ و $\mathbf{C}_t$ مستقیماً از روی ورودی استخراج می‌شوند:
\begin{equation}
[\Delta_{\text{raw}}, \mathbf{B}_t, \mathbf{C}_t] = \operatorname{Linear}_x(\mathbf{X}_{\text{ssm}}), \quad \Delta_t = \operatorname{Softplus}(\operatorname{Linear}_{\Delta}(\Delta_{\text{raw}}))
\end{equation}
ماتریس‌های پیوسته فضا-حالت با تقریب نگاشت نمایی مرتبه اول گسسته می‌شوند:
\begin{equation}
\bar{\mathbf{A}}_t = \exp(\Delta_t \cdot \mathbf{A}), \quad \bar{\mathbf{B}}_t = \Delta_t \cdot \mathbf{B}_t
\end{equation}

\paragraph{۲. اسکن دوطرفه و دروازه‌بانی ضربی:}
برای مدل‌های دیکودر علّی، اسکن رو به جلو $\mathbf{y}^{\text{fwd}}$ محاسبه می‌شود. در انکودرهای دوطرفه، اسکن معکوس $\mathbf{y}^{\text{bwd}}$ نیز اجرا شده و خروجی به صورت میانگین متقارن حاصل می‌گردد:
\begin{equation}
\mathbf{y} = \frac{1}{2}\left(\mathbf{y}^{\text{fwd}} + \mathbf{y}^{\text{bwd}}\right)
\end{equation}
در نهایت، خروجی با شاخه گیت $\mathbf{Z}$ ادغام شده و به بعد اصلی تصویر می‌گردد:
\begin{equation}
\mathbf{O} = \operatorname{LoRA-Linear}_{\text{out}}(\mathbf{y} \odot \operatorname{SiLU}(\mathbf{Z})) \in \mathbb{R}^{B \times L \times D}
\end{equation}

\paragraph{تحلیل طیفی و نامساوی‌های پایداری خطا}
برای بررسی میزان دقت تقریب ماتریس وزن یا نگاشت توجه در ماژول \lr{upi\_stuffed\_mamba}، تفاضل ماتریس فاکتورگیری‌شده از نگاشت متراکم اصلی بر اساس نامساوی هافدینگ-فون‌نیومان ارزیابی می‌گردد. فرض کنید $\mathbf{W} \in \mathbb{R}^{d_{\text{in}} \times d_{\text{out}}}$ ماتریس وزن هدف باشد که توسط تقریب کم‌مرتبه یا پراکنده با رتبه موثر $r$ جایگزین شده است. کران بالای خطای بازسازی بر مبنای نرم فروبنیوس به صورت زیر تصریح می‌شود:
\begin{equation}
\label{eq:spectral_bound_upi_stuffed_mamba}
\| \mathbf{W} - \mathbf{W}_{r} \|_F = \sqrt{\sum_{i=r+1}^{\min(d_{\text{in}}, d_{\text{out}})} \sigma_i^2(\mathbf{W})} \le \epsilon \| \mathbf{W} \|_F
\end{equation}
که در آن $\sigma_i(\mathbf{W})$ بیانگر $i$-امین مقدار منفرد مرتب‌شده ماتریس به صورت نزولی است. نسبت حفظ انرژی طیفی $\eta_r = \frac{\sum_{i=1}^r \sigma_i^2}{\sum_{i=1}^{d} \sigma_i^2}$ تضمین می‌کند که با انتخاب آستانه $\tau \ge 0.90$، بیش از نود درصد از پراکندگی واریانس داده‌ها حفظ شده و افت سیگنال در انتشار گرادیان رو به عقب به حداقل ممکن کاهش می‌یابد.

\paragraph{تحلیل مجانبی و پیچیدگی اشتراک حافظه سخت‌افزاری}
از منظر پیچیدگی محاسباتی و انتقال داده‌ها در سلسومراتب حافظه (حافظه سراسری \lr{HBM} به حافظه محلی \lr{SRAM})، عملیات ضرب ماتریسی در این مکانیزم نیازمند تخصیص بافر موقت با ابعاد فاکتورگیری‌شده است. برای پردازش دنباله‌ای به طول $N$ با بعد مخفی $D$، بار محاسباتی هسته از فرم درجه دو به رابطه زیر تقلیل می‌یابد:
\begin{equation}
\label{eq:complexity_upi_stuffed_mamba}
\mathcal{C}_{\text{upi\_stuffed\_mamba}} = \mathcal{O}\left( B \cdot N \cdot D \cdot r + B \cdot H \cdot N \cdot d_k \right) \ll \mathcal{O}\left( B \cdot H \cdot N^2 \cdot d_k \right)
\end{equation}
این بهبود مرتبه پیچیدگی، با کاهش چشمگیر حجم تبادل تانسورها در گذرگاه‌های ارتباطی کارت گرافیک، مانع از اشباع پهنای باند حافظه (پدیده \lr{Memory-Bound}) در فرآیند استنتاج طولانی‌مدت می‌گردد.

\subsection{معماری و پیاده‌سازی}
پیاده‌سازی ماژول در فایل \lr{Codes/framework/attentions/upi\_stuffed\_mamba.py} دارای خصوصیات زیر است:
\begin{itemize}
    \item \textbf{اسکن پایتونی با مهار ماسک:} تابع \lr{\_selective\_scan} به روزرسانی حالت‌های $\mathbf{h}_t$ را با اعمال ماسک معتبر توکن‌ها جهت ممانعت از آلودگی حافظه با توکن‌های پدینگ اجرا می‌کند.
    \item \textbf{استنتاج گام‌به‌گام (\lr{step\_recurrent}):} تعبیه بافر با اندازه ثابت جهت پردازش تک‌توکنی در زمان تولید متن با پیچیدگی زمانی و حافظه‌ای $\mathcal{O}(1)$.
    \item \textbf{آداپتورهای کم‌مرتبه لورا:} استفاده از \lr{LoRALinear} در پروجکشن‌های ورودی و خروجی با رتبه $r=16$.
\end{itemize}

\subsection{محدودیت‌ها و تحلیل شکست}
تحلیل فنی پیرامون وضعیت \lr{PASS\_LOW} در مدل‌های زبانی و علل کندی توان پردازشی به شرح زیر است:
\begin{enumerate}
    \item \textbf{تحلیل سرعت پایین و توان پردازشی محدود (\lr{1,200 - 2,400 tok/s}):} در مقایسه با سایر ماژول‌ها که از ضرب‌های ماتریسی موازی و شتاب سخت‌افزاری \lr{SDPA} سود می‌برند، حلقه اسکن حالت $\mathbf{h}_t$ در این پیاده‌سازی به صورت یک حلقه ترتیبی در سطح پایتون اجرا می‌شود. این گلوگاه تفسیری پایتون مانع از درگیری کامل هسته‌های \lr{Tensor Core} پردازنده گرافیکی شده و توان عملیاتی را کاهش می‌دهد.
    \item \textbf{تحلیل همگرایی دقت روی ۵۰.۹۲٪ (\lr{Majority Class Collapse}):} پارامترهای فضای حالت $\mathbf{A}, \mathbf{B}, \mathbf{C}$ نیازمند آموزش پیش‌زمینه یا انطباق دوره‌ای طولانی‌مدت برای یادگیری انتقال حالات متوالی هستند. در بازنشانی سریع تک‌دوره‌ای با لایه‌های منجمد، بازنمایی‌های حالت قادر به ثبت روابط ظریف احساسی در جمله نبوده و مدل به سمت پیش‌بینی رده پرتکرار سوق یافته است.
    \item \textbf{عدم پشتیبانی از ساختارهای نوین در \lr{Qwen} (\lr{SKIP\_NO\_SUPPORT}):} بلوک مامبا به صورت پایه‌ای جایگزین کامل توجه استاندارد بوده و هندسه تعبیه مکانی دورانی چندبعدی و ساختار گروه‌بندی سرهای \lr{Qwen} در نسخه فعلی ماژول پیاده‌سازی نشده است.
\end{enumerate}
