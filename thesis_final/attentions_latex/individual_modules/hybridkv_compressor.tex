\section{ماژول hybridkv\_compressor (HybridKV Compressor)}
\label{sec:module_hybridkv_compressor}

\subsection{معرفی و انگیزه}
این مکانیزم بر مبنای مبانی نظری پیشرفته در ادبیات توجه و فشرده‌سازی \cite{ref_hybridkvcompressor_2024} توسعه یافته است.

در پردازش متن‌های طولانی و مدل‌های زبانی چندوجهی (\lr{MLLMs})، حافظه نهان کلید و مقدار (\lr{KV Cache}) به سرعت رشد کرده و به تنگنای اصلی حافظه با پهنای باند بالا (\lr{HBM}) در کارت‌های گرافیکی تبدیل می‌شود. بسیاری از روش‌های کاهش اندازه حافظه نهان، سیاست یکسانی را بر روی تمامی سرهای توجه (\lr{Attention Heads}) اعمال می‌کنند که موجب حذف بخش‌های حیاتی زمینه یا از دست رفتن تمرکز موضعی می‌شود.

موتور فشرده‌سازی \lr{HybridKV} بر مبنای تفکیک ساختاری سرهای توجه به دو دسته سرهای ایستا (\lr{Static Heads}) و سرهای پویا (\lr{Dynamic Heads}) طراحی شده است. این الگوریتم با اعمال سیاست‌های نگهداری متمایز تحت یک بودجه حافظه سلسله‌مراتبی، توکن‌های سینک اولیه (\lr{Attention Sinks}) و عناصر سنگین سراسری را در سرهای ایستا تثبیت نموده، در حالی که در سرهای پویا از پنجره‌های لغزان محلی (\lr{Local Sliding Window}) به همراه هرس خوشه‌ای استفاده می‌کند تا بدون تخریب ظرفیت استدلال مدل، حجم حافظه نهان پویا را به شدت کاهش دهد.

\subsection{مبانی نظری و فرمول‌بندی ریاضی}
فرض کنید برای سر توجه $h \in \{1, \dots, H\}$ در طول توالی $T$، تانسورهای کلید و مقدار به ترتیب $\mathbf{K}_h \in \mathbb{R}^{T \times d_k}$ و $\mathbf{V}_h \in \mathbb{R}^{T \times d_v}$ باشند. با توجه به ماتریس نقاب سرهای ایستا $M_{\text{static}}(h) \in \{0, 1\}$، ظرفیت مجاز حافظه نهان تحت نسبت بودجه $\beta \in (0, 1]$ به صورت زیر تعیین می‌شود:
\begin{equation}
L_{\text{max}} = \max\left(L_{\text{min}}, \lfloor T_{\text{cache}} \cdot \beta \rfloor \right)
\end{equation}
اگر سر توجه ایستا باشد ($M_{\text{static}}(h) = 1$):
\begin{itemize}
    \item تعداد $S$ توکن ابتدایی به عنوان جاذب‌های توجه (\lr{Attention Sinks}) همواره در حافظه تثبیت می‌شوند: $\mathcal{I}_{\text{sink}} = \{1, 2, \dots, S\}$.
    \item مابقی ظرفیت به توکن‌های با بیشترین انرژی توجه یا مقادیر فعال‌سازی بزرگ سراسری (\lr{Heavy Hitters}) اختصاص می‌یابد:
    \begin{equation}
    \mathcal{I}_{\text{heavy}} = \operatorname{arg\,top\,k}_{t > S} \left( \|\mathbf{k}_t\|_2 \right), \quad |\mathcal{I}_{\text{heavy}}| = L_{\text{max}} - S
    \end{equation}
\end{itemize}
اگر سر توجه پویا باشد ($M_{\text{static}}(h) = 0$):
\begin{itemize}
    \item علاوه بر توکن‌های سینک اولیه، یک پنجره لغزان محلی به طول $W$ برای حفظ آخرین توکن‌های تولیدشده در نظر گرفته می‌شود:
    \begin{equation}
    \mathcal{I}_{\text{recent}} = \{T - W + 1, \dots, T\}
    \end{equation}
    \item توکن‌های باقی‌مانده بر اساس امتیاز تشابه یا هرس توده‌ای هدایت می‌شوند.
\end{itemize}
بدین ترتیب، تانسور حافظه نهان فشرده‌شده به صورت زیر تجمیع و بازسازی می‌گردد:
\begin{equation}
\mathbf{K}_{h}^{\text{comp}} = \mathbf{K}_{h}[\mathcal{I}_{\text{retained}}, :], \quad \mathbf{V}_{h}^{\text{comp}} = \mathbf{V}_{h}[\mathcal{I}_{\text{retained}}, :]
\end{equation}
که در آن $\mathcal{I}_{\text{retained}} = \mathcal{I}_{\text{sink}} \cup \mathcal{I}_{\text{recent}} \cup \mathcal{I}_{\text{heavy}}$ است.

\paragraph{تحلیل طیفی و نامساوی‌های پایداری خطا}
برای بررسی میزان دقت تقریب ماتریس وزن یا نگاشت توجه در ماژول \lr{hybridkv\_compressor}، تفاضل ماتریس فاکتورگیری‌شده از نگاشت متراکم اصلی بر اساس نامساوی هافدینگ-فون‌نیومان ارزیابی می‌گردد. فرض کنید $\mathbf{W} \in \mathbb{R}^{d_{\text{in}} \times d_{\text{out}}}$ ماتریس وزن هدف باشد که توسط تقریب کم‌مرتبه یا پراکنده با رتبه موثر $r$ جایگزین شده است. کران بالای خطای بازسازی بر مبنای نرم فروبنیوس به صورت زیر تصریح می‌شود:
\begin{equation}
\label{eq:spectral_bound_hybridkv_compressor}
\| \mathbf{W} - \mathbf{W}_{r} \|_F = \sqrt{\sum_{i=r+1}^{\min(d_{\text{in}}, d_{\text{out}})} \sigma_i^2(\mathbf{W})} \le \epsilon \| \mathbf{W} \|_F
\end{equation}
که در آن $\sigma_i(\mathbf{W})$ بیانگر $i$-امین مقدار منفرد مرتب‌شده ماتریس به صورت نزولی است. نسبت حفظ انرژی طیفی $\eta_r = \frac{\sum_{i=1}^r \sigma_i^2}{\sum_{i=1}^{d} \sigma_i^2}$ تضمین می‌کند که با انتخاب آستانه $\tau \ge 0.90$، بیش از نود درصد از پراکندگی واریانس داده‌ها حفظ شده و افت سیگنال در انتشار گرادیان رو به عقب به حداقل ممکن کاهش می‌یابد.

\paragraph{تحلیل مجانبی و پیچیدگی اشتراک حافظه سخت‌افزاری}
از منظر پیچیدگی محاسباتی و انتقال داده‌ها در سلسومراتب حافظه (حافظه سراسری \lr{HBM} به حافظه محلی \lr{SRAM})، عملیات ضرب ماتریسی در این مکانیزم نیازمند تخصیص بافر موقت با ابعاد فاکتورگیری‌شده است. برای پردازش دنباله‌ای به طول $N$ با بعد مخفی $D$، بار محاسباتی هسته از فرم درجه دو به رابطه زیر تقلیل می‌یابد:
\begin{equation}
\label{eq:complexity_hybridkv_compressor}
\mathcal{C}_{\text{hybridkv\_compressor}} = \mathcal{O}\left( B \cdot N \cdot D \cdot r + B \cdot H \cdot N \cdot d_k \right) \ll \mathcal{O}\left( B \cdot H \cdot N^2 \cdot d_k \right)
\end{equation}
این بهبود مرتبه پیچیدگی، با کاهش چشمگیر حجم تبادل تانسورها در گذرگاه‌های ارتباطی کارت گرافیک، مانع از اشباع پهنای باند حافظه (پدیده \lr{Memory-Bound}) در فرآیند استنتاج طولانی‌مدت می‌گردد.

\subsection{معماری و پیاده‌سازی}
پیاده‌سازی این موتور در فایل \lr{Codes/framework/compression\_engine/hybridkv\_compressor.py} مستقر بوده و تحت نام \lr{hybridkv} در رجیستری فشرده‌سازی ثبت شده است. از ویژگی‌های شاخص معماری می‌توان به موارد زیر اشاره کرد:
\begin{itemize}
    \item \textbf{مدیر پویا حافظه نهان (\lr{HybridKVCache}):} کلاس مدیریت حافظه نهان که عملیات به‌روزرسانی و فشرده‌سازی برخط (\lr{update\_and\_compress}) را بدون تخصیص مکرر حافظه و با حداقل سربار اجرا می‌کند.
    \item \textbf{حفظ کامل وزن‌های ایستا (\lr{Stationary Weights Preservation}):} این موتور هیچ تغییری در وزن‌های لایه‌های شبکه ایجاد نکرده و منحصراً بر روی تانسورهای پویای حافظه نهان \lr{KV} در طول پیش‌پرکردن (\lr{Prefill}) و تولید توکن‌به‌توکن تمرکز دارد.
    \item \textbf{مدیریت ابعاد چندگانه تانسورها:} پشتیبانی خودکار از تانسورهای ورودی دو، سه و چهاربعدی با ترتیب‌های مختلف دسته‌ای و سرهای توجه.
    \item \textbf{پایداری عددی در تخلیه کش:} پیشگیری از خطای سرریز حافظه یا اشاره‌گرهای تهی با استفاده از متد \lr{reset} و تضمین حداقل طول کش مجاز.
\end{itemize}

\subsection{محدودیت‌ها و تحلیل شکست}
تحلیل فنی ساختار و سناریوهای مرزی این ماژول محدودیت‌های زیر را برجسته می‌سازد:
\begin{itemize}
    \item \textbf{عدم تاثیر بر حافظه وزن‌های مدل در توالی‌های کوتاه:} از آنجا که \lr{HybridKV} وزن‌های مدل را فشرده نمی‌کند و تنها بر حافظه نهان فعال‌سازی‌های کلید و مقدار متمرکز است، در پردازش توالی‌های کوتاه سودمندی کاهش حافظه ناچیز بوده و مزیت اصلی آن تنها در متن‌های بسیار طولانی نمایان می‌شود.
    \item \textbf{وابستگی تفکیک سرهای ایستا به کالیبراسیون قبلی:} تعیین دقیق ماسک سرهای ایستا نیازمند عبور داده‌های کالیبراسیون و بررسی پراکندگی ماتریس‌های توجه است؛ در غیاب انتساب دقیق، ممکن است سرهای با رفتار سراسری به عنوان سرهای محلی تلقی شده و توکن‌های حیاتی پیش‌زمینه هرس گردند.
\end{itemize}
