\section{تحلیل علمی و کالبدشکافی ساختاری ماژول \lr{adaptive\_sparse\_attention}}
\label{sec:adaptive_sparse_attention}

\subsection*{بخش ۱: ایده محوری و چرایی معماری \lr{(Architectural Rationale)}}
توزیع احتمالاتی در مکانیزم خودتوجهی چندسره استاندارد \lr{(Multi-Head Attention - MHA)} به دلیل بهره‌گیری از عملگر بیشینه هموار چگال \lr{(Dense Softmax)}، مقادیر اکیداً مثبت را به تمامی نشانه‌های دنباله تخصیص می‌دهد ($\mathbf{A}_{ij} > 0, \forall i, j$). این فرمول‌بندی نمایی، دو نارسایی عمیق محاسباتی و ادراکی به همراه دارد: نخست تحمیل پیچیدگی درجه دوم $\mathcal{O}(N^2)$ در زمان اجرا و مصرف حافظه گرافیکی، و دوم تزریق مداوم آنتروپی زاید و نوفه آماری ناشی از تجمع وزن‌های کوچک نشانه‌های پس‌زمینه (نظیر نشانه‌های نشانه‌گذاری و کلمات دستوری کم‌اهمیت) به درون بردار بازنمایی نهایی توکن‌ها. تحلیل‌های تجربی بر روی فضاهای پنهان ترنسفورمر نشان داده است که وزن‌های مؤثر معنایی تنها بر روی زیرمجموعه بسیار کوچکی از جفت‌توکن‌ها تمرکز دارند و توزیع توجه در لایه‌های میانی و فوقانی ماهیتی شدیداً تنک \lr{(Sparse)} از خود بروز می‌دهد.

رویکردهای تنک‌سازی کلاسیک بر پایه‌های هندسی صلب نظیر پنجره‌های لغزان موضعی \lr{(Local Sliding Windows)} یا الگوهای ساختاریافته از پیش‌تعیین‌شده (مانند \lr{Longformer} و \lr{BigBird}) بنا شده‌اند. این الگوهای استاتیک، بدون توجه به محتوای سمانتیک، ساختار اتصال گراف توجه را محدود می‌کنند؛ در نتیجه، در پیوندهای معنایی دوربرد (مانند ارجاع ضمایر به مرجع در فواصل چندصد توکنی یا ارتباط فاعل و فعل در ساختارهای درهم‌تنیده)، الگوهای هندسی صلب دچار شکست کامل می‌شوند.

معماری \lr{adaptive\_sparse\_attention} با معرفی سازوکار گزینش تنک تطبیقی وابسته به محتوا \lr{(Content-Dependent Dynamic Top-K Routing)}، این چالش را به صورت بنیادین برطرف می‌سازد. در این ماژول، هر سر توجه به صورت مستقل و بر پایه انرژی ضرب داخلی پرس‌وجو--کلید \lr{(Query-Key Dot Product)}، بودجه نشانه‌های کلیدی فعال خود را مشخص می‌کند. نشانه‌های نامرتبط به طور صریح به $-\infty$ ماسک شده و از توزیع سافت‌مکس حذف می‌گردند که معادل وزن صفر قطعی در ماتریس توجه است.

چهار مؤلفه تحلیلی و مهندسی متمایز در طراحی این ماژول تعبیه شده است:
\begin{enumerate}
    \item \textbf{تخصیص بودجه تنک‌سازی متمایز برای انکودرها و دیکودرها:} در مدل‌های انکودر دوطرفه \lr{(Bidirectional Encoders)} مانند \lr{DistilBERT} و \lr{RoBERTa}، نشانه‌ها اطلاعات بافتی را از سرتاسر دنباله گردآوری می‌کنند؛ بنابراین، اعمال تنک‌سازی تهاجمی زودهنگام به سرعت شبکه‌های ادراکی را متلاشی می‌سازد. از این رو، بودجه تنک پیش‌فرض انکودرها به صورت محافظه‌کارانه بر روی \lr{0.1} (حذف ۱۰ درصد کم‌اهمیت‌ترین‌ها) تنظیم گردیده است. در نقطه مقابل، در دیکودرهای خودبازگشت \lr{(Autoregressive Decoders)} نظیر \lr{GPT-2} و \lr{Qwen}، نرخ تنک‌سازی به \lr{0.5} (حذف ۵۰ درصد تاریخچه) ارتقا می‌یابد.
    \item \textbf{رعایت اصل عدم‌نشت علّی قبل از مرتب‌سازی \lr{(Pre-Sort Causal Masking)}:} در دیکودرها، چنانچه فیلتر بیشینه \lr{Top-K} پیش از اعمال ماسک مثلثی علّی \lr{(Causal Mask)} فراخوانی شود، توکن‌های آینده به دلیل انرژی اولیه بالا در گزینش وارد شده و سپس در مرحله ماسک علّی صفر می‌گردند؛ این پدیده موسوم به نشت علّی \lr{(Causal Data Leakage)}، توکن فعلی را از نشانه‌های معتبر گذشته محروم می‌سازد. ماژول حاضر با اعمال صریح ماسک علّی پیش از الگوریتم مرتب‌سازی، اصالت زمانی تولید توکن را تضمین می‌کند.
    \item \textbf{گریز از تله لایه‌نشانی \lr{(Escaping the Padding Trap)}:} در وظایف دسته‌بندی توالی به کمک مدل‌های علّی که ورودی‌ها از سمت راست لایه‌نشانی \lr{(Right-Padded)} می‌شوند، توکن نهایی دسته‌بندی در فاصله‌ای دور از نشانه‌های اصلی متن قرار می‌گیرد. مکانیزم‌های پنجره موضعی در این سناریو به طور کامل در میان توکن‌های تهی به دام می‌افتند. در مقابل، مکانیزم تنک تطبیقی جهانی \lr{(Global Dynamic Top-K)} بدون محدودیت مکانی، برترین کلیدها را از کل طول دنباله استخراج کرده و مسیر گرادیان مستقیم به سرنخ‌های متنی ابتدای جمله باز می‌دارد.
    \item \textbf{تثبیت گرادیان از طریق انجماد لاجیت تنک‌سازی و به‌کارگیری آداپتورهای \lr{LoRA}:} تابع گزینش گسسته رتبه برتر فاقد مشتق‌پذیری هموار است ($\nabla_k = 0$). برای ممانعت از پرش‌های ناگهانی گرادیان، پارامترهای لاجیت تنک‌سازی هر سر \lr{(sparsity\_logits)} به عنوان فراپارامتر ثابت منجمد شده \lr{(requires\_grad = False)} و تطبیق فضاهای برداری منحصراً توسط آداپتورهای \lr{LoRALinear} در طول لایه‌های پروجکشن صورت می‌پذیرد.
\end{enumerate}

\subsection*{بخش ۲: فرمول‌بندی ریاضی و مبانی نظری \lr{(Mathematical Formulation)}}
فرض کنید تانسور فعال‌سازی ورودی پرس‌وجو به صورت $\mathbf{X} \in \mathbb{R}^{B \times N_q \times D}$ و تانسور ورودی کلید-مقدار متناظر به صورت $\mathbf{X}_{kv} \in \mathbb{R}^{B \times N_k \times D}$ تعریف شوند.

\paragraph{۱. نگاشت‌های خطی مجهز به آداپتورهای کم‌مرتبه \lr{(LoRA Projections)}:}
تانسورهای ورودی توسط نگاشت‌های خطی با پارامترهای پایه منجمد $\mathbf{W}_{\text{\lr{base}}}$ و شاخه‌های سازگار کم‌مرتبه به فضاهای پرس‌وجو، کلید و مقدار نگاشت می‌شوند:
\begin{align}
\mathbf{Q}_{\text{\lr{flat}}} &= \left(\mathbf{X} \mathbf{W}_{0, Q}^\top + \frac{\alpha_{\text{\lr{lora}}}}{r_{\text{\lr{lora}}}} (\mathbf{X} \mathbf{A}_Q^\top) \mathbf{B}_Q^\top + \mathbf{b}_Q \right) \odot \gamma_Q \in \mathbb{R}^{B \times N_q \times (H \cdot d)} \\
\mathbf{K}_{\text{\lr{flat}}} &= \left(\mathbf{X}_{kv} \mathbf{W}_{0, K}^\top + \frac{\alpha_{\text{\lr{lora}}}}{r_{\text{\lr{lora}}}} (\mathbf{X}_{kv} \mathbf{A}_K^\top) \mathbf{B}_K^\top + \mathbf{b}_K \right) \odot \gamma_K \in \mathbb{R}^{B \times N_k \times (H_{kv} \cdot d)} \\
\mathbf{V}_{\text{\lr{flat}}} &= \left(\mathbf{X}_{kv} \mathbf{W}_{0, V}^\top + \frac{\alpha_{\text{\lr{lora}}}}{r_{\text{\lr{lora}}}} (\mathbf{X}_{kv} \mathbf{A}_V^\top) \mathbf{B}_V^\top + \mathbf{b}_V \right) \odot \gamma_V \in \mathbb{R}^{B \times N_k \times (H_{kv} \cdot d)}
\end{align}
که در آن $\gamma$ متغیر مقیاس‌سنجی خروجی آداپتور \lr{(output\_scale)} است.

\paragraph{۲. بازآرایی ساختار چندسره و کدگذاری موقعیت \lr{(RoPE \& Relative Bias)}:}
تانسورها به ابعاد ۴-بعدی $[B, H, N, d]$ تفکیک شده و دوران مکانی چرخشی \lr{(RoPE)} بر روی ابعاد سرها اعمال می‌گردد. همچنین چنانچه مدل نیازمند تخصیص سوگیری موقعیت نسبی \lr{(Relative Position Bias)} نظیر معماری \lr{T5} باشد، ماتریس سوگیری بر مبنای فاصله‌بندی لگاریتمی \lr{(Logarithmic Bucketing)} با ۳۲ باکت محاسبه می‌شود:
\begin{equation}
\mathbf{B}_{\text{\lr{rel}}} = \operatorname{RelativeBucketing}(N_q, N_k, \text{\lr{num\_buckets}}=32, \text{\lr{max\_dist}}=128) \in \mathbb{R}^{1 \times H \times N_q \times N_k}
\end{equation}

در مدل‌های دارای توجه پرس‌وجوی گروهی \lr{(GQA)}، تانسورهای کلید و مقدار با نرخ تکرار $G = H / H_{kv}$ در راستای بعد سرها بسط می‌یابند. در یک پیاده‌سازی بهینه از طریق عملگر \lr{.expand()}، همتاسازی حافظه صورت نگرفته و تنها خاصیت \lr{Broadcasting} فعال می‌گردد: $\mathbf{K} \leftarrow \operatorname{expand}(\mathbf{K}, G)$.

\paragraph{۳. قطعه‌بندی پرس‌وجو و محاسبه ماتریس خام امتیازات \lr{(Query Chunking)}:}
برای بهینه‌سازی ترافیک ثبات‌ها و جایگیری ماتریس در کش \lr{L2}، محاسبات بر روی قطعات پرس‌وجو با اندازه ثابت $C_q = 64$ انجام می‌پذیرد. برای قطعه $c \in \{0, \dots, \lceil N_q / C_q \rceil - 1\}$:
\begin{equation}
\mathbf{S}_c = \frac{\mathbf{Q}[:, :, q_{\text{\lr{start}}}:q_{\text{\lr{end}}}, :] \cdot \mathbf{K}^\top}{\sqrt{d}} + \mathbf{B}_{\text{\lr{rel}}, c} \in \mathbb{R}^{B \times H \times C_q \times N_k}
\end{equation}

\paragraph{۴. اعمال همزمان ماسک لایه‌نشانی و ماسک علّی پیش از گزینش تنک:}
ماسک توکن‌های نامعتبر \lr{(Padding Mask)} و ماسک علّی قطعه‌بندی‌شده \lr{(Chunked Causal Mask)} به طور همزمان پیش از فیلترینگ اعمال می‌گردند:
\begin{equation}
\hat{S}_{c, b, h, i, j} = \begin{cases} S_{c, b, h, i, j} & \text{\lr{if }} \mathbf{M}_{\text{\lr{valid}}, j} = 1 \text{\lr{ and }} (j \le q_{\text{\lr{start}}} + i \text{\lr{ for Decoders}}) \\ -10000.0 & \text{\lr{otherwise}} \end{cases}
\end{equation}

\paragraph{۵. گزینش برداری آستانه پویای \lr{Top-K} و تنک‌سازی:}
به ازای هر سر توجه $h \in \{1, \dots, H\}$، نسبت تنک‌سازی $\gamma_h$ و تعداد نشانه‌های مجاز $k_h$ از لاجیت‌های منجمد استخراج می‌شود:
\begin{equation}
\gamma_h = \operatorname{clamp}\left(\sigma(\omega_h), 0.0, 0.95\right), \quad k_h = \operatorname{clamp}\left(\left\lfloor (1.0 - \gamma_h) \cdot N_k \right\rfloor, 1, N_k\right)
\end{equation}
ماتریس نمرات در امتداد بعد کلیدها ($N_k$) به صورت نزولی مرتب‌سازی موازی شده و آستانه قطع سطرها $\tau_{b, h, i}$ با عملگر واکشی برداری \lr{(Gather)} در ایندکس $k_h - 1$ به دست می‌آید:
\begin{equation}
\tau_{b, h, i} = \operatorname{SortDesc}(\hat{\mathbf{S}}_c, \text{\lr{dim}}=-1)[b, h, i, k_h - 1]
\end{equation}
ماسک‌گذاری صریح درایه‌های پایین‌تر از آستانه:
\begin{equation}
\tilde{S}_{c, b, h, i, j} = \begin{cases} \hat{S}_{c, b, h, i, j} & \text{\lr{if }} \hat{S}_{c, b, h, i, j} \ge \tau_{b, h, i} \\ -10000.0 & \text{\lr{if }} \hat{S}_{c, b, h, i, j} < \tau_{b, h, i} \end{cases}
\end{equation}

\paragraph{۶. نرمال‌سازی سافت‌مکس تنک و تجمیع زمینه:}
برای ارتقای پایداری عددی، تبدیل به \lr{FP32} پیش از جمع و اعمال دراپ‌اوت صورت گرفته و سپس در مقادیر ضرب می‌گردد:
\begin{equation}
\mathbf{O}_c = \operatorname{Dropout}\left(\operatorname{Softmax}(\tilde{\mathbf{S}}_c, \text{\lr{dim}}=-1)\right) \cdot \mathbf{V} \in \mathbb{R}^{B \times H \times C_q \times d}
\end{equation}
قطعات در امتداد طول دنباله به هم متصل شده و خروجی نهایی از پروجکشن خروجی عبور می‌کند. در انکودرهای تودرتو \lr{(RoBERTa/BERT)} پروجکشن خروجی بای‌پس می‌شود.

\subsection*{بخش ۴: تحلیل معماری سخت‌افزار و مدل سقف کارایی \lr{(Hardware \& Roofline Analysis)}}
ارزیابی محاسباتی این ماژول بر روی پردازنده گرافیکی \lr{NVIDIA GeForce RTX 4070 Laptop} مبتنی بر معماری \lr{Ada Lovelace} انجام شده است.

\paragraph{۱. مدل سقف کارایی \lr{(Roofline Model)} و رژیم عملیاتی:}
نقطه زانویی سقف کارایی پردازنده گرافیکی حدود $390.6 \text{\lr{ FLOP/Byte}}$ تعیین می‌شود. در الگوریتم توجه استاندارد چگال، شدت محاسباتی در حدود $64 \text{\lr{ FLOP/Byte}}$ بوده و مکانیزم عمیقاً در رژیم \textbf{حافظه‌محور \lr{(Memory-Bound)}} قفل می‌گردد. در این ناحیه، سرعت پردازش توسط پهنای باند حافظه محدود می‌شود.

\paragraph{۲. مهار حافظه از طریق قطعه‌بندی پرس‌وجو (Query Chunking):}
ماژول \lr{adaptive\_sparse\_attention} با قطعه‌بندی سطرها به ابعاد $C_q = 64$، ترافیک داده فعال را کنترل می‌کند. ردپای بافر امتیازات در هر قطعه برابر است با:
\begin{equation}
\mathcal{M}_{\text{\lr{chunk}}} = 8 \times 12 \times 64 \times 512 \times 2 \text{\lr{ Bytes}} \approx 6.29 \text{\lr{ MB}}
\end{equation}
از آنجا که این مقدار بسیار کوچکتر از ۳۲ مگابایت کش \lr{L2} است، از خطای \lr{OOM} و فراخوانی مکرر \lr{GDDR6} به طرز چشمگیری جلوگیری می‌شود.

\subsection*{بخش ۶: نقد فنی، گلوگاه‌های پنهان و افق‌های توسعه \lr{(Bottlenecks \& Optimizations)}}
ساختار و پیاده‌سازی این ماژول نمونه‌ای از مهندسی یادگیری عمیق در سطح صنعتی \lr{(Production-Ready)} است. با این وجود، بهینه‌سازی‌های زیر ظرفیت مدل را به سقف تئوریک آن نزدیک می‌سازد:

\begin{itemize}
    \item \textbf{هزینه سنگین \lr{torch.sort} در دنباله‌های طولانی:} با اینکه ترکیب \lr{sort} و \lr{gather} محدودیت فیلترینگ پویای پایتورچ را دور می‌زند، الگوریتم مرتب‌سازی کامل دارای پیچیدگی $\mathcal{O}(N_k \log N_k)$ است. برای طول‌های بیش از ۴۰۹۶، این بخش به گلوگاه اصلی پردازنده گرافیکی تبدیل می‌شود. توسعه یک کرنل سفارشی \lr{Triton} که مستقیماً $K$-امین عنصر را بدون مرتب‌سازی کامل بیابد، راه‌حل نهایی خواهد بود.
    \item \textbf{یکپارچه‌سازی پویای اندازه قطعه \lr{(Dynamic Chunk Size)}:} مقدار $C_q = 64$ در کد سخت‌افزاری \lr{(Hardcoded)} است. یک بهینه‌سازی ساختاری، افزودن یک گزاره شرطی نظیر \lr{if $N\_q \le 512$} است تا برای طول‌های کوتاه از حلقه \lr{for} صرف‌نظر شده و ماتریس به صورت یکپارچه \lr{(Unchunked)} پردازش گردد تا از سربار راه‌اندازی متوالی کرنل‌های پایتون \lr{(Kernel Launch Overhead)} پرهیز شود.
    \item \textbf{همتاسازی زائد در \lr{GQA}:} در مدل‌هایی با \lr{GQA} بالا نظیر \lr{LLaMA-3}، دستور \lr{repeat\_kv} باعث کپی شدن فیزیکی تانسورها \lr{(Materialization)} در حافظه می‌شود. با بهره‌گیری از خاصیت \lr{Broadcasting} و تابع \lr{.expand()} می‌توان از اتلاف حافظه در این مرحله ممانعت به عمل آورد.
    \item \textbf{پایداری عددی \lr{(Dtype Casting)}:} بازگرداندن داده‌ها به \lr{dtype} اصلی \lr{(FP16)} پیش از اعمال سافت‌مکس ممکن است دقت‌های بسیار کوچک در توزیع تنک را به صفر میل دهد. اجرای ضرب نهایی \lr{attn\_chunk} در $V$ در فضای محاسباتی \lr{FP32} پایداری عددی را به شدت ارتقا خواهد داد.
\end{itemize}
