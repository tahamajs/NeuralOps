\section{ماژول توجه لینفورمر با انقباض رتبه بریده‌شده (TLinFormer Attention)}
\label{sec:module_tlinformer_attention}

\subsection{معرفی و انگیزه}
این مکانیزم بر مبنای مبانی نظری پیشرفته در ادبیات توجه و فشرده‌سازی \cite{ref_tlinformerattention_2024} توسعه یافته است.

مکانیزم توجه لینفورمر بریده‌شده \lr{(Truncated LinFormer Attention - TLinFormer)} با الهام از معماری برجسته \lr{LinFormer} (ارائه‌شده توسط وانگ و همکاران، ۲۰۲۰) و با هدف شکستن دیوار پیچیدگی محاسباتی درجه دو $\mathcal{O}(N^2)$ طراحی شده است. ایده بنیادین لینفورمر بر این فرض ریاضی استوار است که ماتریس شباهت توجه با ابعاد $N \times N$ را می‌توان از طریق پروجکشن‌های خطی با رتبه کم $r \ll N$ بر روی محور طول دنباله تقریب زد، به گونه‌ای که ابعاد دنباله کلید و مقدار به عددی ثابت و بسیار کوچک تقلیل یابد.

ماژول \lr{TLinFormerAttention} (ثبت‌شده در رجیستری با شناسه‌های \lr{tlinformer} و \lr{tlinformer\_attention}) این مفهوم را برای سازگاری کامل با مدل‌های روز مهندسی و سخت‌سازی کرده است. ویژگی اصلی پیاده‌سازی حاضر، تعبیه ماتریس‌های پروجکشن یادگیرنده دنباله، پشتیبانی از تعبیه مکانی دورانی (\lr{RoPE})، تطابق با سرهای گروه‌بندی‌شده (\lr{GQA}) و ماسک‌گذاری پیش‌گیرانه پدینگ جهت ممانعت از نشت اطلاعات توکن‌های زاید است.

\subsection{مبانی نظری و فرمول‌بندی ریاضی}
فرض کنید $\mathbf{X} \in \mathbb{R}^{B \times N \times D}$ ماتریس حالت‌های پنهان ورودی باشد. ابتدا پروجکشن‌های پرس‌وجو، کلید و مقدار با استفاده از نگاشت‌های خطی مجهز به لورا تشکیل می‌شوند:
\begin{align}
\mathbf{Q} &= \operatorname{LoRA-Linear}_Q(\mathbf{X}) \in \mathbb{R}^{B \times H \times N \times d}, \\
\mathbf{K} &= \operatorname{LoRA-Linear}_K(\mathbf{X}) \in \mathbb{R}^{B \times H_{kv} \times N \times d}, \\
\mathbf{V} &= \operatorname{LoRA-Linear}_V(\mathbf{X}) \in \mathbb{R}^{B \times H_{kv} \times N \times d}
\end{align}

\paragraph{۱. انقباض خطی بعد دنباله (Sequence Contraction):}
ماژول مجهز به دو ماتریس پروجکشن دنباله یادگیرنده $\mathbf{P}_K, \mathbf{P}_V \in \mathbb{R}^{r \times N_{\text{max}}}$ است که بعد طول دنباله کلید و مقدار را از $N$ به $r$ منقبض می‌نمایند. برای طول دنباله جاری $N$، ماتریس برش‌خورده $\mathbf{P} \in \mathbb{R}^{r \times N}$ بر روی کلید و مقدار ضرب می‌شود:
\begin{align}
\mathbf{K}_{\text{proj}} &= \mathbf{P}_K \mathbf{K}_{\text{masked}} \in \mathbb{R}^{B \times H \times r \times d}, \\
\mathbf{V}_{\text{proj}} &= \mathbf{P}_V \mathbf{V}_{\text{masked}} \in \mathbb{R}^{B \times H \times r \times d}
\end{align}
پیش از اعمال این ضرب، به منظور ممانعت از نشت نویز توکن‌های پدینگ به داخل فضای فشرده زیرفضا، ماسک باینری معتبرسازی توکن‌ها اعمال می‌گردد:
\begin{equation}
\mathbf{K}_{\text{masked}} = \mathbf{K} \odot \mathbf{M}_{\text{valid}}, \quad \mathbf{V}_{\text{masked}} = \mathbf{V} \odot \mathbf{M}_{\text{valid}}
\end{equation}

\paragraph{۲. محاسبه ماتریس توجه نامتقارن و خروجی:}
ماتریس همبستگی توجه میان پرس‌وجو با ابعاد کامل $N$ و کلید فشرده با طول $r$ با پیچیدگی $\mathcal{O}(N \cdot r \cdot d)$ تشکیل می‌گردد:
\begin{equation}
\mathbf{S} = \frac{\mathbf{Q} \mathbf{K}_{\text{proj}}^\top}{\sqrt{d}} \in \mathbb{R}^{B \times H \times N \times r}
\end{equation}
\begin{equation}
\mathbf{A} = \operatorname{Softmax}(\mathbf{S}), \quad \mathbf{O}_{\text{heads}} = \mathbf{A} \mathbf{V}_{\text{proj}} \in \mathbb{R}^{B \times H \times N \times d}
\end{equation}
در نهایت، سرها ادغام شده و از لایه پروجکشن خروجی عبور داده می‌شوند:
\begin{equation}
\mathbf{Y} = \operatorname{LoRA-Linear}_O(\mathbf{O}_{\text{heads}}) \in \mathbb{R}^{B \times N \times D}
\end{equation}

\paragraph{تحلیل طیفی و نامساوی‌های پایداری خطا}
برای بررسی میزان دقت تقریب ماتریس وزن یا نگاشت توجه در ماژول \lr{tlinformer\_attention}، تفاضل ماتریس فاکتورگیری‌شده از نگاشت متراکم اصلی بر اساس نامساوی هافدینگ-فون‌نیومان ارزیابی می‌گردد. فرض کنید $\mathbf{W} \in \mathbb{R}^{d_{\text{in}} \times d_{\text{out}}}$ ماتریس وزن هدف باشد که توسط تقریب کم‌مرتبه یا پراکنده با رتبه موثر $r$ جایگزین شده است. کران بالای خطای بازسازی بر مبنای نرم فروبنیوس به صورت زیر تصریح می‌شود:
\begin{equation}
\label{eq:spectral_bound_tlinformer_attention}
\| \mathbf{W} - \mathbf{W}_{r} \|_F = \sqrt{\sum_{i=r+1}^{\min(d_{\text{in}}, d_{\text{out}})} \sigma_i^2(\mathbf{W})} \le \epsilon \| \mathbf{W} \|_F
\end{equation}
که در آن $\sigma_i(\mathbf{W})$ بیانگر $i$-امین مقدار منفرد مرتب‌شده ماتریس به صورت نزولی است. نسبت حفظ انرژی طیفی $\eta_r = \frac{\sum_{i=1}^r \sigma_i^2}{\sum_{i=1}^{d} \sigma_i^2}$ تضمین می‌کند که با انتخاب آستانه $\tau \ge 0.90$، بیش از نود درصد از پراکندگی واریانس داده‌ها حفظ شده و افت سیگنال در انتشار گرادیان رو به عقب به حداقل ممکن کاهش می‌یابد.

\paragraph{تحلیل مجانبی و پیچیدگی اشتراک حافظه سخت‌افزاری}
از منظر پیچیدگی محاسباتی و انتقال داده‌ها در سلسومراتب حافظه (حافظه سراسری \lr{HBM} به حافظه محلی \lr{SRAM})، عملیات ضرب ماتریسی در این مکانیزم نیازمند تخصیص بافر موقت با ابعاد فاکتورگیری‌شده است. برای پردازش دنباله‌ای به طول $N$ با بعد مخفی $D$، بار محاسباتی هسته از فرم درجه دو به رابطه زیر تقلیل می‌یابد:
\begin{equation}
\label{eq:complexity_tlinformer_attention}
\mathcal{C}_{\text{tlinformer\_attention}} = \mathcal{O}\left( B \cdot N \cdot D \cdot r + B \cdot H \cdot N \cdot d_k \right) \ll \mathcal{O}\left( B \cdot H \cdot N^2 \cdot d_k \right)
\end{equation}
این بهبود مرتبه پیچیدگی، با کاهش چشمگیر حجم تبادل تانسورها در گذرگاه‌های ارتباطی کارت گرافیک، مانع از اشباع پهنای باند حافظه (پدیده \lr{Memory-Bound}) در فرآیند استنتاج طولانی‌مدت می‌گردد.

\subsection{معماری و پیاده‌سازی}
پیاده‌سازی ماژول در فایل \lr{Codes/framework/attentions/tlinformer\_attention.py} دارای ویژگی‌های زیر است:
\begin{itemize}
    \item \textbf{انتقال چندمعماری وزن‌ها (\lr{copy\_weights}):} قابلیت انتقال پارامترها از انواع معماری‌های \lr{DistilBERT, RoBERTa, GPT-2} و \lr{Qwen} با اعتبارسنجی نُرم فروبنیوس ($\ge 10^{-4}$).
    \item \textbf{انقباض ایمن دنباله:} پیاده‌سازی ضرب انقباضی $\mathbf{K}_{\text{proj}} = \mathbf{P} \mathbf{K}$ با رعایت ترتیب ابعاد تانسوری و حفظ پایداری گرادیان.
    \item \textbf{انطباق با RoPE و ماسک علّی:} پشتیبانی از امبدینگ موقعیتی چرخشی و اعمال ماسک در دیکودرها.
\end{itemize}

\subsection{محدودیت‌ها و تحلیل شکست}
تحلیل رفتار عملکردی ماژول بر روی شاخص‌های دقت و علت قرارگیری در رده \lr{PASS\_LOW} به شرح زیر است:
\begin{enumerate}
    \item \textbf{تحلیل همگرایی دقت روی ۵۰.۹۲٪ و شاخص F1 معادل ۰.۶۷۴۸:} ثبت رقم یکنواخت دقت \lr{50.92\%} در هر سه مدل \lr{DistilBERT}، \lr{RoBERTa} و \lr{GPT-2}، بیانگر پدیده کلاسیک «پیش‌بینی اکثریت رده \lr{(Majority Class Collapse)}» در مجموعه آزمون \lr{SST-2} است (که توزیع کلاس مثبت آن دقیقاً ۵۰.۹۲٪ است).
    \item \textbf{ناسازگاری پروجکشن تصادفی اولیه در گام‌های محدود بازیابی:} ماتریس‌های انقباض دنباله $\mathbf{P}_K$ و $\mathbf{P}_V$ با توزیع تصادفی مقداردهی اولیه می‌شوند. در فرآیند ارزیابی بنچمارک با تعداد گام‌های کم بازیابی \lr{(Few-step Recovery)}، گرادیان‌های لورا قادر به آموزش کامل نگاشت فضای دنباله $N \to r$ نبوده‌اند؛ در نتیجه خروجی توجه به یک بردار میانگین‌گیری شبه‌یکنواخت تبدیل شده و مدل به سمت پیش‌بینی کلاس اکثریت میل کرده است.
    \item \textbf{تخریب ساختار ترتیبی در دیکودر علّی (\lr{GPT-2}):} اعمال ضرب ماتریسی $\mathbf{P} \mathbf{K}$ در سراسر محور توالی، تفکیک زمانی توکن‌های قبل و بعد را مختل نموده و ماهیت جهت‌دار زنجیره استنتاج در مدل‌های خودبازگشتی را تضعیف می‌کند.
\end{enumerate}
