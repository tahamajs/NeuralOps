\section{ماژول توجه خطی گیت‌دار (Gated Linear Attention)}
\label{sec:module_gated_linear_attention}

\subsection{معرفی و انگیزه}
مکانیزم توجه خطی گیت‌دار \lr{(Gated Linear Attention - GLA)} به عنوان پلی نوآورانه میان مدل‌های توجه استاندارد ترنسفورمر با پیچیدگی محاسباتی و حافظه‌ای مرتبه دوم $\mathcal{O}(N^2)$ و مدل‌های حالت فضای خطی \lr{(SSM / RNN)} با پیچیدگی $\mathcal{O}(N)$ توسعه یافته است. در الگوهای سنتی خودتوجهی، هزینه پردازش و ردپای حافظه با افزایش طول دنباله ورودی به صورت مربعی منفجر شده و مانع از مقیاس‌پذیری کارآمد در پردازش متون و اسناد بسیار طولانی می‌گردد. اگرچه تقریب‌های پیشین توجه خطی با استفاده از ویژگی‌های کرنلی ضرب‌پذیر پیچیدگی را به $\mathcal{O}(N)$ رساندند، اما فقدان مکانیزم تضعیف وابسته به داده منجر به فروپاشی ظرفیت حافظه تداعی‌گر و فراموشی شدید اطلاعات متنی دوربرد در مقایسه با تبدیل غیرخطی بیشینه هموار \lr{(Softmax)} می‌شد.

ایده محوری در ماژول \lr{GatedLinearAttention}، تلفیق بازگشت‌پذیری خطی چندسره با گیت‌های زوال وابسته به ورودی \lr{(Data-Dependent Decay Gating)} و گیتینگ ضربی خروجی بر پایه تابع فعال‌ساز \lr{SiLU} است. این معماری به هر سر توجه اجازه می‌دهد به صورت کامپکت و پویا، نرخ نگهداری و میرایی اطلاعات در طول زمان را تنظیم نموده و پایداری ماتریس حافظه را تضمین نماید. همچنین به کارگیری نرمال‌سازی مخرج کسری مانع از سرریز یا ناپایداری‌های مقیاسی در دنباله‌های طولانی می‌گردد.

\subsection{مبانی نظری و فرمول‌بندی ریاضی}
فرض کنید $\mathbf{X} \in \mathbb{R}^{B \times L \times D}$ ماتریس بازنمایی لایه ورودی باشد که در آن $B$ اندازه دسته، $L$ طول دنباله و $D$ بعد پنهان مدل است. در معماری \lr{GatedLinearAttention}، نگاشت‌های پروجکشن پرس‌وجو، کلید، مقدار و گیت خروجی با نگاشت‌های خطی مجهز به تطبیق‌دهنده‌های کم‌مرتبه لورا \lr{(LoRA)} محاسبه می‌شوند:
\begin{align}
\mathbf{Q} &= \operatorname{LoRA-Linear}_Q(\mathbf{X}) \in \mathbb{R}^{B \times L \times (H \cdot d)}, \\
\mathbf{K} &= \operatorname{LoRA-Linear}_K(\mathbf{X}) \in \mathbb{R}^{B \times L \times (H_{kv} \cdot d)}, \\
\mathbf{V} &= \operatorname{LoRA-Linear}_V(\mathbf{X}) \in \mathbb{R}^{B \times L \times (H_{kv} \cdot d)}, \\
\mathbf{Z} &= \operatorname{LoRA-Linear}_Z(\mathbf{X}) \in \mathbb{R}^{B \times L \times (H \cdot d)}
\end{align}
که در آن $H$ تعداد سرهای پرس‌وجو، $H_{kv}$ تعداد سرهای کلید/مقدار (جهت پشتیبانی از معماری \lr{GQA}) و $d$ بعد هر سر توجه است. در صورت نابرابری سرهای کلید با پرس‌وجو، عملیات تکرار \lr{repeat\_kv} اعمال می‌گردد.

برای تضمین اکید نامنفی بودن مولفه‌های فضا و پایداری ماتریس انتقال حالت، کرنل غیرخطی $\phi(x) = \operatorname{ELU}(x) + 1.0$ بر روی پرس‌وجوها و کلیدها اعمال می‌شود:
\begin{equation}
\tilde{\mathbf{q}}_t = \operatorname{ELU}(\mathbf{q}_t) + 1.0, \quad \tilde{\mathbf{k}}_t = \operatorname{ELU}(\mathbf{k}_t) + 1.0
\end{equation}

گیت‌های زوال وابسته به ورودی $\boldsymbol{\lambda} \in (0, 1)^{B \times H \times L}$ از ترکیب پروجکشن خطی محتوایی و پارامتر یادگیرنده لگاریتمی زوال $\mathbf{w}_{\text{decay}}$ استخراج می‌شوند:
\begin{equation}
\boldsymbol{\lambda}_t = \sigma\left(\mathbf{W}_{\text{decay}} \mathbf{x}_t + \mathbf{b}_{\text{decay}} + \log \boldsymbol{\lambda}_{\text{init}}\right)
\end{equation}

در فاز پیشرو علّی (مورد استفاده در مدل‌های دیکودر مانند \lr{GPT-2})، معادله اسکن موازی تجمعی زوال از طریق ماتریس تفاضل لگاریتمی تعریف می‌گردد:
\begin{equation}
c_t = \sum_{\tau=1}^t \log \boldsymbol{\lambda}_\tau, \quad D_{i, j}^{\text{causal}} = \exp\left(c_i - c_j\right) \cdot \mathbb{I}_{[i \ge j]}
\end{equation}
سپس خروجی توجه صورت و مخرج به صورت موازی محاسبه می‌شوند:
\begin{equation}
\mathbf{S}_{i, j} = \frac{\tilde{\mathbf{q}}_i \tilde{\mathbf{k}}_j^T}{\sqrt{d}}, \quad
\mathbf{Num}_i = \sum_{j=1}^i \left(\mathbf{S}_{i, j} \cdot D_{i, j}^{\text{causal}}\right) \mathbf{v}_j, \quad
\operatorname{Denom}_i = \sum_{j=1}^i |\mathbf{S}_{i, j}| \cdot D_{i, j}^{\text{causal}}
\end{equation}
\begin{equation}
\mathbf{y}_i^{\text{fwd}} = \frac{\mathbf{Num}_i}{\operatorname{Denom}_i + \epsilon}
\end{equation}
برای مدل‌های انکودر دوطرفه (نظیر \lr{DistilBERT} و \lr{RoBERTa})، اسکن پادعلّی معکوس $\mathbf{y}^{\text{bwd}}$ با ماتریس زوال بالامثلثی $D_{i, j}^{\text{anti}} = \exp(c_j - c_i) \cdot \mathbb{I}_{[i \le j]}$ نیز محاسبه شده و برآیند دوطرفه متوازن به دست می‌آید:
\begin{equation}
\mathbf{y} = \frac{1}{2} \left(\mathbf{y}^{\text{fwd}} + \mathbf{y}^{\text{bwd}}\right)
\end{equation}
در گام نهایی، پس از عبور از لایه نرمال‌سازی حالت \lr{LayerNorm} روی بعد سرها، مدولاسیون گیت خروجی با تابع \lr{SiLU} فعال شده و تصویر خطی خروجی بازتولید می‌شود:
\begin{equation}
\mathbf{o}_t = \operatorname{LoRA-Linear}_O\left( \operatorname{LayerNorm}(\mathbf{y}_t) \odot \operatorname{SiLU}(\mathbf{z}_t) \right) \cdot \gamma_{\text{scale}}
\end{equation}

\subsection{معماری و پیاده‌سازی}
کلاس پایتونی اصلی این ماژول تحت عنوان \lr{GatedLinearAttention} در فایل \lr{Codes/framework/attentions/gated\_linear\_attention.py} پیاده‌سازی شده و از کلاس پایه \lr{BaseAttention} ارث‌بری می‌کند. این ماژول در رجیستری چارچوب با شناسه‌های \lr{gated\_linear\_attention} و \lr{gated\_linear\_attn} با پشتیبانی رسمی از ماسک‌گذاری علّی و \lr{GQA} ثبت گردیده است.

مهم‌ترین مشخصات فنی و آرگومان‌های کلاس عبارتند از:
\begin{itemize}
    \item \textbf{آرگومان‌های مقداردهی:} متد \lr{\_\_init\_\_} ورودی‌های \lr{config}، \lr{rank}، \lr{d\_model}، \lr{num\_heads}، \lr{lora\_rank=16}، \lr{lora\_alpha=32.0} و \lr{is\_decoder} را دریافت می‌کند. ابعاد سرهای توجه $d = D/H$ و تعداد سرهای کلید/مقدار به شکل خودکار استخراج می‌شوند.
    \item \textbf{لایه‌های خطی لورا:} تمامی پروجکشن‌های اصلی $\mathbf{Q}, \mathbf{K}, \mathbf{V}, \mathbf{Z}$ و خروجی با ماژول اختصاصی \lr{LoRALinear} پیاده‌سازی شده‌اند که وزن‌های پایه پیش‌آموزش‌دیده در آن‌ها منجمد \lr{(freeze)} شده و تنها ماتریس‌های کم‌مرتبه $\mathbf{A}$ و $\mathbf{B}$ با مقیاس‌گذاری $\alpha / r$ آموزش‌پذیر باقی می‌مانند.
    \item \textbf{انتقال وزن‌های پیش‌آموزش‌دیده:} متد \lr{copy\_weights} وزن‌های لایه‌های اصلی مدل‌های پایه گوناگون شامل ساختارهای مسطح (\lr{q\_lin, k\_lin, v\_lin, out\_lin} در \lr{DistilBERT})، ساختارهای تودرتو انکودر (\lr{query, key, value, dense} در \lr{RoBERTa})، ماتریس‌های ادغام‌شده \lr{c\_attn} در \lr{GPT-2} و پروجکشن‌های تجزیه‌شده \lr{q\_proj, k\_proj, v\_proj, o\_proj} در \lr{Qwen} را به دقت استخراج و نگاشت می‌نماید.
    \item \textbf{سازگاری با موقعیت‌یابی چرخشی:} تابع \lr{apply\_rotary\_pos\_emb} تطابق کامل تعبیه‌های مکانی \lr{RoPE} را در مدل‌های مدرن تضمین می‌کند.
    \item \textbf{بهینه‌سازی حافظه گرادیان:} در صورت فعال بودن فاز آموزش، اسکن موازی \lr{\_gla\_scan\_parallel} از طریق مکانیزم بازرسی حافظه \lr{torch.utils.checkpoint} اجرا می‌شود تا مصرف حافظه فعال‌سازها کاهش یابد.
\end{itemize}

\subsection{محدودیت‌ها و تحلیل شکست}
با وجود کارایی چشمگیر محاسباتی و سخت‌افزاری، دقت ارزیابی‌شده در سه مدل انکودر و دیکودر در تراز پایه ۵۰.۹۲٪ با وضعیت \lr{PASS\_LOW} قرار دارد و در مدل \lr{Qwen} مقدار دقت ۰.۰۰٪ با برچسب \lr{SKIP\_NO\_SUPPORT} گزارش شده است. ریشه‌یابی فنی و معماری این پدیده به شرح زیر است:
\begin{enumerate}
    \item \textbf{تحلیل شکست مدل \lr{Qwen} (\lr{Accuracy = 0.0\% / SKIP\_NO\_SUPPORT}):} در نسخه ارزیابی‌شده، ساختار تانسوری سرهای گروه‌بندی‌شده نامتقارن \lr{GQA} و لایه‌بندی انحصاری مدل چت \lr{Qwen} بدون بازآموزی عمیق اولیه تطبیق‌دهنده‌های لورا نتوانسته است خطای گرادیان استنتاج صفر-شات را بازیابی کند. عدم تطابق کامل تعبیه‌های پیشرفته موقعیتی \lr{RoPE} با اسکن ماتریسی لگاریتمی زوال منجر به خروج شاخه از ارزیابی خط لوله LoRA شده و وضعیت شکست سازگاری معماری را رقم زده است.
    \item \textbf{توقف در حد تصادفی ۵۰.۹۲٪ (\lr{PASS\_LOW}):} دقت ۵۰.۹۲٪ دقیقا منطبق بر نسبت اکثریت رده در مجموعه اعتبارسنجی \lr{SST-2} است که نشان می‌دهد لایه بدون فین‌تیونینگ سراسری در بازه همگرایی محدود، پیش‌بینی‌ها را به سمت کلاس با احتمال غالب شیفت داده است. مکانیزم زوال با پروجکشن تصادفی اولیه $\mathbf{W}_{\text{decay}}$ پیش از تنظیم وزن‌ها به عنوان فیلتر بالاگذر شدید عمل کرده و توالی‌های همبستگی متنی بلند را تا حد زیادی تضعیف می‌سازد.
\end{enumerate}
