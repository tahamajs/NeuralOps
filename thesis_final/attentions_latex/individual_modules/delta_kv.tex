\section{ماژول delta\_kv (Delta-KV Compressor)}
\label{sec:module_delta_kv}

\subsection{معرفی و انگیزه}
این مکانیزم بر مبنای مبانی نظری پیشرفته در ادبیات توجه و فشرده‌سازی \cite{ref_deltakv_2024} توسعه یافته است.

در مدل‌های ترنسفورمر، به ویژه در کاربردهای تولید متن با طول توالی بالا، رشد حافظه پنهان کلید و مقدار (\lr{KV Cache}) و لایه‌های پروجکشن خطی متناظر با توجه، اصلی‌ترین گلوگاه حافظه تلقی می‌شوند. از سوی دیگر، تقریب رتبه پایین محض از طریق \lr{SVD} استاندارد غالباً در نگاشت اطلاعات با فرکانس بالا و الگوهای نادر واژگانی دچار کمبود ظرفیت شده و خطای تقریب را افزایش می‌دهد.

ماژول فشرده‌سازی \lr{Delta-KV Compressor} با نام فارسی «فشرده‌ساز پسماند دلتای کلید و مقدار» با هدف ایجاد تعادل ریاضی میان تقریب رتبه پایین و پسماندهای پرانرژی تنک پیاده‌سازی شده است. ایده محوری این روش بر تجزیه ماتریس وزن یا نگاشت‌های پنهان به یک مؤلفه با رتبه پایین ساختاری به همراه یک ماتریس مانده تنک با مقادیر بزرگ (\lr{Low-Rank plus Sparse Decomposition}) استوار است. بدین ترتیب، تقریب خطی رتبه پایین روندهای عمومی و همبستگی‌های دوربرد را جذب کرده و مؤلفه تنک، ناهنجاری‌ها و انحرافات حساس توجه را بازنمایی می‌کند.

\subsection{مبانی نظری و فرمول‌بندی ریاضی}
فرض کنید ماتریس وزن هدف در لایه‌های تصویرسازی توجه یا پروجکشن‌های خطی به صورت $\mathbf{W} \in \mathbb{R}^{d_{\text{out}} \times d_{\text{in}}}$ داده شده باشد. الگوریتم \lr{Delta-KV} ماتریس وزن بازسازی‌شده را به فرم مجموع رتبه پایین و مانده تنک مدل‌سازی می‌کند:
\begin{equation}
\mathbf{W}_{\text{comp}} = \mathbf{L} + \mathbf{R}
\end{equation}
به طوری که:
\begin{enumerate}
    \item مؤلفه با رتبه پایین ساختاری $\mathbf{L}$ با استفاده از تجزیه مقادیر منفرد بر روی ماتریس وزن به دست می‌آید:
    \begin{equation}
    \mathbf{W} = \mathbf{U} \mathbf{\Sigma} \mathbf{V}^\top, \quad \mathbf{L} = \mathbf{U}_{:, :r} \mathbf{\Sigma}_{:r} \mathbf{V}_{:, :r}^\top
    \end{equation}
    که در آن $r = \max(1, \min(r_{\text{target}}, d_{\text{in}}, d_{\text{out}}))$ رتبه فشرده‌سازی انتخاب‌شده است.
    \item ماتریس مانده خام برابراست با تفاضل ماتریس اصلی و تقریب کم‌مرتبه:
    \begin{equation}
    \mathbf{E} = \mathbf{W} - \mathbf{L}
    \end{equation}
    سپس بر مبنای نرخ تنک‌پذیری تعیین‌شده $s \in [0, 1]$، عملگر گزینش بزرگ‌ترین قدرمطلق‌ها (\lr{Top-k Magnitude Operator}) بر روی المان‌های برداری‌شده $|\mathbf{E}|$ اعمال می‌شود. با تعیین آستانه برش $\tau = \text{Quantile}(|\mathbf{E}|, 1 - s)$، ماتریس مانده تنک $\mathbf{R}$ محاسبه می‌گردد:
    \begin{equation}
    \mathbf{R}_{i, j} = \begin{cases} 
    \mathbf{E}_{i, j} & \text{if } |\mathbf{E}_{i, j}| \ge \tau \\
    0 & \text{otherwise}
    \end{cases}
    \end{equation}
\end{enumerate}

خطای بازسازی فروبنیوس به صورت زیر اعتبارسنجی می‌شود:
\begin{equation}
\epsilon_{\text{rel}} = \frac{\|\mathbf{W} - (\mathbf{L} + \mathbf{R})\|_F}{\max(\|\mathbf{W}\|_F, 10^{-12})}
\end{equation}
در صورتی که $\epsilon_{\text{rel}} > \epsilon_{\text{max}}$ باشد، الگوریتم لایه را به ساختار متراکم اولیه بازمی‌گرداند. علاوه بر این، در حالت اجرای برخط کش، ماژول شامل کلاس ذخیره‌ساز پویا \lr{DeltaKVStore} است که با سنجش شباهت کسینوسی بردار پنهان جاری با حافظه تاریخی $K/V$ مراجع، تفاضل بردار را در قالب دلتای فشرده بازنمایی می‌نماید:
\begin{equation}
\mathbf{k}_{\text{residual}} = \mathbf{k} - \mathbf{r}_k, \quad \hat{\mathbf{k}} = \mathbf{r}_k + \mathbf{k}_{\text{residual}} \odot \mathbb{I}(\|\mathbf{k}_{\text{residual}}\|_2 > \theta_{\text{res}})
\end{equation}

\paragraph{تحلیل طیفی و نامساوی‌های پایداری خطا}
برای بررسی میزان دقت تقریب ماتریس وزن یا نگاشت توجه در ماژول \lr{delta\_kv}، تفاضل ماتریس فاکتورگیری‌شده از نگاشت متراکم اصلی بر اساس نامساوی هافدینگ-فون‌نیومان ارزیابی می‌گردد. فرض کنید $\mathbf{W} \in \mathbb{R}^{d_{\text{in}} \times d_{\text{out}}}$ ماتریس وزن هدف باشد که توسط تقریب کم‌مرتبه یا پراکنده با رتبه موثر $r$ جایگزین شده است. کران بالای خطای بازسازی بر مبنای نرم فروبنیوس به صورت زیر تصریح می‌شود:
\begin{equation}
\label{eq:spectral_bound_delta_kv}
\| \mathbf{W} - \mathbf{W}_{r} \|_F = \sqrt{\sum_{i=r+1}^{\min(d_{\text{in}}, d_{\text{out}})} \sigma_i^2(\mathbf{W})} \le \epsilon \| \mathbf{W} \|_F
\end{equation}
که در آن $\sigma_i(\mathbf{W})$ بیانگر $i$-امین مقدار منفرد مرتب‌شده ماتریس به صورت نزولی است. نسبت حفظ انرژی طیفی $\eta_r = \frac{\sum_{i=1}^r \sigma_i^2}{\sum_{i=1}^{d} \sigma_i^2}$ تضمین می‌کند که با انتخاب آستانه $\tau \ge 0.90$، بیش از نود درصد از پراکندگی واریانس داده‌ها حفظ شده و افت سیگنال در انتشار گرادیان رو به عقب به حداقل ممکن کاهش می‌یابد.

\paragraph{تحلیل مجانبی و پیچیدگی اشتراک حافظه سخت‌افزاری}
از منظر پیچیدگی محاسباتی و انتقال داده‌ها در سلسومراتب حافظه (حافظه سراسری \lr{HBM} به حافظه محلی \lr{SRAM})، عملیات ضرب ماتریسی در این مکانیزم نیازمند تخصیص بافر موقت با ابعاد فاکتورگیری‌شده است. برای پردازش دنباله‌ای به طول $N$ با بعد مخفی $D$، بار محاسباتی هسته از فرم درجه دو به رابطه زیر تقلیل می‌یابد:
\begin{equation}
\label{eq:complexity_delta_kv}
\mathcal{C}_{\text{delta\_kv}} = \mathcal{O}\left( B \cdot N \cdot D \cdot r + B \cdot H \cdot N \cdot d_k \right) \ll \mathcal{O}\left( B \cdot H \cdot N^2 \cdot d_k \right)
\end{equation}
این بهبود مرتبه پیچیدگی، با کاهش چشمگیر حجم تبادل تانسورها در گذرگاه‌های ارتباطی کارت گرافیک، مانع از اشباع پهنای باند حافظه (پدیده \lr{Memory-Bound}) در فرآیند استنتاج طولانی‌مدت می‌گردد.

\subsection{معماری و پیاده‌سازی}
کلاس‌های اصلی این ماژول در مسیر \lr{Codes/framework/compression\_engine/delta\_kv.py} قرار داشته و شامل دو بخش اصلی \lr{DeltaKVCompressor} (به عنوان فشرده‌ساز مدل) و \lr{DeltaKVStore} (به عنوان کش فرّار زمان اجرا) می‌باشند. ویژگی‌های برجسته پیاده‌سازی عبارتند از:
\begin{itemize}
    \item \textbf{رجیستری چندگانه:} ثبت رسمی در رجیستری با شناسه‌های \lr{delta\_kv}، \lr{delta\_kv\_class}، \lr{delta\_kv\_compressor} و \lr{delta\_kv\_factory}.
    \item \textbf{فیلتر هدف‌گذاری هوشمند لایه‌ها:} توانایی تفکیک پروجکشن‌های توجه (\lr{q\_proj}، \lr{k\_proj}، \lr{v\_proj}، \lr{o\_proj} و \lr{c\_attn}) از لایه‌های پیش‌خور در صورت تنظیم پارامتر \lr{target\_attention\_only}.
    \item \textbf{پایداری حافظه و بازنویسی مستقیم (\lr{In-place Weight Rewrite}):} ماژول بدون تخریب ساختار گراف اجرایی، وزن متراکم را مستقیماً با ماتریس فیلترشده $\mathbf{W}_{\text{comp}}$ در نوع داده اصلی بازنویسی کرده و گرادیان‌ها را برای لایه‌های هرس‌شده مدیریت می‌کند.
    \item \textbf{محافظت از لایه‌های بحرانی:} جلوگیری از فشرده‌سازی پروجکشن‌های طبقه‌بند، سر مدل زبانی (\lr{lm\_head})، ماتریس‌های تعبیه (\lr{embedding}) و لایه‌های با ابعاد کوچک‌تر از \lr{min\_layer\_dim}.
\end{itemize}

\subsection{محدودیت‌ها و تحلیل شکست}
با وجود نتایج پیشتاز این ماژول، تحلیل ابعاد فنی آن مبین چند محدودیت اساسی است:
\begin{itemize}
    \item \textbf{عدم کاهش پارامتر فیزیکی در مد بازنویسی وزن (\lr{Weight-only In-place}):} در صورتی که از فرم بازنویسی ماتریس چگال بدون تبدیل به لایه‌های دوگانه فاکتورگیری استفاده شود، حجم فیزیکی پارامترها در رم یکسان باقی مانده و صرفاً تنک‌پذیری داخلی وزن‌ها ارتقا می‌یابد؛ کاهش واقعی ردپای حافظه مستلزم ترکیب با فرمت‌های ذخیره‌سازی تنک ساختاریافته است.
    \item \textbf{سربار جستجوی شباهت در \lr{DeltaKVStore}:} در سناریوهای استنتاج با طول زمینه قابل توجه طولانی، محاسبه ضرب ماتریسی به منظور یافتن $k$ مرجع برتر در کش تاریخی می‌تواند تاخیر استنتاج توکن اول (\lr{Time-to-First-Token}) را اندکی تحت تاثیر قرار دهد.
    \item \textbf{حساسیت عملکرد در مدل‌های فوق بزرگ:} در مدل‌های مقیاس بزرگ نظیر \lr{Qwen}، مصرف حافظه در فاز کالیبراسیون و انباشت پسماندها به \lr{5538.93 MB} رسیده که کنترل دقیق‌تر آستانه تنک‌پذیری را ایجاب می‌کند.
\end{itemize}
