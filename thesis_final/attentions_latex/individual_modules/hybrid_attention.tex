\section{ماژول توجه ترکیبی بهینه‌شده (Hybrid Attention)}
\label{sec:module_hybrid_attention}

\subsection{معرفی و انگیزه}
این ماژول در چارچوب پژوهش حاضر پیاده‌سازی و ارزیابی شده و فاقد مقاله مستقل جداگانه در ادبیات تحقیق است.

مکانیزم توجه \lr{Hybrid Attention} (توجه ترکیبی چندشاخه با گیت اختصاصی به ازای هر سر) با هدف تلفیق قابلیت‌های بیان‌گری و تطبیق‌پذیری بالای مکانیزم توجه سنتی بر پایه تابع بیشینه هموار (\lr{Softmax Attention}) و کارایی محاسباتی خطی مکانیزم‌های بازگشتی یا هسته‌محور (\lr{Linear Attention}) در پردازش توالی‌های طویل طراحی شده است. توجه استاندارد دارای پیچیدگی زمانی و مکانی $\mathcal{O}(N^2)$ است که در طول دنباله‌های زیاد منجر به اشباع سریع حافظه کارت گرافیک (\lr{VRAM}) و افت محسوس نرخ پردازش توکن‌ها می‌گردد؛ از سوی دیگر، مکانیزم‌های توجه خطی با تقریب نگاشت هسته، هرچند حافظه و توان محاسباتی را به $\mathcal{O}(N \cdot d)$ تقلیل می‌دهند، اما در ثبت وابستگی‌های نحوی و ساختاری ریزدانه با کاهش ظرفیت مواجه می‌شوند.

نوآوری بنیادین این ماژول در پیاده‌سازی گیت تلفیق یادگیرنده برای هر سر توجه (\lr{Per-Head Learned Gate}) نهفته است. بر خلاف معماری‌هایی نظیر \lr{MEGA} که یک مقدار گیت سراسری مشترک را اعمال می‌کنند، در این ماژول به هر سر اجازه داده می‌شود به طور پویا و متناسب با بافت محتوایی ورودی، اولویت خود را میان شاخه دقیق دانس و شاخه خطی تعیین نماید. علاوه بر این، با راه‌اندازی اولیه بایاس گیت روی مقادیر بزرگ (\lr{bias = 10.0})، شبکه در آغاز فرآیند یادگیری کاملاً بر شاخه \lr{Softmax} منطبق بوده و با حداقل انحراف مقادیر اولیه‌ی مدل پایه، مسیر بهینه‌سازی را بدون افت کیفی اولیه طی می‌کند.

\subsection{مبانی نظری و فرمول‌بندی ریاضی}
فرض کنید $\mathbf{X} \in \mathbb{R}^{B \times N \times D}$ نمایانگر نمایش‌های پنهان ورودی برای دسته به اندازه $B$، طول توالی $N$ و بعد پنهان $D$ باشد. تانسورهای پرس‌وجو ($\mathbf{Q}$)، کلید ($\mathbf{K}$) و مقدار ($\mathbf{V}$) با تعداد سرهای $H$ و بعد هر سر $d$ ($D = H \cdot d$) تعریف می‌شوند:
\begin{equation}
\mathbf{Q}_h = \mathbf{X} \mathbf{W}_h^Q, \quad \mathbf{K}_h = \mathbf{X} \mathbf{W}_h^K, \quad \mathbf{V}_h = \mathbf{X} \mathbf{W}_h^V
\end{equation}

در این ماژول، دو شاخه محاسباتی موازی جریان می‌یابند:
\begin{enumerate}
    \item \textbf{شاخه توجه استاندارد (\lr{Standard Softmax Branch}):}
    \begin{equation}
    \mathbf{O}_h^{\text{softmax}} = \operatorname{Softmax}\left(\frac{\mathbf{Q}_h \mathbf{K}_h^T}{\sqrt{d}} + \mathbf{M}\right) \mathbf{V}_h
    \end{equation}
    که در آن $\mathbf{M}$ نشان‌دهنده ماتریس ماسک توجه (علّی یا ماسک پرکردگی) است.
    
    \item \textbf{شاخه توجه خطی هسته‌محور (\lr{Linear Branch}):}
    با بهره‌گیری از نگاشت ویژگی فعال‌سازی $\phi(\mathbf{x}) = \operatorname{ELU}(\mathbf{x}) + 1.0 \ge 0$، ضرب ماتریسی بر اساس خاصیت شرکت‌پذیری بدون محاسبه صریح ماتریس $N \times N$ به شکل زیر محاسبه می‌شود:
    \begin{equation}
    \mathbf{O}_h^{\text{linear}} = \frac{\phi(\mathbf{Q}_h) \left( \phi(\mathbf{K}_h)^T \mathbf{V}_h \right)}{\phi(\mathbf{Q}_h) \left( \sum_{i=1}^N \phi(\mathbf{K}_{h, i}) \right)^T + \epsilon}
    \end{equation}
    در مدل‌های خودبازگشتی (\lr{Autoregressive Decoders})، این شاخه به صورت تکه‌ای (\lr{Chunked causal linear scan}) با انباشت حالات مخفی $\mathbf{S}_k = \mathbf{S}_{k-1} + \phi(\mathbf{K}_k)^T \mathbf{V}_k$ در ابعاد زمانی به صورت خطی پیاده‌سازی می‌گردد.
\end{enumerate}

گیت تلفیق سرها به ازای بردار ویژگی هر موقعیت $\mathbf{x}_t$ به صورت زیر است:
\begin{equation}
g_h(\mathbf{x}_t) = \sigma\left(\mathbf{w}_h^T \mathbf{x}_t + b_h\right), \quad b_h \approx 10.0
\end{equation}
خروجی نهایی پیش از تصویرسازی نگاشت پایانی از برهم‌نهی وزندار این دو شاخه حاصل می‌شود:
\begin{equation}
\mathbf{O}_h = g_h \odot \mathbf{O}_h^{\text{softmax}} + (1 - g_h) \odot \mathbf{O}_h^{\text{linear}}
\end{equation}
\begin{equation}
\mathbf{Y} = \operatorname{Concat}(\mathbf{O}_1, \dots, \mathbf{O}_H) \mathbf{W}^O
\end{equation}

\subsection{معماری و پیاده‌سازی}
کلاس \lr{HybridAttention} در ماژول \lr{Codes/framework/attentions/hybrid\_attention.py} مستقر شده و از \lr{BaseAttention} ارث‌بری می‌کند. این معماری ویژگی‌های کلیدی زیر را داراست:
\begin{itemize}
    \item \textbf{عدم افزایش چشمگیر پارامترها:} تنها پارامتر افزوده‌شده ماتریس نگاشت گیت با ابعاد $D \times H$ به همراه بایاس آن است ($768 \times 12 + 12 = 9228$ پارامتر اضافی معادل تنها $+0.014\%$ در مدل \lr{DistilBERT}).
    \item \textbf{پشتیبانی کامل از سازوکارهای مدرن:} دربرگیرنده سازوکار چرخش موقعیتی (\lr{RoPE}) با تنظیم متناسب جابه‌جایی در کش (\lr{past\_len offset})، سازگاری با \lr{KV-Cache} پویا و پیاده‌سازی مسیر سریع \lr{PyTorch SDPA} برای کاهش مصرف حافظه.
    \item \textbf{انتقال امن وزن‌ها (\lr{copy\_weights}):} تطبیق کامل با ساختارهای انکودر متداخل (\lr{Nested Encoders}) در \lr{RoBERTa} و \lr{BERT} و دیکودرهای خانواده \lr{GPT-2} و \lr{Qwen}.
\end{itemize}

\subsection{محدودیت‌ها و تحلیل شکست}
بررسی نتایج در سطوح مدل‌های مختلف رفتارهای معماری متفاوتی را آشکار می‌سازد:
\begin{itemize}
    \item \textbf{افت دقت در \lr{RoBERTa} و \lr{GPT-2} (\lr{PASS\_LOW}):} دقت ۵۵.۳۹٪ در \lr{RoBERTa} و ۵۰.۹۲٪ در \lr{GPT-2} ناشی از حساسیت فاز استخراج لایه‌های میانی به برهم‌خوردن هماهنگی سرها در مقیاس‌های عمیق بدون بازتنظیم کامل انکودر متداخل است. همچنین، گرادیان گیت‌ها در رژیم‌های بدون تنظیم دقیق ممکن است شاخه خطی را پیش از موعد در لایه‌های ابتدایی فعال کند.
    \item \textbf{علت ثبت دقت ۰٪ و وضعیت شکست در \lr{Qwen} (\lr{SKIP\_QWEN3\_5\_TIMEOUT}):} دقت صفر ثبت‌شده در مدل \lr{Qwen} ($0.0\%$) ناشی از شکست عملکرد محاسباتی یا انحطاط یادگیری پارامتری نیست؛ بلکه حاصل رویداد وقفه زمانی (\lr{Timeout}) به دلیل سربار الگوریتم توجه خطی تکه‌ای علّی (\lr{Chunked causal linear scan}) در ترکیب با سازوکار \lr{GQA} در طول دنباله‌های بزرگ است که اجرای ارزیابی را پیش از استخراج مقادیر خروجی به اتمام رسانده است (شاخص \lr{F1} با مقدار ۰.۵۵۷۰ تأییدکننده فعال بودن گرادیان‌ها تا لحظه توقف فرآیند است).
\end{itemize}
