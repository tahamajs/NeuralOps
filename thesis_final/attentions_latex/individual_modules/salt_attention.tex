\section{ماژول salt\_attention (Singular Value Adaptation with Low-Rank Tuning)}
\label{sec:module_salt_attention}

\subsection{معرفی و انگیزه}
این مکانیزم بر مبنای مبانی نظری پیشرفته در ادبیات توجه و فشرده‌سازی \cite{ref_saltattention_2024} توسعه یافته است.

تنظیم دقیق کارآمد پارامتری (\lr{Parameter-Efficient Fine-Tuning - PEFT}) و فشرده‌سازی پروجکشن‌های لایه توجه در ترنسفورمرها از اهداف اساسی کاهش هزینه‌های محاسباتی به شمار می‌روند. روش‌های رایج نظیر \lr{LoRA}، ماتریس‌های اصلاحی تصادفی با رتبه پایین را به وزن‌های اولیه اضافه می‌کنند، اما از ساختار طیفی و راستاهای مقادیر منفرد موجود در ماتریس‌های آموزش‌دیده اولیه غافل هستند.

مکانیزم توجه \lr{SALT (Singular Value Adaptation with Low-Rank Tuning)} با هدف ادغام هوشمند تجزیه مقادیر منفرد بر پایه اطلاعات فیشر (\lr{Fisher-Weighted SVD}) و تطبیق کم‌مرتبه باقیمانده ابداع شده است. در ماژول \lr{SALT Attention}، پروجکشن‌های خطی توجه استاندارد با لایه‌های فاکتورگیری‌شده \lr{SALTLinear} جایگزین می‌شوند. این لایه‌ها راستاهای بردارهای منفرد متعامد اولیه را حفظ کرده، مقادیر منفرد را با ضریب مقیاس‌گذاری تطبیقی $\boldsymbol{\gamma}$ تعدیل نموده و هم‌زمان یک شاخه کم‌مرتبه یادگیرنده موازی $\mathbf{A} \mathbf{B}$ را برای جبران خطای باقیمانده به کار می‌گیرند.

\subsection{مبانی نظری و فرمول‌بندی ریاضی}
فرض کنید ماتریس وزن پروجکشن اولیه $\mathbf{W}_0 \in \mathbb{R}^{d_{\text{out}} \times d_{\text{in}}}$ با استفاده از تجزیه مقادیر منفرد وزن‌دار فیشر تجزیه شود. در صورتی که $\mathbf{F} \in \mathbb{R}^{d_{\text{out}} \times d_{\text{in}}}$ ماتریس قطری اطلاعات فیشر باشد، تجزیه بر روی ماتریس وزنی $\tilde{\mathbf{W}}_0 = \mathbf{W}_0 \odot \sqrt{\mathbf{F}}$ اعمال می‌گردد:
\begin{equation}
\mathbf{W}_0 \approx \mathbf{U}_{:, :r} \operatorname{diag}(\boldsymbol{\Sigma}_{1:r}) \mathbf{V}_{1:r, :}^\top
\end{equation}
که در آن $r \le \min(d_{\text{in}}, d_{\text{out}})$ رتبه فاکتورسازی است. در لایه \lr{SALTLinear}، وزن موثر پروجکشن در حین استنتاج و آموزش به صورت زیر فرمول‌بندی می‌شود:
\begin{equation}
\mathbf{W} = \mathbf{U}_{:, :r} \operatorname{diag}(\boldsymbol{\Sigma}_{1:r} \odot \boldsymbol{\gamma}) \mathbf{V}_{1:r, :}^\top + \frac{\alpha}{r} \mathbf{A} \mathbf{B}
\end{equation}
که در آن:
\begin{itemize}
    \item $\mathbf{U}_{:, :r} \in \mathbb{R}^{d_{\text{out}} \times r}$ و $\mathbf{V}_{1:r, :}^\top \in \mathbb{R}^{r \times d_{\text{in}}}$ پایه‌های متعامد مقادیر منفرد هستند که منجمد باقی می‌مانند.
    \item $\boldsymbol{\gamma} \in \mathbb{R}^r$ بردار پارامترهای آموزش‌پذیر مقیاس‌گذاری طیفی است که با ۱ مقداردهی اولیه می‌شود ($\boldsymbol{\gamma}_{\text{init}} = \mathbf{1}$).
    \item $\mathbf{A} \in \mathbb{R}^{d_{\text{out}} \times r}$ و $\mathbf{B} \in \mathbb{R}^{r \times d_{\text{in}}}$ آداپتورهای باقیمانده تصحیح‌کننده کم‌مرتبه هستند.
\end{itemize}
خروجی نگاشت به ازای تانسور ورودی $\mathbf{X}$ برابر است با:
\begin{equation}
\mathbf{Y} = \mathbf{X} \left( \mathbf{V}_{1:r, :} \operatorname{diag}(\boldsymbol{\Sigma}_{1:r} \odot \boldsymbol{\gamma}) \mathbf{U}_{:, :r}^\top \right) + \frac{\alpha}{r} \mathbf{X} \mathbf{B}^\top \mathbf{A}^\top + \mathbf{b}
\end{equation}
تانسورهای پرس‌وجو، کلید و مقدار پس از فرافکنی و اعمال کدگذاری موقعیت چرخشی (\lr{RoPE}) و هم‌ترازی سرهای \lr{GQA}، وارد لایه توجه شده و بافتار نهایی استخراج می‌گردد:
\begin{equation}
\mathbf{O} = \operatorname{softmax}\left(\frac{\mathbf{Q} \mathbf{K}^\top}{\sqrt{d}} + \mathbf{M}\right) \mathbf{V}
\end{equation}

\paragraph{تحلیل طیفی و نامساوی‌های پایداری خطا}
برای بررسی میزان دقت تقریب ماتریس وزن یا نگاشت توجه در ماژول \lr{salt\_attention}، تفاضل ماتریس فاکتورگیری‌شده از نگاشت متراکم اصلی بر اساس نامساوی هافدینگ-فون‌نیومان ارزیابی می‌گردد. فرض کنید $\mathbf{W} \in \mathbb{R}^{d_{\text{in}} \times d_{\text{out}}}$ ماتریس وزن هدف باشد که توسط تقریب کم‌مرتبه یا پراکنده با رتبه موثر $r$ جایگزین شده است. کران بالای خطای بازسازی بر مبنای نرم فروبنیوس به صورت زیر تصریح می‌شود:
\begin{equation}
\label{eq:spectral_bound_salt_attention}
\| \mathbf{W} - \mathbf{W}_{r} \|_F = \sqrt{\sum_{i=r+1}^{\min(d_{\text{in}}, d_{\text{out}})} \sigma_i^2(\mathbf{W})} \le \epsilon \| \mathbf{W} \|_F
\end{equation}
که در آن $\sigma_i(\mathbf{W})$ بیانگر $i$-امین مقدار منفرد مرتب‌شده ماتریس به صورت نزولی است. نسبت حفظ انرژی طیفی $\eta_r = \frac{\sum_{i=1}^r \sigma_i^2}{\sum_{i=1}^{d} \sigma_i^2}$ تضمین می‌کند که با انتخاب آستانه $\tau \ge 0.90$، بیش از نود درصد از پراکندگی واریانس داده‌ها حفظ شده و افت سیگنال در انتشار گرادیان رو به عقب به حداقل ممکن کاهش می‌یابد.

\paragraph{تحلیل مجانبی و پیچیدگی اشتراک حافظه سخت‌افزاری}
از منظر پیچیدگی محاسباتی و انتقال داده‌ها در سلسومراتب حافظه (حافظه سراسری \lr{HBM} به حافظه محلی \lr{SRAM})، عملیات ضرب ماتریسی در این مکانیزم نیازمند تخصیص بافر موقت با ابعاد فاکتورگیری‌شده است. برای پردازش دنباله‌ای به طول $N$ با بعد مخفی $D$، بار محاسباتی هسته از فرم درجه دو به رابطه زیر تقلیل می‌یابد:
\begin{equation}
\label{eq:complexity_salt_attention}
\mathcal{C}_{\text{salt\_attention}} = \mathcal{O}\left( B \cdot N \cdot D \cdot r + B \cdot H \cdot N \cdot d_k \right) \ll \mathcal{O}\left( B \cdot H \cdot N^2 \cdot d_k \right)
\end{equation}
این بهبود مرتبه پیچیدگی، با کاهش چشمگیر حجم تبادل تانسورها در گذرگاه‌های ارتباطی کارت گرافیک، مانع از اشباع پهنای باند حافظه (پدیده \lr{Memory-Bound}) در فرآیند استنتاج طولانی‌مدت می‌گردد.

\subsection{معماری و پیاده‌سازی}
کلاس پایتونی \lr{SALTAttention} در سورس‌کد \lr{Codes/framework/attentions/salt\_attention.py} پیاده‌سازی شده و از کلاس \lr{BaseAttention} ارث‌بری می‌کند. مؤلفه‌های کلیدی پیاده‌سازی عبارتند از:
\begin{itemize}
    \item \textbf{لایه خطی فاکتورگیری‌شده \lr{SALTLinear}:} پیاده‌سازی مستقیم فرمول طیفی با دو مسیر محاسباتی موازی؛ مسیر طیفی منجمد مقیاس‌پذیر با پارامتر $\boldsymbol{\gamma}$ و مسیر کم‌مرتبه تصحیح‌کننده خطای فاکتورسازی.
    \item \textbf{پشتیبانی از تجزیه با وزن‌دهی فیشر (\lr{Fisher-weighted SVD}):} امکان مقداردهی وزن‌ها با اولویت‌دهی به پارامترهای با گرادیان و حساسیت بالا در مدل پایه.
    \item \textbf{سازگاری کامل با معماری‌های ناهمگن:} شناسایی خودکار انکودرهای تودرتو (\lr{RoBERTa/BERT})، دیکودرهای علّی (\lr{GPT-2})، مدل‌های \lr{GQA} و مدیریت ابعاد تانسورهای سرهای کلید/مقدار نامتقارن.
    \item \textbf{مسیر بهینه‌شده هسته \lr{SDPA}:} اجرای سریع بدون سربار حافظه در صورت عدم نیاز به ثبت تانسورهای توجه میانی.
\end{itemize}

\subsection{محدودیت‌ها و تحلیل شکست}
تحلیل نتایج به دست آمده بر روی سایر مدل‌ها شواهد زیر را آشکار می‌سازد:
\begin{itemize}
    \item \textbf{افت دقت روی \lr{RoBERTa} و \lr{GPT-2} (\lr{50.92\%}):} عملکرد ماژول روی مدل‌های \lr{RoBERTa} و \lr{GPT-2} در وضعیت \lr{PASS\_LOW} قرار دارد. این افت عملکرد ناشی از فاکتورسازی شدید با رتبه $r=16$ در لایه‌های چگال پروجکشن بدون اجرای فاز همگرایی تطبیقی عمیق است. انجماد بردارهای تکین $\mathbf{U}$ و $\mathbf{V}$ در صورتی که رتبه $r$ بخش ناچیزی از ابعاد کامل (۷۶۸) را پوشش دهد، انعطاف‌پذیری بازنمایی فضایی را به شدت محدود ساخته و منجر به قفل شدن پیش‌بینی‌های مدل بر روی توزیع کلاس اکثریت می‌گردد.
    \item \textbf{ناسازگاری ساختاری با \lr{Qwen}:} در مدل \lr{Qwen} به علت تفاوت ساختاری در لایه‌های ادغام‌شده پروجکشن \lr{QKV} و ناسازگاری ماسک‌های چندبعدی، ماژول در نسخه پایه با برچسب عدم سازگاری مواجه گردیده است.
\end{itemize}
