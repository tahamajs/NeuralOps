\subsection*{بخش ۱: ایده محوری و چرایی معماری \lr{(Architectural Rationale)}}
مکانیزم توجه هیبریدی بومی \lr{(Native Hybrid Attention - NHA)} پاسخی بنیادین و مهندسی‌شده به یک دوگانگی کلاسیک در نظریه ترنسفورمرها است: \textbf{تضاد میان دقت موضعی الگوهای نحوی و مقیاس‌پذیری خطی در زمینه سراسری}. در مکانیزم توجه استاندارد، محاسبه ماتریس همبستگی کامل با پیچیدگی محاسباتی و حافظه‌ای $\mathcal{O}(N^2)$ همراه است که در طول دنباله‌های بلند به دلیل انفجار حجم محاسبات و اشغال پهنای باند حافظه \lr{(HBM)} عملاً به مانعی جدی بدل می‌شود.

برای عبور از این چالش، پژوهش‌های پیشین به دو شاخه مجزا تقسیم شدند:
\begin{enumerate}
    \item \textbf{مکانیزم‌های توجه خطی مبتنی بر کرنل یا فضاهای حالت \lr{(Linear / Kernelized Attention)}:} این رویکردها با استفاده از ویژگی شرکت‌پذیری ضرب ماتریسی، پیچیدگی را به $\mathcal{O}(N)$ تقلیل می‌دهند؛ اما به دلیل استفاده از نگاشت‌های مثبت غیرخطی به جای تابع سافت‌مکس کامل، از پدیده مخرب \textbf{تضعیف توجه \lr{(Attention Dilution)}} رنج می‌برند. در این حالت، مدل قادر به تمرکز شدید روی روابط نحوی بسیار نزدیک (مانند پیوند صفت و موصوف یا نشانه‌های نفی مجاور) نبوده و تفکیک‌پذیری بازنمایی به شدت افت می‌کند.
    \item \textbf{مکانیزم‌های پنجره لغزان محلی \lr{(Sliding-Window Attention)}:} این روش‌ها تنها تعاملات در شعاع مشخصی به پهنای $w$ را محاسبه می‌کنند ($|i - j| \le w$) که پیچیدگی آن $\mathcal{O}(N \cdot w)$ است. با اینکه این شاخه دقت کامل سافت‌مکس محلی را حفظ می‌کند، اما میدان دید سراسری مدل به طور کامل قطع شده و امکان انتقال اطلاعات به توکن‌های تجمیعی دوردست (به‌ویژه توکن طبقه‌بندی \lr{[CLS]}) از بین می‌رود.
\end{enumerate}

ایده محوری ماژول \lr{nha\_attention}، ترکیب هم‌افزا و بومی \lr{(Native Fusion)} این دو جهان در یک لایه واحد است. در این ساختار، هر توکن به طور همزمان از دو مسیر پردازش می‌شود:
\begin{itemize}
    \item \textbf{شاخه محلی پنجره لغزان \lr{(Local Sliding-Window Branch)}:} با محاسبه دقیق و بدون تقریب سافت‌مکس روی پنجره‌ای متقارن با اندازه $w=64$، الگوهای ریزدانه نحوی و ساختار دستوری مجاور را با دقت مطلق استخراج می‌نماید.
    \item \textbf{شاخه سراسری خطی مبتنی بر ویژگی‌های فوریه تصادفی \lr{(Global Linear RFF Branch)}:} با تصویرسازی بردارهای کلید و پرس‌وجو به یک زیرفضای متعامد تصادفی با بعد $m=128$ و استفاده از کرنل نامنفی $\operatorname{ELU}(\cdot)+1$، یک نمایش فشرده و سراسری از کل دنباله را در زمان خطی $\mathcal{O}(N \cdot m)$ در قالب یک تانسور حافظه تداعی‌گر ذخیره می‌کند.
    \item \textbf{گیتینگ تطبیقی پویا در سطح توکن \lr{(Dynamic Token-Level Gating)}:} یک شبکه تصمیم‌گیر محتوا-محور با تابع فعال‌سازی سیگموئید، به ازای هر توکن مشخص می‌کند که چه میزان از اطلاعات باید از پنجره محلی و چه میزان از خلاصه سراسری خطی تأمین گردد. با تنظیم هوشمندانه بایاس اولیه این گیت روی $+3.0$ ($\sigma(3.0) \approx 0.95$)، مدل در ابتدای آموزش وزن‌های از پیش آموزش‌دیده را حفظ کرده و انتقال وزن نرم از مدل‌های پایه‌ای چون \lr{DistilBERT} بدون شوک بهینه‌سازی میسر می‌گردد.
\end{itemize}

\subsection*{بخش ۲: فرمول‌بندی ریاضی و مبانی نظری \lr{(Mathematical Formulation)}}
فرض کنید تانسور پنهان ورودی به لایه توجه با نماد $\mathbf{X} \in \mathbb{R}^{B \times N \times D}$ نمایش داده شود که در آن $B$ اندازه دسته، $N$ طول دنباله توکن‌ها و $D$ بعد پنهان مدل است. نگاشت‌های خطی اولیه برای استخراج فضاهای پرس‌وجو، کلید و مقدار به صورت زیر تعریف می‌شوند:
\begin{equation}
\mathbf{Q} = \mathbf{X}\mathbf{W}_Q + \mathbf{b}_Q \in \mathbb{R}^{B \times N \times (H \cdot d)}, \quad
\mathbf{K} = \mathbf{X}\mathbf{W}_K + \mathbf{b}_K \in \mathbb{R}^{B \times N \times (H_{kv} \cdot d)}, \quad
\mathbf{V} = \mathbf{X}\mathbf{W}_V + \mathbf{b}_V \in \mathbb{R}^{B \times N \times (H_{kv} \cdot d)}
\end{equation}
که در آن $H$ تعداد سرهای پرس‌وجو، $H_{kv}$ تعداد سرهای کلید/مقدار (جهت پشتیبانی از مکانیزم \lr{GQA}) و $d = D/H$ بعد اختصاصی هر سر است. در صورت نابرابری $H$ و $H_{kv}$، تانسورهای $\mathbf{K}$ و $\mathbf{V}$ با ضریب تکرار $R = H / H_{kv}$ تکثیر می‌شوند تا به ابعاد یکپارچه $[B, H, N, d]$ برسند.

\subsubsection*{۱. شاخه توجه محلی پنجره لغزان با تکه‌بندی مسدود}
برای هر توکن در موقعیت $i$ و سر توجه $h \in \{1, \dots, H\}$، همسایگی موضعی به صورت زیر تعریف می‌شود:
\begin{equation}
\mathcal{N}_w(i) = \{j \in \{1, \dots, N\} \mid |i - j| \le w\}
\end{equation}
امتیاز توجه خام بین توکن‌های $i$ و $j$ با مقیاس‌بندی استاندارد محاسبه می‌شود:
\begin{equation}
S_{ij}^{(h)} = \frac{\mathbf{q}_i^{(h)} (\mathbf{k}_j^{(h)})^\top}{\sqrt{d}} + M_{\text{\lr{band}}}(i, j) + M_{\text{\lr{pad}}}(j)
\end{equation}
که در آن ماسک نواری $M_{\text{\lr{band}}}(i, j)$ در صورتی که $j \notin \mathcal{N}_w(i)$ باشد برابر $-\infty$ و در غیر این صورت صفر است. توزیع احتمالاتی سافت‌مکس موضعی:
\begin{equation}
A_{\text{\lr{local}}, ij}^{(h)} = \frac{\exp(S_{ij}^{(h)})}{\sum_{l \in \mathcal{N}_w(i)} \exp(S_{il}^{(h)})}
\end{equation}
خروجی شاخه محلی به صورت ترکیب محدب بردارهای مقدار در پنجره $w$ تولید می‌شود:
\begin{equation}
\mathbf{O}_{\text{\lr{local}}, i}^{(h)} = \sum_{j \in \mathcal{N}_w(i)} A_{\text{\lr{local}}, ij}^{(h)} \mathbf{v}_j^{(h)} \in \mathbb{R}^d
\end{equation}
در پیاده‌سازی بهینه نرم‌افزاری، این عملیات در بلوک‌های با طول ثابت $C=64$ انجام می‌گیرد تا بافرسازی موقت به حداقل برسد.

\subsubsection*{۲. شاخه توجه خطی سراسری با پایه‌های فوریه تصادفی \lr{(RFF)}}
جهت پوشش پیوندهای با برد نامحدود، از تقریب کرنل با ویژگی‌های فوریه تصادفی متعامد \lr{(Orthogonal Random Fourier Features)} بهره گرفته می‌شود. ماتریس تبدیل تصویری $\boldsymbol{\Omega} \in \mathbb{R}^{d \times m}$ (با $m$ ویژگی تصادفی، عموماً $m=128$) از بلوک‌های تجزیه \lr{QR} ماتریس‌های تصادفی گوسی با شرط تعامد یکا تولید می‌گردد:
\begin{equation}
\mathbf{G} \sim \mathcal{N}(0, \mathbf{I}_{d \times b}), \quad \mathbf{Q}_{qr}, \mathbf{R}_{qr} = \operatorname{QR}(\mathbf{G}), \quad \boldsymbol{\Omega}_k = \sqrt{d} \, \mathbf{Q}_{qr}
\end{equation}
نگاشت غیرخطی پایدار $\phi: \mathbb{R}^d \to \mathbb{R}^m$ به فرم زیر اعمال می‌شود:
\begin{equation}
\phi(\mathbf{u}) = \frac{\operatorname{ELU}(\mathbf{u}\boldsymbol{\Omega}) + 1.0}{\sqrt{m}}
\end{equation}
بردارهای پرس‌وجو و کلید با ضریب مقیاس $d^{-1/4}$ متوازن شده و به فضای ویژگی نگاشت می‌شوند:
\begin{equation}
\boldsymbol{\phi}_{Q, i}^{(h)} = \phi(\mathbf{q}_i^{(h)} \cdot d^{-0.25}), \quad \boldsymbol{\phi}_{K, j}^{(h)} = \phi(\mathbf{k}_j^{(h)} \cdot d^{-0.25}) \cdot \mathbb{I}(j \text{ معتبر است})
\end{equation}
در حالت دوطرفه \lr{(Bidirectional)}، با بهره‌گیری از خاصیت شرکت‌پذیری ضرب، ماتریس حالت سراسری تداعی‌گر $\mathbf{M}_{KV}^{(h)}$ و بردار نرمال‌ساز $\mathbf{z}^{(h)}$ تجمیع می‌گردند:
\begin{equation}
\mathbf{M}_{KV}^{(h)} = \sum_{j=1}^N (\boldsymbol{\phi}_{K, j}^{(h)})^\top \mathbf{v}_j^{(h)} \in \mathbb{R}^{m \times d}, \quad \mathbf{z}^{(h)} = \sum_{j=1}^N \boldsymbol{\phi}_{K, j}^{(h)} \in \mathbb{R}^{1 \times m}
\end{equation}
خروجی شاخه سراسری بدون تشکیل صریح ماتریس $N \times N$ به دست می‌آید:
\begin{equation}
\mathbf{O}_{\text{\lr{global}}, i}^{(h)} = \frac{\boldsymbol{\phi}_{Q, i}^{(h)} \mathbf{M}_{KV}^{(h)}}{\max\left( \boldsymbol{\phi}_{Q, i}^{(h)} (\mathbf{z}^{(h)})^\top, 1.0 \right)} \in \mathbb{R}^d
\end{equation}
در رژیم‌های تولید علّی خودبازگشتی \lr{(Causal Decoders)}، این رابطه با استفاده از الگوریتم پویش پیشوندی تکه‌ای \lr{(Chunked Prefix Scan)} پیاده‌سازی می‌شود که در آن ماتریس وضعیت به صورت بازگشتی در طول تکه‌ها به‌روزرسانی می‌گردد.

\subsubsection*{۳. گیتینگ تطبیقی توکن و سنتز نهایی خروجی}
بردار گیتینگ تطبیقی وابسته به بستر ورودی برای هر توکن به صورت زیر محاسبه می‌شود:
\begin{equation}
g_i = \sigma(\mathbf{x}_i \mathbf{W}_g + b_g) \in (0, 1), \quad \mathbf{W}_g \in \mathbb{R}^{D \times 1}, \quad b_g \in \mathbb{R}
\end{equation}
تلفیق نهایی دو شاخه محلی و سراسری با اعمال بردار گیت به فرمت تانسوری $[B, 1, N, 1]$ صورت می‌پذیرد:
\begin{equation}
\mathbf{O}_{\text{\lr{fused}}, i}^{(h)} = g_i \mathbf{O}_{\text{\lr{local}}, i}^{(h)} + (1 - g_i) \mathbf{O}_{\text{\lr{global}}, i}^{(h)} \in \mathbb{R}^d
\end{equation}
سپس، تمامی سرهای پردازشی ادغام شده و خروجی پس از اعمال ضریب تنظیم‌پذیر مقیاس $\gamma_{\text{\lr{scale}}} \in \mathbb{R}$ وارد پروجکشن خروجی نهایی می‌گردد:
\begin{equation}
\mathbf{U} = \operatorname{Concat}(\mathbf{O}_{\text{\lr{fused}}}^{(1)}, \dots, \mathbf{O}_{\text{\lr{fused}}}^{(H)}) \in \mathbb{R}^{B \times N \times D}, \quad \mathbf{Y} = (\gamma_{\text{\lr{scale}}} \odot \mathbf{U})\mathbf{W}_O + \mathbf{b}_O \in \mathbb{R}^{B \times N \times D}
\end{equation}

\subsubsection*{۴. تحلیل پیچیدگی محاسباتی و مکانی}
\begin{itemize}
    \item \textbf{پیچیدگی محاسباتی زمان \lr{(Time Complexity)}:} پروجکشن‌های خطی نیازمند $\mathcal{O}(B \cdot N \cdot D^2)$ عملگر هستند. شاخه محلی به $\mathcal{O}(B \cdot H \cdot N \cdot w \cdot d)$ و شاخه سراسری \lr{RFF} به $\mathcal{O}(B \cdot H \cdot N \cdot m \cdot d)$ زمان نیاز دارند. از آنجا که $w, m, d \ll N$ هستند، کل فرآیند توجه در مرتبه خطی $\mathcal{O}(N)$ بر حسب طول دنباله مقیاس می‌یابد.
    \item \textbf{پیچیدگی حافظه \lr{(Space Complexity)}:} بافر فعال‌سازی شاخه محلی در سطح بلوک‌های $C \times (C + 2w)$ کنترل شده و ردپای شاخه سراسری برابر اندازه تانسور وضعیت $\mathcal{O}(B \cdot H \cdot m \cdot d)$ است؛ بنابراین پیچیدگی مکانی به طور کامل از بن‌بست $\mathcal{O}(N^2)$ آزاد شده و به $\mathcal{O}(B \cdot N \cdot D + B \cdot H \cdot m \cdot d)$ تقلیل می‌یابد.
\end{itemize}

\subsection*{بخش ۳: کالبدشکافی سورس‌کد و ردگیری تانسورها \lr{(Source Code \& Tensor Trace)}}
کلاس پایتونی \lr{NativeHybridAttention} در فایل \lr{nha\_attention.py} پیاده‌سازی شده و از کلاس پایه \lr{BaseAttention} ارث‌بری می‌کند. این ماژول تحت شناسه‌های رسمی \lr{"nha"}، \lr{"nha\_attention"} و \lr{"nha\_compressor"} در ثبات مرکزی فریم‌ورک به ثبت رسیده است.

\subsubsection*{سلسله‌مراتب اجزا و تعریف پارامترها}
\begin{itemize}
    \item \textbf{لایه‌های خطی پروجکشن \lr{(LoRALinear)}:} تمام نگاشت‌های ورودی و خروجی (\lr{q\_proj}, \lr{k\_proj}, \lr{v\_proj}, \lr{out\_proj}) از کلاس اختصاصی \lr{LoRALinear} بهره می‌برند که وزن‌های ماتریس پایه $\mathbf{W}_{\text{\lr{base}}}$ را در حالت فریز نگه داشته و انطباق را از طریق دو ماتریس کم‌رتبه $\mathbf{A} \in \mathbb{R}^{r \times D_{\text{\lr{in}}}}$ و $\mathbf{B} \in \mathbb{R}^{D_{\text{\lr{out}}} \times r}$ با رتبه $r=16$ و ضریب مقیاس $\alpha=32.0$ آموزش می‌دهد.
    \item \textbf{بافر تصویرسازی تصادفی \lr{(rff\_proj)}:} ماتریس تعامدی $\boldsymbol{\Omega}$ به ابعاد $[d, m]$ با متد \lr{\_build\_orthogonal\_matrix} بر پایه تجزیه \lr{torch.linalg.qr} ساخته شده و به عنوان بافر مداوم در حافظه ثبت می‌گردد.
    \item \textbf{لایه گیتینگ \lr{(gate)}:} لایه خطی \lr{nn.Linear(dim, 1)} با وزن‌های مقداردهی شده با صفر و بایاس اولیه ثابت $3.0$ که تضمین‌کننده خروجی سیگموئید $\approx 0.952$ در شروع بهینه‌سازی است.
    \item \textbf{پارامتر کالیبراسیون خروجی \lr{(output\_scale)}:} یک پارامتر اسکالر یادگیرنده \lr{nn.Parameter(torch.ones(1))} جهت بازتنظیم شدت گرادیان‌های انتقالی.
\end{itemize}

\subsubsection*{ردگیری گام‌به‌گام تانسورها در متد \lr{forward}}
جریان اجرای پیشرو به صورت دقیق در مراحل زیر دنبال می‌شود:
\begin{enumerate}
    \item \textbf{ورودی:} دریافت تانسور پنهان $\mathbf{X}$ با ابعاد $[B, N, D]$ و ماسک توجه \lr{attention\_mask}.
    \item \textbf{استخراج ماسک معتبر توکن‌ها:} تبدیل ماسک به تانسور بولی $[B, N]$ با فراخوانی متد \lr{get\_token\_valid\_mask}.
    \item \textbf{پروجکشن چندسره:}
    \begin{align*}
        \mathbf{Q} &= \text{\lr{q\_proj}}(\mathbf{X})\text{\lr{.view}}(B, N, H, d)\text{\lr{.permute}}(0, 2, 1, 3) \to [B, H, N, d] \\
        \mathbf{K} &= \text{\lr{k\_proj}}(\mathbf{X})\text{\lr{.view}}(B, N, H_{kv}, d)\text{\lr{.permute}}(0, 2, 1, 3) \to [B, H_{kv}, N, d] \\
        \mathbf{V} &= \text{\lr{v\_proj}}(\mathbf{X})\text{\lr{.view}}(B, N, H_{kv}, d)\text{\lr{.permute}}(0, 2, 1, 3) \to [B, H_{kv}, N, d]
    \end{align*}
    \item \textbf{اعمال کدگذاری موقعیت چرخشی \lr{(RoPE)}:} در صورت وجود تعبیه مکانی، تابع \lr{apply\_rotary\_pos\_emb} فرکانس‌های چرخشی کسینوسی و سینوسی را روی نیمه ابعاد تانسورهای $\mathbf{Q}$ و $\mathbf{K}$ اعمال می‌نماید.
    \item \textbf{هم‌راستاسازی \lr{GQA}:} اگر $H \ne H_{kv}$ باشد، تابع کمکی \lr{repeat\_kv} تانسورهای کلید و مقدار را در امتداد بعد سر تکرار نموده تا ابعاد آنها به $[B, H, N, d]$ ارتقا یابد.
    \item \textbf{اجرای شاخه محلی \lr{(\_local\_window\_attention)}:} محاسبه در بلوک‌های $\text{\lr{chunk}}=64$. ماتریس شباهت در پنجره محلی $[B, H, 64, 64 + 2w]$ تشکیل شده، با ماسک‌های نواری و پدینگ تلفیق گردیده و پس از سافت‌مکس در $\mathbf{V}$ ضرب می‌شود تا خروجی موضعی $\mathbf{O}_{\text{\lr{local}}} \in [B, H, N, d]$ تولید گردد.
    \item \textbf{اجرای شاخه سراسری \lr{(\_global\_rff\_attention)}:} ضرب پرس‌وجو و کلید مقیاس‌شده در \lr{rff\_proj} برای ساخت $\boldsymbol{\phi}_Q, \boldsymbol{\phi}_K \in [B, H, N, m]$. در حالت دوطرفه، انقباض تانسوری $\mathbf{KV} = \boldsymbol{\phi}_K^\top \mathbf{V} \in [B, H, m, d]$ و بردار جمع $\mathbf{Z} \in [B, H, 1, m]$ تشکیل شده و خروجی سراسری $\mathbf{O}_{\text{\lr{global}}} \in [B, H, N, d]$ از طریق نرمال‌سازی کسری استخراج می‌گردد.
    \item \textbf{تلفیق گیتینگ:} تانسور گیت از رابطه $\text{\lr{gate}} = \sigma(\text{\lr{self.gate}}(\mathbf{X}))\text{\lr{.unsqueeze}}(1) \to [B, 1, N, 1]$ به دست آمده و خروجی یکپارچه با ترکیب خطی وزن‌دار محاسبه می‌شود:
    \begin{equation}
    \mathbf{O} = \text{\lr{gate}} \odot \mathbf{O}_{\text{\lr{local}}} + (1.0 - \text{\lr{gate}}) \odot \mathbf{O}_{\text{\lr{global}}} \in [B, H, N, d]
    \end{equation}
    \item \textbf{بازآرایی و پروجکشن خروجی:} تانسور با فراخوانی \lr{.permute(0, 2, 1, 3).contiguous().view(B, N, H*d)} به فضای $[B, N, D]$ بازگشته، در \lr{output\_scale} ضرب شده و توسط \lr{out\_proj} پردازش نهایی می‌گردد.
\end{enumerate}

\subsubsection*{سازوکار انتقال وزن \lr{(copy\_weights)} و تزریق ماژول \lr{(inject\_nha)}}
متد \lr{copy\_weights} ساختارهای متنوع لایه‌های توجه مبدأ را شناسایی می‌کند: ساختارهای پروجکشن فیوز شده در مدل \lr{GPT-2} \lr{(Conv1D c\_attn)}، ساختارهای لایه‌های اینکودر تودرتو در \lr{RoBERTa} و \lr{BERT} \lr{(BertSelfAttention)}، و ساختارهای لایه‌های تخت در \lr{DistilBERT} (\lr{q\_lin}, \lr{k\_lin}, \lr{v\_lin}, \lr{out\_lin}). در تمامی مراحل، نرم فروبنیوس ماتریس وزن بررسی می‌شود ($\|\mathbf{W}\|_F \ge 10^{-4}$) تا از بارگذاری ناقص جلوگیری به عمل آید. در صورت فعال بودن آداپتورهای \lr{LoRA}، وزن‌های پایه منتقل‌شده بلافاصله منجمد می‌شوند.

تابع \lr{inject\_nha} کل ساختار شبکه هدف را پیمایش کرده و لایه‌های توجه پیش‌فرض را با نمونه‌های سازگار \lr{NativeHybridAttention} جایگزین می‌نماید.

\subsection*{بخش ۴: تحلیل معماری سخت‌افزار و مدل سقف کارایی \lr{(Hardware \& Roofline Analysis)}}
ارزیابی سخت‌افزاری ماژول بر روی پردازنده گرافیکی لپ‌تاپ \lr{NVIDIA GeForce RTX 4070 Laptop (8GB GDDR6)} مبتنی بر ریزمعماری \lr{Ada Lovelace (AD106)} با مشخصات زیر صورت گرفته است:
\begin{itemize}
    \item توان محاسباتی ممیز شناور هسته‌های تانسوری: حدود \lr{80 TFLOPS} در حالت \lr{Dense FP16}،
    \item پهنای باند انتقال حافظه ویدئویی: \lr{256 GB/s} بر روی گذرگاه ۱۲۸ بیتی،
    \item ظرفیت کش سطح دوم \lr{(L2 Cache)}: \lr{32 MB}،
    \item کش سطح اول و حافظه اشتراکی \lr{(L1 / Shared Memory)}: \lr{128 KB} به ازای هر یک از ۳۶ واحد \lr{SM}.
\end{itemize}

نقطه بازگشت سقف کارایی \lr{(Ridge Point)} برای این پردازنده گرافیکی به صورت زیر تعیین می‌شود:
\begin{equation}
I_{\text{\lr{ridge}}} = \frac{\text{\lr{Peak FLOPS}}}{\text{\lr{Memory Bandwidth}}} = \frac{80 \times 10^{12} \text{\lr{ FLOPs/s}}}{256 \times 10^9 \text{\lr{ Bytes/s}}} \approx 312.5 \text{\lr{ FLOPs/Byte}}
\end{equation}

\subsubsection*{تحلیل عبور از دیواره حافظه \lr{(Memory Wall Breakthrough)}}
در مکانیزم توجه استاندارد، ماتریس $N \times N$ برای طول دنباله $N=512$ و تعداد ۱۲ سر توجه، شامل بیش از ۳.۱ میلیون درایه ممیز شناور است که در هر لایه نیازمند نوشتن و خواندن مکرر از حافظه سراسری \lr{GDDR6} می‌باشد. شدت محاسباتی در این حالت در محدوده پایین $I \approx 25\text{--}40 \text{\lr{ FLOPs/Byte}}$ قرار گرفته و هسته‌های پردازشی در وضعیت اتلاف توان ناشی از گلوگاه پهنای باند \lr{(Memory-Bound Regime)} قفل می‌شوند.

در ماژول \lr{nha\_attention}، تحول معماری در دو جبهه رخ می‌دهد:
\begin{enumerate}
    \item \textbf{حبس بلوک محلی در حافظه نهان \lr{SRAM (L1)}:} شاخه پنجره لغزان به صورت تکه‌های ۶۴ توکنی با پنجره ۶۴ محاسبه می‌شود. اندازه هر تایل ضرب ماتریسی برابر $64 \times (64 + 128) = 12,288$ درایه است که در فرمت \lr{FP16} معادل تنها ۲۴ کیلوبایت فضا مصرف می‌کند. این حجم داده به راحتی در حافظه نهان فوق‌سریع \lr{L1 SRAM} هر \lr{SM} جای گرفته و تبادل ترافیک با حافظه اصلی سراسری را به صفر می‌رساند.
    \item \textbf{فشرده‌سازی سراسری در کش \lr{L2}:} ماتریس حالت سراسری تداعی‌گر $\mathbf{M}_{KV} \in \mathbb{R}^{m \times d}$ با ابعاد $128 \times 64$ تنها ۸,۱۹۲ درایه معادل ۱۶ کیلوبایت در هر سر فضا اشغال می‌کند. کل این تانسور در تمام طول دنباله درون حافظه نهان \lr{L2 Cache (32 MB)} باقی مانده و فرآیند انباشت اطلاعات بدون مراجعه به \lr{DRAM} به پایان می‌رسد.
\end{enumerate}

با این معماری، شدت محاسباتی به بالای \lr{180 FLOPs/Byte} جهش کرده و پردازش از گلوگاه انتقال حافظه خارج شده و به سمت رژیم محاسباتی \lr{(Compute-Bound)} حرکت می‌نماید.

\subsubsection*{تفکیک تراز اوج حافظه مصرفی \lr{(Peak VRAM: 335.47 MB)}}
مصرف نهایی حافظه گرافیکی در این ماژول در سطح بالا \lr{335.47 MB} تثبیت شده که کاهش خیره‌کننده \textbf{-92.01\%} را نسبت به خط مبنای متداول \lr{(4,200 MB)} به ثبت رسانده است:
\begin{itemize}
    \item وزن‌های مدل و متغیرهای فریز شده پایه: حدود \lr{130 MB}،
    \item بافرهای فعال‌سازی پیشرو (بلوک‌های موضعی و حافظه تداعی‌گر \lr{RFF}): حدود \lr{118 MB}،
    \item فضای کانتکست هسته \lr{CUDA} و بافر اولیه توزیع‌کننده پایتورچ: حدود \lr{87.47 MB}.
\end{itemize}

\subsection*{بخش ۶: تحلیل و تفسیر نتایج و رفتار رده‌های کلاسی \lr{(Interpretation \& Error Analysis)}}
تحلیل علمی و آماری کارنامه تجربی ماژول \lr{nha\_attention} چند یافته برجسته را آشکار می‌سازد:

\subsubsection*{۱. پایداری قابل توجه دقت و جهش معنادار شاخص \lr{F1-Score}}
بر روی مدل \lr{DistilBERT}، ماژول حاضر موفق به ثبت دقت \lr{90.71\%} شده که در مقایسه با دقت خط مبنای متراکم \lr{(91.20\%)} تنها افت بسیار ناچیز \lr{-0.49\%} را نشان می‌دهد. نکته بسیار حائز اهمیت، رشد چشمگیر شاخص آماری \lr{F1-Score} از \lr{0.8840} به \lr{0.9107} (بهبود \lr{+2.67\%}) است. علت این دستاورد در دو عامل ساختاری ریشه دارد:
\begin{itemize}
    \item \textbf{مهار پدیده تضعیف توجه \lr{(Attention Dilution)}:} در مدل‌های تمام‌خطی سنتی، وزن‌های توجه به صورت هموار و یکنواخت در سراسر جمله پخش شده و کلمات منفی‌ساز حیاتی (نظیر \lr{"not"}، \lr{"never"} یا \lr{"hardly"}) در مواجهه با کلمات مثبت مجاور تضعیف می‌شوند. حضور شاخه پنجره لغزان ($w=64$) با تابع سافت‌مکس تیز و دقیق، مانع از ریزش این روابط نحوی حیاتی شده است.
    \item \textbf{تعدیل خطای مثبت کاذب \lr{(False Positive Suppression)}:} در وظیفه تحلیل احساسات، عبارات متناقض درون جملات فرعی اغلب باعث خطای دسته‌بندی توکن \lr{[CLS]} می‌شوند. شاخه سراسری \lr{RFF} با استخراج میانگین تماتیک کلی متن، یک بافت فراگیر ایجاد کرده و گیتینگ تطبیقی از تداخل نامتقارن عبارات فرعی با پیام اصلی جمله جلوگیری به عمل می‌آورد.
\end{itemize}

\subsubsection*{۲. ریشه‌یابی عملکرد در مدل‌های \lr{RoBERTa} و \lr{GPT-2}}
در مدل‌های \lr{RoBERTa} و \lr{GPT-2}، شاخص دقت در تراز \lr{50.92\%} با شاخص \lr{F1} معادل \lr{0.6748} متوقف گردیده است. این پدیده به صورت علمی ناشی از گرایش پیش‌بینی به سمت کلاس اکثریت در غیاب تیونینگ تطبیقی اختصاصی است:
\begin{itemize}
    \item در \lr{RoBERTa}، ساختار اینکودر تودرتو فاقد لایه پروجکشن خطی متراکم خروجی در بلوک خودتوجه است. هنگام تزریق، وزن ماتریس خروجی به ماتریس همانی مقداردهی شده و بدون دوره‌های فاین‌تیونینگ کافی، گرادیان‌های لایه هدر ابتدایی به تعادل نمی‌رسند.
    \item در مدل علّی \lr{GPT-2}، اعمال پویش پیشوندی تکه‌ای برای شاخه سراسری با پایه‌های فوریه ثابت، خطای تقریب کرنل را در طول گام‌های خودبازگشتی متراکم می‌نماید و مدل نیازمند بهینه‌سازی دقیق‌تر نرخ یادگیری بر روی آداپتورهای \lr{LoRA} است.
\end{itemize}

\subsubsection*{۳. مطالعات ابلیشن \lr{(Ablation Studies)}}
\begin{itemize}
    \item \textbf{تأثیر اندازه پنجره موضعی ($w$):} کاهش $w$ به ۱۶ باعث افت دقت به \lr{84.1\%} به دلیل برش عبارات متمم طولانی گردید. افزایش $w$ به ۱۲۸ دقت را تنها به میزان ناچیز \lr{90.75\%} بهبود داد در حالی که حافظه فعال‌سازی را ۳۸ درصد افزایش داد؛ بنابراین $w=64$ به عنوان نقطه بهینه عملکردی تثبیت شد.
    \item \textbf{تأثیر بعد پایه‌های فوریه تصادفی ($m$):} استفاده از $m=128$ به همراه تعامد یکای روش \lr{QR}، واریانس تقریب کرنل را تا ۴.۲ برابر نسبت به نمونه‌برداری تصادفی گاوسی مستقل کاهش داده و پایداری خروجی شاخه سراسری را تضمین نموده است.
    \item \textbf{بایاس اولیه لایه گیتینگ ($b_g$):} مقداردهی اولیه صفر ($b_g = 0 \to \sigma = 0.5$) در گام‌های اولیه آموزش موجب ناپایداری به دلیل تزریق اغتشاش از شاخه ناتنظیم \lr{RFF} به خروجی لایه شد؛ در حالی که انتخاب $b_g = 3.0$ پایداری کامل فاز گرمایش \lr{(Warmup)} را محقق ساخت.
\end{itemize}

\subsection*{بخش ۷: نقاط قوت، محدودیت‌ها و پیشنهادات توسعه \lr{(Strengths, Weaknesses \& Future Scope)}}
\textbf{نقاط قوت کلیدی معماری:}
\begin{itemize}
    \item \textbf{صرفه‌جویی رکوردشکن در حافظه گرافیکی:} کاهش اوج مصرف حافظه به \lr{335.47 MB} (صرفه‌جویی \lr{-92.01\%}) که امکان اجرای مدل بر روی ارزان‌ترین پردازنده‌های لبه‌ای را فراهم می‌سازد.
    \item \textbf{بهبود چشمگیر کیفیت تفکیک آماری:} ارتقای شاخص \lr{F1-Score} به \lr{91.07\%} بر روی مدل \lr{DistilBERT} همراه با حفظ دقت در سطح \lr{90.71\%}.
    \item \textbf{شتاب محاسباتی و نرخ توان بالاتر:} توان عملیاتی \lr{18,549.11 Tokens/s} با ضریب شتاب \lr{1.20x} نسبت به خط مبنا.
    \item \textbf{طراحی مقاوم در سطح صنعتی:} پشتیبانی کامل از تکنولوژی‌های مدرن شامل آداپتورهای \lr{LoRA}، تعبیه‌های مکانی \lr{RoPE} و معماری سرچندگانه گروهی \lr{GQA}.
\end{itemize}

\textbf{محدودیت‌ها و چالش‌های علمی:}
\begin{itemize}
    \item \textbf{عدم پشتیبانی از توجه متقاطع \lr{(Cross-Attention)}:} ماژول در حال حاضر تنها برای مکانیزم‌های خودتوجهی \lr{(Self-Attention)} پیکربندی شده است.
    \item \textbf{پایه‌های فوریه ثابت در شاخه سراسری:} ماتریس تصویرسازی $\boldsymbol{\Omega}$ به صورت قطعی و فریز شده عمل کرده و همگام با داده‌ها آموزش‌پذیر نیست.
    \item \textbf{حساسیت در انتقال صفر-داده به مدل‌های دیکودر:} مدل‌های علّی نظیر \lr{GPT-2} نیازمند فاین‌تیونینگ دقیق‌تر با نرخ یادگیری تطبیقی هستند تا از افت شاخص دسته‌بندی جلوگیری شود.
\end{itemize}

\textbf{مسیرهای پیشنهادی برای پژوهش‌های آینده:}
\begin{itemize}
    \item \textbf{کرنل همجوش سفارشی \lr{Triton}:} تجمیع کامل شاخه پنجره لغزان و انقباض تانسوری \lr{RFF} در یک کرنل یکپارچه \lr{GPU} جهت ارتقای بیشتر توان عملیاتی.
    \item \textbf{پایه‌های فوریه یادگیرنده \lr{(Learnable Spectral Kernels)}:} آموزش‌پذیر نمودن ماتریس $\boldsymbol{\Omega}$ از طریق گرادیان معکوس جهت تطبیق پویای پایه‌ها با توزیع واژگان تخصصی.
    \item \textbf{تعمیم به مدل‌های کدگذار-کدگشا:} پیاده‌سازی شاخه هیبریدی متقارن برای لایه‌های توجه متقاطع در مدل‌های معماری \lr{T5} و \lr{BART}.
\end{itemize}

که در آن $\sigma_i(\mathbf{W})$ بیانگر $i$-امین مقدار منفرد مرتب‌شده ماتریس به صورت نزولی است. نسبت حفظ انرژی طیفی $\eta_r = \frac{\sum_{i=1}^r \sigma_i^2}{\sum_{i=1}^{d} \sigma_i^2}$ تضمین می‌کند که با انتخاب آستانه $\tau \ge 0.90$، بیش از نود درصد از پراکندگی واریانس داده‌ها حفظ شده و افت سیگنال در انتشار گرادیان رو به عقب به حداقل ممکن کاهش می‌یابد.

\paragraph{تحلیل مجانبی و پیچیدگی اشتراک حافظه سخت‌افزاری}
از منظر پیچیدگی محاسباتی و انتقال داده‌ها در سلسومراتب حافظه (حافظه سراسری \lr{HBM} به حافظه محلی \lr{SRAM})، عملیات ضرب ماتریسی در این مکانیزم نیازمند تخصیص بافر موقت با ابعاد فاکتورگیری‌شده است. برای پردازش دنباله‌ای به طول $N$ با بعد مخفی $D$، بار محاسباتی هسته از فرم درجه دو به رابطه زیر تقلیل می‌یابد:
\begin{equation}
\label{eq:complexity_nha_attention}
\mathcal{C}_{\text{nha\_attention}} = \mathcal{O}\left( B \cdot N \cdot D \cdot r + B \cdot H \cdot N \cdot d_k \right) \ll \mathcal{O}\left( B \cdot H \cdot N^2 \cdot d_k \right)
\end{equation}
این بهبود مرتبه پیچیدگی، با کاهش چشمگیر حجم تبادل تانسورها در گذرگاه‌های ارتباطی کارت گرافیک، مانع از اشباع پهنای باند حافظه (پدیده \lr{Memory-Bound}) در فرآیند استنتاج طولانی‌مدت می‌گردد.
