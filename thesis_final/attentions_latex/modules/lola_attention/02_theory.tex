\subsection{مبانی نظری و فرمول‌بندی ریاضی}
در لایه‌های پروجکشن $\mathbf{W} \in \mathbb{R}^{d_{\text{out}} \times d_{\text{in}}}$، ماتریس وزن به حاصل‌ضرب دو ماتریس متعامد با رتبه پایین $r \ll \min(d_{\text{in}}, d_{\text{out}})$ تجزیه شده و توسط جمله باقیمانده \lr{LoRA} بازسازی می‌شود:
\begin{equation}
\mathbf{W} = \mathbf{U} \mathbf{V} + \frac{\alpha}{r_{\text{lora}}} \mathbf{B}_{\text{lora}} \mathbf{A}_{\text{lora}}
\end{equation}
که در آن $\mathbf{U} \in \mathbb{R}^{d_{\text{out}} \times r}$ و $\mathbf{V} \in \mathbb{R}^{r \times d_{\text{in}}}$ با شرایط ارتوگونالیتی اولیه $\mathbf{U}^T \mathbf{U} = \mathbf{I}$ مقداردهی شده و در طی فرآیند تنظیم دقیق تثبیت می‌گردند.

در سمت حافظه پنهان دیکودر، تانسورهای کلید و مقدار $\mathbf{K}$ و $\mathbf{V}$ در سه مخزن زمانی مجزا با ساختار چندسطحی مدیریت می‌شوند:
\begin{enumerate}
    \item \textbf{بافر توکن‌های اخیر (\lr{Recent Window}):} نگهداری دقیق $W_{\text{recent}}$ توکن گذشته برای ثبت وابستگی‌های نحوی بومی:
    \begin{equation}
    \mathbf{K}_{\text{recent}} = \mathbf{K}_{t - W_{\text{recent}} : t}
    \end{equation}
    \item \textbf{بافر نمونه‌برداری تُنُک (\lr{Strided Sparse Window}):} ذخیره توکن‌ها با گام زمانی $s > 1$ برای پوشش پنجره‌های میانی:
    \begin{equation}
    \mathbf{K}_{\text{sparse}} = \{ \mathbf{K}_i \mid i \equiv 0 \pmod s, \; i < t - W_{\text{recent}} \}
    \end{equation}
    \item \textbf{بافر تصویرسازی کم‌رتبه (\lr{Low-Rank Projected Buffer}):} کاهش بعد توکن‌های باستانی با استفاده از تجزیه ارتوگونال \lr{QR} روی ماتریس تصویر تصادفی ثابت $\mathbf{P} \in \mathbb{R}^{d \times k}$:
    \begin{equation}
    \mathbf{Q}_r, \mathbf{R}_r = \operatorname{QR}(\mathbf{K}_{\text{history}} \mathbf{P}), \quad \mathbf{K}_{\text{compressed}} = \mathbf{Q}_r \left( \mathbf{Q}_r^T \mathbf{K}_{\text{history}} \right)
    \end{equation}
\end{enumerate}
در نهایت، امتیازات توجه با الحاق این سه مخزن و با پیچیدگی بسیار کمتر از کش کامل محاسبه می‌گردد.


\paragraph{تحلیل طیفی و نامساوی‌های پایداری خطا}
برای بررسی میزان دقت تقریب ماتریس وزن یا نگاشت توجه در ماژول \lr{lola\_attention}، تفاضل ماتریس فاکتورگیری‌شده از نگاشت متراکم اصلی بر اساس نامساوی هافدینگ-فون‌نیومان ارزیابی می‌گردد. فرض کنید $\mathbf{W} \in \mathbb{R}^{d_{\text{in}} \times d_{\text{out}}}$ ماتریس وزن هدف باشد که توسط تقریب کم‌مرتبه یا پراکنده با رتبه موثر $r$ جایگزین شده است. کران بالای خطای بازسازی بر مبنای نرم فروبنیوس به صورت زیر تصریح می‌شود:
\begin{equation}
\label{eq:spectral_bound_lola_attention}
\| \mathbf{W} - \mathbf{W}_{r} \|_F = \sqrt{\sum_{i=r+1}^{\min(d_{\text{in}}, d_{\text{out}})} \sigma_i^2(\mathbf{W})} \le \epsilon \| \mathbf{W} \|_F
\end{equation}
که در آن $\sigma_i(\mathbf{W})$ بیانگر $i$-امین مقدار منفرد مرتب‌شده ماتریس به صورت نزولی است. نسبت حفظ انرژی طیفی $\eta_r = \frac{\sum_{i=1}^r \sigma_i^2}{\sum_{i=1}^{d} \sigma_i^2}$ تضمین می‌کند که با انتخاب آستانه $\tau \ge 0.90$، بیش از نود درصد از پراکندگی واریانس داده‌ها حفظ شده و افت سیگنال در انتشار گرادیان رو به عقب به حداقل ممکن کاهش می‌یابد.

\paragraph{تحلیل مجانبی و پیچیدگی اشتراک حافظه سخت‌افزاری}
از منظر پیچیدگی محاسباتی و انتقال داده‌ها در سلسومراتب حافظه (حافظه سراسری \lr{HBM} به حافظه محلی \lr{SRAM})، عملیات ضرب ماتریسی در این مکانیزم نیازمند تخصیص بافر موقت با ابعاد فاکتورگیری‌شده است. برای پردازش دنباله‌ای به طول $N$ با بعد مخفی $D$، بار محاسباتی هسته از فرم درجه دو به رابطه زیر تقلیل می‌یابد:
\begin{equation}
\label{eq:complexity_lola_attention}
\mathcal{C}_{\text{lola\_attention}} = \mathcal{O}\left( B \cdot N \cdot D \cdot r + B \cdot H \cdot N \cdot d_k \right) \ll \mathcal{O}\left( B \cdot H \cdot N^2 \cdot d_k \right)
\end{equation}
این بهبود مرتبه پیچیدگی، با کاهش چشمگیر حجم تبادل تانسورها در گذرگاه‌های ارتباطی کارت گرافیک، مانع از اشباع پهنای باند حافظه (پدیده \lr{Memory-Bound}) در فرآیند استنتاج طولانی‌مدت می‌گردد.
