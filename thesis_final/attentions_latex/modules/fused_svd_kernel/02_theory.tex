\subsection{مبانی نظری و فرمول‌بندی ریاضی}
در این لایه، تبدیل ماتریسی وزن‌های پروجکشن با تجزیه مقادیر منفرد فاکتورگیری شده به صورت دو مؤلفه متعامد $\mathbf{V} \in \mathbb{R}^{r \times D_{in}}$ و $\mathbf{U} \in \mathbb{R}^{D_{out} \times r}$ بازتعریف می‌شود:
\begin{equation}
\mathbf{W} \approx \mathbf{U} \mathbf{\Sigma} \mathbf{V}
\end{equation}
فرآیند محاسباتی یکپارچه درون ثبات‌ها و حافظه اشتراکی پردازنده (\lr{SRAM}) به صورت زنجیره زیر تعریف می‌گردد:
\begin{enumerate}
    \item تصویرسازی کاهشی به زیرفضای رتبه $r$:
    \begin{equation}
    \mathbf{Z} = \mathbf{X} \mathbf{V}^\top \in \mathbb{R}^{B \times N \times r}
    \end{equation}
    \item اعمال نرمال‌سازی \lr{RMSNorm} بدون خروج داده از حافظه \lr{SRAM}:
    \begin{equation}
    \mathbf{Z}_{\text{norm}} = \frac{\mathbf{Z}}{\sqrt{\frac{1}{r} \sum_{i=1}^r Z_i^2 + \epsilon}} \odot \boldsymbol{\gamma}
    \end{equation}
    \item تصویرسازی افزایشی بلافاصله به فضای خروجی:
    \begin{equation}
    \mathbf{Y} = \mathbf{Z}_{\text{norm}} \mathbf{U}^\top + \mathbf{b} \in \mathbb{R}^{B \times N \times D_{out}}
    \end{equation}
\end{enumerate}
سپس جریان ضرب داخلی خطی یا فاکتورگیری‌شده جهت محاسبه خروجی توجه با عبور از نگاشت‌های کرنل فرمول‌بندی شده و تانسور خروجی نهایی شکل می‌گیرد.


\paragraph{تحلیل طیفی و نامساوی‌های پایداری خطا}
برای بررسی میزان دقت تقریب ماتریس وزن یا نگاشت توجه در ماژول \lr{fused\_svd\_kernel}، تفاضل ماتریس فاکتورگیری‌شده از نگاشت متراکم اصلی بر اساس نامساوی هافدینگ-فون‌نیومان ارزیابی می‌گردد. فرض کنید $\mathbf{W} \in \mathbb{R}^{d_{\text{in}} \times d_{\text{out}}}$ ماتریس وزن هدف باشد که توسط تقریب کم‌مرتبه یا پراکنده با رتبه موثر $r$ جایگزین شده است. کران بالای خطای بازسازی بر مبنای نرم فروبنیوس به صورت زیر تصریح می‌شود:
\begin{equation}
\label{eq:spectral_bound_fused_svd_kernel}
\| \mathbf{W} - \mathbf{W}_{r} \|_F = \sqrt{\sum_{i=r+1}^{\min(d_{\text{in}}, d_{\text{out}})} \sigma_i^2(\mathbf{W})} \le \epsilon \| \mathbf{W} \|_F
\end{equation}
که در آن $\sigma_i(\mathbf{W})$ بیانگر $i$-امین مقدار منفرد مرتب‌شده ماتریس به صورت نزولی است. نسبت حفظ انرژی طیفی $\eta_r = \frac{\sum_{i=1}^r \sigma_i^2}{\sum_{i=1}^{d} \sigma_i^2}$ تضمین می‌کند که با انتخاب آستانه $\tau \ge 0.90$، بیش از نود درصد از پراکندگی واریانس داده‌ها حفظ شده و افت سیگنال در انتشار گرادیان رو به عقب به حداقل ممکن کاهش می‌یابد.

\paragraph{تحلیل مجانبی و پیچیدگی اشتراک حافظه سخت‌افزاری}
از منظر پیچیدگی محاسباتی و انتقال داده‌ها در سلسومراتب حافظه (حافظه سراسری \lr{HBM} به حافظه محلی \lr{SRAM})، عملیات ضرب ماتریسی در این مکانیزم نیازمند تخصیص بافر موقت با ابعاد فاکتورگیری‌شده است. برای پردازش دنباله‌ای به طول $N$ با بعد مخفی $D$، بار محاسباتی هسته از فرم درجه دو به رابطه زیر تقلیل می‌یابد:
\begin{equation}
\label{eq:complexity_fused_svd_kernel}
\mathcal{C}_{\text{fused\_svd\_kernel}} = \mathcal{O}\left( B \cdot N \cdot D \cdot r + B \cdot H \cdot N \cdot d_k \right) \ll \mathcal{O}\left( B \cdot H \cdot N^2 \cdot d_k \right)
\end{equation}
این بهبود مرتبه پیچیدگی، با کاهش چشمگیر حجم تبادل تانسورها در گذرگاه‌های ارتباطی کارت گرافیک، مانع از اشباع پهنای باند حافظه (پدیده \lr{Memory-Bound}) در فرآیند استنتاج طولانی‌مدت می‌گردد.
