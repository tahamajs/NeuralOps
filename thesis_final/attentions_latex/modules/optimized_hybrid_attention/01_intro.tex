\subsection*{بخش ۱: ایده محوری و چرایی معماری \lr{(Architectural Rationale)}}
مکانیزم توجه چندسره استاندارد \lr{(Standard Multi-Head Attention - MHA)} سنگ‌بنای درک بافتار در ترنسفورمرها به شمار می‌رود، اما هزینه محاسباتی و ردپای حافظه آن به صورت مربعی $\mathcal{O}(N^2)$ با افزایش طول دنباله توکن‌ها ($N$) مقیاس‌بندی می‌شود. این وابستگی مرتبه دو ناشی از تشکیل ماتریس ضرب داخلی دوبه‌دو میان تمامی جفت‌های پرس‌وجو و کلید است که در طول‌های توالی بلند ($N > 2048$) به گلوگاه بحرانی پهنای باند حافظه \lr{(Memory Bandwidth Bottleneck)} و فرسودگی حافظه موقت پردازنده گرافیکی منجر می‌گردد.

پیش از معرفی رویکردهای ترکیبی، تلاش‌های پیشین جهت مهار این گلوگاه به دو پارادایم مجزا تقسیم می‌شدند که هر یک با تناقض‌های ساختاری روبرو بودند:
\begin{enumerate}
    \item \textbf{مکانیزم‌های توجه محلی و پنجره لغزان \lr{(Sliding-Window Attention)}:} این رویکردها دامنه پذیرش هر توکن را به یک پنجره محلی با شعاع محدود $W_{\text{\lr{window}}}$ محدود ساخته و پیچیدگی را به $\mathcal{O}(N \cdot W_{\text{\lr{window}}})$ کاهش می‌دهند. با این حال، محروم‌سازی مدل از دریافت سیگنال‌های دوربرد، توانایی درک پیوندهای فرامتنی و استدلال‌های معنایی سراسری متن را به شدت تضعیف می‌کند.
    \item \textbf{مکانیزم‌های توجه فشرده و تقریب‌های کم‌مرتبه \lr{(Sequence Compression \& Low-Rank Pooling)}:} این دسته از روش‌ها با اعمال نگاشت‌های خطی یا ادغام‌های سراسری، بعد توالی را فشرده می‌سازند؛ اما در فرایند متراکم‌سازی، جزئیات موقعیتی، پیوندهای نحوی محلی \lr{(Local Syntactic Bindings)} و ارجاعات همسایگی در میانگین‌گیری سراسری محو و تار می‌شوند.
\end{enumerate}

ماژول \lr{Optimized Hybrid Attention (OHA)} با هدف ابطال این پارادوکس، معماری هیبریدی دو-مسیره \lr{(Dual-Path Hybrid Architecture)} را با هدایت تلفیق دینامیک پیشنهاد می‌دهد. ایده محوری \lr{OHA} بر این فرض اثبات‌شده هندسی استوار است که رفتار توجه زبانی ماهیتی چندمقیاسی \lr{(Multi-Scale Nature)} دارد: الگوهای محلی نیازمند تفکیک‌پذیری کامل ضرب داخلی در همسایگی نزدیک هستند، در حالی که اطلاعات دوربرد را می‌توان در یک زیرفضای فشرده با چگالی طیفی بالا بدون تخریب بار معنایی خلاصه نمود. 

معماری \lr{OHA} به صورت سیستماتیک از سه جزء هم‌افزا تشکیل یافته است:
\begin{itemize}
    \item \textbf{شاخه محلی پنجره لغزان \lr{(Local Sliding-Window Path)}:} متمرکز بر تفکیک‌پذیری موضعی برای بازنمایی الگوهای گرامری و جفت‌شدگی‌های مجاور درون شعاع پنجره $W_{\text{\lr{window}}} = 64$ با بار محاسباتی خطی.
    \item \textbf{شاخه سراسری فشرده‌شده بر مبنای دنباله \lr{(Sequence-Compressed Low-Rank Global Path)}:} نگاشت کل توالی به یک زیرفضای نهفته با رتبه فشرده $R \ll N$ ($R = 32$) از طریق وزن‌های وزن‌دار تجمیعی سافت‌مکس در بعد دنباله، که امکان تبادل اطلاعات میان دورترین توکن‌ها را در زمان خطی $\mathcal{O}(N \cdot R \cdot d)$ فراهم می‌آورد.
    \item \textbf{مدولاسیون گیتینگ پویا در سطح سرها \lr{(Dynamic Head-Wise Gated Modulation)}:} یک شبکه گیتینگ سیگموئیدی پیش‌بین برای هر سر توجه که ضریب تلفیق بهینه میان بافت محلی و بافت سراسری را به شکل توکن‌به‌توکن تنظیم می‌نماید و با بایاس اولیه مثبت ($b_{\text{\lr{gate}}} = 2.0$) پایداری همگرایی را در مراحل آغازین آموزش تضمین می‌کند.
    \item \textbf{ادغام آداپتورهای کم‌مرتبه آموزش‌پذیر \lr{(LoRA Integration)}:} مجهزسازی لایه‌های پروجکشن پایه به مسیرهای فاکتورگیری‌شده کم‌مرتبه $r_{\text{\lr{lora}}} = 16$ بر روی وزن‌های منجمد \lr{(Frozen Weights)} که امکان بازآرایی زیرفضا را با کمترین بار بازآموزی پارامتری میسر می‌سازد.
\end{itemize}

\subsection*{بخش ۲: فرمول‌بندی ریاضی و مبانی نظری \lr{(Mathematical Formulation)}}
در این بخش، کلیه مراحل تبدیل تانسوری، نگاشت‌های خطی و انقباض‌های فضای توجه با دقت نمادگذاری ریاضی تبیین می‌گردد.

\subsubsection*{تعریف تانسورها و پارامترهای ورودی}
فرض کنید تانسور تعبیه پنهان ورودی لایه توجه به صورت $\mathbf{X} \in \mathbb{R}^{B \times N \times D}$ باشد که در آن $B$ نمایانگر اندازه دسته \lr{(Batch Size)}، $N$ طول دنباله توکن‌ها \lr{(Sequence Length)}، و $D$ بعد پنهان مدل \lr{(Hidden Dimension)} است. لایه دارای $H$ سر توجه با بعد سر $d = D / H$ است. برای پشتیبانی از معماری‌های مدرن، تعداد سرهای کلید و مقدار با $H_{KV}$ نشان داده می‌شود که نسبت گروه‌بندی کوئری \lr{(GQA)} به صورت $G = H / H_{KV}$ تعریف می‌گردد.

\subsubsection*{پروجکشن‌های خطی با لایه‌های \lr{LoRALinear}}
ماتریس‌های تبدیل پرس‌وجو، کلید و مقدار با استفاده از فرمولاسیون فاکتورسازی تطبیقی کم‌مرتبه تعریف می‌گردند:
\begin{equation}
\mathbf{W}_{\text{\lr{eff}}} = \mathbf{W}_0 + \frac{\alpha}{r_{\text{\lr{lora}}}} \mathbf{B} \mathbf{A}, \quad \mathbf{W}_0 \in \mathbb{R}^{D_{\text{\lr{out}}} \times D_{\text{\lr{in}}}}, \ \mathbf{A} \in \mathbb{R}^{r_{\text{\lr{lora}}} \times D_{\text{\lr{in}}}}, \ \mathbf{B} \in \mathbb{R}^{D_{\text{\lr{out}}} \times r_{\text{\lr{lora}}}}
\end{equation}
که در آن وزن‌های پایه $\mathbf{W}_0$ منجمد بوده ($\nabla_{\mathbf{W}_0} \mathcal{L} = 0$) و ماتریس‌های $\mathbf{A}$ و $\mathbf{B}$ با رتبه $r_{\text{\lr{lora}}} = 16$ و مقیاس $\alpha = 32.0$ آموزش می‌بینند. پروجکشن‌های سه‌گانه به شرح زیر محاسبه می‌شوند:
\begin{align}
\mathbf{Q} &= \mathbf{X} \mathbf{W}_Q + \mathbf{b}_Q \in \mathbb{R}^{B \times N \times (H \cdot d)} \xrightarrow{\text{\lr{Reshape \& Permute}}} \mathbb{R}^{B \times H \times N \times d} \\
\mathbf{K} &= \mathbf{X} \mathbf{W}_K + \mathbf{b}_K \in \mathbb{R}^{B \times N \times (H_{KV} \cdot d)} \xrightarrow{\text{\lr{Reshape \& Permute}}} \mathbb{R}^{B \times H_{KV} \times N \times d} \\
\mathbf{V} &= \mathbf{X} \mathbf{W}_V + \mathbf{b}_V \in \mathbb{R}^{B \times N \times (H_{KV} \cdot d)} \xrightarrow{\text{\lr{Reshape \& Permute}}} \mathbb{R}^{B \times H_{KV} \times N \times d}
\end{align}
در صورت برقراری $G > 1$، تانسورهای $\mathbf{K}$ و $\mathbf{V}$ از طریق تکرار ابعادی با متد \lr{repeat\_kv} به ابعاد متناظر $\mathbb{R}^{B \times H \times N \times d}$ گسترش می‌یابند.

\subsubsection*{کدگذاری موقعیتی چرخشی \lr{(Rotary Position Embeddings - RoPE)}}
در صورت پیکربندی جاسازی موقعیت با بردار فرکانسی چرخشی $(\cos, \sin)$، تبدیل هرمیتین متعامد به صورت برداری بر روی نیمه‌های بعد سر اعمال می‌شود:
\begin{equation}
\operatorname{RoPE}(\mathbf{Z}) = (\mathbf{Z} \odot \cos) + (\operatorname{rotate\_half}(\mathbf{Z}) \odot \sin)
\end{equation}
که در آن $\operatorname{rotate\_half}(\mathbf{Z}) = [-\mathbf{Z}_{:, :, :, d/2:}, \ \mathbf{Z}_{:, :, :, :d/2}]$ بوده و بر روی $\mathbf{Q}$ و $\mathbf{K}$ نگاشته می‌شود.

\subsubsection*{مسیر اول: توجه پنجره لغزان محلی \lr{(Sliding-Window Local Branch)}}
برای هر توکن در موقعیت $i$ و کلید در موقعیت $j$، ماتریس امتیازدهی محلی با در نظر گرفتن پنجره همسایگی $W_{\text{\lr{window}}}$ به صورت زیر فرمول‌بندی می‌شود:
\begin{equation}
S_{b, h, i, j}^{\text{\lr{local}}} = \frac{\mathbf{Q}_{b, h, i} \mathbf{K}_{b, h, j}^\top}{\sqrt{d}} + M_{i, j}^{\text{\lr{band}}} + M_{b, j}^{\text{\lr{pad}}}
\end{equation}
که در آن ماسک نواری $M^{\text{\lr{band}}}$ بر اساس ماهیت مدل (دوجهته یا علّی خودبازگشتی) تعریف می‌گردد:
\begin{equation}
M_{i, j}^{\text{\lr{band}}} = \begin{cases} 
0 & \text{\lr{if }} |i - j| \le W_{\text{\lr{window}}} \quad \text{\lr{(Bidirectional Encoder)}} \\
0 & \text{\lr{if }} 0 \le i - j \le W_{\text{\lr{window}}} \quad \text{\lr{(Causal Decoder)}} \\
-\infty & \text{\lr{otherwise}}
\end{cases}
\end{equation}
توزیع احتمالاتی و بردار زمینه محلی از طریق تابع بیشینه هموار به دست می‌آیند:
\begin{equation}
\mathbf{A}^{\text{\lr{local}}} = \operatorname{Softmax}\left( \mathbf{S}^{\text{\lr{local}}} \right), \quad \mathbf{O}^{\text{\lr{local}}} = \mathbf{A}^{\text{\lr{local}}} \mathbf{V} \in \mathbb{R}^{B \times H \times N \times d}
\end{equation}

\subsubsection*{مسیر دوم: توجه سراسری فشرده‌شده با رتبه پایین \lr{(Compressed Global Branch)}}
جهت استخراج نمای سراسری کل توالی بدون پرداخت هزینه مربعی، تانسور ورودی $\mathbf{X}$ به فضاهای فشرده‌سازی با رتبه $R$ نگاشته می‌شود:
\begin{equation}
\mathbf{C}_K = \mathbf{X} \mathbf{W}_K^{\text{\lr{comp}}} \in \mathbb{R}^{B \times N \times R}, \quad \mathbf{C}_V = \mathbf{X} \mathbf{W}_V^{\text{\lr{comp}}} \in \mathbb{R}^{B \times N \times R}
\end{equation}
که در آن $\mathbf{W}_K^{\text{\lr{comp}}}, \mathbf{W}_V^{\text{\lr{comp}}} \in \mathbb{R}^{D \times R}$ نگاشت‌های خطی فاقد بایاس می‌باشند. سپس با اعمال ماسک توکن‌های معتبر ($M^{\text{\lr{pad}}}$)، توزیع وزن‌دهی فشرده‌سازی در امتداد بعد دنباله ($N$) محاسبه می‌گردد:
\begin{equation}
\mathbf{P}_K = \operatorname{Softmax}(\mathbf{C}_K + M^{\text{\lr{pad}}}, \operatorname{dim}=1) \in \mathbb{R}^{B \times N \times R}
\end{equation}
\begin{equation}
\mathbf{P}_V = \operatorname{Softmax}(\mathbf{C}_V + M^{\text{\lr{pad}}}, \operatorname{dim}=1) \in \mathbb{R}^{B \times N \times R}
\end{equation}
تانسورهای متراکم کلید و مقدار سراسری با ادغام انقباضی در طول بعد دنباله تولید می‌شوند:
\begin{equation}
\widetilde{\mathbf{K}}_{b, h, r, :} = \sum_{n=1}^N P_{K, b, n, r} \mathbf{K}_{b, h, n, :} \implies \widetilde{\mathbf{K}} \in \mathbb{R}^{B \times H \times R \times d}
\end{equation}
\begin{equation}
\widetilde{\mathbf{V}}_{b, h, r, :} = \sum_{n=1}^N P_{V, b, n, r} \mathbf{V}_{b, h, n, :} \implies \widetilde{\mathbf{V}} \in \mathbb{R}^{B \times H \times R \times d}
\end{equation}
سپس، پرس‌وجوها نسبت به پایه‌های فشرده سراسری مورد سنجش قرار می‌گیرند:
\begin{equation}
\mathbf{S}^{\text{\lr{global}}} = \frac{\mathbf{Q} \widetilde{\mathbf{K}}^\top}{\sqrt{d}} \in \mathbb{R}^{B \times H \times N \times R}
\end{equation}
در مدل‌های دیکودر علّی، ماسک مثلثی رتبه $\mathbf{M}^{\text{\lr{causal\_global}}} \in \mathbb{R}^{N \times R}$ بر روی ابعاد مربوطه اعمال می‌گردد تا از نشت اطلاعات توکن‌های متناظر جلوگیری شود. در نهایت، بافتار سراسری به شرح زیر بازسازی می‌شود:
\begin{equation}
\mathbf{A}^{\text{\lr{global}}} = \operatorname{Softmax}\left( \mathbf{S}^{\text{\lr{global}}} \right) \in \mathbb{R}^{B \times H \times N \times R}, \quad \mathbf{O}^{\text{\lr{global}}} = \mathbf{A}^{\text{\lr{global}}} \widetilde{\mathbf{V}} \in \mathbb{R}^{B \times H \times N \times d}
\end{equation}

\subsubsection*{تلفیق گیتینگ پویا و پروجکشن پایانی}
ترکیب محدب دو بردار بافتار محلی و سراسری از طریق یک شبکه گیتینگ خطی سبک با تابع فعال‌سازی لجستیک صورت می‌پذیرد:
\begin{equation}
\mathbf{g} = \sigma\left( \mathbf{X} \mathbf{W}_{\text{\lr{gate}}} + \mathbf{b}_{\text{\lr{gate}}} \right) \in \mathbb{R}^{B \times N \times H} \xrightarrow{\text{\lr{Permute \& Unsqueeze}}} \mathbb{R}^{B \times H \times N \times 1}
\end{equation}
بردار بافتار نهایی و خروجی لایه توجه از فرمولاسیون زیر حاصل می‌آیند:
\begin{equation}
\mathbf{O}^{\text{\lr{fused}}} = \mathbf{g} \odot \mathbf{O}^{\text{\lr{local}}} + (1 - \mathbf{g}) \odot \mathbf{O}^{\text{\lr{global}}} \in \mathbb{R}^{B \times H \times N \times d}
\end{equation}
\begin{equation}
\mathbf{Y} = \operatorname{Reshape}\left( \mathbf{O}^{\text{\lr{fused}}} \right) \mathbf{W}_O + \mathbf{b}_O \in \mathbb{R}^{B \times N \times D}
\end{equation}
در ساختارهای انکودر تودرتو \lr{(Nested Encoders)}، پروجکشن خروجی با نگاشت همانی تنظیم شده و وظیفه اعمال پروجکشن پایانی به لایه متناظر خارجی \lr{(output.dense)} واگذار می‌گردد تا از انباشت خطاهای ماتریسی جلوگیری شود.

\subsubsection*{سلسله‌مراتب کلاس‌ها و طراحی نرم‌افزاری}
ماژول بر پایه دو کلاس اصلی سازمان‌دهی شده است:
\begin{enumerate}
    \item \textbf{کلاس \lr{LoRALinear}:} مشتق از \lr{nn.Module} با نگهداری تانسورهای وزنی مبنا \lr{weight} و بایاس، و پارامترهای جبرانی کم‌مرتبه \lr{lora\_A} و \lr{lora\_B}. در فاز آموزش، متد \lr{freeze\_base\_weights} ویژگی \lr{requires\_grad = False} را روی وزن‌های اصلی فعال می‌سازد و تنها پارامترهای افزایشی کم‌مرتبه گرادیان دریافت می‌نمایند. ضریب مقیاس‌گذاری با $\alpha / r_{\text{\lr{lora}}}$ تضمین‌کننده مقیاس مناسب خروجی در حین مقداردهی اولیه کایمینگ \lr{(Kaiming Uniform)} است.
    \item \textbf{کلاس اصلی \lr{OptimizedHybridAttention}:} مشتق از کلاس انتزاعی \lr{BaseAttention} و ثبت‌شده در رجیستری مرکزی معماری با شناسه رسمی \lr{"oha"} و \lr{"optimized\_hybrid\_attention"}. این کلاس متدهای تخصصی انتقال وزن \lr{copy\_weights} و کنترل ابعادی را دربر می‌گیرد.
\end{enumerate}

\subsubsection*{مکانیزم جامع انتقال وزن پیش‌آموزش‌دیده \lr{(copy\_weights)}}
متد \lr{copy\_weights} برای انطباق بدون درز با چهار خانواده اصلی مدل‌های زبانی طراحی شده است:
\begin{itemize}
    \item \textbf{مدل‌های با پروجکشن ادغام‌شده \lr{(GPT-2 Conv1D)}:} استخراج لایه \lr{c\_attn} با ابعاد $[D, 3D]$، ترانهاده‌سازی ابعادی ماتریس و تفکیک دقیق به سه بلوک مساوی برای لایه‌های کوئری، کلید و مقدار، و تفکیک \lr{c\_proj} برای پروجکشن خروجی.
    \item \textbf{مدل‌های انکودر تودرتو \lr{(RoBERTa / BERT Nested)}:} بازیابی وزن‌ها از زیربلوک‌های داخلی \lr{self.query}، \lr{self.key} و \lr{self.value}. با تشخیص این ساختار، متغیر بولین \lr{is\_nested\_encoder} فعال شده و وزن‌های لایه \lr{out\_proj} ماژول با ماتریس همانی مقداردهی می‌شوند تا تداخلی با لایه خارجی \lr{output.dense} ایجاد نگردد.
    \item \textbf{مدل‌های با پروجکشن‌های مسطح \lr{(DistilBERT, Qwen, LLaMA)}:} مپینگ مستقیم از ماژول‌های تک‌لایه نظیر \lr{q\_lin}، \lr{k\_lin}، \lr{v\_lin} و \lr{out\_lin}.
    \item \textbf{سنجش ایمنی نُرم فروبنیوس:} پس از کپی وزن‌ها، نرم فروبنیوس تانسورها سنجیده می‌شود:
    \begin{equation}
    \|\mathbf{W}\|_F = \sqrt{\sum_{i, j} w_{i, j}^2} \ge 10^{-4}
    \end{equation}
    در صورت کمتر بودن نُرم از آستانه عددی فوق، خطای زمان اجرا صادر شده و تزریق نامعتبر خنثی می‌گردد.
\end{itemize}
