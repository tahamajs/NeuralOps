\subsection{B.1 SALTLinear}

\begin{lstlisting}
class SALTLinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, rank=16,
                 original\_weight=None, original\_bias=None):
        super().\_\_init\_\_()
        if not hasattr(self, 'output\_scale'):
            self.output\_scale = nn.Parameter(torch.ones(1))  # DEAD CODE
        
        self.in\_features = in\_features
        self.out\_features = out\_features
        self.rank = min(rank, min(in\_features, out\_features))

        W = original\_weight.float() if original\_weight is not None else ...
        U, S, Vt = torch.linalg.svd(W, full\_matrices=False)
        
        # Buffers
        self.register\_buffer("U", U[:, :self.rank])
        self.register\_buffer("Vt", Vt[:self.rank, :])
        self.register\_buffer("S", S[:self.rank])
        
        # Trainable
        self.gamma = nn.Parameter(torch.ones(self.rank))
        self.A = nn.Parameter(torch.randn(out\_features, self.rank) * 0.01)
        self.B = nn.Parameter(torch.zeros(self.rank, in\_features))
        
        if original\_bias is not None:
            self.bias = nn.Parameter(original\_bias.clone())
        else:
            self.bias = nn.Parameter(torch.zeros(out\_features))
    
    @property
    def weight(self):
        adapted\_S = self.S * self.gamma
        W\_svd = (self.U * adapted\_S.unsqueeze(0)) @ self.Vt
        W\_lora = self.A @ self.B
        return W\_svd + W\_lora
    
    def forward(self, x):
        adapted\_S = self.S * self.gamma
        W\_svd = (self.U * adapted\_S.unsqueeze(0)) @ self.Vt
        W\_lora = self.A @ self.B
        W = (W\_svd + W\_lora).to(dtype=x.dtype)
        return F.linear(x, W, self.bias.to(dtype=x.dtype))
\end{lstlisting}

    z = z * (self.S * self.gamma)                     # scale
    y\_svd = F.linear(z, self.U)                       # (B, L, d\_out)
    y\_lora = F.linear(F.linear(x, self.B), self.A)    # (B, L, d\_out)
