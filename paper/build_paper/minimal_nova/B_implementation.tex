\subsection{B.1 The `FactorisedLinear` Module}

\begin{lstlisting}
class FactorisedLinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, rank=64, bias=True,
                 energy=0.95, lora\_rank=16, lora\_alpha=32.0, ...):
        super().\_\_init\_\_()
        self.in\_features = in\_features
        self.out\_features = out\_features
        self.rank = min(rank, in\_features, out\_features)
        self.energy = energy
        self.lora\_rank = lora\_rank
        self.lora\_scaling = lora\_alpha / lora\_rank if lora\_rank \textgreater{} 0 else 1.0
        
        self.down = nn.Linear(in\_features, self.rank, bias=False)
        self.up = nn.Linear(self.rank, out\_features, bias=bias)
        
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.randn(lora\_rank, in\_features) * 0.01)
            self.lora\_B = nn.Parameter(torch.zeros(out\_features, lora\_rank))
        
        self.reset\_parameters()
    
    def forward(self, x):
        out = self.up(self.down(x))
        if self.lora\_rank \textgreater{} 0:
            out = out + F.linear(F.linear(x, self.lora\_A), self.lora\_B) * self.lora\_scaling
        return out
\end{lstlisting}

\subsection{B.2 The `decompose\_from\_weight` Method}

\begin{lstlisting}
def decompose\_from\_weight(self, weight, bias=None, energy=None):
    with torch.no\_grad():
        W = weight.float()
        
        # Handle KV dimension matching for GQA
        if W.shape[0] \textless{} self.out\_features and self.out\_features \% W.shape[0] == 0:
            rep = self.out\_features // W.shape[0]
            W = W.repeat(rep, 1)
        elif W.shape[0] \textgreater{} self.out\_features and W.shape[0] \% self.out\_features == 0:
            rep = W.shape[0] // self.out\_features
            W = W.view(self.out\_features, rep, -1).mean(dim=1)
        
        # SVD
        U, S, Vt = torch.linalg.svd(W, full\_matrices=False)
        
        # Energy-based rank selection
        target\_energy = energy if energy is not None else self.energy
        total\_energy = (S ** 2).sum()
        cum\_energy = torch.cumsum(S ** 2, dim=0) / (total\_energy + 1e-12)
        calc\_rank = int((cum\_energy \textless{} target\_energy).sum().item()) + 1
        r = min(self.rank, calc\_rank, U.size(1), Vt.size(0))
        r = max(1, r)
        
        # Symmetric factorization
        sqrt\_s = torch.sqrt(S[:r])
        A = (U[:, :r] * sqrt\_s).to(self.up.weight.dtype)
        B = (sqrt\_s.unsqueeze(1) * Vt[:r, :]).to(self.down.weight.dtype)
        
        # Store
        self.up.weight.data.zero\_()
        self.down.weight.data.zero\_()
        self.up.weight.data[:, :r] = A
        self.down.weight.data[:r, :] = B
        
        self.freeze\_base\_weights()
\end{lstlisting}

if is\_conv1d:
    w = w.t()    # (out, in)
