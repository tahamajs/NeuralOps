class PrimalAttention(BaseAttention):
    def \_\_init\_\_(self, config=None, rank=64, pinv\_iters=6, pretrained\_attn=None, **kwargs):

\subsection{B.2 Landmark Pooling}

\begin{lstlisting}
def pool\_to\_landmarks(x, m):
    """Segment pooling: divide into m chunks, average each."""
    B, H, L, d = x.shape
    if L \% m == 0:
        s = L // m
        return x.reshape(B, H, m, s, d).mean(dim=3)
    else:
        idx = torch.linspace(0, L - 1, m, dtype=torch.long, device=x.device)
        return x[:, :, idx, :]
\end{lstlisting}

\subsection{B.3 Forward Pass}

\begin{lstlisting}
def forward(self, hidden\_states, attention\_mask=None, ...):
    B, L, D = hidden\_states.shape
    
    # 1. Project
    Q = self.query(hidden\_states).view(B, L, H, d).permute(0, 2, 1, 3)
    K = self.key(hidden\_states).view(B, L, G, d).permute(0, 2, 1, 3)
    V = self.value(hidden\_states).view(B, L, G, d).permute(0, 2, 1, 3)
    
    # 2. RoPE
    if position\_embeddings is not None:
        Q, K = apply\_rotary\_pos\_emb(Q, K, cos, sin)
    
    # 3. GQA repeat
    if H != G:
        K = repeat\_kv(K, H // G)
        V = repeat\_kv(V, H // G)
    
    # 4. Landmark pooling
    m = min(self.rank, L)
    Q\_land = pool\_to\_landmarks(Q, m)
    K\_land = pool\_to\_landmarks(K, m)
    
    # 5. Compute factors
    scale = 1.0 / (d ** 0.5)
    F1 = torch.matmul(Q, K\_land.transpose(-2, -1)) * scale
    F2 = torch.matmul(Q\_land, K\_land.transpose(-2, -1)) * scale
    F3 = torch.matmul(Q\_land, K.transpose(-2, -1)) * scale
    
    # 6. Pseudoinverse
    F2\_inv = torch.linalg.pinv(F2.float(), rcond=1e-4)
    I = torch.eye(m, device=Q.device, dtype=torch.float32).unsqueeze(0).unsqueeze(0)
    for \_ in range(min(self.pinv\_iters, 4)):
        F2\_inv = torch.matmul(F2\_inv, 2 * I - torch.matmul(F2.float(), F2\_inv))
    
    # 7. Approximate attention
    attn = torch.matmul(F1, torch.matmul(F2\_inv, F3))
    
    # 8. Causal mask
    if self.is\_decoder:
        causal = torch.triu(torch.full((L, L), -1e4), diagonal=1)
        attn = attn + causal.unsqueeze(0).unsqueeze(0)
    
    # 9. Attention mask
    if attention\_mask is not None:
        attn = apply\_attention\_mask(attn, attention\_mask)
    
    # 10. Softmax
    probs = F.softmax(attn, dim=-1).to(hidden\_states.dtype)
    probs = torch.nan\_to\_num(probs)
    probs = self.dropout(probs)
    
    # 11. Context
    context = torch.matmul(probs, V)
    context = context.permute(0, 2, 1, 3).contiguous().view(B, L, H * d)
    
    if self.is\_nested\_encoder:
        return context, None
    
    return self.out\_proj(context), None
\end{lstlisting}

\subsection{B.4 `copy\_weights`}

Handles four architectures: GPT-2 (fused Conv1D), RoBERTa (nested), DistilBERT

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 FP16 Pseudoinverse}

\texttt{torch.linalg.pinv} requires FP32/FP64. Cast to FP32, compute, cast back.

\subsubsection{B.5.2 Schulz Divergence}

Starting from \texttt{X\_0 = 0} often diverges. Start from \texttt{X\_0 = pinv(F2)}.

\subsubsection{B.5.3 RoPE After Pooling}

Apply RoPE \textbf{before} pooling. Otherwise, pooled positions lose their 
absolute position information.

\subsubsection{B.5.4 GQA Repeat Before Pooling}

Repeat KV heads \textbf{before} pooling. Otherwise, pooled landmarks don't match 
query head count.

\subsubsection{B.5.5 Causal Mask After Approximation}

Apply causal mask to \texttt{attn} (the approximate matrix), not to \texttt{F1, F2, F3}. 
This is essential for the linear complexity.
