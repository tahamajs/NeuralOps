\subsection{B.1 FactorisedLinear}

\begin{lstlisting}
class FactorisedLinear(nn.Module):
    def \_\_init\_\_(self, in\_features\_or\_weight, out\_features\_or\_bias=None, ...):
        super().\_\_init\_\_()
        if not hasattr(self, 'output\_scale'):
            self.output\_scale = nn.Parameter(torch.ones(1))
        ...
    
    def decompose\_from\_weight(self, weight, bias=None, energy=None):
        with torch.no\_grad():
            W = weight.float()
            U, S, Vt = torch.linalg.svd(W, full\_matrices=False)
            # Energy-based rank
            target\_energy = energy if energy is not None else self.energy
            cum = torch.cumsum(S**2, dim=0) / ((S**2).sum() + 1e-5)
            calc\_rank = int((cum \textless{} target\_energy).sum().item()) + 1
            r = min(self.rank, calc\_rank, U.size(1), Vt.size(0))
            
            # Store factors
            A = U[:, :r] * S[:r].unsqueeze(0)
            B = Vt[:r, :]
            self.down.weight.data.zero\_()
            self.up.weight.data.zero\_()
            self.down.weight.data[:r, :] = B
            self.up.weight.data[:, :r] = A
            self.freeze\_base\_weights()
    
    def forward(self, x):
        out = self.up(self.down(x))
        if self.lora\_rank \textgreater{} 0:
            out = out + F.linear(F.linear(x, self.lora\_A), self.lora\_B) * self.lora\_scaling
        return out * self.output\_scale
\end{lstlisting}

\subsection{B.2 Softpick Fix}

\begin{lstlisting}
def softpick(x, dim=-1, temperature=1.0):
    """Stable softmax with temperature. Replaces broken ReLU-based softpick."""
    x\_f = x.float()
    if temperature \textless{}= 0 or temperature == 1.0:
        return F.softmax(x\_f, dim=dim).to(x.dtype)
    return F.softmax(x\_f / temperature, dim=dim).to(x.dtype)
\end{lstlisting}

\subsection{B.3 Local Window Attention}

\begin{lstlisting}
def local\_window\_attention(Q, K, V, window, softpick\_temperature=1.0, 
                          token\_valid=None, is\_decoder=False):
    B, H, N, D = Q.shape
    idx\_q = torch.arange(N).unsqueeze(1)
    idx\_k = torch.arange(N).unsqueeze(0)
    dist = (idx\_q - idx\_k).abs()
    mask = (dist \textless{}= window // 2).unsqueeze(0).unsqueeze(0)
    if is\_decoder and N \textgreater{} 1:
        causal = (idx\_k \textless{}= idx\_q).unsqueeze(0).unsqueeze(0)
        mask = mask \& causal
    
    scores = torch.matmul(Q.float(), K.float().transpose(-2, -1)) * (D**-0.5)
    scores = scores.masked\_fill(~mask, -10000.0)
    if token\_valid is not None:
        scores = scores.masked\_fill(~token\_valid.unsqueeze(1).unsqueeze(2), -10000.0)
    attn = softpick(scores, dim=-1, temperature=softpick\_temperature)
    return torch.matmul(attn, V)
\end{lstlisting}

\subsection{B.4 Kernel Linear Attention}

\begin{lstlisting}
def kernel\_linear\_attention(Q, K, V, rff\_proj, use\_zero\_sum=True, 
                            max\_clamp=11.0, is\_causal=False):
    B, H, N, D = Q.shape
    F\_dim = rff\_proj.size(-1)
    
    Q\_f, K\_f, V\_f = Q.float(), K.float(), V.float()
    if use\_zero\_sum:
        K\_f = K\_f - K\_f.mean(dim=2, keepdim=True)
    
    Q\_norm = F.normalize(Q\_f, p=2, dim=-1)
    K\_norm = F.normalize(K\_f, p=2, dim=-1)
    
    Q\_proj = torch.einsum("bhnd,hdf-\textgreater{}bhnf", Q\_norm, rff\_proj.float()).clamp(max=max\_clamp)
    K\_proj = torch.einsum("bhnd,hdf-\textgreater{}bhnf", K\_norm, rff\_proj.float()).clamp(max=max\_clamp)
    Q\_feat = torch.exp(Q\_proj) / math.sqrt(F\_dim)
    K\_feat = torch.exp(K\_proj) / math.sqrt(F\_dim)
    
    if is\_causal:
        kv\_outer = torch.einsum("bhnf,bhnd-\textgreater{}bhnfd", K\_feat, V\_f)
        KV\_cum = torch.cumsum(kv\_outer, dim=2)
        z\_cum = torch.cumsum(K\_feat, dim=2)
        num = torch.einsum("bhnf,bhnfd-\textgreater{}bhnd", Q\_feat, KV\_cum)
        denom = torch.einsum("bhnf,bhnf-\textgreater{}bhn", Q\_feat, z\_cum).unsqueeze(-1).clamp(min=1e-5)
        ctx = num / denom
    else:
        KV = torch.einsum("bhnf,bhnd-\textgreater{}bhfd", K\_feat, V\_f)
        K\_sum = K\_feat.sum(dim=2)
        denom = torch.einsum("bhnf,bhf-\textgreater{}bhn", Q\_feat, K\_sum).unsqueeze(-1)
        ctx = torch.einsum("bhnf,bhfd-\textgreater{}bhnd", Q\_feat, KV) / (denom + 1e-5)
    
    return torch.nan\_to\_num(ctx, nan=0.0).to(Q.dtype)
\end{lstlisting}

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 ReLU-Based Softpick}

The V4 formula is broken. Always use stable softmax.

\subsubsection{B.5.2 Energy Rank = Full Rank}

With \texttt{max\_rank=384} and \texttt{d=768}, SVD gives no compression. Use 
\texttt{max\_rank \textless{} d/2}.

\subsubsection{B.5.3 Decoder Fallback}

Without the fallback, GPT-2 would see future tokens. Always check 
\texttt{is\_decoder}.

\subsubsection{B.5.4 Conv1D Orientation}

GPT-2's Conv1D stores (in, out). Transpose before chunking.

\subsubsection{B.5.5 RFF Numeric Overflow}

RFF features can overflow FP16. Compute in FP32 and clamp.
