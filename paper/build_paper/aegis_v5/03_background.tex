\section{Background}

\subsection{Standard Attention}

For input $X \in \mathbb{R}^{B \times L \times d}$:

\[
Q = X W\_Q, \quad K = X W\_K, \quad V = X W\_V
\]

\[
\text{Attn}(Q, K, V) = \text{softmax}\!\left(\frac{Q K^\top}{\sqrt{d\_h}}\right) V
\]

Complexity: $\mathcal{O}(L^2 d_h)$ for scores.

\subsection{Sliding Window Attention}

Local window attention restricts each query to a window of size $W$ around 
it:

\[
\text{Attn}\_\text{local}(q\_i, K, V) = \sum\_{j : |i - j| \leq W/2} \alpha\_{ij} v\_j
\]

Complexity: $\mathcal{O}(L \cdot W \cdot d_h)$.

For decoders, the causal variant restricts to $j \leq i$:

\[
\text{Attn}\_\text{local, causal}(q\_i, K, V) = \sum\_{j : i - W/2 \leq j \leq i} \alpha\_{ij} v\_j
\]

\subsection{Kernel Linear Attention}

Kernel attention factorizes the softmax kernel:

\[
e^{q \cdot k} \approx \phi(q)^\top \phi(k)
\]

The output becomes:

\[
\text{Out}\_\text{kernel}(q\_i, K, V) = \frac{\phi(q\_i)^\top \sum\_j \phi(k\_j) v\_j^\top}{\phi(q\_i)^\top \sum\_j \phi(k\_j)}
\]

Complexity: $\mathcal{O}(L \cdot F \cdot d_h)$ where $F$ is the feature 
dimension.

\subsubsection{Random Fourier Features}

For RFF:

\[
\phi(x) = \frac{1}{\sqrt{F}} \exp\!\left(W\_\text{rff} x\right)
\]

where $W_\text{rff} \in \mathbb{R}^{d_h \times F}$ is a random projection.

\subsubsection{Zero-Sum Keys}

Zero-sum keys (Katharopoulos et al., 2020) subtract the mean:

\[
\tilde{K} = K - \bar{K}
\]

This improves the approximation for positive-definite kernels.

\subsubsection{Causal Kernel Attention}

For decoders, the cumulative sum formulation:

\[
S\_i = \sum\_{j \leq i} \phi(k\_j) v\_j^\top, \quad
Z\_i = \sum\_{j \leq i} \phi(k\_j)
\]

\[
\text{Out}\_i = \frac{\phi(q\_i)^\top S\_i}{\phi(q\_i)^\top Z\_i}
\]

\subsection{Singular Value Decomposition}

For $W \in \mathbb{R}^{m \times n}$:

\[
W = U \Sigma V^\top
\]

Truncated at rank $r$:

\[
W\_r = U\_{:, :r} \Sigma\_{:r, :r} V\_{:r, :}^\top
\]

\subsection{Energy-Based Rank Selection}

The \textbf{energy fraction} captured by rank $r$:

\[
\text{Energy}(r) = \frac{\sum\_{i \leq r} \sigma\_i^2}{\sum\_i \sigma\_i^2}
\]

We select $r^* = \min \{ r : \text{Energy}(r) \geq \eta \}$.

For $\eta = 0.95$: $r^*$ captures 95\% of the weight energy.

\subsection{LoRA}

LoRA parameterizes weight updates:

\[
\Delta W = \frac{\alpha}{r\_L} B A, \quad B \in \mathbb{R}^{d \times r\_L}, \quad A \in \mathbb{R}^{r\_L \times k}
\]

with $B = 0$ at initialization.

\subsection{Grouped-Query Attention (GQA)}

Modern models (Qwen, LLaMA) use $H_{kv} < H$:

\[
K, V \in \mathbb{R}^{B \times L \times H\_{kv} \times d\_h}
\]

Expanded to $H$ via \texttt{repeat\_kv}.

\subsection{Rotary Position Embeddings (RoPE)}

RoPE rotates $Q, K$:

\[
\tilde{q}\_i = q\_i \cos(\theta\_i) + \text{rot}(q\_i) \sin(\theta\_i)
\]

\subsection{Softpick (Fixed)}

The original AEGIS used a ReLU-based softpick:

\[
\text{softpick}(x)\_i = \text{ReLU}\!\left(e^{(x\_i - m)/T} - e^{-m/T}\right)
\]

where $m = \max_j x_j$. \textbf{Problem:} When $m \leq 0$ (all-masked rows), 
this returns all zeros, causing attention collapse.

AEGIS-V5 replaces this with a stable softmax:

\[
\text{softpick}(x)\_i = \frac{e^{x\_i / T}}{\sum\_j e^{x\_j / T}}
\]

which is always numerically stable.
