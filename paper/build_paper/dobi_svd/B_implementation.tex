class DobiSVDLinearLayer(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, rank, bias=True,

        super().\_\_init\_\_()

        self.out\_features = out\_features

        self.U = nn.Parameter(torch.empty(out\_features, self.rank, ...))
        self.gate = nn.Parameter(torch.full((self.rank,), init\_gate\_val, ...))
        
        if bias:
            self.bias = nn.Parameter(torch.empty(out\_features, ...))
        
        self.reset\_parameters()
    
    def forward(self, x):
        if x.shape[-1] != self.in\_features:
            # Pad or truncate
            ...
        g = torch.sigmoid(self.gate).unsqueeze(1)   # (rank, 1)
        V\_gated = self.V * g
        latent = F.linear(x, V\_gated)
        out = F.linear(latent, self.U, self.bias)
        return out

The \texttt{.weight} property computes:

@property
def weight(self):
    g = torch.sigmoid(self.gate).unsqueeze(1)
    return torch.matmul(self.U, self.V * g)

class DobiSVDCompressor(BaseCompressor):
    def compress(self):
        for name, module in target\_layers:
            new\_layer, err, r = self.compress\_linear(module)
            
            if new\_layer is None or err \textgreater{} self.max\_error:
                if self.fallback\_to\_original:
                    self.fallback\_layers.append(name)
                    for p in module.parameters():
                        p.requires\_grad = False
                    continue
            
            # Replace
            if hasattr(module, "base\_layer"):
                module.base\_layer = new\_layer
            elif hasattr(module, "linear") and hasattr(module, "lora\_A"):
                module.linear = new\_layer
            else:
                replace\_module(self.model, name, new\_layer)

\subsection{B.6 Post-Hoc Pruning Script}

\begin{lstlisting}
def prune\_gates(model, tau=0.5):
    for name, module in model.named\_modules():
        if isinstance(module, DobiSVDLinearLayer):
            g = torch.sigmoid(module.gate)
            keep = g \textgreater{} tau
            r\_eff = keep.sum().item()
            
            if r\_eff \textless{} module.rank:
                # Physically prune
                module.V.data = module.V.data[keep, :]
                module.U.data = module.U.data[:, keep]
                module.gate.data = module.gate.data[keep]
                module.rank = r\_eff
\end{lstlisting}

\subsection{B.7 Common Pitfalls}

\subsubsection{B.7.1 Forgetting Gate Compensation}

Without dividing $V$ by $\sigma(\alpha_0)$:
\begin{itemize}
\item Initial forward is \textbf{scaled} by $\sigma(\alpha_0)$
\item Model starts with reduced magnitude
\item Slower convergence

\end{itemize}
\subsubsection{B.7.2 Wrong Init Gate Value}

Too small ($\alpha_0 < 1$): initial forward is significantly scaled down.
Too large ($\alpha_0 > 5$): gates saturate, gradients vanish.

Default $\alpha_0 = 2.5$ is a good balance.

\subsection{B.9 Fallback Behavior}

When \texttt{fallback\_to\_original = True} and error exceeds threshold:

\begin{lstlisting}
self.fallback\_layers.append(name)
for p in module.parameters():
    p.requires\_grad = False   # Freeze the original layer
\end{lstlisting}

This ensures recovery training does not touch fallback layers.
