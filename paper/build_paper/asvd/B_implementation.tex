\subsection{B.1 The ASVDLinear Module}

\begin{lstlisting}
class ASVDLinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, rank, bias=True,
                 lora\_rank=0, freeze\_base=False):
        super().\_\_init\_\_()
        self.in\_features = in\_features
        self.out\_features = out\_features
        self.rank = rank
        
        self.U = nn.Parameter(torch.empty(out\_features, rank))
        self.S = nn.Parameter(torch.empty(rank))
        self.V = nn.Parameter(torch.empty(rank, in\_features))
        
        if bias:
            self.bias = nn.Parameter(torch.empty(out\_features))
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.empty(lora\_rank, in\_features))
            self.lora\_B = nn.Parameter(torch.zeros(out\_features, lora\_rank))
        
        self.reset\_parameters()
        if freeze\_base:
            self.freeze\_base\_weights()
    
    def forward(self, x):
        Vx = F.linear(x, self.V)          # x @ V^T
        SVx = Vx * self.S                  # scale by $\Sigma$
        out = F.linear(SVx, self.U)        # @ U^T
        if self.bias is not None:
            out = out + self.bias
        if self.lora\_rank \textgreater{} 0:
            out = out + F.linear(F.linear(x, self.lora\_A), self.lora\_B)
        return out * self.output\_scale
\end{lstlisting}

\subsection{B.2 Energy-Based Rank Selection}

\begin{lstlisting}
def \_decompose\_and\_set(self, target, src\_weight, src\_bias, proj\_name):
    W = src\_weight.detach().float()
    U, S, Vt = torch.linalg.svd(W, full\_matrices=False)
    
    # Cumulative energy
    total\_energy = torch.sum(S ** 2)
    cum\_energy = torch.cumsum(S ** 2, dim=0) / (total\_energy + 1e-12)
    
    # Select rank
    target\_energy = max(self.energy\_threshold, 
                       1.0 - self.error\_threshold ** 2)
    indices = (cum\_energy \textgreater{}= target\_energy).nonzero()
    r = indices[0].item() + 1 if len(indices) \textgreater{} 0 else min(U.size(1), Vt.size(0))
    
    if self.dynamic\_rank:
        r = max(16, min(r, U.size(1), Vt.size(0)))
    else:
        r = min(self.rank, U.size(1), Vt.size(0))
    
    # Validate
    w\_rec = (U[:, :r] * S[:r].unsqueeze(0)) @ Vt[:r, :]
    rel\_err = float((torch.norm(W - w\_rec) / (torch.norm(W) + 1e-9)).item())
    
    if rel\_err \textgreater{} self.error\_threshold:
        raise RuntimeError(...)
    
    target.set\_svd\_factors(U[:, :r], Vt[:r, :], S[:r], bias=src\_bias)
    return rel\_err
\end{lstlisting}

\subsection{B.3 The `set\_identity` Method}

\begin{lstlisting}
def set\_identity(self):
    with torch.no\_grad():
        self.U.data.zero\_()
        self.V.data.zero\_()
        self.S.data.fill\_(1.0)
        r = min(self.rank, self.in\_features, self.out\_features)
        self.U.data[:r, :r].fill\_diagonal\_(1.0)
        self.V.data[:r, :r].fill\_diagonal\_(1.0)
        if self.bias is not None:
            self.bias.data.zero\_()
        self.freeze\_base\_weights()
\end{lstlisting}

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 Forgetting `set\_identity` for Nested Encoders}

Without \texttt{set\_identity}, RoBERTa double-projects the output.

\subsubsection{B.5.2 Wrong Conv1D Transpose}

GPT-2's Conv1D stores weights as \texttt{(in, out)}. Must transpose.

\subsubsection{B.5.3 Missing `freeze\_base\_weights`}

Without freezing, recovery training modifies the base factors and undoes 
compression.

\subsubsection{B.5.4 Not Handling `output\_scale`}

The \texttt{output\_scale} parameter is initialized to 1.0 but unused. Removing 
it would simplify the module.
