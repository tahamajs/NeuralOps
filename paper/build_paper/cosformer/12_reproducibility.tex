\subsection{Diagnostics}

\subsubsection{Verify Cos/Sin Reweighting}

\begin{lstlisting}
import math
N = 128
max\_L = 512
theta = [math.pi * i / (2 * max\_L) for i in range(N)]
assert theta[0] == 0.0
assert theta[N-1] \textless{} math.pi / 2
\end{lstlisting}

\subsubsection{Verify Causality}

\begin{lstlisting}
x = torch.randn(1, 128, 768, requires\_grad=True)
y = model(x)
y[0, 50].sum().backward()
assert x.grad[0, 51:].abs().max() \textless{} 1e-6
\end{lstlisting}

\subsection{Known Issues}

\begin{itemize}
\item \textbf{RoBERTa instability}: Default hyperparameters cause drop.
\item \textbf{Slow decoder}: chunked scan adds overhead.
\item \textbf{Fragile RoPE extraction}: Triple-nested try/except.
\item \textbf{Dead \texttt{tau} parameter}.

\end{itemize}
