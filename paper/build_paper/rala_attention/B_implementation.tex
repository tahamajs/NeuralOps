\subsection{B.2 Forward Pass (Causal)}

\begin{lstlisting}
if is\_causal:
    S = torch.zeros(B, H, d, d, device=..., dtype=q.dtype)
    z = torch.zeros(B, H, d, device=..., dtype=q.dtype)
    num\_list, denom\_list = [], []
    
    for i in range(seq\_len):
        q\_i = q\_prime[:, :, i, :]  # (B, H, d)
        k\_i = k\_prime[:, :, i, :]
        v\_i = v\_masked[:, :, i, :]
        
        S = S + torch.einsum("bhd,bhe-\textgreater{}bhde", k\_i, v\_i)
        z = z + k\_i
        
        num\_i = torch.einsum("bhd,bhde-\textgreater{}bhe", q\_i, S)
        denom\_i = torch.einsum("bhd,bhd-\textgreater{}bh", q\_i, z).unsqueeze(-1)
        num\_list.append(num\_i)
        denom\_list.append(denom\_i)
    
    num = torch.stack(num\_list, dim=2)
    denom = torch.stack(denom\_list, dim=2).clamp(min=1e-4)
    lin\_out = torch.nan\_to\_num(num / denom)
    
    # Augmentation
    rank\_mat = torch.einsum("hdr,her-\textgreater{}hde", self.U, self.V)  # (H, d, d)
    aug\_out = torch.nan\_to\_num(
        torch.einsum("bhnd,hde-\textgreater{}bhne", q\_prime, rank\_mat) / denom
    )
\end{lstlisting}

\subsection{B.6 Common Pitfalls}

\subsubsection{B.6.1 Frozen U, V}

When LoRA is active, \texttt{freeze\_base\_weights} sets \texttt{U.requires\_grad = False}. 
This disables learning of the augmentation. To unfreeze:

\begin{lstlisting}
self.U.requires\_grad = True
self.V.requires\_grad = True
\end{lstlisting}

\subsubsection{B.6.2 Missing Mask}

Without \texttt{mask\_4d}, padding tokens contribute to prefix sums:

\begin{lstlisting}
k\_prime = F.softplus(k) * mask\_4d  # zero out padding
v\_masked = v * mask\_4d
\end{lstlisting}

\subsubsection{B.6.3 Denominator Clamp}

The denominator $\phi(q)^\top z$ can be near zero. Clamp:

\begin{lstlisting}
denom = denom.clamp(min=1e-4)
\end{lstlisting}
