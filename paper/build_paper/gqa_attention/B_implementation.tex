\subsection{B.3 Group Averaging}

\begin{lstlisting}
def group\_average(w\_mha, H\_src, G\_tgt, d\_h):
    """Average KV heads into G groups."""
    r = H\_src // G\_tgt
    w\_grouped = w\_mha.view(d\_h * G\_tgt, r, ...).mean(dim=1)
    return w\_grouped
\end{lstlisting}

\subsection{B.4 `repeat\_kv`}

\begin{lstlisting}
def repeat\_kv(x, n\_rep):
    if n\_rep == 1:
        return x
    B, G, L, d\_h = x.shape
    x = x[:, :, None, :, :].expand(B, G, n\_rep, L, d\_h)
    return x.reshape(B, G * n\_rep, L, d\_h)
\end{lstlisting}

\subsection{B.5 Forward Pass}

\begin{lstlisting}
def forward(self, hidden\_states, attention\_mask=None, ...):
    B, N\_q, D = hidden\_states.shape
    
    Q = self.query(hidden\_states).view(B, N\_q, H, d).transpose(1, 2)
    K = self.key(hidden\_states).view(B, N\_q, G, d).transpose(1, 2)
    V = self.value(hidden\_states).view(B, N\_q, G, d).transpose(1, 2)
    
    if RoPE:
        Q, K = apply\_rotary\_pos\_emb(Q, K, ...)
    
    if self.kv\_repeat \textgreater{} 1:
        K = repeat\_kv(K, self.kv\_repeat)
        V = repeat\_kv(V, self.kv\_repeat)
    
    scores = Q @ K.transpose(-2, -1) / sqrt(d)
    
    if causal:
        scores = scores.masked\_fill(~causal\_tril, -10000.0)
    
    attn = softmax(scores)
    context = attn @ V
    context = context.transpose(1, 2).view(B, N\_q, H * d)
    
    if self.is\_nested\_encoder:
        return context, None
    
    output = self.out\_proj(context)
    return output, None
\end{lstlisting}

\subsection{B.6 Common Pitfalls}

\subsubsection{B.6.1 Wrong Transpose for Conv1D}

\begin{lstlisting}
# WRONG
w = src.c\_attn.weight  # (in, out) -- no transpose

# RIGHT
w = src.c\_attn.weight.t()  # (out, in)
\end{lstlisting}

\subsubsection{B.6.2 Missing Divide Check}

$G$ must divide $H$. If not, snap to largest divisor.

\subsubsection{B.6.3 Causal Mask Before Repetition}

Same result, but apply after for clarity.

\subsubsection{B.6.4 RoPE After Repetition}

RoPE must be applied \textbf{before} repetition (to shared KV), not after.
