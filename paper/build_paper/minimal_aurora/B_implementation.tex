\section{Appendix B. Implementation Details}

\subsection{B.1 The FactorisedLinear Module}

\begin{lstlisting}
class FactorisedLinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, rank=64, bias=True,
                 energy=0.95, lora\_rank=16, lora\_alpha=32.0):
        super().\_\_init\_\_()
        self.rank = min(rank, in\_features, out\_features)
        self.down = nn.Linear(in\_features, self.rank, bias=False)
        self.up = nn.Linear(self.rank, out\_features, bias=bias)
        
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.randn(lora\_rank, in\_features) * 0.01)
            self.lora\_B = nn.Parameter(torch.zeros(out\_features, lora\_rank))
        
        self.reset\_parameters()
    
    def decompose\_from\_weight(self, W, bias=None, energy=None):
        U, S, Vt = torch.linalg.svd(W.float())
        target\_energy = energy if energy is not None else self.energy
        cum\_energy = torch.cumsum(S**2, dim=0) / (S**2).sum()
        calc\_rank = int((cum\_energy \textless{} target\_energy).sum().item()) + 1
        r = min(self.rank, calc\_rank, U.size(1), Vt.size(0))
        r = max(1, r)
        
        sqrt\_s = torch.sqrt(S[:r])
        A = U[:, :r] * sqrt\_s
        B = sqrt\_s[:, None] * Vt[:r, :]
        
        self.up.weight.data[:, :r] = A
        self.down.weight.data[:r, :] = B
        
        if bias is not None:
            self.up.bias.data[:len(bias)] = bias
        
        self.freeze\_base\_weights()
\end{lstlisting}

def forward(self, x):
    out = self.up(self.down(x))

    return out

\subsection{B.6 Common Pitfalls}

\subsubsection{B.6.1 Forgetting to Freeze Base}

After \texttt{decompose\_from\_weight}, \texttt{freeze\_base\_weights} must be called.

\subsubsection{B.6.2 Nested Encoder Mis-Detection}

DistilBERT's \texttt{MultiHeadSelfAttention} ends with "SelfAttention" but has 
\texttt{out\_lin} $\to$ NOT nested. Detection must check \texttt{has\_out\_proj}.

\subsubsection{B.6.3 `output\_scale` Drift}

Initialize to 1.0; monitor during training.

\subsubsection{B.6.4 Silent Injection}

\texttt{inject\_minimal\_aurora} returns \texttt{None} on failure.
