
\section{Appendix B. Implementation Details}

\subsection{B.1 State Update Inner Loop}

\begin{lstlisting}
def \_delta\_chunk\_forward(S, q\_i, k\_i, v\_i, beta\_i, alpha\_i):
    """Inner loop over chunk\_size tokens."""
    C = q\_i.shape[2]
    chunk\_outs = []
    curr\_S = S
    for t in range(C):
        kt = k\_i[:, :, t:t+1, :]
        vt = v\_i[:, :, t:t+1, :]
        qt = q\_i[:, :, t:t+1, :]
        bt = beta\_i[:, :, t:t+1].unsqueeze(-1)
        at = alpha\_i[:, :, t:t+1].unsqueeze(-1)
        
        # Prediction error
        inter\_v = torch.matmul(kt, curr\_S)
        err = vt - inter\_v
        
        # Normalized step
        norm\_k\_sq = (kt * kt).sum(dim=-1, keepdim=True)
        denom = (1.0 + norm\_k\_sq).clamp(min=1.0)
        step = (at / denom).clamp(min=0.0, max=1.0)
        
        # State update
        delta\_update = step * torch.matmul(kt.transpose(-1, -2), err)
        curr\_S = bt * curr\_S + delta\_update
        curr\_S = torch.nan\_to\_num(curr\_S, nan=0.0, posinf=1e4, neginf=-1e4)
        
        # Output
        yt = torch.matmul(qt, curr\_S)
        chunk\_outs.append(yt)
    
    return torch.cat(chunk\_outs, dim=2), curr\_S
\end{lstlisting}

\subsection{B.2 Outer Chunk Loop}

\begin{lstlisting}
def chunked\_delta\_rule\_scan(q, k, v, beta, alpha, chunk\_size=8, use\_checkpoint=True):
    B, H, L, d = q.shape
    
    # FP32 for numerical stability
    q\_f = q.float(); k\_f = k.float(); v\_f = v.float()
    beta\_f = beta.float().clamp(1e-4, 1-1e-4)
    alpha\_f = alpha.float().clamp(1e-4, 1)
    
    # Normalize
    k\_norm = torch.norm(k\_f, p=2, dim=-1, keepdim=True).clamp(min=1e-6)
    k\_unit = k\_f / k\_norm
    q\_norm = torch.norm(q\_f, p=2, dim=-1, keepdim=True).clamp(min=1e-6)
    q\_unit = q\_f / q\_norm
    
    # Pad
    pad\_len = (chunk\_size - L \% chunk\_size) \% chunk\_size
    if pad\_len \textgreater{} 0:
        q\_unit = F.pad(q\_unit, (0, 0, 0, pad\_len))
        k\_unit = F.pad(k\_unit, (0, 0, 0, pad\_len))
        v\_f = F.pad(v\_f, (0, 0, 0, pad\_len))
        beta\_f = F.pad(beta\_f, (0, pad\_len), value=0.0)
        alpha\_f = F.pad(alpha\_f, (0, pad\_len), value=1.0)
    
    # Reshape
    L\_pad = L + pad\_len
    num\_chunks = L\_pad // chunk\_size
    q\_c = q\_unit.view(B, H, num\_chunks, chunk\_size, d)
    # ... same for k, v, beta, alpha
    
    # Recurrence
    S = torch.zeros((B, H, d, d), device=device, dtype=torch.float32)
    outputs = []
    
    from torch.utils.checkpoint import checkpoint
    for c in range(num\_chunks):                    # $\leftarrow$ outer Python loop
        q\_i, k\_i, v\_i = q\_c[:,:,c], k\_c[:,:,c], v\_c[:,:,c]
        beta\_i, alpha\_i = beta\_c[:,:,c], alpha\_c[:,:,c]
        
        if use\_checkpoint and q.requires\_grad:
            y\_i, S = checkpoint(\_delta\_chunk\_forward, S, q\_i, k\_i, v\_i, 
                                 beta\_i, alpha\_i, use\_reentrant=False)
        else:
            y\_i, S = \_delta\_chunk\_forward(S, q\_i, k\_i, v\_i, beta\_i, alpha\_i)
        outputs.append(y\_i)
    
    y = torch.cat(outputs, dim=2)
    if pad\_len \textgreater{} 0:
        y = y[:, :, :L, :]
    return torch.nan\_to\_num(y, nan=0.0)
\end{lstlisting}

\subsection{B.3 The Z Projection}

The Z projection is used for SiLU gating:

\begin{lstlisting}
z = self.z\_proj(hidden\_states)
# ... after recurrence:
gated\_y = y * F.silu(z)
\end{lstlisting}

It is \textbf{not present} in baseline attention. We initialize it from $W_Q$:

\begin{lstlisting}
self.z\_proj.weight.data.copy\_(w\_q)
self.z\_proj.bias.data.copy\_(b\_q)
\end{lstlisting}

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 Forgetting `.float()`}

The recurrence \textbf{must} be done in FP32. FP16 accumulates error:

\begin{lstlisting}
q\_f = q.float(); k\_f = k.float(); v\_f = v.float()
\end{lstlisting}

\subsubsection{B.5.2 Missing Normalization}

Without normalization, the state can explode or vanish:

\begin{lstlisting}
k\_norm = torch.norm(k\_f, p=2, dim=-1, keepdim=True).clamp(min=1e-6)
k\_unit = k\_f / k\_norm
\end{lstlisting}

\subsubsection{B.5.3 Padding Handling}

Padding with \texttt{pad\_len \textgreater{} 0} requires:

\begin{lstlisting}
if pad\_len \textgreater{} 0:
    q\_unit = F.pad(q\_unit, (0, 0, 0, pad\_len))
    beta\_f = F.pad(beta\_f, (0, pad\_len), value=0.0)
\end{lstlisting}

\subsubsection{B.5.4 Gradient Checkpointing Requires `.requires\_grad`}

\begin{lstlisting}
if use\_checkpoint and q.requires\_grad:
    ...checkpoint...
else:
    ...normal forward...
\end{lstlisting}

Without \texttt{requires\_grad}, checkpointing fails or is a no-op.

\subsection{B.6 Diagnostics}

\subsubsection{Verify State Update}

\begin{lstlisting}
# Single-token
k = torch.randn(1, 1, 1, 64)
v = torch.randn(1, 1, 1, 64)
S\_prev = torch.zeros(1, 1, 64, 64)
beta, alpha = torch.tensor([[[0.9]]]), torch.tensor([[[0.5]]])

S\_new, \_ = \_delta\_chunk\_forward(S\_prev, k, k, v, beta, alpha)

# Manual check
err = v - torch.matmul(k, S\_prev)
norm\_k\_sq = (k * k).sum(-1, keepdim=True)
step = alpha / (1 + norm\_k\_sq).clamp(min=1.0)
expected = beta * S\_prev + step * torch.matmul(k.transpose(-1, -2), err)

assert torch.allclose(S\_new, expected, atol=1e-5)
\end{lstlisting}

@ATTENTION\_REGISTRY.register("gated\_delta\_net\_inject")
def inject\_gated\_delta\_net(layer, model\_config, comp\_config, old\_attn=None, **kwargs):
    # Find source attention
    src\_attn = old\_attn or extract\_attention(layer)
    
    # Create new
    try:
        new\_block = GatedDeltaNet(
            config=model\_config, comp\_config=comp\_config,
            d\_model=hidden\_size, num\_heads=num\_heads,
            old\_attn=src\_attn, **kwargs,
        )
    except Exception as e:
        logger.error(f"Injection failed: {e}")
        return None
    
    # Replace
    replace\_attention(layer, new\_block)
    return new\_block
