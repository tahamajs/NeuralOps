
\section{Reproducibility Statement}

\subsection{Environment}

\begin{lstlisting}
git clone https://github.com/Efficiency Lab/SVD\_Project
cd SVD\_Project
python -m venv .venv
.venv\Scripts\activate
pip install -r requirements.txt
\end{lstlisting}

\subsection{Running}

\begin{lstlisting}
python run\_all\_4\_nsa.py \
    --models distilbert,roberta,gpt2 \
    --window\_size 128 \
    --top\_k 2 \
    --recovery\_steps 50
\end{lstlisting}

\subsection{Reproducing Tables}

\subsubsection{Main (Table 7.1)}
\begin{lstlisting}
python run\_all\_4\_nsa.py --window\_size 32 --top\_k 16
\end{lstlisting}

\subsubsection{Window Ablation (Table 7.4)}
\begin{lstlisting}
for w in 8 16 32 64; do
    python run\_all\_4\_nsa.py --window\_size $w --output_suffix w$w
done
\end{lstlisting}

\subsubsection{Top-K Ablation (Table 7.5)}
\begin{lstlisting}
for k in 4 8 16 32; do
    python run\_all\_4\_nsa.py --top\_k $k --output_suffix k$k
done
\end{lstlisting}
