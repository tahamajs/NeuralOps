\subsection{Reproduction}

\begin{lstlisting}
python run_all_4_nystrom.py \
    --models distilbert,roberta,gpt2 \
    --num_landmarks 32 \
    --recovery_steps 50 \
    --lora_rank 8
\end{lstlisting}

Runtime: ~PyTorch 2.0 minutes.

\subsection{Long Sequence Evaluation}

\textbf{Note.} At $N = 128$, all models use dense fallback. To activate Nystr m, 
sequences must be $N > 128$. We test this in a separate experiment with 
$N = 512$:

\begin{lstlisting}
python run_all_4_nystrom.py \
    --models distilbert \
    --seq_len 512 \
    --num_landmarks 32
\end{lstlisting}
