\section{Appendix B. Implementation Details}

\subsection{B.1 The `REAPMoE` Module}

\begin{lstlisting}
class REAPMoE(nn.Module):
    def \_\_init\_\_(self, d\_model, n\_experts, top\_k, noise\_std, use\_svd):
        self.router\_ln = nn.LayerNorm(d\_model)
        self.router = nn.Linear(d\_model, n\_experts, bias=False)
        nn.init.uniform\_(self.router.weight, -0.01, 0.01)
        
        expert\_layer = lambda: nn.Sequential(
            SVDLinear(d\_model, 2*d\_model, rank=d\_model//4),
            nn.GELU(),
            SVDLinear(2*d\_model, d\_model, rank=d\_model//4),
        )
        self.experts = nn.ModuleList([expert\_layer() for \_ in range(n\_experts)])
        
        # Zero-init final layer
        for exp in self.experts:
            final = exp[-1]
            if isinstance(final, SVDLinear):
                nn.init.zeros\_(final.U)
                nn.init.zeros\_(final.S)
                nn.init.zeros\_(final.bias)
\end{lstlisting}

\subsection{B.2 The `SVDLinear` Module}

\begin{lstlisting}
class SVDLinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, rank):
        super().\_\_init\_\_()
        self.U = nn.Parameter(torch.randn(out\_features, rank) * 0.02)
        self.S = nn.Parameter(torch.ones(rank))
        self.V = nn.Parameter(torch.randn(in\_features, rank) * 0.02)
        self.bias = nn.Parameter(torch.zeros(out\_features))
    
    @property
    def weight(self):
        return (self.U * self.S.unsqueeze(0)) @ self.V.t()
    
    def forward(self, x):
        return (x @ self.V) * self.S @ self.U.t() + self.bias
\end{lstlisting}

\subsection{B.3 Forward Pass}

\begin{lstlisting}
def forward(self, x):
    B, L, D = x.shape
    
    # Router
    logits = self.router(self.router\_ln(x))
    if self.training and self.noise\_std \textgreater{} 0:
        logits = logits + torch.randn\_like(logits) * self.noise\_std
    
    # Top-k
    topk\_w, topk\_i = torch.topk(logits, self.top\_k, dim=-1)
    topk\_w = F.softmax(topk\_w, dim=-1)
    
    # Aux loss
    router\_probs = F.softmax(logits, dim=-1).view(-1, self.n\_experts)
    expert\_mask = F.one\_hot(topk\_i, num\_classes=self.n\_experts).float()
    density = expert\_mask.sum(dim=2).view(-1, self.n\_experts).mean(dim=0)
    prob\_mean = router\_probs.mean(dim=0)
    self.aux\_loss = self.n\_experts * torch.sum(density * prob\_mean)
    
    # Expert computation
    experts\_out = torch.stack([e(x) for e in self.experts], dim=2)  # [B, L, E, D]
    topk\_expanded = topk\_i.unsqueeze(-1).expand(-1, -1, -1, D)
    gathered = torch.gather(experts\_out, 2, topk\_expanded)
    out = torch.sum(gathered * topk\_w.unsqueeze(-1), dim=2)
    
    return out * self.output\_scale, None
\end{lstlisting}

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 Not Zero-Initializing}

Without zero-init, the MoE adds random noise at step 0, disrupting training.

\subsubsection{B.5.2 Wrong Aux Loss Scaling}

Aux loss should be scaled by \texttt{lambda\_aux} (typically 0.01).

\subsubsection{B.5.3 Training-Eval Mismatch}

Noise should be added only during training:

\begin{lstlisting}
if self.training and self.noise\_std \textgreater{} 0:
    logits = logits + torch.randn\_like(logits) * self.noise\_std
\end{lstlisting}

\subsubsection{B.5.4 Expert Output Stacking}

Use \texttt{torch.stack} for [B, L, E, D] layout:

\begin{lstlisting}
experts\_out = torch.stack([e(x) for e in self.experts], dim=2)
\end{lstlisting}
