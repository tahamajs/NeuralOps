\subsection{B.1 The `FactorisedLinear\_CUR` Module}

\begin{lstlisting}
class FactorisedLinear\_CUR(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, cur\_rank=16, ...):
        super().\_\_init\_\_()
        self.down = nn.Linear(in\_features, cur\_rank, bias=False)
        self.up = nn.Linear(cur\_rank, out\_features, bias=True)
        # LoRA (optional)
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.empty(lora\_rank, in\_features))
            self.lora\_B = nn.Parameter(torch.zeros(out\_features, lora\_rank))
        self.output\_scale = nn.Parameter(torch.ones(1))
    
    def forward(self, x):
        out = self.up(self.down(x))
        if self.lora\_rank \textgreater{} 0:
            out = out + F.linear(F.linear(x, self.lora\_A), self.lora\_B) * self.lora\_scaling
        return out * self.output\_scale
\end{lstlisting}

\subsection{B.2 The `copy\_cur` Method}

\begin{lstlisting}
def copy\_cur(self, weight, bias):
    with torch.no\_grad():
        W = weight.float()
        # SVD
        U, S, Vt = torch.linalg.svd(W, full\_matrices=False)
        # Leverage
        col\_lev = (Vt ** 2).sum(dim=0)
        row\_lev = (U ** 2).sum(dim=1)
        # Top-r selection
        rank = min(self.cur\_rank, W.shape[0], W.shape[1])
        col\_idx = torch.topk(col\_lev, k=rank).indices
        row\_idx = torch.topk(row\_lev, k=rank).indices
        # CUR factors
        C = W[:, col\_idx]
        R = W[row\_idx, :]
        W\_int = W[row\_idx][:, col\_idx]
        U\_int\_pinv = torch.linalg.pinv(W\_int)
        # Assign
        self.down.weight.data.zero\_()
        self.down.weight.data[:rank].copy\_(R)
        self.up.weight.data.zero\_()
        self.up.weight.data[:, :rank].copy\_(C @ U\_int\_pinv)
        if bias is not None:
            self.up.bias.data.copy\_(bias)
    if self.freeze\_base:
        self.freeze\_base\_weights()
\end{lstlisting}

\subsection{B.3 LSH Attention}

\begin{lstlisting}
def lsh\_attention(Q, K, V, rounds, bucket\_size, softpick\_temp, ...):
    B, H, N, D = Q.shape
    scale = D ** -0.5
    n\_bits = max(1, min(bucket\_size, 8))
    bit\_shifts = torch.tensor([2**i for i in range(n\_bits)], device=Q.device)
    
    Q\_norm = F.normalize(Q, p=2, dim=-1, eps=1e-5)
    K\_norm = F.normalize(K, p=2, dim=-1, eps=1e-5)
    
    scores = torch.matmul(Q.float(), K.float().transpose(-2, -1)) * scale
    # mask padding
    if token\_valid is not None:
        scores = scores.masked\_fill(~token\_valid.unsqueeze(1).unsqueeze(2), -1e4)
    # causal
    if is\_causal:
        causal = torch.triu(torch.ones(N, N, device=Q.device, dtype=torch.bool), diagonal=1)
        scores = scores.masked\_fill(causal.unsqueeze(0).unsqueeze(0), -1e4)
    
    accum = torch.zeros\_like(V)
    for \_ in range(rounds):
        proj = F.normalize(torch.randn(D, n\_bits, device=Q.device, dtype=Q.dtype), p=2, dim=0)
        hq = (torch.matmul(Q\_norm, proj) \textgreater{} 0).float() @ bit\_shifts
        hk = (torch.matmul(K\_norm, proj) \textgreater{} 0).float() @ bit\_shifts
        match = (hq.unsqueeze(-1) == hk.unsqueeze(-2))
        eye = torch.eye(N, device=Q.device, dtype=torch.bool)
        match = match | eye
        if is\_causal:
            match = (match & ~causal.unsqueeze(0).unsqueeze(0)) | eye
        scores\_r = scores.masked\_fill(~match, -1e4)
        attn = softpick(scores\_r, dim=-1, temperature=softpick\_temp).to(Q.dtype)
        if dropout is not None:
            attn = dropout(attn)
        accum += torch.matmul(attn, V)
    
    return accum / rounds
\end{lstlisting}

\subsection{B.4 The `softpick` Function}

\begin{lstlisting}
def softpick(x, dim=-1, temperature=1.0):
    orig = x.dtype
    x = x.float()
    m = x.max(dim=dim, keepdim=True).values
    temp = max(temperature, 1e-4)
    e = torch.exp((x - m) / temp)
    m\_scaled = torch.clamp(-m / temp, max=80.0)
    denom = e - torch.exp(m\_scaled)
    p = F.relu(denom)
    p\_sum = p.sum(dim=dim, keepdim=True)
    fallback = F.softmax(x / temp, dim=dim)
    res = torch.where(p\_sum \textgreater{} 1e-5, p / torch.clamp(p\_sum, min=1e-5), fallback)
    return res.to(orig)
\end{lstlisting}

\subsection{B.8 Common Pitfalls}

\subsubsection{B.8.1 Missing L2 Normalization}

LSH requires normalized vectors for angular hashing:

\begin{lstlisting}
Q\_norm = F.normalize(Q, p=2, dim=-1, eps=1e-5)
\end{lstlisting}

\subsubsection{B.8.2 Wrong Bit Shifts}

Bits must be weighted by powers of 2:

\begin{lstlisting}
bit\_shifts = torch.tensor([2**i for i in range(n\_bits)])
\end{lstlisting}

\subsubsection{B.8.3 No Diagonal Mask}

Without the eye matrix, a token may not attend to itself:

\begin{lstlisting}
match = match | eye
\end{lstlisting}

\subsubsection{B.8.4 Wrong Causal Mask Direction}

Causal mask for decoder:

\begin{lstlisting}
causal = torch.triu(torch.ones(N, N), diagonal=1)
match = match & ~causal.unsqueeze(0).unsqueeze(0)
\end{lstlisting}
