\subsection{B.1 The SVDMamba Module Structure}

\begin{lstlisting}
@ATTENTION\_REGISTRY.register("svd\_mamba")
class SVDMamba(BaseAttention):
    def \_\_init\_\_(self, config, rank=None, d\_model=768, d\_state=16,
                 d\_conv=4, expand=1, global\_rank=8, lora\_rank=16, ...):
        super().\_\_init\_\_(config, rank)
        self.d\_model = d\_model
        self.d\_state = d\_state
        self.d\_conv = d\_conv
        self.inner\_dim = expand * d\_model
        
        # Attention path
        self.q\_proj = SVDLinear(d\_model, all\_head\_size, ...)
        self.k\_proj = SVDLinear(d\_model, kv\_head\_size, ...)
        self.v\_proj = SVDLinear(d\_model, kv\_head\_size, ...)
        self.attn\_out\_proj = SVDLinear(all\_head\_size, d\_model, ...)
        
        # SSM path
        self.in\_proj = SVDLinear(d\_model, 2 * inner\_dim, ...)
        self.out\_proj = SVDLinear(inner\_dim, d\_model, ...)
        self.x\_proj = SVDLinear(inner\_dim, dt\_rank + 2 * d\_state, ...)
        self.dt\_proj = SVDLinear(dt\_rank, inner\_dim, ...)
        self.conv1d = nn.Conv1d(inner\_dim, inner\_dim, kernel\_size=d\_conv, groups=inner\_dim)
        self.A\_log = nn.Parameter(...)
        self.D = nn.Parameter(torch.ones(inner\_dim))
        self.dt\_bias = nn.Parameter(torch.zeros(inner\_dim))
        
        # Global SVD mixing
        self.global\_V = nn.Parameter(torch.randn(d\_model, global\_rank) * 0.02)
        self.global\_out = SVDLinear(global\_rank, inner\_dim, ...)
        
        # Gating (init to silence SSM)
        self.gating = nn.Linear(d\_model, d\_model)
        nn.init.zeros\_(self.gating.weight)
        nn.init.constant\_(self.gating.bias, -7.0)
\end{lstlisting}

\subsection{B.2 The SVDLinear Module}

\begin{lstlisting}
class SVDLinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, rank=None, lora\_rank=16, ...):
        super().\_\_init\_\_()
        self.U = nn.Parameter(torch.randn(out\_features, rank) * 0.02)
        self.S = nn.Parameter(torch.ones(rank))
        self.V = nn.Parameter(torch.randn(in\_features, rank) * 0.02)
        self.bias = nn.Parameter(torch.zeros(out\_features))
        
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.empty(lora\_rank, in\_features))
            self.lora\_B = nn.Parameter(torch.zeros(out\_features, lora\_rank))
    
    def forward(self, x):
        out = ((x @ self.V) * self.S) @ self.U.t() + self.bias
        if self.lora\_rank \textgreater{} 0:
            out = out + F.linear(F.linear(x, self.lora\_A), self.lora\_B) * self.scaling
        return out * self.output\_scale
\end{lstlisting}

\subsection{B.3 The SSM Forward}

\begin{lstlisting}
def ssm\_forward(self, x):
    # x: [B, L, inner\_dim]
    x\_t = x.transpose(1, 2)
    x\_pad = F.pad(x\_t, (self.d\_conv - 1, 0))
    x\_conv = self.conv1d(x\_pad)[..., :seqlen].transpose(1, 2)
    x\_conv = F.silu(x\_conv)
    
    x\_proj = self.x\_proj(x\_conv)
    delta = F.softplus(self.dt\_proj(x\_proj[:, :, :dt\_rank]) + self.dt\_bias)
    delta = delta.clamp(min=1e-3, max=0.1)
    B\_ssm = x\_proj[:, :, dt\_rank:dt\_rank+d\_state]
    C\_ssm = x\_proj[:, :, dt\_rank+d\_state:]
    
    A = -torch.exp(self.A\_log.clamp(min=-20, max=10))
    
    # Chunked scan
    h\_running = zeros(B, inner\_dim, d\_state)
    y\_chunks = []
    for c in chunks:
        # ... intra-chunk scan + inter-chunk carry
        y\_chunks.append(y\_c)
    y = cat(y\_chunks, dim=1)
    y = y + self.D * x\_conv
    return y * F.silu(z)

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 Gate Bias Must Be Negative}

If $b_g \geq 0$, the SSM contributes from the start and destabilizes training.

\subsubsection{B.5.2 Conv1d Causal Padding}

The convolution must be causal: pad only on the left.

\subsubsection{B.5.3 SSM Scan Clamps}

Without clamps on \texttt{log\_alpha}, \texttt{h\_chunk}, \texttt{decay\_inv}, \texttt{denom}, the SSM 
produces NaN.

\subsubsection{B.5.4 Nested Encoder Detection}

RoBERTa requires \texttt{is\_nested\_encoder = True}.

\subsubsection{B.5.5 SVD Factorization Rank}

The rank must satisfy $r < \min(d_\text{in}, d_\text{out})$ for 
compression. If $r = \min$, no compression.
