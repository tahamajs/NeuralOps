
\section{Appendix B. Implementation Details}

\subsection{B.1 The `ProgressiveTokenPruning` Class}

\begin{lstlisting}
class ProgressiveTokenPruning(BaseAttention):
    def \_\_init\_\_(self, config=None, keep\_ratio=0.8, rank=None, ...):
        super().\_\_init\_\_(config, rank)
        
        # ... resolve dims ...
        
        # Score network
        self.score\_network = nn.Sequential(
            nn.Linear(self.dim, 64),
            nn.GELU(),
            nn.Linear(64, 1),
        )
        
        # Q/K/V/O with LoRA
        self.q\_lin = LoRALinear(...)
        self.k\_lin = LoRALinear(...)
        self.v\_lin = LoRALinear(...)
        self.out\_lin = LoRALinear(...)
\end{lstlisting}

\subsection{B.2 Scoring and Selection}

\begin{lstlisting}
# Score tokens
scores = torch.sigmoid(self.score\_network(x)).squeeze(-1)  # (B, L)

# CLS preservation
scores[:, 0] = 1e4

# Padding mask
valid\_mask = get\_token\_valid\_mask(attention\_mask, B, L, device)
scores = scores.masked\_fill(~valid\_mask, -1e4)

# Re-apply CLS (padding mask may have overwritten)
scores[:, 0] = 1e4

# Top-k
k\_tokens = max(1, min(L, int(math.ceil(L * self.keep\_ratio))))
\_, top\_indices = torch.topk(scores, k\_tokens, dim=-1)
top\_indices, \_ = torch.sort(top\_indices, dim=-1)
\end{lstlisting}

\subsection{B.3 KV Gather}

\begin{lstlisting}
# Repeat KV for GQA if needed
if self.num\_heads != self.num\_kv\_heads:
    repeats = self.num\_heads // self.num\_kv\_heads
    k\_all = repeat\_kv(k\_all, repeats)
    v\_all = repeat\_kv(v\_all, repeats)

# Gather K, V at top\_indices
top\_idx\_exp = top\_indices.unsqueeze(1).unsqueeze(-1).expand(
    -1, self.num\_heads, -1, self.head\_dim
)
k\_pruned = torch.gather(k\_all, 2, top\_idx\_exp)
v\_pruned = torch.gather(v\_all, 2, top\_idx\_exp)
\end{lstlisting}

\subsection{B.4 Causal Masking After Pruning}

\begin{lstlisting}
q\_pos = torch.arange(L, device=device).view(1, 1, L, 1)
k\_orig\_pos = top\_indices.unsqueeze(1).unsqueeze(2)  # (B, 1, 1, k)
causal\_mask = k\_orig\_pos \textgreater{} q\_pos
attn\_scores = attn\_scores.masked\_fill(causal\_mask, -1e4)
\end{lstlisting}

\subsection{B.5 Padding Mask for Pruned Keys}

\begin{lstlisting}
valid\_keys = torch.gather(valid\_mask, 1, top\_indices)  # (B, k)
attn\_scores = attn\_scores.masked\_fill(
    ~valid\_keys.unsqueeze(1).unsqueeze(2), -1e4
)
\end{lstlisting}

\subsection{B.6 Common Pitfalls}

\subsubsection{B.6.1 Forgetting [CLS]}

Without \texttt{scores[:, 0] = 1e4}, [CLS] may be pruned and classification fails.

\subsubsection{B.6.2 Padding Mask Overwrites [CLS]}

If [CLS] position is not guaranteed valid, the padding mask overrides it. 
Re-apply the [CLS] clamp \textbf{after} the padding mask.

\subsubsection{B.6.3 Causal Mask Uses Pruned Positions}

Use \textbf{original} positions (\texttt{top\_indices}) for causality, not pruned 
positions.

\subsubsection{B.6.4 Wrong GQA Order}

Repeat KV \textbf{before} pruning, not after. Pruning before repeat would prune 
fewer effective heads.

\subsubsection{B.6.5 No Sort After Top-K}

Sorting \texttt{top\_indices} preserves order for gather and downstream layers.

\subsubsection{B.6.6 Unsorted Indices Break Downstream}

Without sorting, gather produces an unsorted sequence, which breaks 
downstream layers that assume ordering.
