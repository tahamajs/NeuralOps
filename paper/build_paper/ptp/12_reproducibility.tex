
\section{Reproducibility Statement}

\subsection{Environment}

\begin{lstlisting}
git clone https://github.com/Efficiency Lab/SVD\_Project
cd SVD\_Project
python -m venv .venv
.venv\Scripts\activate
pip install -r requirements.txt
\end{lstlisting}

\subsection{Running}

\begin{lstlisting}
python run\_all\_4\_ptp.py \
    --models distilbert,roberta,gpt2 \
    --keep\_ratio 0.5 \
    --recovery\_steps 50
\end{lstlisting}

\subsection{Reproducing Tables}

\subsubsection{Main (Table 7.1)}
\begin{lstlisting}
python run\_all\_4\_ptp.py --keep\_ratio 0.8
\end{lstlisting}

\subsubsection{Keep Ratio Ablation (Table 7.3)}
\begin{lstlisting}
for rho in 0.5 0.7 0.8 0.9 0.95; do
    python run\_all\_4\_ptp.py --keep\_ratio $rho --output_suffix rho$rho
done
\end{lstlisting}

\subsubsection{[CLS] Ablation (Table 7.4)}
\begin{lstlisting}
python run\_all\_4\_ptp.py --preserve\_cls true --output\_suffix cls
python run\_all\_4\_ptp.py --preserve\_cls false --output\_suffix nopcls
\end{lstlisting}
