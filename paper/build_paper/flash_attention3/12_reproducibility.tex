\section{Reproducibility Statement}

\subsection{Environment}

\begin{lstlisting}
git clone https://github.com/Efficiency Lab/SVD_Project
cd SVD_Project
python -m venv .venv
.venv\Scripts\activate
pip install -r requirements.txt
\end{lstlisting}

\subsection{Optional: FlashAttention Installation}

\begin{lstlisting}
# For Hopper (H100/H200):
pip install flash-attn --no-build-isolation

# For older GPUs, no installation needed (SDPA fallback works).
\end{lstlisting}

\subsection{Running the Wrapper}

\begin{lstlisting}
python run_all_4_flash3.py \
    --models distilbert,roberta,gpt2 \
    --use_fp8 True \
    --recovery_steps 50
\end{lstlisting}

\subsection{Verifying Path Selection}

\begin{lstlisting}
from framework.attentions.flash_attention3 import FLASH_ATTN_AVAILABLE
print("FlashAttention available:", FLASH_ATTN_AVAILABLE)
\end{lstlisting}

Expected: \texttt{True} on H100, \texttt{False} otherwise.
