\section{Background}

\subsection{Standard Multi-Head Attention}

For input $X \in \mathbb{R}^{B \times N \times d}$:

\[
Q = X W\_Q, \quad K = X W\_K, \quad V = X W\_V
\tag{3.1}
\]

\[
\text{Attn}(Q, K, V) = \text{softmax}\!\left(\frac{Q K^\top}{\sqrt{d\_h}}\right) V
\tag{3.2}
\]

Complexity: $\mathcal{O}(N^2 d_h)$.

\subsection{Mixture-of-Block Attention (MoBA)}

MoBA partitions the sequence into $N_p = \lceil N / P \rceil$ pages of 
size $P$. For each query page $i$, a router selects the top-$k_\text{exp}$ 
key pages:

\[
\mathcal{S}\_i = \text{TopK}(\{\ell\_{ij}\}\_j, k\_\text{exp})
\tag{3.3}
\]

Attention is computed only over the selected pages:

\[
\text{Out}\_i = \text{softmax}\!\left(\frac{Q\_i K\_{\mathcal{S}\_i}^\top}{\sqrt{d\_h}}\right) V\_{\mathcal{S}\_i}
\tag{3.4}
\]

Complexity: $\mathcal{O}(N_p \cdot k_\text{exp} \cdot P^2 d_h)$.

\subsection{The MoBA Router}

The router logit between query page $i$ and key page $j$:

\[
\ell\_{ij} = \frac{(W\_\text{rg} \bar{q}\_i) \cdot \bar{k}\_j}{\sqrt{d\_h}}
\tag{3.5}
\]

where:

\[
\bar{q}\_i = \frac{1}{P} \sum\_{t \in \text{page}\_i} q\_t, \quad
\bar{k}\_j = \frac{1}{P} \sum\_{t \in \text{page}\_j} k\_t
\tag{3.6}
\]

$W_\text{rg} \in \mathbb{R}^{d_h \times d_h}$ is a learnable routing matrix, 
initialized to identity.

\subsection{Delta-KV Compression}

Given consecutive key vectors $k_t, k_{t-1}$, the temporal delta is:

\[
\Delta k\_t = k\_t - k\_{t-1}
\tag{3.7}
\]

For smooth sequences (natural language often is), $\|\Delta k_t\| \ll \|k_t\|$.

\textbf{Compression.} Reshape deltas to $\Delta \in \mathbb{R}^{(B H (N-1)) \times d_h}$, 
then compute SVD:

\[
\Delta \approx U\_r \Sigma\_r V\_r^\top
\tag{3.8}
\]

Reconstruct:

\[
\hat{\Delta} = U\_r \Sigma\_r V\_r^\top
\tag{3.9}
\]

Then reconstruct $K$:

\[
\hat{k}\_t = k\_0 + \sum\_{\tau=1}^t \hat{\Delta}\_\tau
\tag{3.10}
\]

\subsection{Straight-Through Estimator (STE)}

For differentiable quantization:

\[
\text{STE}(x) = x + (Q(x) - x).\text{detach}()
\tag{3.11}
\]

\textbf{Forward:} $Q(x)$ (quantized).
\textbf{Backward:} $\nabla \text{STE}(x) = \nabla x$ (identity).

\subsection{4-Bit Uniform Quantization}

Given $x$, compute min/max per row:

\[
x\_\text{min} = \min(x), \quad x\_\text{max} = \max(x)
\tag{3.12}
\]

Scale:

\[
s = \frac{x\_\text{max} - x\_\text{min}}{15}
\tag{3.13}
\]

Quantize:

\[
x\_q = \text{round}\!\left(\frac{x - x\_\text{min}}{s}\right) \in \{0, 1, \dots, 15\}
\tag{3.14}
\]

Dequantize:

\[
\hat{x} = x\_q \cdot s + x\_\text{min}
\tag{3.15}
\]

\subsection{Load Balancing Loss}

For MoE routers, a common auxiliary loss:

\[
\mathcal{L}\_\text{aux} = N\_p \cdot \sum\_{j=1}^{N\_p} \rho\_j \cdot \pi\_j
\tag{3.16}
\]

where:

\begin{itemize}
\item $\rho_j$ = fraction of queries that selected page $j$
\item $\pi_j$ = mean router probability for page $j$

\end{itemize}
\textbf{Effect.} Encourages uniform page selection, preventing router collapse.

\subsection{CLS Token Preservation}

In classification models (BERT family), the \texttt{[CLS]} token at position 0 
aggregates sequence information. We ensure page 0 is always selected:

\[
\ell\_{i, 0} \leftarrow \ell\_{i, 0} + 10^4
\tag{3.17}
\]

This boost is large enough that page 0 always appears in the top-$k$.

\subsection{Grouped-Query Attention (GQA)}

For $H_{kv} < H$, K/V are shared across query heads. Our implementation 
expands via \texttt{repeat\_kv} before attention.

\subsection{Rotary Position Embeddings (RoPE)}

Applied to $Q, K$ after projection, before delta compression.
