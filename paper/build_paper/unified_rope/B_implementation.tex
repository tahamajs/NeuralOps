class UnifiedRoPE(BaseAttention):
    def \_\_init\_\_(self, config=None, rank=None, dim=None, num\_heads=None,
                 max\_seq\_len=4096, base=10000, ...):
        super().\_\_init\_\_(config=config, rank=rank)
        
        # Dimensions
        self.hidden\_size = ...
        self.num\_heads = ...
        self.num\_kv\_heads = ...
        self.head\_dim = hidden\_size // num\_heads
        
        # Projections
        self.q\_proj = LoRALinear(hidden\_size, q\_head\_size, lora\_rank=..., freeze\_base=True)
        self.k\_proj = LoRALinear(hidden\_size, kv\_head\_size, lora\_rank=..., freeze\_base=True)
        self.v\_proj = LoRALinear(hidden\_size, kv\_head\_size, lora\_rank=..., freeze\_base=True)

        # RoPE cache
        self.rotary\_dim = 2 * (head\_dim // 2)
        inv\_freq = 1.0 / (base ** (torch.arange(0, rotary\_dim, 2).float() / rotary\_dim))
        self.register\_buffer("inv\_freq", inv\_freq, persistent=False)
        self.\_build\_cache(max\_seq\_len)
\end{lstlisting}

\subsection{B.2 The `\_build\_cache` Method}

\begin{lstlisting}
def \_build\_cache(self, seq\_len):
    t = torch.arange(seq\_len, device=self.inv\_freq.device, dtype=self.inv\_freq.dtype)
    freqs = torch.einsum("i,j-\textgreater{}ij", t, self.inv\_freq)
    emb = torch.cat((freqs, freqs), dim=-1)
    self.register\_buffer("cos\_cached", emb.cos(), persistent=False)
    self.register\_buffer("sin\_cached", emb.sin(), persistent=False)
\end{lstlisting}

\textbf{Non-persistent.} The cache is not saved with \texttt{state\_dict()}. On load, 
it is rebuilt from \texttt{inv\_freq}.

\subsection{B.3 The `\_apply\_rope` Method}

\begin{lstlisting}
def \_apply\_rope(self, x, seq\_len):
    if self.rotary\_dim == 0 or self.cos\_cached.shape[-1] == 0:
        return x
    
    if seq\_len \textgreater{} self.max\_seq\_len:
        self.\_build\_cache(seq\_len)
        self.max\_seq\_len = seq\_len
    
    rotary\_dim = min(self.rotary\_dim, self.cos\_cached.shape[-1], x.shape[-1])
    rotary\_dim = 2 * (rotary\_dim // 2)
    
    cos = self.cos\_cached[:seq\_len, :rotary\_dim].unsqueeze(0).unsqueeze(0)
    sin = self.sin\_cached[:seq\_len, :rotary\_dim].unsqueeze(0).unsqueeze(0)
    
    if rotary\_dim \textless{} x.shape[-1]:
        x\_rot = x[..., :rotary\_dim]
        x\_pass = x[..., rotary\_dim:]
        rotated = (x\_rot * cos) + (\_rotate\_half(x\_rot) * sin)
        return torch.cat((rotated, x\_pass), dim=-1)
    else:
        return (x * cos) + (\_rotate\_half(x) * sin)
\end{lstlisting}

\subsection{B.5 The Injection Hook}

\begin{lstlisting}
def inject\_unified\_rope(layer, model\_config=None, comp\_config=None,
                        old\_attn=None, **kwargs):
    try:
        src = old\_attn
        if src is None:
            for attr in ("attention", "self\_attn", "attn", "self"):
                if hasattr(layer, attr):
                    src = getattr(layer, attr)
                    break
        
        if src is None:
            return None
        
        if model\_config is None and hasattr(layer, "config"):
            model\_config = layer.config
        
        lora\_rank = kwargs.pop("lora\_rank", 16)
        rank = kwargs.pop("rank", 32)
        
        new\_attn = UnifiedRoPE(
            config=model\_config, rank=rank, lora\_rank=lora\_rank,
            pretrained\_attn=src, **kwargs,
        )
        
        # Replace
        for attr in ("self\_attn", "attn", "self"):
            if hasattr(layer, attr):
                setattr(layer, attr, new\_attn)
                return new\_attn
        if hasattr(layer, "attention"):
            if hasattr(layer.attention, "self"):
                layer.attention.self = new\_attn
            else:
                layer.attention = new\_attn
            return new\_attn
        layer.attention = new\_attn
        return new\_attn
    except Exception as e:
        logger.error("inject\_unified\_rope failed: \%s", e)
        raise RuntimeError("Injection failed")
\end{lstlisting}

\subsection{B.6 Common Pitfalls}

\subsubsection{B.6.1 Conv1D Not Transposed}

GPT-2's Conv1D stores \texttt{(in, out)}. Must transpose.

\subsubsection{B.6.2 Nested Encoder Flag}

RoBERTa requires \texttt{is\_nested\_encoder = True}.

\subsubsection{B.6.3 GQA Not Expanded}

Qwen requires \texttt{repeat\_kv}.

\subsubsection{B.6.4 Cache Not Rebuilt}

If sequence length exceeds \texttt{max\_seq\_len}, cache must be rebuilt.
