
\section{Method}

\subsection{Overview}

UnifiedRoPE implements RoPE-based scaled dot-product attention with a 
single \texttt{nn.Module}. It provides:

\begin{enumerate}
\item \textbf{LoRALinear projections} for Q, K, V, O.
\item \textbf{Precomputed RoPE caches} for fast frequency lookup.
\item \textbf{Robust weight transfer} across architectures.
\item \textbf{GQA expansion} via \texttt{repeat\_kv}.
\item \textbf{Nested encoder bypass} for RoBERTa-style models.

\end{enumerate}
\textbf{Figure 4.1} shows the pipeline.

\subsection{The `UnifiedRoPE` Module}

\subsubsection{Initialization}

\begin{lstlisting}
class UnifiedRoPE(BaseAttention):
    def \_\_init\_\_(self, config=None, rank=None, dim=None, num\_heads=None,
                 max\_seq\_len=4096, base=10000, dropout=0.1,
                 pretrained\_attn=None, lora\_rank=16, lora\_alpha=32.0,
                 is\_decoder=None, **kwargs):
        # Resolve dimensions
        self.hidden\_size = ...
        self.num\_heads = ...
        self.num\_kv\_heads = ...
        self.head\_dim = hidden\_size // num\_heads
        
        # LoRA projections
        self.q\_proj = LoRALinear(hidden\_size, q\_head\_size, lora\_rank=..., freeze\_base=True)
        self.k\_proj = LoRALinear(hidden\_size, kv\_head\_size, lora\_rank=..., freeze\_base=True)
        self.v\_proj = LoRALinear(hidden\_size, kv\_head\_size, lora\_rank=..., freeze\_base=True)
        self.out\_proj = LoRALinear(q\_head\_size, hidden\_size, lora\_rank=..., freeze\_base=True)
        
        # RoPE cache
        self.rotary\_dim = 2 * (head\_dim // 2)
        self.register\_buffer("inv\_freq", ..., persistent=False)
        self.\_build\_cache(max\_seq\_len)
\end{lstlisting}

\subsubsection{Precomputed RoPE Cache}

\begin{lstlisting}
def \_build\_cache(self, seq\_len):
    t = torch.arange(seq\_len, dtype=self.inv\_freq.dtype)
    freqs = torch.einsum("i,j-\textgreater{}ij", t, self.inv\_freq)
    emb = torch.cat((freqs, freqs), dim=-1)
    self.register\_buffer("cos\_cached", emb.cos(), persistent=False)
    self.register\_buffer("sin\_cached", emb.sin(), persistent=False)
\end{lstlisting}

\textbf{Note.} Buffers are \texttt{persistent=False}, so they are not stored in 
\texttt{state\_dict()}. This avoids checkpoint pollution.

\subsubsection{Applying RoPE}

\begin{lstlisting}
def \_apply\_rope(self, x, seq\_len):
    rotary\_dim = min(self.rotary\_dim, self.cos\_cached.shape[-1], x.shape[-1])
    rotary\_dim = 2 * (rotary\_dim // 2)
    
    cos = self.cos\_cached[:seq\_len, :rotary\_dim].unsqueeze(0).unsqueeze(0)
    sin = self.sin\_cached[:seq\_len, :rotary\_dim].unsqueeze(0).unsqueeze(0)
    
    if rotary\_dim \textless{} x.shape[-1]:
        x\_rot = x[..., :rotary\_dim]
        x\_pass = x[..., rotary\_dim:]
        rotated = (x\_rot * cos) + (\_rotate\_half(x\_rot) * sin)
        return torch.cat((rotated, x\_pass), dim=-1)
    else:
        return (x * cos) + (\_rotate\_half(x) * sin)
\end{lstlisting}

\subsection{Forward Pass}

\begin{lstlisting}
def forward(self, hidden\_states, attention\_mask=None, ...):
    B, L, \_ = hidden\_states.shape
    
    # 1. Project
    q = self.q\_proj(hidden\_states).view(B, L, H, d\_h).transpose(1, 2)
    k = self.k\_proj(hidden\_states).view(B, L, Hkv, d\_h).transpose(1, 2)
    v = self.v\_proj(hidden\_states).view(B, L, Hkv, d\_h).transpose(1, 2)
    
    # 2. Apply RoPE
    pos\_emb = kwargs.get("position\_embeddings")
    if pos\_emb is not None:
        cos, sin = pos\_emb
        q, k = apply\_rotary\_pos\_emb(q, k, cos, sin)
    else:
        q = self.\_apply\_rope(q, L)
        k = self.\_apply\_rope(k, L)
    
    # 3. GQA expansion
    if self.num\_kv\_heads != self.num\_heads:
        k = repeat\_kv(k, self.num\_key\_value\_groups)
        v = repeat\_kv(v, self.num\_key\_value\_groups)
    
    # 4. Scaled dot-product attention
    scores = q @ k^T / sqrt(d\_h)
    
    # 5. Causal mask for decoders
    if self.is\_decoder and L \textgreater{} 1:
        scores = scores + causal\_mask
    
    # 6. Padding mask
    if attention\_mask is not None:
        scores = apply\_attention\_mask(scores, attention\_mask)
    
    # 7. Softmax, dropout, context
    attn = softmax(scores)
    context = attn @ v
    
    # 8. Output projection (skip if nested)
    if self.is\_nested\_encoder:
        output = context
    else:
        output = self.out\_proj(context)
    
    return output, attn
\end{lstlisting}

\subsection{Two RoPE Entry Points}

\subsubsection{Internal `\_apply\_rope`}

Uses the cached \texttt{cos\_cached}, \texttt{sin\_cached} buffers. \textbf{Fast path} for models 
that do not supply their own RoPE cache.

\subsubsection{External `apply\_rotary\_pos\_emb`}

Accepts \texttt{(cos, sin)} from HuggingFace models that precompute their own 
RoPE. Used when \texttt{position\_embeddings} is passed in \texttt{forward}.

\textbf{Why two paths?} Some models (Qwen) supply their RoPE cache; others do 
not. UnifiedRoPE supports both.

\subsection{GQA Expansion}

For $H_{kv} < H$:

\begin{lstlisting}
if self.num\_kv\_heads != self.num\_heads:
    k = repeat\_kv(k, self.num\_key\_value\_groups)
    v = repeat\_kv(v, self.num\_key\_value\_groups)
\end{lstlisting}

\texttt{repeat\_kv} broadcasts each KV head to \texttt{H / H\_{kv}} query heads.

\subsection{Nested Encoder Bypass}

For RoBERTa-style models:

\begin{lstlisting}
if self.is\_nested\_encoder:
    output = context   # Raw context
else:
    output = self.out\_proj(context)
\end{lstlisting}

The outer \texttt{BertSelfOutput.dense} applies the projection.

\subsection{Theoretical Notes}

\subsubsection{RoPE Rotation Correctness}

\textbf{Proposition.} The rotation operator preserves L2 norm:

\[
\|\tilde{q}\_i\|\_2 = \|q\_i\|\_2
\tag{4.1}
\]

\textbf{Proof.} Rotation matrices are orthogonal. $\square$

\subsubsection{Relative Position Property}

\textbf{Proposition.} The attention score satisfies:

\[
\tilde{q}\_i^\top \tilde{k}\_j = f(q\_i, k\_j, i - j)
\tag{4.2}
\]

\textbf{Proof.} Classical (Su et al., 2021). $\square$

\subsubsection{GQA Equivalence}

\textbf{Proposition.} GQA with \texttt{repeat\_kv} produces the same output as if the 
KV heads were duplicated.

\textbf{Proof.} \texttt{repeat\_kv} literally duplicates. $\square$
