\subsection{Known Issues}

\begin{itemize}
\item \textbf{Encoder defaults to dense.} Cascade not exercised.
\item \textbf{Batch-level gate.} No per-token adaptivity.
\item \textbf{RFF cumsum OOM} at $L > 512$.
\item \textbf{Sparse block Python loop.}
\item \textbf{\texttt{inject\_nasc} raises on failure.}

\end{itemize}
