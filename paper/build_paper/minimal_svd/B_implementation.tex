\subsection{B.1 The `FactorisedLinear` Module}

\begin{lstlisting}
class FactorisedLinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, rank=64, bias=True,
                 energy=0.95, lora\_rank=16, lora\_alpha=32.0):
        super().\_\_init\_\_()
        self.in\_features = in\_features
        self.out\_features = out\_features
        self.rank = min(rank, in\_features, out\_features)
        self.energy = energy
        self.lora\_rank = lora\_rank
        self.lora\_scaling = lora\_alpha / lora\_rank if lora\_rank \textgreater{} 0 else 1.0
        
        self.down = nn.Linear(in\_features, self.rank, bias=False)
        self.up = nn.Linear(self.rank, out\_features, bias=bias)
        
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.randn(lora\_rank, in\_features) * 0.01)
            self.lora\_B = nn.Parameter(torch.zeros(out\_features, lora\_rank))
        
        self.reset\_parameters()
    
    def forward(self, x):
        out = self.up(self.down(x))
        if self.lora\_rank \textgreater{} 0:
            out += F.linear(F.linear(x, self.lora\_A), self.lora\_B) * self.lora\_scaling
        return out
\end{lstlisting}

\subsection{B.2 `decompose\_from\_weight`}

\begin{lstlisting}
def decompose\_from\_weight(self, weight, bias=None, energy=None):
    with torch.no\_grad():
        W = weight.float()
        
        # Handle GQA dimension mismatch
        if W.shape[0] \textless{} self.out\_features and self.out\_features \% W.shape[0] == 0:
            rep = self.out\_features // W.shape[0]
            W = W.repeat(rep, 1)
        elif W.shape[0] \textgreater{} self.out\_features and W.shape[0] \% self.out\_features == 0:
            rep = W.shape[0] // self.out\_features
            W = W.view(self.out\_features, rep, -1).mean(dim=1)
        
        # SVD
        U, S, Vt = torch.linalg.svd(W, full\_matrices=False)
        
        # Energy-based rank
        target\_energy = energy if energy is not None else self.energy
        total\_energy = (S ** 2).sum()
        cum\_energy = torch.cumsum(S ** 2, dim=0) / (total\_energy + 1e-12)
        calc\_rank = int((cum\_energy \textless{} target\_energy).sum().item()) + 1
        r = min(self.rank, calc\_rank, U.size(1), Vt.size(0))
        r = max(1, r)
        
        # Symmetric distribution
        sqrt\_s = torch.sqrt(S[:r])
        A = (U[:, :r] * sqrt\_s).to(self.up.weight.dtype)
        B = (sqrt\_s.unsqueeze(1) * Vt[:r, :]).to(self.down.weight.dtype)
        
        # Store
        self.up.weight.data.zero\_()
        self.down.weight.data.zero\_()
        self.up.weight.data[:, :r] = A
        self.down.weight.data[:r, :] = B
        
        self.freeze\_base\_weights()
\end{lstlisting}

\subsection{B.3 The Injection Hook (Minimal)}

\begin{lstlisting}
def inject\_minimal\_svd(layer, model\_config, comp\_config=None,
                       layer\_idx=0, total\_layers=1, old\_attn=None, **kwargs):
    if model\_config is None:
        model\_config = type('Cfg', (), {'hidden\_size': 64, 'num\_heads': 4, 'rank': 32})()
    
    energy = getattr(comp\_config, "svd\_energy", 0.95) if comp\_config else 0.95
    dropout = getattr(comp\_config, "dropout", 0.1) if comp\_config else 0.1
    
    new\_attn = MinimalSVDAttention(
        model\_config=model\_config,
        energy=energy,
        dropout=dropout,
    )
    
    src\_attn = old\_attn if old\_attn is not None else \
               getattr(layer, "attention", getattr(layer, "self", None))
    
    if src\_attn is not None:
        try:
            new\_attn.copy\_weights(src\_attn)
        except Exception as e:
            logger.error(f"copy\_weights failed: {e}")
            return None
        p = next(src\_attn.parameters(), None)
        if p is not None:
            new\_attn.to(device=p.device, dtype=p.dtype)
    
    if hasattr(layer, "attention"):
        if hasattr(layer.attention, "self"):
            layer.attention.self = new\_attn
            new\_attn.is\_nested\_encoder = True
        else:
            layer.attention = new\_attn
            new\_attn.is\_nested\_encoder = False
    elif hasattr(layer, "self"):
        layer.self = new\_attn
        new\_attn.is\_nested\_encoder = True
    else:
        new\_attn.is\_nested\_encoder = False
    
    return new\_attn
\end{lstlisting}

\textbf{Note.} This function is intentionally minimal:

\begin{itemize}
\item \textbf{No whole-model traversal} -- caller handles
\item \textbf{No pre-test} -- simplicity over safety
\item \textbf{No double-injection guard} -- assume well-formed caller

\end{itemize}

\subsection{B.6 Nested Encoder Detection}

MinimalSVD sets \texttt{is\_nested\_encoder} in \textbf{two places}:

\textbf{In \texttt{\_\_init\_\_}:}

\begin{lstlisting}
model\_type = getattr(cfg, "model\_type", "").lower() if cfg else ""
self.is\_nested\_encoder = (
    getattr(cfg, "is\_nested\_encoder", False)
    or model\_type in ("roberta", "bert", "xlm-roberta")
)
\end{lstlisting}

\textbf{In \texttt{copy\_weights}:}

\begin{lstlisting}
cls\_name = src.\_\_class\_\_.\_\_name\_\_
has\_out\_proj = any(hasattr(src, attr) for attr in ("dense", "out\_proj", "c\_proj", "out\_lin", "o\_proj"))

is\_nested = False
if hasattr(src, "self") and hasattr(getattr(src, "self"), "query"):
    is\_nested = True
elif (
    cls\_name in ("BertSelfAttention", "RobertaSelfAttention", "XLMRobertaSelfAttention")
    or (cls\_name.endswith("SelfAttention") and not has\_out\_proj)
):
    is\_nested = True

self.is\_nested\_encoder = is\_nested
\end{lstlisting}

\textbf{Note.} \texttt{copy\_weights} overrides the \texttt{\_\_init\_\_} value. This is important 
because \texttt{\_\_init\_\_} only has the config, while \texttt{copy\_weights} inspects the 
actual module.
