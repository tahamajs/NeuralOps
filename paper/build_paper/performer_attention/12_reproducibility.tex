\subsection{Diagnostics}

\subsubsection{Verify Feature Positivity}

\begin{lstlisting}
Q\_feat = model.\_feature\_map(Q)
assert (Q\_feat \textgreater{}= 0).all(), "Feature map produced negative values!"
\end{lstlisting}

\subsubsection{Verify Omega Persistence}

\begin{lstlisting}
sd = model.state\_dict()
assert any('omega' in k for k in sd), "omega NOT persisted!"
\end{lstlisting}

\subsubsection{Verify Causal Path}

\begin{lstlisting}
x = torch.randn(1, 128, 768, requires\_grad=True)
y = model(x)
y[0, 50].sum().backward()
assert x.grad[0, 51:].abs().max() \textless{} 1e-6
\end{lstlisting}
