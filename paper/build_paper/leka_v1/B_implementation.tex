\section{Appendix B. Implementation Details}

\subsection{B.1 The Kernel Feature Map}

\begin{lstlisting}
def kernel\_feature\_map(self, x):
    x\_norm = self.norm(x)  # LayerNorm on head\_dim
    features = torch.einsum("bhnd,hdf-\textgreater{}bhnf", x\_norm, self.proj.to(x.dtype))
    scale = 1.0 / math.sqrt(self.feature\_dim)
    phi\_cos = torch.cos(features) * scale
    phi\_sin = torch.sin(features) * scale
    return torch.cat([phi\_cos, phi\_sin], dim=-1)  # (B, H, N, 2F)
\end{lstlisting}

\subsection{B.2 QR Projection Init}

\begin{lstlisting}
proj\_data = torch.randn(self.num\_heads, self.head\_dim, self.feature\_dim)
for h in range(self.num\_heads):
    if self.head\_dim \textless{} self.feature\_dim:
        q\_h, \_ = torch.linalg.qr(proj\_data[h].T)
        proj\_data[h] = q\_h.T
    else:
        q\_h, \_ = torch.linalg.qr(proj\_data[h])
        proj\_data[h] = q\_h[:, :self.feature\_dim]
self.register\_buffer("proj", proj\_data, persistent=True)
\end{lstlisting}

\subsection{B.3 Causal Forward}

\begin{lstlisting}
kv\_outer = torch.einsum("bhnf,bhnd-\textgreater{}bhnfd", K\_features.float(), v.float())
KV\_cum = torch.cumsum(kv\_outer, dim=2)
num = torch.einsum("bhnf,bhnfd-\textgreater{}bhnd", Q\_features.float(), KV\_cum)
z\_cum = torch.cumsum(K\_features.float(), dim=2)
denom = torch.einsum("bhnf,bhnf-\textgreater{}bhn", Q\_features.float(), z\_cum).unsqueeze(-1).clamp(min=1e-5)
context = (num / denom).to(hidden\_states.dtype)
\end{lstlisting}

\subsection{B.4 Bidirectional Forward}

\begin{lstlisting}
KV = torch.einsum("bhnf,bhnd-\textgreater{}bhfd", K\_features.float(), v.float())
K\_sum = K\_features.float().sum(dim=2, keepdim=True)
num = torch.einsum("bhnf,bhfd-\textgreater{}bhnd", Q\_features.float(), KV)
denom = torch.einsum("bhnf,bhf-\textgreater{}bhn", Q\_features.float(), K\_sum.squeeze(2)).unsqueeze(-1).clamp(min=1e-5)
context = (num / denom).to(hidden\_states.dtype)
\end{lstlisting}

\subsection{B.7 Common Pitfalls}

\subsubsection{B.7.1 Cumsum Memory}

The causal cumsum materializes \texttt{(B, H, L, 2F, d\_h)} -- can OOM at 
large L.

\subsubsection{B.7.2 FP32 Precision}

We use \texttt{.float()} for the einsum to avoid FP16 precision loss in the 
cumsum.

\subsubsection{B.7.3 Mask Application}

Mask must be applied \textbf{before} aggregation:

\begin{lstlisting}
K\_features = K\_features * mask
v = v * mask
\end{lstlisting}

\subsubsection{B.7.4 Feature Map Output}

$\phi(x)$ has shape $(B, H, N, 2F)$, not $(B, H, N, F)$.

\subsection{B.8 Diagnostic Checks}

assert not torch.isnan(context).any()
\end{lstlisting}
