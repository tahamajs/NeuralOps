
\section{Background}

\subsection{Sliding-Window Attention}

Given $Q, K, V \in \mathbb{R}^{L \times d_h}$ and window size $w$:

\[
S\_{ij} = \frac{Q\_i^\top K\_j}{\sqrt{d\_h}}, \quad
\text{mask}\_{ij} = \begin{cases} 1 & |i - j| \leq w \\ 0 & \text{otherwise} \end{cases}
\tag{3.1}
\]

\[
O\_i^\text{local} = \frac{\sum\_j \text{mask}\_{ij} \exp(S\_{ij}) V\_j}{\sum\_j \text{mask}\_{ij} \exp(S\_{ij})}
\tag{3.2}
\]

\textbf{Complexity.} $\mathcal{O}(L w d_h)$ time, $\mathcal{O}(L w)$ memory.

\textbf{Causal version.} Only $j \le i$ and $i - j \leq w$ are attended.

\subsection{Random Fourier Features}

Rahimi \& Recht (2007) show that:

\[
k(q, k) = \exp\!\left(-\frac{\|q - k\|^2}{2}\right) \approx \phi(q)^\top \phi(k)
\tag{3.3}
\]

where $\phi(x) = \frac{1}{\sqrt{m}} \left[\cos(\omega_1^\top x), \sin(\omega_1^\top x), \ldots\right]$ 
with $\omega_i \sim \mathcal{N}(0, I)$.

\textbf{For attention.} Performer uses $\phi(x) = \frac{1}{\sqrt{m}} \exp(\omega_i^\top x)$ 
with positive orthogonal features.

\textbf{NHA uses a simplified non-negative variant:}

\[
\phi(x) = \frac{1}{\sqrt{m}} (\mathrm{elu}(\Omega^\top x) + 1)
\tag{3.4}
\]

with $\Omega \in \mathbb{R}^{d_h \times m}$ orthogonal.

\subsection{Orthogonal Projection Matrix}

We construct $\Omega$ by block QR:

\[
\Omega = [Q\_1, Q\_2, \ldots, Q\_b] \cdot \sqrt{d\_h}
\tag{3.5}
\]

where each $Q_i \in \mathbb{R}^{d_h \times d_h}$ is from \texttt{torch.linalg.qr} 
on a random Gaussian matrix, and $b = \lceil m / d_h \rceil$.

\textbf{Why orthogonal?} Orthogonal projections give \textbf{lower variance} kernel 
approximation than i.i.d. Gaussians, at no extra cost.

\subsection{Kernel Linear Attention}

Given $\phi$, linear attention is:

\[
O\_i^\text{global} = \frac{\phi(Q\_i)^\top \left(\sum\_j \phi(K\_j) V\_j^\top\right)}{\phi(Q\_i)^\top \left(\sum\_j \phi(K\_j)\right)}
\tag{3.6}
\]

\textbf{Complexity.} $\mathcal{O}(L m d_h)$ time, $\mathcal{O}(m d_h)$ memory.

\subsection{Causal Prefix Scan}

For causal models, the cumulative state is:

\[
S^{(i)} = \sum\_{j \leq i} \phi(K\_j) V\_j^\top \in \mathbb{R}^{m \times d\_h}
\tag{3.7}
\]

\[
Z^{(i)} = \sum\_{j \leq i} \phi(K\_j) \in \mathbb{R}^m
\tag{3.8}
\]

\textbf{NHA's chunked implementation.} We process in chunks of 64:

\[
S \leftarrow S + \sum\_{j \in \text{chunk}} \phi(K\_j) V\_j^\top
\tag{3.9}
\]

This bounds memory to $\mathcal{O}(m d_h)$ regardless of $L$.

\subsection{Sigmoid Gate}

Per-token gate:

\[
g\_i = \sigma(w^\top x\_i + b) \in (0, 1)
\tag{3.10}
\]

\textbf{Initialization.} $w = 0$, $b = 3.0$ gives $g_i = \sigma(3) \approx 0.95$ 
for all tokens. The model starts close to the local stream.
