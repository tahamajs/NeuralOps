\subsection{B.2 Orthogonal Matrix Construction}

\begin{lstlisting}
def \_build\_orthogonal\_matrix(self, d, m):
    blocks = []
    remaining = m
    while remaining \textgreater{} 0:
        block\_size = min(d, remaining)
        g = torch.randn(d, block\_size)
        q, \_ = torch.linalg.qr(g)
        blocks.append(q)
        remaining -= block\_size
    W = torch.cat(blocks, dim=1) * math.sqrt(d)
    return W
\end{lstlisting}

\subsection{B.3 Local Window Attention (Chunked)}

\begin{lstlisting}
def \_local\_window\_attention(self, Q, K, V, valid\_mask, is\_causal=False):
    B, H, N, d = Q.shape
    scale = 1.0 / math.sqrt(d)
    chunk = 64
    window = self.window\_size
    out = torch.zeros\_like(V)
    
    for i in range(0, N, chunk):
        end = min(i + chunk, N)
        q\_block = Q[:, :, i:end]
        k\_start = max(0, i - window)
        k\_end = min(N, end + window) if not is\_causal else end
        k\_block = K[:, :, k\_start:k\_end]
        v\_block = V[:, :, k\_start:k\_end]
        
        scores = torch.matmul(q\_block.float(), k\_block.float().transpose(-2, -1)) * scale
        # ... band mask, causal mask, padding mask ...
        attn = F.softmax(scores, dim=-1).to(Q.dtype)
        attn = torch.nan\_to\_num(attn, nan=0.0)
        out[:, :, i:end] = torch.matmul(attn, v\_block)
    return out
\end{lstlisting}

\subsection{B.4 Global RFF Attention (Chunked Prefix Scan)}

\begin{lstlisting}
def \_global\_rff\_attention(self, Q, K, V, valid\_mask, is\_causal=False):
    B, H, N, d = Q.shape
    scale = d ** -0.25
    Q\_scaled = Q * scale
    K\_scaled = K * scale
    
    proj = self.rff\_proj.to(dtype=Q.dtype, device=Q.device)
    Qf = torch.matmul(Q\_scaled, proj)
    Kf = torch.matmul(K\_scaled, proj)
    
    norm = 1.0 / math.sqrt(self.rff\_features)
    phi\_Q = (F.elu(Qf) + 1.0) * norm
    phi\_K = (F.elu(Kf) + 1.0) * norm
    
    pad = valid\_mask.unsqueeze(1).unsqueeze(-1).to(Q.dtype)
    phi\_K = phi\_K * pad
    phi\_Q = phi\_Q * pad
    V\_masked = V * pad
    
    with torch.autocast(device\_type=Q.device.type, enabled=False):
        # ... FP32 computation ...
        if is\_causal:
            # Chunked prefix scan
            chunk\_size = min(64, N)
            S = torch.zeros(B, H, self.rff\_features, d, dtype=torch.float32)
            Z = torch.zeros(B, H, self.rff\_features, dtype=torch.float32)
            out\_list = []
            for c in range(num\_chunks):
                # ... intra + inter chunk computation ...
                S = S + torch.matmul(k\_c.transpose(-1, -2), v\_c)
                Z = Z + k\_c.sum(dim=2)
            out = torch.cat(out\_list, dim=2) * pad
        else:
            KV = torch.matmul(phi\_K\_f.transpose(-2, -1), V\_f)
            Z = phi\_K\_f.sum(dim=-2, keepdim=True)
            num = torch.matmul(phi\_Q\_f, KV)
            den = torch.matmul(phi\_Q\_f, Z.transpose(-2, -1)).clamp(min=1e-6)
            out = (num / den).to(Q.dtype) * pad
    return out
\end{lstlisting}

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 Missing FP32 in RFF}

FP16 overflows for $L > 512$. Always cast to FP32.

\subsubsection{B.5.2 Zero Padding Contamination}

Zero out invalid K, V before cumsum to avoid accumulating padding.

\subsubsection{B.5.3 Gate Initialization}

\texttt{nn.init.zeros\_(gate.weight)} and \texttt{nn.init.constant\_(gate.bias, 3.0)} -- 
ensures $\sigma(3) \approx 0.95$ at start.

\subsubsection{B.5.4 RoPE Before Streams}

Apply RoPE to Q, K before both streams.

\subsubsection{B.5.5 GQA Repeat Before Streams}

Apply \texttt{repeat\_kv} before local and global streams.

\subsubsection{B.5.6 Nested Encoder Output}

For RoBERTa, \texttt{is\_nested\_encoder=True} $\to$ skip own \texttt{out\_proj}.
