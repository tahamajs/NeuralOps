\subsection{B.1 The `AdaSplashLinear` Module}

\begin{lstlisting}
class AdaSplashLinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, lora\_rank=16, bias=True):
        super().\_\_init\_\_()
        self.in\_features = in\_features
        self.out\_features = out\_features
        self.lora\_rank = min(lora\_rank, in\_features, out\_features)
        
        # Frozen base
        self.weight = nn.Parameter(
            torch.empty(out\_features, in\_features),
            requires\_grad=False
        )
        if bias:
            self.bias = nn.Parameter(
                torch.empty(out\_features),
                requires\_grad=False
            )
        
        # Trainable LoRA
        if self.lora\_rank \textgreater{} 0:
            self.lora\_B = nn.Parameter(torch.empty(out\_features, self.lora\_rank))
            self.lora\_A = nn.Parameter(torch.empty(self.lora\_rank, in\_features))
        
        self.reset\_parameters()
    
    def forward(self, x):
        out = F.linear(x, self.weight, self.bias)
        if self.lora\_A is not None:
            out = out + F.linear(F.linear(x, self.lora\_A), self.lora\_B)
        return out
\end{lstlisting}

\subsection{B.2 Entmax Bisection}

\begin{lstlisting}
def entmax\_bisect(scores, alpha=1.5, valid\_mask=None, dim=-1, n\_iter=25):
    device = scores.device
    dtype = scores.dtype
    neg\_inf\_val = torch.tensor(-10000.0, device=device, dtype=dtype)
    
    # Ensure finite
    scores = torch.where(torch.isfinite(scores), scores, neg\_inf\_val)
    
    # Softmax fallback for alpha = 1
    if abs(alpha - 1.0) \textless{} 1e-4:
        if valid\_mask is not None:
            scores = scores.masked\_fill(~valid\_mask, -10000.0)
        sf = torch.nan\_to\_num(F.softmax(scores.float(), dim=dim), nan=0.0).to(dtype)
        if valid\_mask is not None:
            sf = sf.masked\_fill(~valid\_mask, 0.0)
        sf\_denom = sf.sum(dim=dim, keepdim=True).clamp(min=1e-9)
        return (sf / sf\_denom).to(dtype)
    
    alpha\_minus\_1 = max(1e-4, alpha - 1.0)
    inv\_alpha\_minus\_1 = 1.0 / alpha\_minus\_1
    
    # Mask invalid keys
    if valid\_mask is not None:
        scores\_valid = scores.masked\_fill(~valid\_mask, -10000.0)
    else:
        scores\_valid = scores
    
    # Shift scores
    max\_scores = scores\_valid.amax(dim=dim, keepdim=True)
    max\_scores = torch.where(max\_scores \textless{} -5000.0, torch.zeros\_like(max\_scores), max\_scores)
    scores\_shift = scores\_valid - max\_scores
    scaled = scores\_shift * alpha\_minus\_1
    
    # Initial bounds
    if valid\_mask is not None:
        scaled\_for\_min = scaled.masked\_fill(~valid\_mask, 10000.0)
        tau\_lo = scaled\_for\_min.amin(dim=dim, keepdim=True) - 1.0
        tau\_lo = torch.where(tau\_lo \textgreater{} 5000.0, torch.zeros\_like(tau\_lo) - 1.0, tau\_lo)
    else:
        tau\_lo = scaled.amin(dim=dim, keepdim=True) - 1.0
    tau\_hi = torch.zeros\_like(tau\_lo)
    
    # Bisection
    for \_ in range(n\_iter):
        tau = (tau\_lo + tau\_hi) / 2.0
        p = torch.clamp(scaled - tau, min=0.0) ** inv\_alpha\_minus\_1
        if valid\_mask is not None:
            p = p.masked\_fill(~valid\_mask, 0.0)
        sum\_p = p.sum(dim=dim, keepdim=True)
        mask = sum\_p \textgreater{} 1.0
        tau\_lo = torch.where(mask, tau, tau\_lo)
        tau\_hi = torch.where(mask, tau\_hi, tau)
    
    # Final
    tau = (tau\_lo + tau\_hi) / 2.0
    p = torch.clamp(scaled - tau, min=0.0) ** inv\_alpha\_minus\_1
    if valid\_mask is not None:
        p = p.masked\_fill(~valid\_mask, 0.0)
    
    denom = p.sum(dim=dim, keepdim=True)
    
    # Softmax fallback
    scores\_for\_sm = scores.masked\_fill(~valid\_mask, -10000.0) if valid\_mask is not None else scores
    sf = torch.nan\_to\_num(F.softmax(scores\_for\_sm.float(), dim=dim), nan=0.0).to(dtype)
    if valid\_mask is not None:
        sf = sf.masked\_fill(~valid\_mask, 0.0)
    sf\_denom = sf.sum(dim=dim, keepdim=True).clamp(min=1e-9)
    sf = sf / sf\_denom
    
    is\_empty = (denom \textless{} 1e-7) | torch.isnan(denom)
    p\_normalized = torch.nan\_to\_num(p / denom.clamp(min=1e-9), nan=0.0)
    p = torch.where(is\_empty, sf, p\_normalized)
    return torch.nan\_to\_num(p, nan=0.0)
\end{lstlisting}

\subsection{B.4 Weight Extraction}

\begin{lstlisting}
def copy\_weights(self, src):
    from framework.core.adapters import UniversalAdapter
    
    extracted\_weights = {}
    
    for proj in ["q", "k", "v", "o"]:
        w, b = None, None
        
        # Try UniversalAdapter
        try:
            w, b = UniversalAdapter.get\_weight\_bias(src, proj)
        except Exception:
            pass
        
        # Fallbacks
        if w is None:
            if proj == "o":
                # Check RoBERTa/BERT outer, GPT-2 c\_proj, DistilBERT out\_lin
                if hasattr(src, "output") and hasattr(src.output, "dense"):
                    w = src.output.dense.weight
                    b = getattr(src.output.dense, "bias", None)
                elif hasattr(src, "c\_proj"):
                    w = src.c\_proj.weight.T
                    b = getattr(src.c\_proj, "bias", None)
                elif hasattr(src, "out\_lin"):
                    w = src.out\_lin.weight
                    b = getattr(src.out\_lin, "bias", None)
            elif proj in ["q", "k", "v"] and hasattr(src, "c\_attn"):
                fused = src.c\_attn.weight.T
                split = fused.size(0) // 3
                idx = {"q": 0, "k": 1, "v": 2}[proj]
                w = fused[idx*split:(idx+1)*split]
                b = src.c\_attn.bias[idx*split:(idx+1)*split] if src.c\_attn.bias is not None else None
            elif hasattr(src, "self") and hasattr(src.self, "query"):
                s = src.self
                name\_map = {"q": "query", "k": "key", "v": "value"}
                w = getattr(s, name\_map[proj]).weight
                b = getattr(getattr(s, name\_map[proj]), "bias", None)
            # ... more cases
        
        extracted\_weights[proj] = (w, b)
    
    # Copy into target projections
    for proj, target in [("q", self.q\_proj), ("k", self.k\_proj), ("v", self.v\_proj), ("o", self.out\_proj)]:
        w, b = extracted\_weights[proj]
        if w is not None:
            with torch.no\_grad():
                target.weight.data.copy\_(w)
                if b is not None and target.bias is not None:
                    target.bias.data.copy\_(b)
                # Freeze base, init LoRA
                target.weight.requires\_grad = False
                if target.bias is not None:
                    target.bias.requires\_grad = False
                if target.lora\_A is not None:
                    nn.init.kaiming\_uniform\_(target.lora\_A, a=math.sqrt(5))
                    target.lora\_A.requires\_grad = True
                if target.lora\_B is not None:
                    nn.init.zeros\_(target.lora\_B)
                    target.lora\_B.requires\_grad = True
\end{lstlisting}

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 Support Empty}

If bisection fails to find valid $\tau$, softmax fallback is essential.

\subsubsection{B.5.2 NaN in Scores}

Mask to -10000 before bisection.

\subsubsection{B.5.3 Wrong Mask for Encoders}

Splash masking is \textbf{only} for causal decoders. Encoders use full attention.

\subsubsection{B.5.4 Conv1D Transpose}

GPT-2 Conv1D stores \texttt{(in, out)}. Must transpose.

\subsubsection{B.5.5 Attention Sinks}

Preserving first 4 tokens is not optional -- accuracy drops without it.
