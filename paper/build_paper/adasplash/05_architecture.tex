\subsection{Weight Extraction}

\subsubsection{DistilBERT (Flat)}

Direct extraction:

\begin{lstlisting}
W_Q = src.q_lin.weight  # Linear(768, 768)
W_K = src.k_lin.weight
W_V = src.v_lin.weight
W_O = src.out_lin.weight
\end{lstlisting}

\texttt{is\_nested\_encoder = False}. All four projections compressed.

\subsubsection{RoBERTa (Nested)}

Extract from \texttt{self}:

\begin{lstlisting}
W_Q = src.self.query.weight
W_K = src.self.key.weight
W_V = src.self.value.weight
# W_O is not extracted -- outer output.dense applies it
\end{lstlisting}

\texttt{is\_nested\_encoder = True}. During forward, \texttt{context} is returned without 
\texttt{out\_proj}; the outer \texttt{BertSelfOutput.dense} applies its projection.

\subsubsection{GPT-2 (Fused Conv1D)}

Split the fused \texttt{c\_attn} weight:

\begin{lstlisting}
fused = src.c_attn.weight.T  # (2304, 768)
W_Q, W_K, W_V = fused.chunk(3, dim=0)
W_O = src.c_proj.weight.T
\end{lstlisting}

Handles the HuggingFace Conv1D layout \texttt{(in, out)}.

\subsubsection{Qwen3.5 (GQA + RoPE)}

Direct extraction with GQA awareness:

\begin{lstlisting}
W_Q = src.q_proj.weight  # (H * d_h, d)
W_K = src.k_proj.weight  # (H_kv * d_h, d)
W_V = src.v_proj.weight
W_O = src.o_proj.weight
\end{lstlisting}

RoPE applied after projection via \texttt{position\_embeddings}.

\subsection{The `AdaSplashLinear` Module}

\begin{lstlisting}
class AdaSplashLinear(nn.Module):
    def __init__(self, in_features, out_features, lora_rank=16, bias=True):
        super().__init__()
        self.lora_rank = min(lora_rank, in_features, out_features)
        
        # Frozen base weight
        self.weight = nn.Parameter(
            torch.empty(out_features, in_features),
            requires_grad=False
        )
        
        if bias:
            self.bias = nn.Parameter(
                torch.empty(out_features),
                requires_grad=False
            )
        
        # Trainable LoRA
        if self.lora_rank > 0:
            self.lora_B = nn.Parameter(
                torch.empty(out_features, self.lora_rank)
            )
            self.lora_A = nn.Parameter(
                torch.empty(self.lora_rank, in_features)
            )
        
        self.reset_parameters()
    
    def forward(self, x):
        out = F.linear(x, self.weight, self.bias)
        if self.lora_A is not None:
            out = out + F.linear(F.linear(x, self.lora_A), self.lora_B)
        return out
\end{lstlisting}

\subsection{Injection Algorithm}

\textbf{Algorithm 3:} Inject AdaSplash into a model
\begin{lstlisting}
Input:  model M, config c, hyperparameters $\alpha$, W, r_L
Output: M with attention replaced

1:  for   in transformer_layers(M) do
2:      src $\leftarrow$ find_attention( )
3:      if src is None: continue
4:      new $\leftarrow$ AdaSplashAttention(c, $\alpha$, W, r_L, pretrained_attn=src)
5:      new.copy_weights(src)
6:      replace_attention( , new)
7: end for
8: return M
\end{lstlisting}

\subsection{Handling Special Cases}

\subsubsection{Attention Sinks}

For causal decoders, we preserve the first $S = 4$ tokens regardless of 
splash mask:

\begin{lstlisting}
splash_local = ((dist >= 0) & (dist <= W)) | (k_idx < 4)
\end{lstlisting}

\subsubsection{GQA Expansion}

\begin{lstlisting}
if self.kv_repeat > 1:
    k = repeat_kv(k, self.kv_repeat)
    v = repeat_kv(v, self.kv_repeat)
\end{lstlisting}

\subsubsection{RoPE}

\begin{lstlisting}
if position_embeddings is not None:
    cos, sin = position_embeddings
    q, k = apply_rotary_emb(q, k, cos, sin)
\end{lstlisting}

\subsubsection{Causal Masking}

\begin{lstlisting}
if is_decoder:
    causal_tril = torch.tril(torch.ones((L_q, L_k), device=device))
    logits = logits.masked_fill(~causal_tril, -10000)
\end{lstlisting}

\subsubsection{Combined Masking}

The causal mask and splash mask are combined:

\begin{lstlisting}
if is_decoder and W > 0:
    dist = q_idx - k_idx
    splash = ((dist >= 0) & (dist <= W)) | (k_idx < 4)
    causal_and_splash = splash  # already includes causality via dist >= 0
    logits = logits.masked_fill(~causal_and_splash, -10000)
\end{lstlisting}

\subsection{Compatibility}

\subsubsection{Encoder-Decoder Models}

For T5, the same class handles encoder (bidirectional, no splash) and decoder 
(causal, splash). The \texttt{is\_decoder} flag switches behavior.

\subsubsection{LoRA Wrappers}

If the source module is a \texttt{LoRALinear}, AdaSplash unwraps to \texttt{base\_layer} and 
preserves the outer LoRA.
