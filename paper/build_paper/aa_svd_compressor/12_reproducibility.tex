\begin{itemize}

\subsection{Known Issues}

\begin{itemize}
\item Windows: \texttt{PYTHONIOENCODING=utf-8} recommended
\item CUDA $\ge$ 11.8 for FP16 SVD
\item Some baselines may need re-download
\end{itemize}
