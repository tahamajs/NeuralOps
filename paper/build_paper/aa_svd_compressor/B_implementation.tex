\subsection{B.1 The `SVDLinear` Module}

\begin{lstlisting}
class SVDLinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, rank, bias=True, ...):
        super().\_\_init\_\_()
        self.in\_features = in\_features
        self.out\_features = out\_features
        self.rank = max(1, min(rank, in\_features, out\_features))
        
        # Base factors
        self.V = nn.Parameter(torch.empty(self.rank, in\_features))
        self.U = nn.Parameter(torch.empty(out\_features, self.rank))
        
        if bias:
            self.bias = nn.Parameter(torch.empty(out\_features))
        
        # Optional LoRA
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.empty(lora\_rank, in\_features))
            self.lora\_B = nn.Parameter(torch.zeros(out\_features, lora\_rank))
        
        self.reset\_parameters()
    
    def forward(self, x):
        latent = F.linear(x, self.V)
        out = F.linear(latent, self.U)
        if self.lora\_rank \textgreater{} 0:
            out += F.linear(F.linear(x, self.lora\_A), self.lora\_B) * self.lora\_scaling
        if self.bias is not None:
            out += self.bias
        return out * self.output\_scale
\end{lstlisting}

\subsection{B.2 Streaming EMA Calibration}

\begin{lstlisting}
def make\_hook(name):
    def hook(module, inp, outp):
        x = inp[0].detach()
        with torch.autocast(device\_type=x.device.type, enabled=False):
            x\_fp32 = x.float()
            x\_flat = x\_fp32.reshape(-1, in\_f)
            mask = x\_flat.abs().sum(dim=-1) \textgreater{} 1e-5
            x\_active = x\_flat[mask]
            if x\_active.size(0) == 0:
                return
            batch\_cov = (
                torch.matmul(x\_active.T, x\_active) / max(1, x\_active.size(0))
            ).cpu()
        
        if name not in covariances:
            covariances[name] = batch\_cov
        else:
            covariances[name] = (
                self.ema\_beta * covariances[name]
                + (1.0 - self.ema\_beta) * batch\_cov
            )
    return hook
\end{lstlisting}

\textbf{Key implementation notes:}

\begin{enumerate}
\item \textbf{FP32 compute:} We cast to FP32 inside the hook to avoid numerical issues.
\item \textbf{Padding filter:} Mask tokens with $\|x_i\|_1 < 10^{-5}$.
\item \textbf{CPU transfer:} \texttt{batch\_cov.cpu()} prevents GPU memory accumulation.

\end{enumerate}
\subsection{B.3 Eigendecomposition with Regularization}

\begin{lstlisting}
def compute\_sqrt\_and\_inv(cov, device):
    cov = cov.to(device=device, dtype=torch.float32)
    cov\_sym = 0.5 * (cov + cov.T)
    eigvals, eigvecs = torch.linalg.eigh(cov\_sym)
    
    # Regularize
    eps = 1e-6 * torch.max(eigvals).clamp(min=1e-12) + 1e-8
    eigvals\_reg = eigvals.clamp(min=eps)
    
    sqrt\_eigvals = torch.sqrt(eigvals\_reg)
    inv\_sqrt\_eigvals = 1.0 / sqrt\_eigvals
    
    cov\_sqrt = eigvecs @ torch.diag(sqrt\_eigvals) @ eigvecs.T
    cov\_inv\_sqrt = eigvecs @ torch.diag(inv\_sqrt\_eigvals) @ eigvecs.T
    
    return cov\_sqrt, cov\_inv\_sqrt
\end{lstlisting}

\subsection{B.4 Weighted SVD}

\begin{lstlisting}
def compress\_linear(self, layer, cov\_factors=None):
    W = weight(layer).float()
    
    if cov\_factors is not None:
        cov\_sqrt, cov\_inv\_sqrt = cov\_factors
        W\_weighted = W @ cov\_sqrt
    else:
        W\_weighted = W
    
    U, S, Vt = torch.linalg.svd(W\_weighted, full\_matrices=False)
    
    # Truncate
    r = min(self.rank, ...)
    sqrt\_S = torch.sqrt(S[:r].clamp(min=1e-12))
    U\_r = U[:, :r] * sqrt\_S.unsqueeze(0)
    V\_r = sqrt\_S.unsqueeze(1) * Vt[:r, :]
    
    # Unweight
    if cov\_inv\_sqrt is not None:
        V\_r = V\_r @ cov\_inv\_sqrt
    
    # Validate
    W\_rec = U\_r @ V\_r
    err = torch.norm(W - W\_rec).item() / max(torch.norm(W).item(), 1e-12)
    
    return SVDLinear.from\_factors(U\_r, V\_r), err, r
\end{lstlisting}

\subsection{B.5 Rank Cap Constraint}

\begin{lstlisting}
max\_allowed\_rank = max(1, min(
    in\_features,
    out\_features,
    int(0.75 * (in\_features * out\_features) / (in\_features + out\_features))
))
r = min(r, max\_allowed\_rank)
\end{lstlisting}

This ensures $r(d_\text{in} + d_\text{out}) < 0.75 d_\text{in} d_\text{out}$, 
guaranteeing at least 25\% parameter reduction per layer.

\subsection{B.7 Common Pitfalls}

\subsubsection{B.7.1 Forgetting CPU Transfer}

Without \texttt{.cpu()}, GPU memory grows linearly with calibration batches.

\subsubsection{B.7.2 No Regularization}

Without regularizing $\lambda_i$, near-zero eigenvalues produce inf in 
$C_X^{-1/2}$.

\subsubsection{B.7.3 Wrong Conv1D Orientation}

GPT-2's Conv1D stores weights as \texttt{(in, out)}. Must transpose.

\subsubsection{B.7.4 No Padding Filter}

Without masking padding tokens, covariance is biased.
