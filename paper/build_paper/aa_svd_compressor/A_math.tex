
\section{Appendix A. Mathematical Derivations}

\subsection{A.1 Weighted Low-Rank Approximation}

\subsubsection{A.1.1 Problem Statement}

Given $W \in \mathbb{R}^{m \times n}$ and PSD weight matrix $C_X \in \mathbb{R}^{n \times n}$:

\[
W\_r^* = \arg\min\_{\text{rank}(W\_r) = r} \mathrm{Tr}\!\left[(W - W\_r) C\_X (W - W\_r)^\top\right]
\tag{A.1}
\]

\subsubsection{A.1.2 Transform}

Since $C_X = C_X^{1/2} C_X^{1/2}$:

\[
\mathrm{Tr}[(W - W\_r) C\_X (W - W\_r)^\top] = \|(W - W\_r) C\_X^{1/2}\|\_F^2
\tag{A.2}
\]

Define $\tilde{W} = W C_X^{1/2}$, $\tilde{W}_r = W_r C_X^{1/2}$. Since 
$\text{rank}(\tilde{W}_r) \leq r$, the optimal $\tilde{W}_r$ is the rank-$r$ 
SVD truncation of $\tilde{W}$:

\[
\tilde{W}\_r = U\_r \Sigma\_r V\_r^\top
\tag{A.3}
\]

Therefore:

\[
W\_r^* = U\_r \Sigma\_r V\_r^\top C\_X^{-1/2}
\tag{A.4}
\]

Setting $U = U_r \Sigma_r^{1/2}$, $V = \Sigma_r^{1/2} V_r^\top C_X^{-1/2}$ 
gives the AASVD factorization. $\square$

\subsection{A.2 Reconstruction Error Bound}

\textbf{Proposition.} Under AASVD at rank $r$:

\[
\|W - U\_r V\_r\|\_F \leq \frac{\sqrt{\sum\_{i \textgreater{} r} \sigma\_i^2}}{\sqrt{\lambda\_\text{min}(C\_X)}}
\tag{A.5}
\]

where $\sigma_i$ are the singular values of $W C_X^{1/2}$.

\textbf{Proof.} From the transformed problem:

\[
\|W C\_X^{1/2} - U\_r V\_r C\_X^{1/2}\|\_F^2 = \sum\_{i \textgreater{} r} \sigma\_i^2
\tag{A.6}
\]

Meanwhile:

\[
\|W C\_X^{1/2} - U\_r V\_r C\_X^{1/2}\|\_F^2 \geq \lambda\_\text{min}(C\_X) \|W - U\_r V\_r\|\_F^2
\tag{A.7}
\]

since $C_X \succeq \lambda_\text{min}(C_X) I$. Combining:

\[
\|W - U\_r V\_r\|\_F^2 \leq \frac{\sum\_{i \textgreater{} r} \sigma\_i^2}{\lambda\_\text{min}(C\_X)}
\tag{A.8}
\]

Taking square roots gives (A.5). $\square$

\subsection{A.3 EMA Convergence}

\textbf{Proposition.} Under constant true covariance $\mathbb{E}[C_k] = \bar{C}$, 
the EMA estimate satisfies:

\[
\mathbb{E}[\hat{C}^{(k)}] - \bar{C} = \beta^{k-1} (C\_1 - \bar{C})
\tag{A.9}
\]

\textbf{Proof.} The EMA recursion:

\[
\hat{C}^{(k)} = \beta \hat{C}^{(k-1)} + (1-\beta) C\_k
\]

has closed form (unrolled):

\[
\hat{C}^{(k)} = \beta^{k-1} C\_1 + (1-\beta) \sum\_{j=2}^{k} \beta^{k-j} C\_j
\]

Taking expectations:

\[
\mathbb{E}[\hat{C}^{(k)}] = \beta^{k-1} C\_1 + (1-\beta) \sum\_{j=2}^{k} \beta^{k-j} \bar{C}
\]

The sum $\sum_{j=2}^{k} \beta^{k-j} = \frac{1 - \beta^{k-1}}{1 - \beta}$. So:

\[
\mathbb{E}[\hat{C}^{(k)}] = \beta^{k-1} C\_1 + (1 - \beta^{k-1}) \bar{C}
\]

Rearranging:

\[
\mathbb{E}[\hat{C}^{(k)}] - \bar{C} = \beta^{k-1} (C\_1 - \bar{C}) \qquad \square
\]

\subsection{A.4 Eigendecomposition for Square Root}

For $C_X = Q \Lambda Q^\top$ with $\Lambda = \mathrm{diag}(\lambda_1, \dots, \lambda_n)$:

\[
C\_X^{1/2} = Q \mathrm{diag}(\sqrt{\lambda\_1}, \dots, \sqrt{\lambda\_n}) Q^\top
\tag{A.10}
\]

\textbf{Verification:}

\[
C\_X^{1/2} \cdot C\_X^{1/2} = Q \Lambda^{1/2} Q^\top Q \Lambda^{1/2} Q^\top = Q \Lambda Q^\top = C\_X
\tag{A.11}
\]

Similarly for $C_X^{-1/2}$. $\square$

\subsection{A.5 Numerical Stability}

Regularization threshold:

\[
\epsilon\_\text{reg} = 10^{-6} \cdot \max\_i \lambda\_i + 10^{-8}
\tag{A.12}
\]

Clamping:

\[
\lambda\_i \leftarrow \max(\lambda\_i, \epsilon\_\text{reg})
\tag{A.13}
\]

This ensures $\lambda_i \geq \epsilon_\text{reg} > 0$, so $1/\sqrt{\lambda_i}$ is bounded.
