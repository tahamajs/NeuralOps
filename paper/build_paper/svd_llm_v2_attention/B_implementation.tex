\subsection{B.2 The `\_project` Helper}

\begin{lstlisting}
def \_project(self, x, vh, s, u, b, lora\_A, lora\_B):
    base\_out = ((x @ vh.t()) * s.to(dtype=x.dtype)) @ u.t() + b
    if self.lora\_rank \textgreater{} 0 and lora\_A is not None and lora\_B is not None:
        lora\_out = F.linear(F.linear(x, lora\_A), lora\_B) * self.scaling
        base\_out = base\_out + lora\_out
    return base\_out
\end{lstlisting}

\subsection{B.3 The `\_decompose\_and\_set` Method}

\begin{lstlisting}
def \_decompose\_and\_set(self, u\_param, s\_param, vh\_param, bias\_param,
                       weight, bias):
    with torch.no\_grad():
        W = weight.float()
        # Pad or truncate to expected shape
        if W.shape != (u\_param.shape[0], vh\_param.shape[1]):
            padded\_W = torch.zeros(u\_param.shape[0], vh\_param.shape[1], ...)
            padded\_W[:min(...), :min(...)] = W[:min(...), :min(...)]
            W = padded\_W
        
        U, S, Vt = torch.linalg.svd(W, full\_matrices=False)
        total\_energy = (S ** 2).sum().clamp(min=1e-12)
        cum\_energy = torch.cumsum(S ** 2, dim=0) / total\_energy
        r\_energy = int((cum\_energy \textless{} self.energy\_threshold).sum().item()) + 1
        r = max(1, min(r\_energy, self.rank, len(S)))
        
        u\_param.data.zero\_()
        s\_param.data.zero\_()
        vh\_param.data.zero\_()
        u\_param.data[:, :r] = U[:, :r].to(u\_param.dtype)
        s\_param.data[:r] = S[:r].to(s\_param.dtype)
        vh\_param.data[:r, :] = Vt[:r, :].to(vh\_param.dtype)
        if bias is not None:
            bias\_param.data.copy\_(bias.to(bias\_param.dtype))
\end{lstlisting}

\subsection{B.4 Weight Transfer}

SVD-LLM V2 handles four patterns:

\begin{lstlisting}
def copy\_weights(self, old\_attn):
    src = old\_attn
    if hasattr(old\_attn, "self") and hasattr(old\_attn, "output"):
        src = old\_attn.self
        if hasattr(old\_attn.output, "dense"):
            w\_o, b\_o = \_unpack\_weight\_bias(old\_attn.output.dense)
    
    # 1. GPT-2 Fused c\_attn
    if hasattr(src, "c\_attn") and hasattr(src, "c\_proj"):
        c\_attn\_w, c\_attn\_b = \_unpack\_weight\_bias(src.c\_attn)
        split\_size = c\_attn\_w.size(0) // 3
        w\_q = c\_attn\_w[:split\_size, :]
        w\_k = c\_attn\_w[split\_size:2*split\_size, :]
        w\_v = c\_attn\_w[2*split\_size:, :]
        # ...
    
    # 2. RoBERTa / BERT
    elif hasattr(src, "query") and hasattr(src, "key") and hasattr(src, "value"):
        w\_q, b\_q = \_unpack\_weight\_bias(src.query)
        # ...
    
    # 3. Qwen / LLaMA
    elif hasattr(src, "q\_proj") and hasattr(src, "k\_proj") and hasattr(src, "v\_proj"):
        # ...
    
    # 4. DistilBERT
    elif hasattr(src, "q\_lin") and hasattr(src, "k\_lin") and hasattr(src, "v\_lin"):
        # ...
    
    # Fallback: UniversalAdapter
    if w\_q is None or w\_k is None or w\_v is None:
        adapter = UniversalAdapter(old\_attn)
        extracted = adapter.extract\_qkv\_out()
        # ...
    
    # GQA pooling for K and V
    if w\_k.shape[0] != self.kv\_head\_size:
        rep = w\_k.shape[0] // self.kv\_head\_size
        w\_k\_pooled = (
            w\_k.view(self.num\_kv\_heads, rep, self.head\_dim, -1)
            .mean(dim=1)
            .view(self.kv\_head\_size, -1)
        )
        # ...
    
    # Decompose and set
    self.\_decompose\_and\_set(self.query\_u, ..., w\_q, b\_q)
    self.\_decompose\_and\_set(self.key\_u, ..., w\_k\_pooled, b\_k\_pooled)
    self.\_decompose\_and\_set(self.value\_u, ..., w\_v\_pooled, b\_v\_pooled)
    if w\_o is not None and not self.is\_nested\_encoder:
        self.\_decompose\_and\_set(self.out\_proj\_u, ..., w\_o, b\_o)
\end{lstlisting}

def inject\_svd\_llm\_v2(layer=None, model=None, model\_config=None,
                      comp\_config=None, ...):
    target = layer if layer is not None else model
    
    # Whole-model traversal
    layer\_list = ...
    if layer\_list is not None:
        # Pre-test
        test\_src = getattr(layer\_list[0], "attention", None)
        if test\_src is not None:
            try:
                SVDLLMV2Attention(cfg, old\_attn=test\_src, ...)
            except Exception as e:
                logger.warning(f"Pre-test failed: {e}")
                return None
        
        # Inject each layer
        count = 0
        for idx, lyr in enumerate(layer\_list):
            res = inject\_svd\_llm\_v2(layer=lyr, ...)
            if res is not None:
                count += 1
        return target if count \textgreater{} 0 else None
    
    # Single-layer injection
    src = get\_attention(target)
    
    # Double-injection guard
    if isinstance(src, SVDLLMV2Attention):
        return None
    
    new\_attn = SVDLLMV2Attention(cfg, old\_attn=src, ...)
    
    # Set nested flag and replace
    for attr in ("attention", "attn", "self\_attn", "self"):
        if hasattr(target, attr):
            if attr == "attention" and hasattr(target.attention, "self"):
                new\_attn.is\_nested\_encoder = True
                target.attention.self = new\_attn
            else:
                new\_attn.is\_nested\_encoder = False
                setattr(target, attr, new\_attn)
            return new\_attn
\end{lstlisting}
