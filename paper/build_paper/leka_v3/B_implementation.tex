\section{Appendix B. Implementation Details}

\subsection{B.1 Kernel Feature Map}

\begin{lstlisting}
def kernel\_feature\_map(self, x):
    proj1 = self.proj1.to(device=x.device, dtype=x.dtype)
    f1 = torch.matmul(x, proj1)
    f1 = torch.clamp(f1, min=-10.0, max=10.0)
    return torch.cat([torch.cos(f1), torch.sin(f1)], dim=-1) / math.sqrt(self.feature\_dim)
\end{lstlisting}

The \texttt{clamp} prevents overflow in \texttt{cos}/\texttt{sin} for extreme inputs.

\subsection{B.2 Landmark Selection}

\begin{lstlisting}
def select\_landmarks(self, K\_features, V, valid\_mask=None, num\_landmarks=None, return\_indices=False):
    B, H, N, feat\_dim = K\_features.shape
    target\_landmarks = num\_landmarks or self.num\_landmarks
    L = min(target\_landmarks, N)
    
    scores = self.landmark\_scorer(K\_features)[..., 0]  # (B, H, N)
    
    if valid\_mask is not None:
        pad = (~valid\_mask).unsqueeze(1)
        scores = scores.masked\_fill(pad.bool(), -1e4)
    
    anchor\_boost = torch.zeros\_like(scores)
    anchor\_boost[:, :, 0] = 1e4
    scores = scores + anchor\_boost
    
    \_, idx = scores.topk(L, dim=-1)
    idx[:, :, 0] = 0  # Force [CLS]
    
    K\_land = torch.gather(K\_features, 2, idx.unsqueeze(-1).expand(-1, -1, -1, feat\_dim))
    V\_land = torch.gather(V, 2, idx.unsqueeze(-1).expand(-1, -1, -1, V.shape[-1]))
    
    if return\_indices:
        return K\_land, V\_land, idx
    return K\_land, V\_land
\end{lstlisting}

\subsection{B.3 Forward Pass}

\begin{lstlisting}
def forward(self, hidden\_states, attention\_mask=None, ...):
    B, N, D = hidden\_states.shape
    H = self.num\_heads
    
    valid\_mask = get\_token\_valid\_mask(attention\_mask, B, N, hidden\_states.device)
    
    q = self.query(hidden\_states).view(B, N, H, self.head\_dim).transpose(1, 2)
    k = self.key(hidden\_states).view(B, N, self.num\_kv\_heads, self.head\_dim).transpose(1, 2)
    v = self.value(hidden\_states).view(B, N, self.num\_kv\_heads, self.head\_dim).transpose(1, 2)
    
    if position\_embeddings is not None:
        q, k = apply\_rotary\_pos\_emb(q, k, cos, sin)
    
    if self.kv\_repeat \textgreater{} 1:
        k = repeat\_kv(k, self.kv\_repeat)
        v = repeat\_kv(v, self.kv\_repeat)
    
    Q\_features = self.kernel\_feature\_map(q)
    K\_features = self.kernel\_feature\_map(k)
    
    effective\_landmarks = min(N, max(self.num\_landmarks, N // 2))
    K\_land, V\_land, land\_idx = self.select\_landmarks(
        K\_features, v, valid\_mask, num\_landmarks=effective\_landmarks, return\_indices=True
    )
    
    W = self.W\_corr.unsqueeze(0).expand(B, -1, -1, -1)
    V\_land\_adj = torch.matmul(V\_land, W)
    
    attn\_weights = torch.matmul(Q\_features.float(), K\_land.transpose(-2, -1).float())
    # ... (masking, softpick, etc.)
\end{lstlisting}

if self.is\_decoder:

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 Forgetting to Clamp Features}

Without \texttt{clamp(f1, -10, 10)}, extreme inputs can cause \texttt{cos}/\texttt{sin} to 
overflow in FP16.

\subsubsection{B.5.2 Wrong Landmark Index in Mask}

Using compressed indices (0..L\_land-1) instead of original positions 
breaks causality.

\subsubsection{B.5.3 Missing [CLS] Boost}

Without \texttt{anchor\_boost}, \texttt{[CLS]} may be dropped, hurting classification.

\subsubsection{B.5.4 W\_corr Memory}

$W_\text{corr}$ is $H \times d_h \times d_h$. For $H = 12, d_h = 64$, 
this is 49K params per layer $\times$ 12 layers = ~590K for RoBERTa.
