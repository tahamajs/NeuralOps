\subsection{B.1 InfoMambaConceptBottleneck}

\begin{lstlisting}
class InfoMambaConceptBottleneck(nn.Module):
    def \_\_init\_\_(self, dim, bottleneck\_dim=64):
        super().\_\_init\_\_()
        self.W\_down = nn.Linear(dim, bottleneck\_dim, bias=False)
        self.W\_up = nn.Linear(bottleneck\_dim, dim, bias=False)
    
    def forward(self, x):
        return self.W\_up(F.gelu(self.W\_down(x)))
\end{lstlisting}

\subsection{B.2 Forward Pass}

\begin{lstlisting}
def forward(self, x, attention\_mask=None, position\_embeddings=None):
    B, L, D = x.shape
    
    # Differential attention branch
    q1 = self.W\_q1(x).view(B, L, H, Dh).transpose(1, 2)
    k1 = self.W\_k1(x).view(B, L, Hkv, Dh).transpose(1, 2)
    q2 = self.W\_q2(x).view(B, L, H, Dh).transpose(1, 2)
    k2 = self.W\_k2(x).view(B, L, Hkv, Dh).transpose(1, 2)
    v = self.W\_v(x).view(B, L, Hkv, Dh).transpose(1, 2)
    
    if position\_embeddings is not None:
        cos, sin = position\_embeddings
        q1, k1 = apply\_rotary\_pos\_emb(q1, k1, cos, sin)
        q2, k2 = apply\_rotary\_pos\_emb(q2, k2, cos, sin)
    
    if self.kv\_repeat \textgreater{} 1:
        k1 = repeat\_kv(k1, self.kv\_repeat)
        k2 = repeat\_kv(k2, self.kv\_repeat)
        v = repeat\_kv(v, self.kv\_repeat)
    
    scores1 = torch.matmul(q1.float(), k1.float().transpose(-2, -1)) * scale
    scores2 = torch.matmul(q2.float(), k2.float().transpose(-2, -1)) * scale
    
    # masks
    if self.is\_decoder:
        causal = torch.tril(...)
        scores1 = scores1.masked\_fill(~causal, -1e4)
        scores2 = scores2.masked\_fill(~causal, -1e4)
    if attention\_mask is not None:
        ...
    
    attn1 = F.softmax(scores1, dim=-1)
    attn2 = F.softmax(scores2, dim=-1)
    diff\_attn = F.relu(attn1 - 0.2 * attn2)
    diff\_attn = diff\_attn / (diff\_attn.sum(dim=-1, keepdim=True) + 1e-8)
    
    attn\_out = torch.matmul(diff\_attn, v).transpose(1, 2).contiguous().view(B, L, self.all\_head\_size)
    
    # Concept-bottlenecked SSM
    z = self.bottleneck(x)  # (B, L, D)
    if self.is\_decoder:
        ssm\_branch = self.ssm(z, attention\_mask=attention\_mask)[0]
    else:
        z\_rev = torch.flip(z, dims=[1])
        M\_fwd = self.ssm(z, attention\_mask=attention\_mask)[0]
        M\_bwd = self.ssm(z\_rev, attention\_mask=attention\_mask)[0]
        ssm\_branch = 0.5 * (M\_fwd + torch.flip(M\_bwd, dims=[1]))
    
    # Fusion
    alpha = torch.sigmoid(self.alpha\_raw)
    combined = (1 - alpha) * attn\_out + alpha * ssm\_branch
    
    if self.is\_nested\_encoder:
        return combined, None
    return self.out\_proj(combined), None
\end{lstlisting}

\subsection{B.3 Bottleneck Dimension}

Default \texttt{b = 256} when \texttt{d = 768}. General rule:

\[
b = \min(d, \max(64, d // 3))
\tag{B.1}
\]

\subsection{B.4 Common Pitfalls}

\subsubsection{B.4.1 Forgetting GELU}

Without nonlinearity, the bottleneck is just a low-rank linear map.

\subsubsection{B.4.2 $\alpha$ Init Sign}

We use \texttt{w  = -2} so $\alpha$  $\approx$ 0.12. \textbf{Not} +2 (which would make $\alpha$ $\approx$ 0.88).

\subsubsection{B.4.3 Fusion Order}

Our fusion is \texttt{(1-$\alpha$) A + $\alpha$ M}, not \texttt{$\alpha$ A + (1-$\alpha$) M}. The gate parameter 
and the fusion order are tied.
