\begin{lstlisting}
git clone https://github.com/Efficiency Lab/SVD\_Project
cd SVD\_Project
python -m venv .venv
.venv\Scripts\activate
pip install -r requirements.txt
\end{lstlisting}

    --recovery\_steps 50 \
    --lora\_rank 8

\subsection{Enabling Inactive Components}

\begin{lstlisting}

python run\_all\_4\_nexus.py --max\_seq\_len 512 --stripe\_enabled true

# Enable RFF rectification
python run\_all\_4\_nexus.py --enable\_spectraformer true
\end{lstlisting}
