\section{Background}

\subsection{Standard Attention}

\[
\text{Attn}(Q, K, V) = \text{softmax}\!\left(\frac{Q K^\top}{\sqrt{d\_h}}\right) V
\tag{3.1}
\]

Complexity: $\mathcal{O}(N^2 d_h)$.

\subsection{Block Partitioning}

Partition $N$ tokens into $M = \lceil N / B_s \rceil$ blocks of size $B_s$. 
Token $t$ belongs to block $\lfloor t / B_s \rfloor$.

\subsection{XAttention Block Scoring}

Sample tokens at stride $\sigma$:

\[
\tilde{Q} = Q[:, :, ::\sigma, :], \quad \tilde{K} = K[:, :, ::\sigma, :]
\tag{3.2}
\]

Block score between blocks $b_i, b_j$:

\[
S\_{b\_i, b\_j} = \text{mean}(\tilde{Q}\_{[b\_i]} \tilde{K}\_{[b\_j]}^\top)
\tag{3.3}
\]

Top-$k$ key blocks per query block are retained.

\subsection{SeerAttention Gating}

Per-head query and key pooling:

\[
\bar{q}\_h = \text{mean}\_t(W\_q q\_t), \quad \bar{k}\_h = \text{mean}\_t(W\_k k\_t)
\tag{3.4}
\]

Gate:

\[
g\_h = \sigma(\bar{q}\_h \cdot \bar{k}\_h / T)
\tag{3.5}
\]

Keep count per head:

\[
k\_h = \text{clamp}(g\_h \cdot M, 1, M)
\tag{3.6}
\]

\subsection{DuSA Dual-Stage}

Intra-block (diagonal) is dense:

\[
\text{Out}\_\text{intra}[i] = \text{softmax}\!\left(\frac{q\_i K\_{B(i)}^\top}{\sqrt{d\_h}}\right) V\_{B(i)}
\tag{3.7}
\]

Inter-block top-$k$:

\[
\text{Out}\_\text{inter}[i] = \text{softmax}\!\left(\frac{q\_i K\_{\mathcal{S}\_i}^\top}{\sqrt{d\_h}}\right) V\_{\mathcal{S}\_i}
\tag{3.8}
\]

\subsection{AnchorAttention Stripe}

Token $t$ belongs to stripe group $\lfloor t / S_w \rfloor$. Mask keeps 
stripe groups within distance 1:

\[
\text{stripe\_mask}[i, j] = \left|\lfloor i / S\_w \rfloor - \lfloor j / S\_w \rfloor\right| \leq 1
\tag{3.9}
\]

\subsection{Spectraformer RFF}

RFF feature map:

\[
\phi(x) = \frac{1}{\sqrt{F}} \exp(W\_\text{rff} \hat{x})
\tag{3.10}
\]

where $\hat{x} = x / \|x\|_2$. Global attention:

\[
\text{Out}[i] = \frac{\phi(q\_i)^\top \sum\_j \phi(k\_j) v\_j^\top}{\phi(q\_i)^\top \sum\_j \phi(k\_j)}
\tag{3.11}
\]

\subsection{Softpick}

Rectified softmax:

\[
\text{softpick}(x)\_i = \frac{\text{ReLU}(e^{x\_i} - 1)}{\sum\_j \text{ReLU}(e^{x\_j} - 1)}
\tag{3.12}
\]

\textbf{Property.} For $x_i < 0$, the numerator is 0. This produces natural 
sparsity (tokens with negative logits get zero weight).
