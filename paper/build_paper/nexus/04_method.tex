\section{Method}

\subsection{Overview}

NEXUS combines six sparse-attention mechanisms in a single forward pass:

\begin{lstlisting}
        Input X
           |
            
    +---------------------+
    | Q/K/V projections   |
    | + RoPE              |
    +----------+----------+
               |
    +---------- ----------+
    | Block partition     |
    +----------+----------+
               |
    +---------- ----------+
    | XAttention scoring  |
    |   top-k blocks      |
    +----------+----------+
               |
    +---------- ----------+
    | Seer gating (learn) |
    |   add/override      |
    +----------+----------+
               |
    +---------- ----------+
    | Dual-stage mask     |
    |   intra: dense      |
    |   inter: top-k      |
    +----------+----------+
               |
    +---------- ----------+
    | Stripe mask         |
    |   (if N > 256)      |
    +----------+----------+
               |
    +---------- ----------+
    | Softpick            |
    +----------+----------+
               |
    +---------- ----------+
    | RFF rectification   |
    |   (every N layers)  |
    +----------+----------+
               |
    +---------- ----------+
    | out_proj            |
    +----------+----------+
               |
        Output
\end{lstlisting}

\subsection{Block Partitioning}

\begin{lstlisting}
bs = self.cfg.block_size
nb = (N + bs - 1) // bs
\end{lstlisting}

\subsection{XAttention Block Scoring}

\begin{lstlisting}
Q_sample = Q[:, :, ::stride, :]
K_sample = K[:, :, ::stride, :]
attn_sample = torch.matmul(Q_sample, K_sample.transpose(-2, -1)) * scale

for bi in range(n_blocks):
    for bj in range(n_blocks):
        block = attn_sample[:, :, q_start:q_end, k_start:k_end]
        scores[:, :, bi, bj] = block.mean(dim=(2, 3))
\end{lstlisting}

\subsection{Top-K Block Selection}

\begin{lstlisting}
scores_inter = scores.clone()
scores_inter[:, :, diag, diag] = -inf
_, idx = scores_inter.topk(top_k, dim=-1)
mask = zeros_like(scores).scatter_(-1, idx, True)
mask[:, :, diag, diag] = True  # Always include diagonal
\end{lstlisting}

\subsection{Seer Gating}

\begin{lstlisting}
Q_pooled = Q.mean(dim=2)
K_pooled = K.mean(dim=2)
q_gate = self.q_pool(Q_pooled)  # [B, H, gate_rank]
k_gate = self.k_pool(K_pooled)
gate_score = torch.sigmoid(q_gate @ k_gate.T / T)  # [B, H]
keep_per_head = clamp(gate_score * nb, 1, nb)
\end{lstlisting}

\subsection{Combined Mask}

\begin{lstlisting}
inter_mask = inter_mask | seer_mask
inter_mask[:, :, :, 0] = True  # CLS block
inter_mask[:, :, diag, diag] = True
\end{lstlisting}

\subsection{Stripe Mask}

\begin{lstlisting}
if self.cfg.stripe_enabled and N > 256:
    stripe_group = idx // sw
    stripe_mask = (q_stripe - k_stripe).abs() <= 1
    scores = scores.masked_fill(~stripe_mask, -10000.0)
\end{lstlisting}

\textbf{Note.} For $N \leq 256$, this is inactive.

\subsection{Softpick}

\begin{lstlisting}
def softpick(x, dim=-1, temperature=1.0):
    x_scaled = torch.clamp(x.float() / temp, max=85.0)
    p = F.relu(torch.exp(x_scaled) - 1.0)
    denom = p.sum(dim=-1, keepdim=True) + 1e-8
    return (p / denom).to(orig_dtype)
\end{lstlisting}

\textbf{Concern.} For all-negative inputs, \texttt{p=0}, \texttt{denom=1e-8}, output $\approx$ 0. 
This can cause attention collapse in masked rows.

\subsection{RFF Rectification}

\begin{lstlisting}
Q_proj = torch.matmul(F.normalize(Q), proj).clamp(max=11.0)
K_proj = torch.matmul(F.normalize(K), proj).clamp(max=11.0)
Q_feat = torch.exp(Q_proj) / sqrt(F)
K_feat = torch.exp(K_proj) / sqrt(F)
KV = einsum("bhnf,bhnd->bhfd", K_feat, V)
Z = K_feat.sum(dim=2)
ctx = einsum("bhnf,bhfd->bhnd", Q_feat, KV)
norm = einsum("bhnf,bhf->bhn", Q_feat, Z)
return ctx / (norm + 1e-5)
\end{lstlisting}

\textbf{Note.} Only active when \texttt{layer\_idx \% rectify\_interval == 0} AND (\texttt{pretrained\_attn is None} OR \texttt{enable\_spectraformer=True}).

\subsection{Full Algorithm}

\textbf{Algorithm 1:} NEXUS forward
\begin{lstlisting}
Input:  X, attention_mask, position_embeddings, is_decoder
Output: Y

 1: Q, K, V $\leftarrow$ projections(X)
 2: if RoPE: (Q, K) $\leftarrow$ apply_rotary(Q, K)
 3: if GQA: K, V $\leftarrow$ repeat_kv(K, V)
 4: token_valid $\leftarrow$ extract_valid_mask(attention_mask)
 5: use_dense $\leftarrow$ is_causal OR is_cross OR N $\le$ 8
 6: if use_dense:
 7:     return standard_attention(Q, K, V)
 8: if is_rectify and spec_rectify:
 9:     return spec_rectify(Q, K, V)
10: block_scores $\leftarrow$ XAttention_score(Q, K)
11: inter_mask $\leftarrow$ topk(block_scores)
12: if seer_gate: seer_mask $\leftarrow$ seer_gate(Q, K)
13: inter_mask $\leftarrow$ inter_mask | seer_mask
14: inter_mask[:, :, :, 0] $\leftarrow$ True
15: inter_mask[:, :, diag, diag] $\leftarrow$ True
16: scores $\leftarrow$ QK^T / $\sqrt{}$d_h
17: scores $\leftarrow$ expand_mask(inter_mask, scores)
18: if N > 256: scores $\leftarrow$ stripe_mask(scores)
19: scores $\leftarrow$ padding_mask(scores, token_valid)
20: attn $\leftarrow$ softpick(scores)
21: context $\leftarrow$ attn @ V
22: context $\leftarrow$ mask_padding(context)
23: return out_proj(context)
\end{lstlisting}

\subsection{Complexity Analysis}

\subsubsection{Block-Sparse Complexity}

\begin{itemize}
\item XAttention scoring: $\mathcal{O}((N/\sigma)^2 d_h)$
\item Top-K: $\mathcal{O}(M^2 \log k)$
\item Block attention: $\mathcal{O}(N \cdot k \cdot B_s \cdot d_h)$
\item \textbf{Total:} $\mathcal{O}(N k B_s d_h)$

\end{itemize}
For $k B_s \ll N$: \textbf{sub-quadratic}.

\subsubsection{Dense Fallback}

For causal, cross, or short sequences: $\mathcal{O}(N^2 d_h)$.

\subsection{Theoretical Notes}

\subsubsection{Softpick Sparsity}

\textbf{Claim.} softpick produces sparsity when logits are negative.

\textbf{Proof.} For $x_i < 0$: $e^{x_i} < 1$, so $e^{x_i} - 1 < 0$, ReLU gives 0. $\square$

\textbf{Concern.} When \textbf{all} logits are negative, softpick collapses to uniform.

\subsubsection{Causality}

\textbf{Claim.} For decoders, causality is preserved via dense fallback.

\textbf{Proof.} The \texttt{use\_dense} path applies a standard triangular mask. $\square$
