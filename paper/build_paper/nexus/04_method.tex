\section{Method}

\subsection{Overview}

NEXUS combines six sparse-attention mechanisms in a single forward pass:

\begin{lstlisting}
        Input X
           |
            
    +---------------------+
    | Q/K/V projections   |
    | + RoPE              |
    +----------+----------+
               |
    +---------- ----------+
    | Block partition     |
    +----------+----------+
               |
    +---------- ----------+
    | XAttention scoring  |
    |   top-k blocks      |
    +----------+----------+
               |
    +---------- ----------+
    | Seer gating (learn) |
    |   add/override      |
    +----------+----------+
               |
    +---------- ----------+
    | Dual-stage mask     |
    |   intra: dense      |
    |   inter: top-k      |
    +----------+----------+
               |
    +---------- ----------+
    | Stripe mask         |
    |   (if N \textgreater{} 256)      |
    +----------+----------+
               |
    +---------- ----------+
    | Softpick            |
    +----------+----------+
               |
    +---------- ----------+
    | RFF rectification   |
    |   (every N layers)  |
    +----------+----------+
               |
    +---------- ----------+
    | out\_proj            |
    +----------+----------+
               |
        Output
\end{lstlisting}

\subsection{Block Partitioning}

\begin{lstlisting}
bs = self.cfg.block\_size
nb = (N + bs - 1) // bs
\end{lstlisting}

\subsection{XAttention Block Scoring}

\begin{lstlisting}
Q\_sample = Q[:, :, ::stride, :]
K\_sample = K[:, :, ::stride, :]
attn\_sample = torch.matmul(Q\_sample, K\_sample.transpose(-2, -1)) * scale

for bi in range(n\_blocks):
    for bj in range(n\_blocks):
        block = attn\_sample[:, :, q\_start:q\_end, k\_start:k\_end]
        scores[:, :, bi, bj] = block.mean(dim=(2, 3))
\end{lstlisting}

\subsection{Top-K Block Selection}

\begin{lstlisting}
scores\_inter = scores.clone()
scores\_inter[:, :, diag, diag] = -inf
\_, idx = scores\_inter.topk(top\_k, dim=-1)
mask = zeros\_like(scores).scatter\_(-1, idx, True)
mask[:, :, diag, diag] = True  # Always include diagonal
\end{lstlisting}

\subsection{Seer Gating}

\begin{lstlisting}
Q\_pooled = Q.mean(dim=2)
K\_pooled = K.mean(dim=2)
q\_gate = self.q\_pool(Q\_pooled)  # [B, H, gate\_rank]
k\_gate = self.k\_pool(K\_pooled)
gate\_score = torch.sigmoid(q\_gate @ k\_gate.T / T)  # [B, H]
keep\_per\_head = clamp(gate\_score * nb, 1, nb)
\end{lstlisting}

\subsection{Combined Mask}

\begin{lstlisting}
inter\_mask = inter\_mask | seer\_mask
inter\_mask[:, :, :, 0] = True  # CLS block
inter\_mask[:, :, diag, diag] = True
\end{lstlisting}

\subsection{Stripe Mask}

\begin{lstlisting}
if self.cfg.stripe\_enabled and N \textgreater{} 256:
    stripe\_group = idx // sw
    stripe\_mask = (q\_stripe - k\_stripe).abs() \textless{}= 1
    scores = scores.masked\_fill(~stripe\_mask, -10000.0)
\end{lstlisting}

\textbf{Note.} For $N \leq 256$, this is inactive.

\subsection{Softpick}

\begin{lstlisting}
def softpick(x, dim=-1, temperature=1.0):
    x\_scaled = torch.clamp(x.float() / temp, max=85.0)
    p = F.relu(torch.exp(x\_scaled) - 1.0)
    denom = p.sum(dim=-1, keepdim=True) + 1e-8
    return (p / denom).to(orig\_dtype)
\end{lstlisting}

\textbf{Concern.} For all-negative inputs, \texttt{p=0}, \texttt{denom=1e-8}, output $\approx$ 0. 
This can cause attention collapse in masked rows.

\subsection{RFF Rectification}

\begin{lstlisting}
Q\_proj = torch.matmul(F.normalize(Q), proj).clamp(max=11.0)
K\_proj = torch.matmul(F.normalize(K), proj).clamp(max=11.0)
Q\_feat = torch.exp(Q\_proj) / sqrt(F)
K\_feat = torch.exp(K\_proj) / sqrt(F)
KV = einsum("bhnf,bhnd-\textgreater{}bhfd", K\_feat, V)
Z = K\_feat.sum(dim=2)
ctx = einsum("bhnf,bhfd-\textgreater{}bhnd", Q\_feat, KV)
norm = einsum("bhnf,bhf-\textgreater{}bhn", Q\_feat, Z)
return ctx / (norm + 1e-5)
\end{lstlisting}

\textbf{Note.} Only active when \texttt{layer\_idx \% rectify\_interval == 0} AND (\texttt{pretrained\_attn is None} OR \texttt{enable\_spectraformer=True}).

\subsection{Full Algorithm}

\textbf{Algorithm 1:} NEXUS forward
\begin{lstlisting}
Input:  X, attention\_mask, position\_embeddings, is\_decoder
Output: Y

 1: Q, K, V $\leftarrow$ projections(X)
 2: if RoPE: (Q, K) $\leftarrow$ apply\_rotary(Q, K)
 3: if GQA: K, V $\leftarrow$ repeat\_kv(K, V)
 4: token\_valid $\leftarrow$ extract\_valid\_mask(attention\_mask)
 5: use\_dense $\leftarrow$ is\_causal OR is\_cross OR N $\le$ 8
 6: if use\_dense:
 7:     return standard\_attention(Q, K, V)
 8: if is\_rectify and spec\_rectify:
 9:     return spec\_rectify(Q, K, V)
10: block\_scores $\leftarrow$ XAttention\_score(Q, K)
11: inter\_mask $\leftarrow$ topk(block\_scores)
12: if seer\_gate: seer\_mask $\leftarrow$ seer\_gate(Q, K)
13: inter\_mask $\leftarrow$ inter\_mask | seer\_mask
14: inter\_mask[:, :, :, 0] $\leftarrow$ True
15: inter\_mask[:, :, diag, diag] $\leftarrow$ True
16: scores $\leftarrow$ QK^T / $\sqrt{}$d\_h
17: scores $\leftarrow$ expand\_mask(inter\_mask, scores)
18: if N \textgreater{} 256: scores $\leftarrow$ stripe\_mask(scores)
19: scores $\leftarrow$ padding\_mask(scores, token\_valid)
20: attn $\leftarrow$ softpick(scores)
21: context $\leftarrow$ attn @ V
22: context $\leftarrow$ mask\_padding(context)
23: return out\_proj(context)
\end{lstlisting}

\subsection{Complexity Analysis}

\subsubsection{Block-Sparse Complexity}

\begin{itemize}
\item XAttention scoring: $\mathcal{O}((N/\sigma)^2 d_h)$
\item Top-K: $\mathcal{O}(M^2 \log k)$
\item Block attention: $\mathcal{O}(N \cdot k \cdot B_s \cdot d_h)$
\item \textbf{Total:} $\mathcal{O}(N k B_s d_h)$

\end{itemize}
For $k B_s \ll N$: \textbf{sub-quadratic}.

\subsubsection{Dense Fallback}

For causal, cross, or short sequences: $\mathcal{O}(N^2 d_h)$.

\subsection{Theoretical Notes}

\subsubsection{Softpick Sparsity}

\textbf{Claim.} softpick produces sparsity when logits are negative.

\textbf{Proof.} For $x_i < 0$: $e^{x_i} < 1$, so $e^{x_i} - 1 < 0$, ReLU gives 0. $\square$

\textbf{Concern.} When \textbf{all} logits are negative, softpick collapses to uniform.

\subsubsection{Causality}

\textbf{Claim.} For decoders, causality is preserved via dense fallback.

\textbf{Proof.} The \texttt{use\_dense} path applies a standard triangular mask. $\square$
