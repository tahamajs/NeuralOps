\section{Appendix B. Implementation Details}

\subsection{B.1 Core Forward}

\begin{lstlisting}
def forward(self, hidden\_states, attention\_mask=None, ...):
    B, N, \_ = hidden\_states.shape
    
    q = self.q\_proj(hidden\_states).view(B, N, H, d).transpose(1, 2)
    k = self.k\_proj(hidden\_states).view(B, N, H\_kv, d).transpose(1, 2)
    v = self.v\_proj(hidden\_states).view(B, N, H\_kv, d).transpose(1, 2)
    
    if position\_embeddings is not None:
        q, k = apply\_rotary\_pos\_emb(q, k, cos, sin)
    
    if self.num\_heads != self.num\_kv\_heads:
        k = repeat\_kv(k, self.num\_heads // self.num\_kv\_heads)
        v = repeat\_kv(v, self.num\_heads // self.num\_kv\_heads)
    
    scale = 1.0 / math.sqrt(self.head\_dim)
    scores = torch.matmul(q, k.transpose(-2, -1)) * scale
    
    invalid\_mask = torch.zeros(B, 1, N, N, dtype=torch.bool)
    if attention\_mask is not None:
        valid\_tok = get\_token\_valid\_mask(attention\_mask, B, N)
        invalid\_mask = invalid\_mask | (~valid\_tok.unsqueeze(1).unsqueeze(2))
    
    if self.is\_decoder:
        causal = torch.tril(torch.ones(N, N, dtype=torch.bool))
        invalid\_mask = invalid\_mask | (~causal.unsqueeze(0).unsqueeze(0))
    
    fill\_val = -1e4 if scores.dtype in (torch.float16, torch.bfloat16) else -1e9
    masked = scores.masked\_fill(invalid\_mask, fill\_val)
    max\_scores = masked.max(dim=-1, keepdim=True).values
    
    thresh = -5e3 if scores.dtype in (torch.float16, torch.bfloat16) else -1e8
    max\_scores = torch.where(max\_scores \textless{}= thresh, torch.zeros\_like(max\_scores), max\_scores)
    
    centered = (scores - max\_scores) / max(self.temperature, 1e-6)
    base = torch.clamp(1.0 + centered / self.order, min=0.0)
    attn = base ** self.order
    attn = attn.masked\_fill(invalid\_mask, 0.0)
    attn = attn / (attn.sum(dim=-1, keepdim=True) + 1e-8)
    
    if head\_mask is not None:
        attn = attn * head\_mask
    
    attn = self.dropout(attn)
    context = torch.matmul(attn, v)
    context = context.permute(0, 2, 1, 3).contiguous().view(B, N, self.all\_head\_size)
    
    if self.is\_nested\_encoder:
        return context, None
    return self.out\_proj(context), None
\end{lstlisting}

\subsection{B.2 Common Pitfalls}

\subsubsection{B.2.1 No Centering $\to$ Overflow}

Without max-centering, base can exceed FP16 max.

\subsubsection{B.2.2 No ReLU $\to$ NaN}

Negative base to fractional power gives NaN.

\subsubsection{B.2.3 All-Masked $\to$ NaN}

If a query has no valid keys, max = -1e4, base = 0, sum = 0 $\to$ NaN. 
Guard with \texttt{max = 0}.

\subsubsection{B.2.4 Wrong Mask Order}

Mask for max BEFORE polynomial; zero mask AFTER polynomial.

\subsubsection{B.2.5 Fragile RoPE Check}

The \texttt{'q\_rot' in locals()} check is fragile. Should compute \texttt{q\_rot, q\_pass} 
unconditionally.
