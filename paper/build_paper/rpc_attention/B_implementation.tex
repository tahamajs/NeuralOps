
\section{Appendix B. Implementation Details}

\subsection{B.1 The Module Structure}

\begin{lstlisting}
@ATTENTION\_REGISTRY.register("rpc")
class RPCAttention(BaseAttention):
    def \_\_init\_\_(self, config, rank=64, pretrained\_attn=None, **kwargs):
        super().\_\_init\_\_(config, rank)
        self.dim = 768
        self.num\_heads = 12
        self.head\_dim = 64
        self.rank = min(rank, self.head\_dim)
        
        # Projections with LoRA
        self.W\_q = LoRALinear(768, 768, lora\_rank=16)
        self.W\_k = LoRALinear(768, 768, lora\_rank=16)
        self.W\_v = LoRALinear(768, 768, lora\_rank=16)
        self.out\_proj = LoRALinear(768, 768, lora\_rank=16)
        
        # Random projection buffer (NOT a parameter)
        self.register\_buffer(
            "proj",
            torch.randn(self.num\_heads, self.head\_dim, self.rank) * (self.head\_dim ** -0.5)
        )
        
        if pretrained\_attn is not None:
            self.copy\_weights(pretrained\_attn)
\end{lstlisting}

\subsection{B.2 Forward Pass}

\begin{lstlisting}
def forward(self, hidden\_states, attention\_mask=None, **kwargs):
    B, L, D = hidden\_states.shape
    
    # 1. Project
    q = self.W\_q(hidden\_states).view(B, L, H, d\_h).permute(0, 2, 1, 3)
    k = self.W\_k(hidden\_states).view(B, L, H, d\_h).permute(0, 2, 1, 3)
    v = self.W\_v(hidden\_states).view(B, L, H, d\_h).permute(0, 2, 1, 3)
    
    # 2. RoPE
    if position\_embeddings is not None:
        q, k = apply\_rotary\_pos\_emb(q, k, *position\_embeddings)
    
    # 3. GQA
    if H != H\_kv:
        k = repeat\_kv(k, H // H\_kv)
        v = repeat\_kv(v, H // H\_kv)
    
    # 4. Sketch
    proj = self.proj.to(dtype=q.dtype, device=q.device)
    q\_sk = torch.einsum("bhld,hdr-\textgreater{}bhlr", q, proj)
    k\_sk = torch.einsum("bhld,hdr-\textgreater{}bhlr", k, proj)
    
    # 5. Score
    scale = 1.0 / math.sqrt(self.rank)
    scores = torch.matmul(q\_sk.float(), k\_sk.float().transpose(-2, -1)) * scale
    
    # 6. Masks
    if self.is\_decoder:
        causal\_mask = torch.triu(torch.full((L, L), -1e4), diagonal=1)
        scores = scores + causal\_mask
    
    if attention\_mask is not None:
        scores = apply\_attention\_mask(scores, attention\_mask)
    
    # 7. Attend
    probs = F.softmax(scores, dim=-1).to(hidden\_states.dtype)
    context = torch.matmul(probs, v)
    context = context.permute(0, 2, 1, 3).contiguous().view(B, L, self.all\_head\_size)
    
    # 8. Output
    if self.is\_nested\_encoder:
        out = context
    else:
        out = self.out\_proj(context)
    
    return out, None
\end{lstlisting}

\subsection{B.3 The Projection Buffer}

\begin{lstlisting}
self.register\_buffer(
    "proj",
    torch.randn(self.num\_heads, self.head\_dim, self.rank) * (self.head\_dim ** -0.5)
)
\end{lstlisting}

\textbf{Notes:}
\begin{itemize}
\item \texttt{randn} $\to$ Gaussian entries
\item \texttt{\textit{ (head\_dim }* -0.5)} $\to$ variance $1/d_h$
\item \texttt{register\_buffer} $\to$ saved with model, not counted in \texttt{parameters()}

\end{itemize}

\textbf{Advantages:}
\begin{itemize}
\item No silent fallback
\item Broken models not produced

\end{itemize}
\textbf{Disadvantages:}
\begin{itemize}
\item Single missing projection aborts injection
\item Less graceful than A3/AASVD

\end{itemize}
\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 Confusing Buffer with Parameter}

The \texttt{proj} buffer is not counted in \texttt{model.parameters()}. If you need 
to count it, use \texttt{model.state\_dict()}.

\subsubsection{B.5.2 RoPE Dimension Mismatch}

RoPE must be applied before sketching, since $\Omega$ assumes a specific 
head dimension.

\subsubsection{B.5.3 Scale Factor}

The scale is $1/\sqrt{r}$, not $1/\sqrt{d_h}$. This is because the sketched 
vectors $\tilde{Q}, \tilde{K}$ have variance $\|Q\|_2^2 \cdot r / d_h$ (not 
$\|Q\|_2^2$). The choice is subtle.

\subsubsection{B.5.4 Random Seed}

The projection is sampled once. Different seeds give different results. 
For reproducibility, set \texttt{torch.manual\_seed()} before initialization.
