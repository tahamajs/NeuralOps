\subsection{B.1 Core Forward Pass}

\begin{lstlisting}
def forward(self, hidden\_states, attention\_mask=None, **kwargs):
    B, L, D = hidden\_states.shape
    
    # 1. Project
    q = self.q\_lin(hidden\_states).view(B, L, H, Dh).transpose(1, 2)
    k = self.k\_lin(hidden\_states).view(B, L, H\_kv, Dh).transpose(1, 2)
    v = self.v\_lin(hidden\_states).view(B, L, H\_kv, Dh).transpose(1, 2)
    
    # 2. RoPE
    if position\_embeddings is not None:
        q, k = apply\_rotary\_pos\_emb(q, k, cos, sin)
    
    # 3. Compress
    k, v, top\_idx, comp\_mask = self.\_select\_compressed\_tokens(
        hidden\_states, k, v, attention\_mask
    )
    
    # 4. GQA
    if H\_kv != H:
        k = repeat\_kv(k, H // H\_kv)
        v = repeat\_kv(v, H // H\_kv)
    
    # 5. Attention with causal mask
    scores = torch.matmul(q.float(), k.float().transpose(-1, -2)) * scale
    if self.is\_decoder:
        q\_pos = torch.arange(L).view(1, 1, L, 1)
        k\_orig\_pos = top\_idx.unsqueeze(1).unsqueeze(2)
        causal\_mask = (k\_orig\_pos \textgreater{} q\_pos)
        scores = scores.masked\_fill(causal\_mask, float("-inf"))
    
    # 6. Padding mask
    if comp\_mask is not None:
        scores = apply\_attention\_mask(scores, comp\_mask)
    
    # 7. Softmax, context, output
    attn = F.softmax(scores, dim=-1)
    context = torch.matmul(attn, v)
    context = context.transpose(1, 2).contiguous().view(B, L, H * Dh)
    
    return self.out\_lin(context), None
\end{lstlisting}

\subsection{B.2 Score Computation}

\begin{lstlisting}
def \_select\_compressed\_tokens(self, x, k\_tensor, v\_tensor, mask):
    B, L, D = x.shape
    
    energy = torch.norm(x.float(), p=2, dim=-1)          # (B, L)
    learned = self.score\_net(x.float()).squeeze(-1)       # (B, L)
    total\_importance = energy + learned
    
    k\_keep = max(1, min(L, max(2, int(math.ceil(L * self.compression\_ratio)))))
    
    total\_importance[:, 0] = 1e4  # Pin [CLS]
    
    if mask is not None:
        if mask.dim() == 2:
            total\_importance = total\_importance.masked\_fill(mask == 0, -1e4)
        total\_importance[:, 0] = 1e4
    
    \_, top\_indices = torch.topk(total\_importance, k\_keep, dim=-1)
    top\_indices, \_ = torch.sort(top\_indices, dim=-1)
    
    top\_idx\_exp = top\_indices.unsqueeze(1).unsqueeze(-1).expand(-1, H\_kv, -1, Dh)
    k\_comp = torch.gather(k\_tensor, 2, top\_idx\_exp)
    v\_comp = torch.gather(v\_tensor, 2, top\_idx\_exp)
    
    return k\_comp, v\_comp, top\_indices, comp\_mask
\end{lstlisting}

\subsection{B.4 Common Pitfalls}

\subsubsection{B.4.1 Forgetting to Sort Indices}

Without \texttt{torch.sort}, causal masking still works but the order of keys is 
scrambled, causing numerical noise.

\subsubsection{B.4.2 Masking Before Compression}

The padding mask must be applied both to score selection \textbf{and} to the 
compressed attention. We do both.

\subsubsection{B.4.3 Wrong [CLS] Index}

For BERT-family models, \texttt{[CLS]} is at index 0. For other models, the "sink" 
token may vary. Our implementation assumes index 0.
