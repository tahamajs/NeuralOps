\section{Method}

\subsection{Overview}

HGRN2Layer replaces standard attention with a single element-wise recurrence. 
The pipeline is:

\begin{lstlisting}
    Input X $\in$ R^{B$\times$L$\times$d}
              |
       +------+------+
                      
    f\_gate  i\_gate  v\_proj
       |      |      |
       +------+------+
              |
        g = (1-f) * i * v
              |
               
    +-------------------+
    | Chunked scan      |
    | (FP32, chunk=32)  |
    +---------+---------+
              |
               
       h\_ctx = scan(f, g)
              |
               
    +-------------------+
    | h\_norm = LayerNorm|
    | gated = (0.5v +   |
    |   0.5h\_norm)*silu(z)|
    +---------+---------+
              |
               
        out\_proj
              |
               
           Output Y
\end{lstlisting}

\subsection{Gate Computation}

\textbf{Forget gate:}

\[
\log f\_t = -\text{softplus}(W\_f x\_t + b\_f)
\tag{4.1}
\]

\[
\log f\_t \leftarrow \text{clamp}(\log f\_t, -3, -10^{-4})
\tag{4.2}
\]

\[
f\_t = \exp(\log f\_t) \in [e^{-3}, 1] \approx [0.05, 1]
\tag{4.3}
\]

\textbf{Initialization.} $W_f$ is normal $N(0, 0.01)$, $b_f = -2.5$ giving 
$\text{softplus}(-2.5) \approx 0.076$, so $f_0 \approx e^{-0.076} \approx 0.93$ 
-- a slow decay.

\textbf{Input gate:}

\[
i\_t = \sigma(W\_i x\_t + b\_i)
\tag{4.4}
\]

\textbf{Value:}

\[
v\_t = W\_v x\_t
\tag{4.5}
\]

\textbf{Input contribution:}

\[
g\_t = (1 - f\_t) \odot i\_t \odot v\_t
\tag{4.6}
\]

\subsection{Chunked Parallel Scan}

We partition the sequence into chunks of size $C = 32$. Within each chunk, 
we compute the intra-chunk recurrence in parallel:

\textbf{Step 1:} Compute cumulative log-forget within the chunk:

\[
A\_k = \sum\_{j=1}^{k} \log f\_j
\tag{4.7}
\]

\textbf{Step 2:} Compute pairwise decay:

\[
D\_{k,j} = \exp(A\_k - A\_j) \quad \text{for } j \leq k
\tag{4.8}
\]

Zero for $j > k$.

\textbf{Step 3:} Intra-chunk state:

\[
h\_k^\text{intra} = \sum\_{j=1}^{k} D\_{k,j} \cdot g\_j
\tag{4.9}
\]

\textbf{Step 4:} Combine with previous chunk state:

\[
h\_k^\text{chunk} = h\_\text{prev} \cdot \exp(A\_k) + h\_k^\text{intra}
\tag{4.10}
\]

\textbf{Step 5:} Clip for stability:

\[
h\_k^\text{chunk} \leftarrow \text{clamp}(h\_k^\text{chunk}, -30, 30)
\tag{4.11}
\]

\textbf{Step 6:} Update $h_\text{prev} = h_C^\text{chunk}$ and proceed to next chunk.

\subsection{Bidirectional Recurrence}

For encoders, we compute both forward and backward scans:

\[
h\_t^\text{fwd} = \text{scan}(f\_{1:L}, g\_{1:L})\_t
\tag{4.12}
\]

\[
h\_t^\text{bwd} = \text{scan}(\text{flip}(f), \text{flip}(g))\_{L-t+1}
\tag{4.13}
\]

\[
h\_t = h\_t^\text{fwd} + h\_t^\text{bwd}
\tag{4.14}
\]

For decoders (causal), we use only the forward scan.

\subsection{Output Modulation}

The raw state $h_t$ is normalized and gated:

\[
h\_t^\text{norm} = \text{LayerNorm}(h\_t)
\tag{4.15}
\]

\[
\text{combined}\_t = 0.5 \cdot v\_t + 0.5 \cdot h\_t^\text{norm}
\tag{4.16}
\]

\[
\text{gated}\_t = \text{combined}\_t \odot \text{SiLU}(z\_t)
\tag{4.17}
\]

where $z_t = W_g x_t + b_g$ is a gating projection initialized with $b_g = 1.28$.

\textbf{Why combine with $v_t$?} Directly using $h_t$ can cause the output to 
drift far from the current value $v_t$. Averaging $0.5 v_t + 0.5 h_t$ 
preserves a self-value path and stabilizes training.

\textbf{Why SiLU gating?} A nonlinear gating on the combined signal allows the 
model to suppress or amplify the recurrence.

\subsection{Output Projection}

\[
\text{out} = W\_o \text{gated} + b\_o + \text{LoRA}(\text{gated})
\tag{4.18}
\]

\subsection{Algorithm}

\textbf{Algorithm 1:} HGRN2Layer forward pass
\begin{lstlisting}
Input:  X $\in$ R^{B$\times$L$\times$d}, attention\_mask
        chunk\_size C = 32
Output: Y $\in$ R^{B$\times$L$\times$d}

 1: f $\leftarrow$ exp(clamp(-softplus(W\_f X + b\_f), -3, -1e-4))
 2: i $\leftarrow$ sigmoid(W\_i X + b\_i)
 3: v $\leftarrow$ W\_v X
 4: z $\leftarrow$ W\_g X + b\_g
 5: g $\leftarrow$ (1 - f)   i   v
 6: h\_fwd $\leftarrow$ chunked\_scan(f, g, C)
 7: if not is\_decoder:
 8:     h\_bwd $\leftarrow$ flip(chunked\_scan(flip(f), flip(g), C))
 9:     h $\leftarrow$ h\_fwd + h\_bwd
10: else:
11:     h $\leftarrow$ h\_fwd
12: h\_norm $\leftarrow$ LayerNorm(h)
13: gated $\leftarrow$ (0.5 v + 0.5 h\_norm)   SiLU(z)
14: Y $\leftarrow$ W\_o gated + b\_o + LoRA(gated)
15: return Y
\end{lstlisting}

\subsection{Complexity Analysis}

For a single layer:

\begin{table}[H]
\centering
\small
\begin{tabular}{lll}
\toprule
\textbf{Component} & \textbf{Time} & \textbf{Memory} \\
\midrule
Forget gate & $\mathcal{O}(L d^2)$ & $\mathcal{O}(L d)$ \\
Input gate & $\mathcal{O}(L d^2)$ & $\mathcal{O}(L d)$ \\
Value & $\mathcal{O}(L d^2)$ & $\mathcal{O}(L d)$ \\
Gate $z$ & $\mathcal{O}(L d^2)$ & $\mathcal{O}(L d)$ \\
Chunked scan & $\mathcal{O}(L d)$ & $\mathcal{O}(C^2 + L d)$ \\
Output & $\mathcal{O}(L d^2)$ & $\mathcal{O}(L d)$ \\
\textbf{Total} & $\mathcal{O}(L d^2)$ & $\mathcal{O}(L d)$ \\
\bottomrule
\end{tabular}
\end{table}

Compare to attention: $\mathcal{O}(L^2 d)$. Our method is \textbf{linear} in $L$.

\subsection{Training}

Trainable parameters:
\begin{itemize}
\item Forget gate $W_f, b_f$
\item Input gate $W_i, b_i$
\item Gate $W_g, b_g$
\item LoRA on $W_v$ and $W_o$

\end{itemize}
Frozen:
\begin{itemize}
\item $W_v, W_o$ base weights
\item All LoRA base weights from pretrained attention

\end{itemize}
\textbf{Trainable count:} for $d = 768$, $r_L = 16$:
\begin{itemize}
\item Gates: $\sim 2.4M$
\item LoRA: $\sim 98K$
\item Total: $\sim 2.5M$ per layer
\end{itemize}
