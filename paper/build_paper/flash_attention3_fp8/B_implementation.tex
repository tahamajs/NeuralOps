
\section{Appendix B. Implementation Details}

\subsection{B.1 FP8 Import Detection}

\begin{lstlisting}
try:
    from flash\_attn\_interface import flash\_attn\_func
    FLASH\_ATTN\_AVAILABLE = True
except ImportError:
    try:
        from flash\_attn import flash\_attn\_func
        FLASH\_ATTN\_AVAILABLE = True
    except ImportError:
        FLASH\_ATTN\_AVAILABLE = False
\end{lstlisting}

Two import paths: \texttt{flash\_attn\_interface} (Hopper-optimized) and \texttt{flash\_attn} 
(community).

\subsection{B.2 FP8 Forward Path}

\begin{lstlisting}
if FLASH\_ATTN\_AVAILABLE and self.use\_fp8 and q.is\_cuda and hasattr(torch, "float8\_e4m3fn"):
    try:
        q\_fa = q.transpose(1, 2).contiguous().to(torch.float8\_e4m3fn)
        k\_fa = k.transpose(1, 2).contiguous().to(torch.float8\_e4m3fn)
        v\_fa = v.transpose(1, 2).contiguous().to(torch.float8\_e4m3fn)
        out = flash\_attn\_func(q\_fa, k\_fa, v\_fa, causal=causal\_flag).to(dtype)
        out = out.contiguous().view(B, L, self.all\_head\_size)
    except Exception:
        out = None
\end{lstlisting}

\subsection{B.3 SDPA Fallback}

\begin{lstlisting}
if out is None:
    if causal\_flag and attn\_mask is None:
        out\_sdpa = F.scaled\_dot\_product\_attention(
            q, k, v, attn\_mask=None, is\_causal=True, dropout\_p=self.dropout.p
        )
    else:
        # Combine causal and padding
        if causal\_flag:
            causal\_m = torch.triu(
                torch.full((L, kv\_len), float("-inf"), device=device),
                diagonal=1
            )
            attn\_mask = (attn\_mask + causal\_m.unsqueeze(0).unsqueeze(0)) \
                        if attn\_mask is not None else causal\_m.unsqueeze(0).unsqueeze(0)
        out\_sdpa = F.scaled\_dot\_product\_attention(
            q, k, v, attn\_mask=attn\_mask, is\_causal=False, dropout\_p=self.dropout.p
        )
    out = out\_sdpa.transpose(1, 2).contiguous().view(B, L, self.all\_head\_size)
\end{lstlisting}

\subsection{B.7 Nested vs Flat Detection}

\begin{lstlisting}
if hasattr(src, "out\_lin"):        # DistilBERT
    is\_nested = False
elif hasattr(src, "self") and hasattr(src.self, "query"):  # RoBERTa
    is\_nested = True
elif cls\_name in ("BertSelfAttention", "RobertaSelfAttention"):
    is\_nested = True
else:
    is\_nested = False
\end{lstlisting}

\textbf{Critical:} Check \texttt{out\_lin} first. DistilBERT's \texttt{MultiHeadSelfAttention} 
ends with "SelfAttention" and would be misclassified otherwise.

\subsection{B.8 Common Pitfalls}

\subsubsection{B.8.1 Missing FP8 Detection}

Without detection:

\begin{lstlisting}
# WRONG -- crashes on RTX 4070
out = flash\_attn\_func(q.to(torch.float8\_e4m3fn), ...)

# CORRECT
if FLASH\_ATTN\_AVAILABLE and hasattr(torch, "float8\_e4m3fn"):
    ...
\end{lstlisting}

\subsubsection{B.8.2 Wrong Conv1D Orientation}

\begin{lstlisting}
# WRONG
w = src.c\_attn.weight.data  # (768, 2304) for GPT-2
q\_w, k\_w, v\_w = w.chunk(3, dim=0)  # wrong shapes

# CORRECT
w = src.c\_attn.weight.data.t()  # (2304, 768)
q\_w, k\_w, v\_w = w.chunk(3, dim=0)
\end{lstlisting}

\subsubsection{B.8.3 Missing causal\_flag}

\begin{lstlisting}
# WRONG -- applies causal to cross-attention
causal\_flag = self.is\_decoder

# CORRECT
causal\_flag = self.is\_decoder and not is\_cross\_attention and L == kv\_len
\end{lstlisting}

\subsection{B.11 Injection Factory}

\begin{lstlisting}
@ATTENTION\_REGISTRY.register("flash3\_fp8\_inject")
def inject\_flash3\_fp8(layer, model\_config, comp\_config, old\_attn=None, **kwargs):
    src\_attn = old\_attn or extract\_attention(layer)
    try:
        new\_attn = FlashAttention3FP8(config=model\_config, old\_attn=src\_attn)
        replace\_attention(layer, new\_attn)
        return new\_attn
    except Exception as e:
        logger.error(f"Injection failed: {e}")
        return None
\end{lstlisting}
