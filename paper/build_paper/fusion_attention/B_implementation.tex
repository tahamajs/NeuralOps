\subsection{B.3 MoBA Block Attention}

\begin{lstlisting}
def \_moba\_block\_attention(self, q, k, v, gate\_scores, valid\_mask, is\_causal):
    B, H, N, D = q.shape
    
    # Pad
    pad\_len = (self.block\_size - (N \% self.block\_size)) \% self.block\_size
    q\_pad = F.pad(q, (0, 0, 0, pad\_len)) if pad\_len \textgreater{} 0 else q
    k\_pad = F.pad(k, (0, 0, 0, pad\_len)) if pad\_len \textgreater{} 0 else k
    v\_pad = F.pad(v, (0, 0, 0, pad\_len)) if pad\_len \textgreater{} 0 else v
    
    num\_blocks = q\_pad.size(2) // self.block\_size
    top\_k = min(self.top\_k\_blocks, num\_blocks)
    
    \_, top\_indices = torch.topk(gate\_scores, top\_k, dim=-1)
    top\_indices[:, :, :, 0] = 0  # Always block 0
    
    # Reshape
    q\_blocks = q\_pad.reshape(B, H, num\_blocks, self.block\_size, D)
    k\_blocks = k\_pad.reshape(B, H, num\_blocks, self.block\_size, D)
    v\_blocks = v\_pad.reshape(B, H, num\_blocks, self.block\_size, D)
    
    # Attention per query block
    out\_blocks = []
    for b in range(num\_blocks):
        idx\_b = top\_indices[:, :, b, :]
        idx\_exp = idx\_b.unsqueeze(-1).unsqueeze(-1).expand(B, H, top\_k, self.block\_size, D)
        k\_sel = torch.gather(k\_blocks, 2, idx\_exp).reshape(B, H, top\_k * self.block\_size, D)
        v\_sel = torch.gather(v\_blocks, 2, idx\_exp).reshape(B, H, top\_k * self.block\_size, D)
        
        q\_curr = q\_blocks[:, :, b, :, :]
        attn\_b = torch.matmul(q\_curr.float(), k\_sel.float().transpose(-2, -1)) / math.sqrt(D)
        
        if is\_causal:
            causal\_b = torch.triu(torch.full((self.block\_size, top\_k * self.block\_size), float("-inf")), diagonal=1)
            attn\_b = attn\_b + causal\_b.unsqueeze(0).unsqueeze(0)
        
        if valid\_mask is not None:
            pad\_m = (~valid\_mask).unsqueeze(1).unsqueeze(2)
            if pad\_m.shape[-1] \textless{} k\_sel.shape[2]:
                pad\_m = F.pad(pad\_m, (0, k\_sel.shape[2] - pad\_m.shape[-1]), value=True)
            attn\_b = attn\_b.masked\_fill(pad\_m[:, :, :, :k\_sel.shape[2]], float("-inf"))
        
        attn\_b = F.softmax(attn\_b, dim=-1)
        attn\_b = torch.nan\_to\_num(attn\_b, nan=0.0)
        attn\_b = self.dropout(attn\_b)
        out\_b = torch.matmul(attn\_b.to(v\_sel.dtype), v\_sel)
        out\_blocks.append(out\_b)
    
    out\_pad = torch.cat(out\_blocks, dim=2)
    return out\_pad[:, :, :N, :]
\end{lstlisting}

\subsection{B.4 Common Pitfalls}

\subsubsection{B.4.1 Threshold too High}

If threshold \textgreater{} test L, sparsity never activates.

\subsubsection{B.4.2 Gate Temperature Collapse}

Clamp gate\_temp to avoid division by zero.

\subsubsection{B.4.3 Padding Mask Shape}

Padding mask must be padded to match selected block length.

\subsubsection{B.4.4 Conv1D Transpose}

GPT-2 Conv1D stores (in, out). Must transpose.

\subsubsection{B.4.5 Nested Encoder Bypass}

RoBERTa: do not apply out\_proj.
