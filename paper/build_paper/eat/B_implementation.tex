\subsection{B.1 The EATBlock Module}

\begin{lstlisting}
@ATTENTION\_REGISTRY.register("eat")
@ATTENTION\_REGISTRY.register("eat\_attention")
class EATBlock(BaseAttention):
    def \_\_init\_\_(self, config, rank=64, ..., prune\_layer=None,
                 pretrained\_attn=None, ...):
        super().\_\_init\_\_(config, rank)
        self.q\_lin = LoRALinear(d, H*d\_h, lora\_rank=16)
        self.k\_lin = LoRALinear(d, H\_kv*d\_h, lora\_rank=16)
        self.v\_lin = LoRALinear(d, H\_kv*d\_h, lora\_rank=16)
        self.out\_lin = LoRALinear(H*d\_h, d, lora\_rank=16)
        self.prune\_layer = prune\_layer
        if pretrained\_attn is not None:
            self.copy\_weights(pretrained\_attn)
    
    def forward(self, X, attention\_mask=None, head\_mask=None,
                encoder\_hidden\_states=None, position\_embeddings=None, **kwargs):
        B, L, D = X.shape
        # 1. Project
        q = self.q\_lin(X).view(B, L, H, d\_h).transpose(1, 2)
        k = self.k\_lin(X).view(B, L, H\_kv, d\_h).transpose(1, 2)
        v = self.v\_lin(X).view(B, L, H\_kv, d\_h).transpose(1, 2)
        # 2. RoPE
        if position\_embeddings is not None:
            q, k = apply\_rotary\_pos\_emb(q, k, *position\_embeddings)
        # 3. GQA
        if self.kv\_repeat \textgreater{} 1:
            k = repeat\_kv(k, self.kv\_repeat)
            v = repeat\_kv(v, self.kv\_repeat)
        # 4. Attention
        scores = torch.matmul(q.float(), k.float().transpose(-2, -1)) * scale
        if self.is\_decoder:
            scores = scores + causal\_mask
        if attention\_mask is not None:
            scores = scores + pad\_mask
        probs = F.softmax(scores, dim=-1)
        attn\_out = torch.matmul(probs.to(v.dtype), v)
        attn\_out = attn\_out.transpose(1, 2).contiguous().view(B, L, H*d\_h)
        # 5. Output projection
        if self.is\_nested\_encoder:
            out = attn\_out
        else:
            out = self.out\_lin(attn\_out)
        # 6. Optional pruning (inference only)
        if self.prune\_layer is not None and not self.training:
            out, \_ = self.prune\_layer(out)
        return out, None
\end{lstlisting}

\subsection{B.2 The LoRALinear Module}

\begin{lstlisting}
class LoRALinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, bias=True,
                 lora\_rank=16, lora\_alpha=32.0, freeze\_base=False):
        super().\_\_init\_\_()
        self.weight = nn.Parameter(torch.empty(out\_features, in\_features))
        if bias:
            self.bias = nn.Parameter(torch.empty(out\_features))
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.empty(lora\_rank, in\_features))
            self.lora\_B = nn.Parameter(torch.zeros(out\_features, lora\_rank))
        self.reset\_parameters()
        if freeze\_base:
            self.freeze\_base\_weights()
    
    def forward(self, x):
        out = F.linear(x, self.weight, self.bias)
        if self.lora\_rank \textgreater{} 0:
            lora = F.linear(F.linear(x, self.lora\_A), self.lora\_B) * self.lora\_scaling
            out = out + lora
        return out
\end{lstlisting}

\subsection{B.4 The Silent Failure Path}

\textbf{Critical code path} (in \texttt{copy\_weights}):

\begin{lstlisting}
missing\_after = [k for k, (w, \_) in extracted.items() 
                 if w is None and (k != "o" or not self.is\_nested\_encoder)]
if missing\_after:
    logger.warning(f"EATBlock copy\_weights: missing projections {missing\_after} "
                   f"from {pretrained\_attn.\_\_class\_\_.\_\_name\_\_}; "
                   f"proceeding with random init for those")
    # $\leftarrow$ continues, doesn't raise
\end{lstlisting}

This is the origin of the "silent failure" problem. We recommend adding:

\begin{lstlisting}
if missing\_after and os.environ.get("EAT\_STRICT", "0") == "1":
    raise RuntimeError(f"Missing projections: {missing\_after}")
\end{lstlisting}

\subsection{B.5 Post-Injection Sanity Check}

\begin{lstlisting}
def sanity\_check(model, device, seq\_len=32):
    model.eval()
    ids = torch.randint(0, 1000, (2, seq\_len), device=device)
    with torch.no\_grad():
        out = model(ids).logits
    std = out.std(dim=0).mean().item()
    uniq = out.argmax(-1).unique().numel()
    if std \textless{} 1e-3 or uniq \textless{} 2:
        raise RuntimeError(f"Collapse detected: std={std}, uniq={uniq}")
\end{lstlisting}

Run this after every injection.

\subsection{B.6 Common Pitfalls}

\subsubsection{B.6.1 Forgetting `is\_nested\_encoder`}

For RoBERTa, must be \texttt{True}. Otherwise double projection.

\subsubsection{B.6.2 Missing Conv1D Transpose}

GPT-2's Conv1D stores weights as \texttt{(in, out)}. Must transpose.
