\section{Introduction}

\subsection{Motivation}

The transformer's parameter count is dominated by four weight matrices 
per layer: $W_Q, W_K, W_V, W_O \in \mathbb{R}^{d \times d}$ for self-attention, 
plus two FFN projections. For a $d = 768$, 12-layer model, this amounts 
to ~85M parameters -- most of the model.

Two orthogonal strategies can reduce this:

\begin{enumerate}
\item \textbf{Low-rank factorization:} Replace each $W \in \mathbb{R}^{d_\text{out} \times d_\text{in}}$
  with a product $W_\text{up} W_\text{down}$ where $W_\text{down} \in \mathbb{R}^{r \times d_\text{in}}$
  and $W_\text{up} \in \mathbb{R}^{d_\text{out} \times r}$. The parameter
  count drops from $d_\text{in} d_\text{out}$ to $r(d_\text{in} + d_\text{out})$.

\item \textbf{Efficient attention:} Replace the $\mathcal{O}(L^2)$ softmax attention
  with a linear or sparse alternative.

\end{enumerate}
Most methods address one or the other. \textbf{ORION combines both.}

\subsection{Our Approach}

ORION has three distinct components:

\subsubsection{Zero-Shot SVD Factorization}

Each Q/K/V/O projection is factorized via SVD at rank $r = 64$, with 
the top-$r$ singular vectors retained:

\[
W \approx U\_r \Sigma\_r V\_r^\top = (U\_r \Sigma\_r^{1/2})(\Sigma\_r^{1/2} V\_r^\top)
\]

The factors are frozen; LoRA adapters provide recovery.

\subsubsection{Head-Wise Bifurcation}

Attention heads are split into two groups:

\begin{itemize}
\item \textbf{Kernel heads} (2/3): linear attention via random Fourier features.
\item \textbf{Full heads} (1/3): full softmax-equivalent attention via softpick.

\end{itemize}
This is motivated by the observation that some heads learn low-rank 
patterns (kernel-friendly), while others require full attention.

\subsubsection{Random Fourier Features}

For the kernel heads:

\[
\phi(x) = \frac{1}{\sqrt{2F}} [\cos(\Omega x), \sin(\Omega x)]
\]

with $\Omega \in \mathbb{R}^{d_h \times F}$ orthogonally initialized.

\subsection{Contributions}

\begin{itemize}
\item \textbf{A tri-component hybrid attention} combining SVD factorization,
  head-wise bifurcation, and RFF.
\item \textbf{Genuine parameter reduction} (~6$\times$ for square projections, ~9.6$\times$ for
  FFN-like projections).
\item \textbf{An honest disclosure} of the \texttt{copy\_weights} raise-bug that can cause
  catastrophic collapse.
\item \textbf{Full reproducibility} including code and checkpoints.

\end{itemize}
\subsection{A Critical Bug Disclosure}

The \texttt{copy\_weights} method contains:

\begin{lstlisting}
except Exception as e:
    print(f'copy\_weights failed: {e}')
    raise RuntimeError(f"copy\_weights failed: {e}")
\end{lstlisting}

If \texttt{UniversalAdapter.get\_weight\_bias} fails for any of the four 
projections (which can happen for architectures it does not recognize), 
the entire injection aborts with a \texttt{RuntimeError}. This matches the 
pattern observed in other modules (AEGIS V4/V5, EAT) that caused 
catastrophic accuracy collapse.

\textbf{This is a bug, not a feature.} We document it here and in Section 8 
for reproducibility.

\section{Background}

\subsection{SVD Decomposition}

For $W \in \mathbb{R}^{m \times n}$:

\[
W = U \Sigma V^\top
\tag{3.1}
\]

with $U \in \mathbb{R}^{m \times m}$, $\Sigma \in \mathbb{R}^{m \times n}$, 
$V \in \mathbb{R}^{n \times n}$.

\textbf{Truncated:} $W_r = U_r \Sigma_r V_r^\top$. Error $\|W - W_r\|_F = \sqrt{\sum_{i > r} \sigma_i^2}$.

\textbf{Symmetric factorization:}

\[
W\_r = (U\_r \Sigma\_r^{1/2})(\Sigma\_r^{1/2} V\_r^\top)
\tag{3.2}
\]

with $W_\text{up} = U_r \Sigma_r^{1/2} \in \mathbb{R}^{m \times r}$, 
$W_\text{down} = \Sigma_r^{1/2} V_r^\top \in \mathbb{R}^{r \times n}$.

\textbf{Parameter count:} $r(m + n)$ vs $mn$.

\subsection{Energy-Based Rank Selection}

Given singular values $\sigma_1 \geq \sigma_2 \geq \dots$, the energy 
captured by the top-$r$:

\[
E(r) = \frac{\sum\_{i=1}^{r} \sigma\_i^2}{\sum\_i \sigma\_i^2}
\tag{3.3}
\]

ORION selects the smallest $r$ such that $E(r) \geq \eta$ (energy threshold, 
default 0.95).

\subsection{Random Fourier Features}

Approximate $\exp(q^\top k)$ via:

\[
\exp(q^\top k) \approx \phi(q)^\top \phi(k), \quad
\phi(x) = \frac{1}{\sqrt{2F}} [\cos(\Omega x), \sin(\Omega x)]
\tag{3.4}
\]

with $\Omega \in \mathbb{R}^{d_h \times F}$.

\subsection{Linear Attention with RFF}

\[
O\_i = \frac{\phi(q\_i)^\top \sum\_j \phi(k\_j) v\_j^\top}{\phi(q\_i)^\top \sum\_j \phi(k\_j)}
\tag{3.5}
\]

Cost: $\mathcal{O}(L F d)$.

\subsection{Softpick (Simplified)}

ORION's softpick:

\[
\text{softpick}(x)\_i = \frac{\text{ReLU}(e^{x\_i} - 1)}{\sum\_j \text{ReLU}(e^{x\_j} - 1) + \epsilon}
\tag{3.6}
\]

\textbf{Key property:} For $x_i = -10^4$ (masked), $e^{x_i} - 1 \approx -1$, 
$\text{ReLU}(-1) = 0$. Masked positions contribute zero.

\section{Method}

\subsection{Overview}

ORION replaces the standard attention block with:

\begin{enumerate}
\item \textbf{Factorized projections} ($W_\text{up} W_\text{down}$) instead of full
  matrices.
\item \textbf{Bifurcated heads}: kernel and full.
\item \textbf{RFF-based linear attention} for kernel heads.
\item \textbf{Softpick full attention} for full heads.

\end{enumerate}
\begin{lstlisting}
                Input X
                   |
        +----------+----------+
        |                     |
                               
    +-------+            +-------+
    |  Q    |            |  K/V  |
    | Factor|            | Factor|
    +---+---+            +---+---+
        |                    |
        +------+-------------+
               |
       +-------+--------+
       |                |
                         
   Kernel heads    Full heads
   (2/3)           (1/3)
       |                |
                         
   RFF linear      Softpick
   attention       full attention
       |                |
       +-------+--------+
               |
                
        Concatenate + out\_proj (factorized)
               |
                
            Output Y
\end{lstlisting}

\subsection{Factorized Projection}

For each projection $W \in \mathbb{R}^{d_\text{out} \times d_\text{in}}$:

\begin{enumerate}
\item Compute SVD: $W = U \Sigma V^\top$.
\item Energy threshold: select smallest $r$ with $E(r) \geq \eta$.
\item Set $W_\text{up} = U_{:, :r} \Sigma_{:r, :r}^{1/2}$, $W_\text{down} = \Sigma_{:r, :r}^{1/2} V_{:r, :}^\top$.
\item Store as \texttt{up} and \texttt{down} linear layers.

\end{enumerate}
\textbf{Default:} $\eta = 0.95$, min rank 4, max rank 64.

\textbf{Forward:}

\[
y = W\_\text{up}(W\_\text{down} x) + b
\tag{4.1}
\]

\textbf{LoRA recovery:}

\[
y = W\_\text{up}(W\_\text{down} x) + \frac{\alpha\_L}{r\_L} B(A x) + b
\tag{4.2}
\]

\subsection{Head Bifurcation}

For $H$ heads:

\[
H\_f = \max(1, \lfloor H / 3 \rfloor), \quad H\_k = H - H\_f
\tag{4.3}
\]

For $H = 12$: $H_f = 4$, $H_k = 8$.

Kernel heads index $0, \ldots, H_k - 1$; full heads index $H_k, \ldots, H - 1$.

\subsection{Kernel Heads (Linear Attention with RFF)}

For the first $H_k$ heads:

\[
Q\_\phi = \phi(Q\_k), \quad K\_\phi = \phi(K\_k)
\tag{4.4}
\]

where $\phi(x) = [\cos(\Omega x), \sin(\Omega x)] / \sqrt{2F}$.

\textbf{Non-causal path:}

\[
KV = K\_\phi^\top V, \quad Z = \sum\_j K\_\phi^{(j)}
\tag{4.5}
\]

\[
O\_{\text{kernel}, i} = \frac{\phi(q\_i)^\top KV}{\phi(q\_i)^\top Z}
\tag{4.6}
\]

\textbf{Causal path (chunked):}

For chunk size $C = 64$, iterate over chunks. Within each chunk, compute 
intra-chunk causal attention; across chunks, use the running state $S, Z$.

\subsection{Full Heads (Softpick Attention)}

For the last $H_f$ heads:

\[
\text{scores}\_{ij} = \frac{q\_i^\top k\_j}{\sqrt{d\_h}}
\tag{4.7}
\]

Mask causal and padding:

\[
\text{scores}\_{ij} \leftarrow \begin{cases} \text{scores}\_{ij} & j \leq i \text{ and valid} \\ -10^4 & \text{otherwise} \end{cases}
\tag{4.8}
\]

Softpick:

\[
\text{attn}\_{ij} = \frac{\text{ReLU}(e^{\text{scores}\_{ij}} - 1)}{\sum\_k \text{ReLU}(e^{\text{scores}\_{ik}} - 1) + \epsilon}
\tag{4.9}
\]

\textbf{Note on masked positions:} For masked positions with score $-10^4$, 
$e^{-10^4} \approx 0$, $\text{ReLU}(-1) = 0$. So masked keys contribute 
zero. \textbf{But} if all keys in a row are masked, the denominator becomes 0 
(plus epsilon) and the output is all zeros -- no attention.

\subsection{Output Fusion}

Concatenate kernel and full heads:

\[
O = \text{concat}(O\_{\text{kernel}}, O\_{\text{full}})
\tag{4.10}
\]

Then apply the factorized output projection.

\subsection{Algorithm}

\textbf{Algorithm 1:} ORION forward
\begin{lstlisting}
Input:  X $\in$ R^{B$\times$L$\times$d}
Output: Y $\in$ R^{B$\times$L$\times$d}

1: # Factorized projections
2: Q\_all $\leftarrow$ W\_Q\_up(W\_Q\_down(X))
3: K\_all $\leftarrow$ W\_K\_up(W\_K\_down(X))
4: V\_all $\leftarrow$ W\_V\_up(W\_V\_down(X))
5:
6: if RoPE: (Q, K) $\leftarrow$ Rotary(Q, K)
7: if GQA: (K, V) $\leftarrow$ repeat\_kv(K, V)
8:
9: # Split heads
10: Q\_kernel, Q\_full $\leftarrow$ Q[:, :H\_k], Q[:, H\_k:]
11: K\_kernel, K\_full $\leftarrow$ K[:, :H\_k], K[:, H\_k:]
12: V\_kernel, V\_full $\leftarrow$ V[:, :H\_k], V[:, H\_k:]
13:
14: # Kernel heads
15: O\_kernel $\leftarrow$ RFF\_linear\_attention(Q\_kernel, K\_kernel, V\_kernel)
16:
17: # Full heads
18: O\_full $\leftarrow$ softpick\_attention(Q\_full, K\_full, V\_full)
19:
20: # Concatenate + output
21: O $\leftarrow$ concat(O\_kernel, O\_full)
22: Y $\leftarrow$ W\_O\_up(W\_O\_down(O))
23: return Y
\end{lstlisting}

\subsection{The `copy\_weights` Bug}

\textbf{Critical.} The \texttt{copy\_weights} method contains:

\begin{lstlisting}
missing = [k for k, (w, \_) in extracted.items() if w is None and (k != "o" or not self.is\_nested\_encoder)]
if missing:
    try:
        from framework.core.adapters import UniversalAdapter
        adapter = UniversalAdapter()
        for k in missing:
            w, b = adapter.get\_weight\_bias(src, k)
            if w is not None:
                extracted[k] = (w.clone(), b.clone() if b is not None else None)
    except Exception as e:
        print(f'copy\_weights failed: {e}')
        raise RuntimeError(f"copy\_weights failed: {e}")  # $\leftarrow$ BUG
\end{lstlisting}

\textbf{Two failure modes:}

\begin{enumerate}
\item \textbf{UniversalAdapter raises:} $\to$ RuntimeError propagates.
\item \textbf{UniversalAdapter returns None for all missing:} $\to$ \texttt{extracted[k]} stays None, then \texttt{missing\_after} check triggers another RuntimeError.

\end{enumerate}
In both cases, the injection aborts. If the caller catches the exception, 
the module keeps \textbf{random init weights} $\to$ collapse.

\textbf{Fix:}

\begin{lstlisting}
except Exception as e:
    logger.debug(f"UniversalAdapter failed: {e}")
    # Continue
\end{lstlisting}

\subsection{Theoretical Notes}

\subsubsection{SVD Optimality}

The truncated SVD is the optimal rank-$r$ approximation in Frobenius norm 
(Eckart-Young).

\subsubsection{RFF Approximation}

RFF approximates the Gaussian kernel $\exp(-\|q - k\|^2 / 2)$. The 
approximation error decreases as $\mathcal{O}(1/\sqrt{F})$.

\subsubsection{Softpick Properties}

ORION's softpick produces \textbf{sparse} attention: for $x_i$ sufficiently 
negative, $\text{ReLU}(e^{x_i} - 1) = 0$. This differs from softmax which 
always produces nonzero weights.

\textbf{Potential issue:} If all $x_i < 0$, all weights are zero $\to$ output is 
zero. This happens on masked rows.

\subsection{Weight Transfer}

Standard four-architecture copy with additional factorization step:

\begin{lstlisting}
with torch.no\_grad():
    for proj\_key, target\_mod in target\_map.items():
        w, b = extracted[proj\_key]
        if w is None:
            continue
        if w.ndim != 2:
            w = w.reshape(w.size(0), -1)
        if w.shape != (target\_mod.out\_features, target\_mod.in\_features) \
           and (target\_mod.out\_features, target\_mod.in\_features) == w.t().shape:
            w = w.t()
        target\_mod.decompose\_from\_weight(w[:target\_mod.out\_features, :target\_mod.in\_features], ...)
\end{lstlisting}

The \texttt{decompose\_from\_weight} method:
\begin{enumerate}
\item Computes SVD.
\item Selects rank by energy threshold.
\item Sets \texttt{down.weight = $\Sigma$^{1/2} V^T}, \texttt{up.weight = U $\Sigma$^{1/2}}.

\end{enumerate}

\subsection{The `copy\_weights` Failure Path}

\textbf{Case 1: UniversalAdapter succeeds for all projections.}
\begin{itemize}
\item Standard extraction works.
\item No issue.

\end{itemize}
\textbf{Case 2: UniversalAdapter fails, but the direct extraction already succeeded.}
\begin{itemize}
\item \texttt{missing} is empty $\to$ no try/except triggered.
\item No issue.

\end{itemize}
\textbf{Case 3: UniversalAdapter raises.}
\begin{itemize}
\item \texttt{raise RuntimeError} aborts \texttt{copy\_weights}.
\item \texttt{inject\_orion} catches it (the outer try/except) $\to$ returns None.
\item \textbf{Safe} -- the module is not installed.

\end{itemize}
\textbf{Case 4: UniversalAdapter returns None for all \texttt{missing}.}
\begin{itemize}
\item \texttt{missing\_after} is non-empty $\to$ RuntimeError.
\item Same as case 3.

\end{itemize}
\textbf{Case 5: UniversalAdapter raises but injection proceeds anyway.}
\begin{itemize}
\item Not possible -- \texttt{inject\_orion} has its own try/except.

\end{itemize}

\subsection{Common Pitfalls}

\subsubsection{`FactorisedLinear.weight` Materializes Full Matrix}

\begin{lstlisting}
@property
def weight(self):
    return torch.matmul(self.up.weight, self.down.weight)
\end{lstlisting}

For 768$\times$768 this is fine. For 3072$\times$768 with rank 64: \texttt{64 $\times$ 3072 + 64 $\times$ 768} 
= 245K params, materializing to 2.36M -- a 10$\times$ overhead. Used in Frobenius 
check inside \texttt{copy\_weights}.

\subsubsection{The `raise` Bug}

See Section 4.10.

\subsubsection{Softpick Masked Rows}

As noted, if a row is entirely masked, output is zero.

\subsubsection{The `\_kernel\_linear\_attention` Python Loop (Decoder)}

For $N = 128$, chunk\_size = 64: 2 iterations. Acceptable but not 
vectorized.

\subsubsection{`rff\_proj` is not persistent}

\begin{lstlisting}
self.register\_buffer("rff\_proj", torch.randn(...), persistent=False)
\end{lstlisting}

With \texttt{persistent=False}, \texttt{rff\_proj} is not saved in \texttt{state\_dict()}. On 
reload, it is re-initialized randomly. \textbf{This is a bug} -- the RFF 
projection must be preserved.

\section{Analysis}

\subsection{The `copy\_weights` Bug}

\textbf{The most important finding.} The \texttt{copy\_weights} method contains:

\begin{lstlisting}
except Exception as e:
    print(f'copy\_weights failed: {e}')
    raise RuntimeError(f"copy\_weights failed: {e}")
\end{lstlisting}

This matches the pattern in \texttt{aegis\_v4\_module.py}, \texttt{aegis\_v5\_module.py}, 
\texttt{eat.py}, \texttt{rala.py}, etc. \textbf{A UniversalAdapter failure aborts injection.}

\subsubsection{When Does This Trigger?}

For the four tested architectures, direct extraction succeeds (Q/K/V/O 
are found via named attributes). \texttt{missing} is empty $\to$ no try/except is 
triggered.

\textbf{However}, for hypothetical architectures where \texttt{UniversalAdapter} is 
the only extraction path (e.g., certain hybrid models), the exception 
path can trigger.

\subsubsection{Impact}

If triggered:
\begin{itemize}
\item \texttt{copy\_weights} raises.
\item \texttt{inject\_orion}'s outer \texttt{try/except} catches it.
\item \textbf{\texttt{inject\_orion} returns None} -- module is not installed.
\item No collapse, but also no ORION.

\end{itemize}
\textbf{This is safer than the pure \texttt{raise} pattern} because \texttt{inject\_orion} 
has its own try/except.

\subsection{The `rff\_proj` Persistence Bug}

\begin{lstlisting}
self.register\_buffer("rff\_proj", torch.randn(...), persistent=False)
\end{lstlisting}

\textbf{Problem.} \texttt{persistent=False} means \texttt{rff\_proj} is \textbf{not saved} in 
\texttt{state\_dict()}.

\textbf{Impact.} After saving and reloading, the model uses a \textbf{different} 
random projection. This causes a mismatch between train-time and test-time 
behavior.

\textbf{Verification.} Save and reload the model, then compare:

\begin{lstlisting}
model1 = ORIONAttention(...)
sd = model1.state\_dict()
# rff\_proj not in sd

model2 = ORIONAttention(...)
model2.load\_state\_dict(sd, strict=False)
# model2.rff\_proj is different from model1.rff\_proj
\end{lstlisting}

\textbf{Fix.} Set \texttt{persistent=True}:

\begin{lstlisting}
self.register\_buffer("rff\_proj", torch.randn(...), persistent=True)
\end{lstlisting}

\subsection{Softpick Behavior}

For ORION's softpick:

\[
\text{softpick}(x)\_i = \frac{\text{ReLU}(e^{x\_i} - 1)}{\sum\_j \text{ReLU}(e^{x\_j} - 1) + \epsilon}
\]

\textbf{Test.} Compare softpick vs softmax on masked rows.

\begin{table}[H]
\centering
\small
\begin{tabular}{ll}
\toprule
\textbf{Method} & \textbf{Masked row output} \\
\midrule
softmax & All zeros (masked keys get ~0 weight) \\
ORION softpick & All zeros (ReLU zeros) \\
Standard softmax + valid keys & Normal \\
\bottomrule
\end{tabular}
\end{table}

\textbf{Observation.} Both methods produce zero output for fully-masked rows. 
The difference: softmax always has a valid denominator (from valid keys), 
while ORION softpick's denominator can be zero if all keys are masked.

\textbf{Fix.} Add a fallback:

\begin{lstlisting}
denom = p.sum(dim=dim, keepdim=True)
p = torch.where(denom \textgreater{} 1e-8, p / denom, F.softmax(x, dim=dim))
\end{lstlisting}

\section{Limitations}

\subsection{`copy\_weights` Raises on Failure}

Matches the pattern in AEGIS V4/V5, EAT, RALA, NHA. **Robustness 
regression** compared to modules that return None.

\subsection{`rff\_proj` is Not Persistent}

\begin{lstlisting}
persistent=False
\end{lstlisting}

The RFF projection is re-initialized randomly on reload. **Train/test 
mismatch** on checkpoint restoration.

\subsection{Softpick Collapses on Masked Rows}

Without a fallback, fully masked rows produce zero attention. Contrast 
with NASC which has a batch-level fallback.

\subsection{Slower than Baseline}

~2.3 s$\times$ slowdown from more kernel launches.

\subsection{Head Split is Heuristic}

The 2/3 : 1/3 split is chosen by \texttt{H // 3}. No principled justification.

\subsection{Rank Selection is Global}

The same \texttt{svd\_max\_rank} is used for all projections. Per-layer rank 
selection could be better.

\subsection{No Multi-Seed}

Single seed.

\subsection{Task Coverage}

Only SST-2. Behavior on other tasks is untested.

\subsection{Model Coverage}

Only three models. Qwen3.5 untested.

\subsection{RFF Memory}

\texttt{rff\_proj} shape is $(H_k, d_h, F)$. For $H_k = 8$, $d_h = 64$, $F = 256$: 
131K floats = ~0.5 MB per layer. Small but not negligible.

\subsection{Python Loop in Decoder Kernel}

For $N > 64$, the causal kernel path uses a Python loop. At $N = 128$: 2 
iterations. At $N = 512$: 8 iterations.

\subsection{The Full Matrix Materialization}

\begin{lstlisting}
@property
def weight(self):
    return torch.matmul(self.up.weight, self.down.weight)
\end{lstlisting}

Materializes the full $d_\text{out} \times d_\text{in}$ matrix. For large 
FFN projections, this is a memory spike.

\subsection{Code Quality}

\begin{itemize}
\item Two bugs (documented)
\item No logging of head split
\item No logging of rank selection

\end{itemize}
\subsection{What This Paper Is Not}

\begin{itemize}
\item Not a perfectly robust module (two bugs)
\item Not the fastest method (slower than baseline)
\item Not validated beyond SST-2
\end{itemize}

\section{Appendix A. Mathematical Derivations}

\subsection{A.1 SVD Factorization}

$W = U \Sigma V^\top$. Truncated at rank $r$:

\[
W\_r = U\_r \Sigma\_r V\_r^\top
\]

Split:

\[
W\_\text{up} = U\_r \Sigma\_r^{1/2}, \quad W\_\text{down} = \Sigma\_r^{1/2} V\_r^\top
\]

Then $W_r = W_\text{up} W_\text{down}$.

\textbf{Parameter count:} $\text{rank}(W_\text{up}) + \text{rank}(W_\text{down}) = r \cdot d_\text{out} + r \cdot d_\text{in} = r(d_\text{out} + d_\text{in})$.

\subsection{A.2 Energy-Based Rank Selection}

Given singular values $\sigma_1 \geq \sigma_2 \geq \dots$:

\[
E(r) = \frac{\sum\_{i=1}^{r} \sigma\_i^2}{\sum\_i \sigma\_i^2}
\]

Select smallest $r$ such that $E(r) \geq \eta$.

\subsection{A.3 RFF Feature Map}

\[
\phi(x) = \frac{1}{\sqrt{2F}} [\cos(\Omega x), \sin(\Omega x)] \in \mathbb{R}^{2F}
\]

where $\Omega \in \mathbb{R}^{d_h \times F}$.

\textbf{Approximation:} $\phi(q)^\top \phi(k) \approx \exp(q^\top k)$ for 
$\Omega$ sampled from $\mathcal{N}(0, I)$.

\subsection{A.4 Linear Attention}

\[
O\_i = \frac{\phi(q\_i)^\top \sum\_j \phi(k\_j) v\_j^\top}{\phi(q\_i)^\top \sum\_j \phi(k\_j)}
\]

Precompute $KV = \sum_j \phi(k_j) v_j^\top \in \mathbb{R}^{2F \times d_h}$ 
and $Z = \sum_j \phi(k_j) \in \mathbb{R}^{2F}$.

\subsection{A.5 Softpick}

\[
\text{softpick}(x)\_i = \frac{\text{ReLU}(e^{x\_i} - 1)}{\sum\_j \text{ReLU}(e^{x\_j} - 1) + \epsilon}
\]

\textbf{Property:} For $x_i < 0$, $\text{ReLU}(e^{x_i} - 1) = 0$. Sparsification 
is natural.

\subsection{A.6 Head Split}

\[
H\_f = \max(1, \lfloor H/3 \rfloor), \quad H\_k = H - H\_f
\]

For $H = 12$: $(H_k, H_f) = (8, 4)$.

\subsection{A.7 Complexity}

Attention cost:

\[
T = H\_k \cdot \mathcal{O}(L F d) + H\_f \cdot \mathcal{O}(L^2 d)
\]

For $H_k/H = 2/3$: attention reduced by ~1/3.

\subsection{A.8 Parameter Reduction}

Per projection: $d^2 \to 2rd$. Ratio: $d / (2r)$.

For $d = 768, r = 64$: $6	imes$.

For FFN: $768 \times 3072 \to 64 \times (768 + 3072) = 245,760$. 
Ratio: $9.6	imes$.

\section{Appendix B. Implementation Details}

\subsection{B.1 FactorisedLinear}

\begin{lstlisting}
class FactorisedLinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, ..., max\_rank=64):
        self.rank = max(min\_rank, min(max\_rank, min(in\_features, out\_features)))
        self.down = nn.Linear(in\_features, self.rank, bias=False)
        self.up = nn.Linear(self.rank, out\_features, bias=True)

    def decompose\_from\_weight(self, weight, bias=None):
        U, S, Vt = torch.linalg.svd(weight.float(), full\_matrices=False)
        total = (S ** 2).sum()
        cum = torch.cumsum(S ** 2, dim=0) / total.clamp(min=1e-8)
        rank = int((cum \textless{} self.energy).sum().item()) + 1
        rank = max(self.min\_rank, min(rank, min(weight.shape), self.max\_rank))
        
        A = U[:, :rank] * S[:rank].unsqueeze(0)
        B = Vt[:rank, :]
        self.down.weight.data = B
        self.up.weight.data = A
\end{lstlisting}

def \_kernel\_linear\_attention(self, Q, K, V):
    proj = self.rff\_proj[:Hk].float()
    Q\_proj = torch.einsum("bhnd,hdf-\textgreater{}bhnf", Q * scale, proj)
    K\_proj = torch.einsum("bhnd,hdf-\textgreater{}bhnf", K * scale, proj)
    Q\_feat = torch.cat([torch.cos(Q\_proj), torch.sin(Q\_proj)], dim=-1) / math.sqrt(2 * F\_dim)
    K\_feat = torch.cat([torch.cos(K\_proj), torch.sin(K\_proj)], dim=-1) / math.sqrt(2 * F\_dim)
    
    if self.is\_decoder:
        for c in range(num\_chunks):
            # chunked causal attention
            ...
    else:
        KV = torch.einsum("bhnf,bhnd-\textgreater{}bhfd", K\_feat, V)
        K\_sum = K\_feat.sum(dim=2)
        denom = torch.einsum("bhnf,bhf-\textgreater{}bhn", Q\_feat, K\_sum).unsqueeze(-1)
        ctx = torch.einsum("bhnf,bhfd-\textgreater{}bhnd", Q\_feat, KV) / denom
    return ctx

\begin{lstlisting}
def softpick(x, dim=-1):
    p = F.relu(torch.exp(x.float()) - 1.0)
    denom = p.sum(dim=dim, keepdim=True)
    fallback = F.softmax(x.float(), dim=dim)
    p = torch.where(denom \textgreater{} 1e-8, p / (denom + 1e-8), fallback)
    return p.to(x.dtype)
\end{lstlisting}

\subsection{D.6 Lessons Learned}

\begin{enumerate}
\item \textbf{\texttt{copy\_weights} robustness matters.} Raise-on-failure is dangerous.
\item \textbf{Persistent buffers matter.} \texttt{persistent=False} breaks save/reload.

\item \textbf{Head bifurcation works.} 2/3 : 1/3 split is a good default.
