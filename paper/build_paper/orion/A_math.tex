
\section{Appendix A. Mathematical Derivations}

\subsection{A.1 SVD Factorization}

$W = U \Sigma V^\top$. Truncated at rank $r$:

\[
W\_r = U\_r \Sigma\_r V\_r^\top
\]

Split:

\[
W\_\text{up} = U\_r \Sigma\_r^{1/2}, \quad W\_\text{down} = \Sigma\_r^{1/2} V\_r^\top
\]

Then $W_r = W_\text{up} W_\text{down}$.

\textbf{Parameter count:} $\text{rank}(W_\text{up}) + \text{rank}(W_\text{down}) = r \cdot d_\text{out} + r \cdot d_\text{in} = r(d_\text{out} + d_\text{in})$.

\subsection{A.2 Energy-Based Rank Selection}

Given singular values $\sigma_1 \geq \sigma_2 \geq \dots$:

\[
E(r) = \frac{\sum\_{i=1}^{r} \sigma\_i^2}{\sum\_i \sigma\_i^2}
\]

Select smallest $r$ such that $E(r) \geq \eta$.

\subsection{A.3 RFF Feature Map}

\[
\phi(x) = \frac{1}{\sqrt{2F}} [\cos(\Omega x), \sin(\Omega x)] \in \mathbb{R}^{2F}
\]

where $\Omega \in \mathbb{R}^{d_h \times F}$.

\textbf{Approximation:} $\phi(q)^\top \phi(k) \approx \exp(q^\top k)$ for 
$\Omega$ sampled from $\mathcal{N}(0, I)$.

\subsection{A.4 Linear Attention}

\[
O\_i = \frac{\phi(q\_i)^\top \sum\_j \phi(k\_j) v\_j^\top}{\phi(q\_i)^\top \sum\_j \phi(k\_j)}
\]

Precompute $KV = \sum_j \phi(k_j) v_j^\top \in \mathbb{R}^{2F \times d_h}$ 
and $Z = \sum_j \phi(k_j) \in \mathbb{R}^{2F}$.

\subsection{A.5 Softpick}

\[
\text{softpick}(x)\_i = \frac{\text{ReLU}(e^{x\_i} - 1)}{\sum\_j \text{ReLU}(e^{x\_j} - 1) + \epsilon}
\]

\textbf{Property:} For $x_i < 0$, $\text{ReLU}(e^{x_i} - 1) = 0$. Sparsification 
is natural.

\subsection{A.6 Head Split}

\[
H\_f = \max(1, \lfloor H/3 \rfloor), \quad H\_k = H - H\_f
\]

For $H = 12$: $(H_k, H_f) = (8, 4)$.

\subsection{A.7 Complexity}

Attention cost:

\[
T = H\_k \cdot \mathcal{O}(L F d) + H\_f \cdot \mathcal{O}(L^2 d)
\]

For $H_k/H = 2/3$: attention reduced by ~1/3.

\subsection{A.8 Parameter Reduction}

Per projection: $d^2 \to 2rd$. Ratio: $d / (2r)$.

For $d = 768, r = 64$: $6	imes$.

For FFN: $768 \times 3072 \to 64 \times (768 + 3072) = 245,760$. 
Ratio: $9.6	imes$.
