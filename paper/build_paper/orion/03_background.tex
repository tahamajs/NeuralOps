
\section{Background}

\subsection{SVD Decomposition}

For $W \in \mathbb{R}^{m \times n}$:

\[
W = U \Sigma V^\top
\tag{3.1}
\]

with $U \in \mathbb{R}^{m \times m}$, $\Sigma \in \mathbb{R}^{m \times n}$, 
$V \in \mathbb{R}^{n \times n}$.

\textbf{Truncated:} $W_r = U_r \Sigma_r V_r^\top$. Error $\|W - W_r\|_F = \sqrt{\sum_{i > r} \sigma_i^2}$.

\textbf{Symmetric factorization:}

\[
W\_r = (U\_r \Sigma\_r^{1/2})(\Sigma\_r^{1/2} V\_r^\top)
\tag{3.2}
\]

with $W_\text{up} = U_r \Sigma_r^{1/2} \in \mathbb{R}^{m \times r}$, 
$W_\text{down} = \Sigma_r^{1/2} V_r^\top \in \mathbb{R}^{r \times n}$.

\textbf{Parameter count:} $r(m + n)$ vs $mn$.

\subsection{Energy-Based Rank Selection}

Given singular values $\sigma_1 \geq \sigma_2 \geq \dots$, the energy 
captured by the top-$r$:

\[
E(r) = \frac{\sum\_{i=1}^{r} \sigma\_i^2}{\sum\_i \sigma\_i^2}
\tag{3.3}
\]

ORION selects the smallest $r$ such that $E(r) \geq \eta$ (energy threshold, 
default 0.95).

\subsection{Random Fourier Features}

Approximate $\exp(q^\top k)$ via:

\[
\exp(q^\top k) \approx \phi(q)^\top \phi(k), \quad
\phi(x) = \frac{1}{\sqrt{2F}} [\cos(\Omega x), \sin(\Omega x)]
\tag{3.4}
\]

with $\Omega \in \mathbb{R}^{d_h \times F}$.

\subsection{Linear Attention with RFF}

\[
O\_i = \frac{\phi(q\_i)^\top \sum\_j \phi(k\_j) v\_j^\top}{\phi(q\_i)^\top \sum\_j \phi(k\_j)}
\tag{3.5}
\]

Cost: $\mathcal{O}(L F d)$.

\subsection{Softpick (Simplified)}

ORION's softpick:

\[
\text{softpick}(x)\_i = \frac{\text{ReLU}(e^{x\_i} - 1)}{\sum\_j \text{ReLU}(e^{x\_j} - 1) + \epsilon}
\tag{3.6}
\]

\textbf{Key property:} For $x_i = -10^4$ (masked), $e^{x_i} - 1 \approx -1$, 
$\text{ReLU}(-1) = 0$. Masked positions contribute zero.
