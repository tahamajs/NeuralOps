\subsection{B.1 Module Structure}

\begin{lstlisting}
class A3FactorizedLinear(nn.Module):
    """Factorized linear layer: y = (xV^T)U^T + b + LoRA"""
    
    def \_\_init\_\_(self, in\_features, out\_features, rank, ...):
        super().\_\_init\_\_()
        self.in\_features = in\_features
        self.out\_features = out\_features
        self.rank = rank
        self.lora\_rank = lora\_rank
        
        # Base factors (frozen after copy)
        self.V = nn.Parameter(torch.empty(rank, in\_features))
        self.U = nn.Parameter(torch.empty(out\_features, rank))
        
        # Bias
        if bias:
            self.bias = nn.Parameter(torch.empty(out\_features))
        
        # Optional LoRA
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.empty(lora\_rank, in\_features))
            self.lora\_B = nn.Parameter(torch.empty(out\_features, lora\_rank))
        
        self.reset\_parameters()
    
    def forward(self, x):
        latent = F.linear(x, self.V)
        out = F.linear(latent, self.U)
        if self.lora\_rank \textgreater{} 0:
            out += F.linear(F.linear(x, self.lora\_A), self.lora\_B) * self.lora\_scaling
        if self.bias is not None:
            out += self.bias
        return out
\end{lstlisting}

\subsection{B.2 The Main Compression Loop}

\begin{lstlisting}
def compress(self):
    # 1. Collect statistics
    cov\_map, fisher\_map = self.\_collect\_activations\_and\_fisher()
    
    # 2. For each layer
    for name, module in self.model.named\_modules():
        if not self.\_is\_target(module, name):
            continue
        
        cov = cov\_map.get(name)
        fisher = fisher\_map.get(name)
        
        factorized, err, rank = self.compress\_linear(
            module, name=name, cov=cov, fisher=fisher
        )
        
        if factorized is not None:
            # Replace
            if hasattr(module, 'base\_layer'):
                module.base\_layer = factorized
                replacement = module
            else:
                replacement = factorized
            replace\_module(self.model, name, replacement)
        # else: fallback, keep original
    
    # 3. LN recalibration
    if self.recalibrate\_ln:
        recalibrate\_layernorms(self.model, ...)
    
    return self.model, self.get\_stats()
\end{lstlisting}

\subsection{B.3 Streaming Statistics}

\begin{lstlisting}
def \_collect\_activations\_and\_fisher(self):
    cov\_running = {}
    fisher\_running = {}
    count\_running = {}
    
    def make\_hook(name):
        def hook(module, inp, outp):
            with torch.no\_grad():
                x = inp[0].detach().float().reshape(-1, inp[0].shape[-1])
                mask = x.abs().sum(dim=-1) \textgreater{} 1e-5
                x\_valid = x[mask]
                if x\_valid.size(0) \textgreater{} 0:
                    c = torch.matmul(x\_valid.T, x\_valid)
                    f = x\_valid.pow(2).sum(dim=0)
                    if name not in cov\_running:
                        cov\_running[name] = c.cpu()
                        fisher\_running[name] = f.cpu()
                        count\_running[name] = x\_valid.size(0)
                    else:
                        cov\_running[name] += c.cpu()
                        fisher\_running[name] += f.cpu()
                        count\_running[name] += x\_valid.size(0)
        return hook
    
    hooks = []
    for name, module in self.model.named\_modules():
        if isinstance(module, nn.Linear):
            hooks.append(module.register\_forward\_hook(make\_hook(name)))
    
    # Run calibration
    self.model.eval()
    with torch.no\_grad():
        for batch in self.calibration\_loader:
            self.model(**batch)
    
    for h in hooks:
        h.remove()
    
    # Normalize
    cov\_map = {n: cov\_running[n] / count\_running[n] for n in cov\_running}
    fisher\_map = {n: fisher\_running[n] / count\_running[n] for n in fisher\_running}
    
    return cov\_map, fisher\_map
\end{lstlisting}

\subsection{B.4 Weighted SVD}

\begin{lstlisting}
def compress\_linear(self, module, name, cov=None, fisher=None):
    W = module.weight.data.float()
    
    # Extract shape
    out\_f, in\_f = W.shape[0], W.shape[1]
    
    # Compute weighting
    if self.use\_fisher and fisher is not None:
        scale = torch.sqrt(fisher.clamp(min=1e-8) + 1e-6)
    elif cov is not None:
        scale = torch.sqrt(torch.diag(cov).clamp(min=1e-8) + 1e-6)
    else:
        scale = None
    
    # Weighted matrix
    if scale is not None:
        if self.normalize\_scale:
            rms = torch.norm(scale) / math.sqrt(in\_f)
            scale = scale / (rms + 1e-8)
        scale = scale.clamp(min=1e-3, max=1e3)
        W\_w = W * scale.unsqueeze(0)
    else:
        W\_w = W
    
    # SVD
    U, S, Vt = torch.linalg.svd(W\_w, full\_matrices=False)
    
    # Retry loop
    norm\_orig = torch.linalg.norm(W).item()
    best\_err = float('inf')
    best\_factors = None
    
    for r in adaptive\_ranks:
        sqrt\_S = torch.sqrt(S[:r].clamp(min=1e-12))
        U\_r = U[:, :r] * sqrt\_S.unsqueeze(0)
        V\_r = sqrt\_S.unsqueeze(1) * Vt[:r, :]
        if scale is not None:
            V\_r = V\_r / scale.unsqueeze(0)
        
        W\_rec = U\_r @ V\_r
        err = torch.linalg.norm(W - W\_rec).item() / (norm\_orig + 1e-8)
        
        if err \textless{} best\_err:
            best\_err = err
            best\_factors = (U\_r, V\_r)
        
        if err \textless{}= self.max\_error:
            break
    
    if best\_err \textgreater{} self.max\_error and self.fallback\_to\_original:
        return None, best\_err, 0
    
    return A3FactorizedLinear(
        in\_features=in\_f, out\_features=out\_f, rank=best\_r,
        U\_factor=best\_factors[0], V\_factor=best\_factors[1]
    ), best\_err, best\_r
\end{lstlisting}

\subsection{B.5 LayerNorm Recalibration}

\begin{lstlisting}
def recalibrate\_layernorms(model, tokenizer, calibration\_loader, ...):
    # 1. Record target std
    target\_stds = {}
    hooks = []
    for name, module in model.named\_modules():
        if isinstance(module, nn.LayerNorm):
            def make\_hook(n):
                def hook(mod, inp, outp):
                    x = inp[0].detach().float()
                    target\_stds[n] = x.std(dim=-1).mean().item()
                return hook
            hooks.append(module.register\_forward\_hook(make\_hook(name)))
    
    # Forward pass
    with torch.no\_grad():
        for batch in calibration\_loader:
            model(**batch)
    for h in hooks: h.remove()
    
    # 2. Compute current std (after compression)
    current\_stds = {}
    # (same procedure)
    
    # 3. Rescale
    for name, module in model.named\_modules():
        if isinstance(module, nn.LayerNorm) and name in target\_stds:
            ratio = current\_stds[name] / (target\_stds[name] + 1e-8)
            with torch.no\_grad():
                module.weight.data *= ratio
\end{lstlisting}

\subsubsection{B.6.2 Wrong Conv1D Orientation}

GPT-2's \texttt{Conv1D} stores \texttt{(in, out)}, but \texttt{nn.Linear} stores \texttt{(out, in)}:

\begin{lstlisting}
if "conv1d" in cls\_name or hasattr(module, "nf"):
    W = raw\_w.t()  # transpose!
\end{lstlisting}

\subsubsection{B.6.3 Missing Padding Filter}

Without filtering padding tokens, the Fisher estimate is biased:

\begin{lstlisting}
mask = x.abs().sum(dim=-1) \textgreater{} 1e-5
x\_valid = x[mask]
\end{lstlisting}

\subsubsection{B.6.4 Numerical Overflow in Weighting}

Clamp $\sqrt{F}$ to avoid extreme values:

\begin{lstlisting}
scale = scale.clamp(min=1e-3, max=1e3)
\end{lstlisting}

\subsection{B.7 Full Pseudocode for End-to-End Compression}
\label{sec:app\_algorithm1}

\textbf{Algorithm 1:} A3 End-to-End Compression Pipeline
\begin{lstlisting}
Input:  Pretrained model M, initial rank r, threshold $\tau$, calibration dataset D
Output: Compressed model M', compression statistics

 1: for layer in linear\_layers(M) do
 2:     F^(layer), C\_X^(layer) $\leftarrow$ running\_accumulate(layer, D)
 3: end for
 4: $\sigma$\_target $\leftarrow$ record\_layernorm\_scales(M, D)
 5:
 6: for layer in linear\_layers(M) do
 7:     W $\leftarrow$ layer.weight.data.clone()
 8:     r\_curr $\leftarrow$ r
 9:     for attempt in 0..max\_retries do
10:         s $\leftarrow$ normalize\_scale(sqrt(F^(layer) + $\epsilon$)).clamp(1e-3, 1e3)
11:         W\_weighted $\leftarrow$ W * diag(s)
12:         U\_r, $\Sigma$\_r, V\_r $\leftarrow$ cpu\_fp32\_svd(W\_weighted, r\_curr)
13:         V\_r $\leftarrow$ V\_r * diag(1 / s)
14:         err $\leftarrow$ norm(W - U\_r * V\_r, 'fro') / (norm(W, 'fro') + 1e-8)
15:         if err $\le$ $\tau$: break
16:         r\_curr $\leftarrow$ escalate\_rank(r\_curr, r\_ceiling=256, attempt)
17:     end for
18:     if err $\le$ $\tau$:
19:         replace\_with\_factorized(M, layer, U\_r, V\_r)
20:     else:
21:         retain\_dense\_fallback(M, layer)
22:     end if
23: end for
24:
25: if recalibrate\_ln:
26:     $\sigma$\_current $\leftarrow$ record\_layernorm\_scales(M', D)
27:     for ln in layernorms(M') do
28:         ln.weight.data *= ($\sigma$\_current[ln] / ($\sigma$\_target[ln] + 1e-8))
29:     end for
30: end if
31: return M'
\end{lstlisting}

\subsection{B.8 LoRALinear Wrapper Integration and Injection Algorithm}
\label{sec:app\_lora\_wrapper}

\begin{lstlisting}
class LoRALinear(nn.Module):
    def \_\_init\_\_(self, base\_layer, lora\_rank=16, lora\_alpha=32):
        super().\_\_init\_\_()
        self.base\_layer = base\_layer
        self.lora\_A = nn.Parameter(torch.empty(lora\_rank, base\_layer.in\_features))
        self.lora\_B = nn.Parameter(torch.zeros(base\_layer.out\_features, lora\_rank))
        self.scaling = lora\_alpha / lora\_rank
        
    def forward(self, x):
        return self.base\_layer(x) + (x @ self.lora\_A.T @ self.lora\_B.T) * self.scaling
\end{lstlisting}
