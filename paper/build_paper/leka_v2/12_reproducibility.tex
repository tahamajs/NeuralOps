\section{Reproducibility}

\subsection{Environment}

\begin{lstlisting}
git clone https://github.com/Efficiency Lab/SVD\_Project
cd SVD\_Project
python -m venv .venv
.venv\Scripts\activate
pip install -r requirements.txt
\end{lstlisting}

\subsection{Running}

\begin{lstlisting}
python run\_all\_4\_leka\_v2.py \
    --models distilbert,roberta,gpt2 \
    --feature\_dim 64 \
    --landmark\_ratio 0.25 \
    --recovery\_steps 50
\end{lstlisting}
