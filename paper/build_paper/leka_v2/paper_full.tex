\section{Introduction}

\subsection{Motivation}

Long-context inference is dominated by attention's $\mathcal{O}(L^2)$ cost. 
\textbf{Linear attention} (Katharopoulos et al., 2020) reduces this to 
$\mathcal{O}(L)$ via a kernel feature map $\phi$:

\[
\text{Attn}(Q, K, V) \approx \frac{\phi(Q)(\phi(K)^\top V)}{\phi(Q)(\phi(K)^\top \mathbf{1})}
\tag{1.1}
\]

However, linear attention has two limitations:

\begin{enumerate}
\item \textbf{Uniform aggregation.} The KV state $\phi(K)^\top V$ is a single global
  summary. Queries cannot selectively attend to specific tokens -- they
  only see the aggregate.
\item \textbf{Kernel approximation error.} RFF with $F$ features gives $\mathcal{O}(1/F)$
  approximation error; small $F$ degrades accuracy.

\end{enumerate}
\textbf{LEKAv1} (companion work) addressed the second via QR-orthogonalized 
projections. \textbf{LEKAv2} additionally addresses the first.

\subsection{Problem Statement}

Given $X \in \mathbb{R}^{B \times L \times d}$, we seek a hybrid attention 
that:

\begin{enumerate}
\item \textbf{Scales linearly} in $L$.
\item \textbf{Retains selective focus} on important tokens.
\item \textbf{Approximates a smooth kernel} (Gaussian RBF).
\item \textbf{Handles masking} without NaN or uniform collapse.
\item \textbf{Transfers pretrained weights} zero-shot.

\end{enumerate}
\subsection{Our Approach: LEKAv2}

LEKAv2 combines four components:

\subsubsection{Dual-Scale Random Features}

Two independent RFF projections with dimensions $F$ and $F/2$:

\[
\phi(x) = \frac{1}{\sqrt{F}} [\cos(x W\_1) \| \sin(x W\_1)] 
\| \frac{1}{\sqrt{F/2}} [\cos(x W\_2) \| \sin(x W\_2)]
\tag{1.2}
\]

with $W_1 \in \mathbb{R}^{d_h \times F}$, $W_2 \in \mathbb{R}^{d_h \times F/2}$.

\subsubsection{Landmark Distillation}

A learned scorer $s_\theta$ selects $L_\text{max} = \min(\rho L, L)$ 
landmark tokens:

\[
\mathcal{L}\_\text{idx} = \text{TopK}(s\_\theta(\phi(K)), L\_\text{max})
\tag{1.3}
\]

Position 0 (\texttt{[CLS]}) is always retained.

\subsubsection{Softpick}

A ReLU-damped softmax that handles masked rows gracefully:

\[
\text{softpick}(x)\_i = \frac{\text{ReLU}(e^{(x\_i - m)/T} - e^{-m/T})}{\sum\_j \text{ReLU}(e^{(x\_j - m)/T} - e^{-m/T})}
\tag{1.4}
\]

with fallback to uniform when all entries are zero.

\subsubsection{Local-Global Fusion}

A per-head learnable gate blends local sliding-window attention and 
global landmark attention:

\[
\text{out} = (1 - g) \cdot \text{Local} + g \cdot \text{Global}, \quad g = \sigma(w\_h)
\tag{1.5}
\]

with $w_h = -2$ initially, biasing toward local attention.

\subsection{Contributions}

\begin{itemize}
\item \textbf{Dual-scale RFF.} We extend RFF with a second projection at $F/2$,
  enriching the kernel representation.
\item \textbf{Learned landmark distillation.} A small scorer selects landmarks
  whose features are used in the global branch.
\item \textbf{Softpick.} A masked-attention-stable variant of softmax.
\item \textbf{Local-global fusion.} Per-head gate blends local and global paths.
\item \textbf{Per-head correction matrix.} $W_\text{corr} \in \mathbb{R}^{H \times d_h \times d_h}$
  applied to landmark values.
\item \textbf{Multi-architecture support.} DistilBERT, RoBERTa, GPT-2, Qwen3.5
  with GQA and RoPE.
\item \textbf{An honest evaluation.} We report both accuracy and parameter cost.

\end{itemize}

\section{Related Work}

\subsection{Linear Attention}

\textbf{Linear Transformer} (Katharopoulos et al., 2020) first replaced softmax 
with a kernel feature map. \textbf{Performer} (Choromanski et al., 2021) uses 
positive orthogonal random features. \textbf{cosFormer} (Qin et al., 2022) uses 
ReLU with cosine reweighting. \textbf{Nystr mformer} (Xiong et al., 2021) uses 
Nystr m approximation with landmarks.

\subsection{Landmark Methods}

\textbf{Nystr mformer} uses Nystr m landmarks selected via a downsampling rule. 
\textbf{Set Transformer} (Lee et al., 2019) uses inducing points. \textbf{Perceiver} 
(Jaegle et al., 2021) uses a learned latent array.

\textbf{Our distinction.} We select landmarks \textbf{per-example} via a learned 
scorer that operates on the RFF features. The selection is content-dependent 
and trained end-to-end.

\subsection{Local-Global Hybrid}

\textbf{Longformer} (Beltagy et al., 2020) combines local sliding-window 
attention with global tokens. \textbf{BigBird} (Zaheer et al., 2020) adds 
random attention. \textbf{Hymba} (Dong et al., 2024) and \textbf{Jamba} (Lieber 
et al., 2024) fuse attention and SSM.

\textbf{Our distinction.} We fuse \textbf{local sliding window} with **landmark 
linear attention**, both derived from the same Q/K/V projections.

\subsection{Softmax Variants}

\textbf{Softpick} (Gu \& Dao, 2024) uses ReLU-damped softmax for stable 
masked attention. \textbf{Sparsemax} (Martins \& Astudillo, 2016) projects 
onto the simplex. \textbf{Entmax} (Peters et al., 2019) uses $\alpha$-entmax.

\textbf{Our use.} We use softpick to handle the case where all rows of the 
attention matrix are masked (which causes standard softmax to give 
uniform attention).

\subsection{Random Fourier Features}

\textbf{Rahimi \& Recht (2007)} introduced RFF for kernel approximation. 
\textbf{Yu et al. (2016)} proposed orthogonal random features with lower 
variance. We extend RFF with \textbf{dual-scale} features.

\section{Background}

\subsection{Kernel Attention}

Standard softmax attention:

\[
\text{Attn}(Q, K, V) = \frac{\exp(QK^\top / \sqrt{d\_h}) V}{\exp(QK^\top / \sqrt{d\_h}) \mathbf{1}}
\tag{3.1}
\]

If $\kappa(q, k) = \phi(q)^\top \phi(k)$:

\[
\text{Attn}(Q, K, V)\_i \approx \frac{\phi(q\_i)^\top \sum\_j \phi(k\_j) v\_j^\top}{\phi(q\_i)^\top \sum\_j \phi(k\_j)}
\tag{3.2}
\]

Complexity: $\mathcal{O}(L F d_h)$.

\subsection{Random Fourier Features}

For the Gaussian kernel $\kappa(x - y) = \exp(-\|x - y\|^2 / 2)$:

\[
\kappa(x - y) = \mathbb{E}\_{\omega \sim \mathcal{N}(0, I)}[\cos(\omega^\top (x - y))]
\tag{3.3}
\]

Estimator with $F$ frequencies:

\[
\hat{\kappa}(x - y) = \frac{1}{F} \sum\_{i=1}^F [\cos(\omega\_i^\top x)\cos(\omega\_i^\top y) + \sin(\omega\_i^\top x)\sin(\omega\_i^\top y)]
\tag{3.4}
\]

Feature map: $\phi(x) = \frac{1}{\sqrt{F}} [\cos(xW) \| \sin(xW)]$.

\subsection{Landmark Distillation}

Given a set of $L$ token features $\{\phi(k_i)\}$ and a scorer $s_\theta$:

\[
\mathcal{S}\_\text{land} = \text{TopK}(\{s\_\theta(\phi(k\_i))\}, L\_\text{max})
\tag{3.5}
\]

The landmark features $\phi(K)_\text{land}$ replace the full aggregate:

\[
\text{Attn}\_\text{land}(q\_i) = \frac{\phi(q\_i)^\top \phi(K\_\text{land})^\top V\_\text{land}}{\phi(q\_i)^\top \phi(K\_\text{land})^\top \mathbf{1}}
\tag{3.6}
\]

Complexity: $\mathcal{O}(L \cdot L_\text{max} \cdot 2F)$ for the scoring 
attention.

\subsection{Softpick}

A variant of softmax that handles masked rows:

\[
\text{softpick}(x)\_i = \text{ReLU}\!\left(e^{(x\_i - m)/T} - e^{-m/T}\right)
\tag{3.7}
\]

where $m = \max_j x_j$ and $T$ is the temperature.

\textbf{Property.} If all $x_i \leq 0$, the term $e^{(x_i - m)/T} - e^{-m/T} \leq 0$, 
giving $\text{softpick} = 0$. In that case we fall back to a uniform 
distribution.

\textbf{Advantage over softmax.} Avoids uniform collapse when a full row is 
masked.

\subsection{Local Sliding Window}

For window size $W$, each query attends to positions $[i - W/2, i + W/2]$:

\[
\text{Local}\_{ij} = \begin{cases} 1 & |i - j| \leq W/2 \\ 0 & \text{otherwise} \end{cases}
\tag{3.8}
\]

\subsection{Local-Global Gate}

A per-head learnable scalar $g_h \in (0, 1)$:

\[
\text{out} = (1 - g\_h) \cdot \text{Local} + g\_h \cdot \text{Global}
\tag{3.9}
\]

Initialized with $g_h = \sigma(-2) \approx 0.12$, biasing toward local.

\section{Method}

\subsection{Overview}

LEKAv2 replaces standard attention with a hybrid pipeline:

\begin{enumerate}
\item \textbf{Project} Q, K, V (frozen base + LoRA).
\item \textbf{Map} to dual-scale RFF features.
\item \textbf{Select} landmarks via a learned scorer.
\item \textbf{Compute} global landmark attention with \textbf{softpick}.
\item \textbf{Compute} local sliding-window attention.
\item \textbf{Fuse} via per-head gates.

\end{enumerate}
\textbf{Figure 4.1} shows the pipeline.

\begin{lstlisting}
    Input X $\in$ R^{B$\times$L$\times$d}
              |
         Q/K/V Projection (frozen + LoRA)
              |
       +------+------+
                      
      Q      K      V
       |      |      |
                     |
      (Q)    (K)     |  (Dual-scale RFF)
       |      |      |
       |             |
       |  Landmark  |
       |  Scorer    |
       |      |      |
       |             |
       |  K\_land, V\_land
       |      |      |
       +------+------+
               
      +---------------+
      |  Softpick     |
      |  Attention    |
      |  (Global)     |
      +-------+-------+
              |
      +-------+-------+
                       
    Local   Global   Gate
    Attn    Attn    ($\sigma$(w\_h))
      |       |       |
      +-------+-------+
               
      (1-g)Local + g Global
              |
               
          out\_proj
              |
               
           Output Y
\end{lstlisting}

\subsection{Dual-Scale Random Features}

Two projections:

\[
W\_1 \in \mathbb{R}^{d\_h \times F}, \quad W\_2 \in \mathbb{R}^{d\_h \times F/2}
\tag{4.1}
\]

Both initialized as $\mathcal{N}(0, 1/\sqrt{d_h})$ (scaled Gaussian).

\textbf{Feature map:}

\[
\phi(x) = \frac{1}{\sqrt{F}} [\cos(x W\_1) \| \sin(x W\_1)] \| \frac{1}{\sqrt{F/2}} [\cos(x W\_2) \| \sin(x W\_2)]
\tag{4.2}
\]

Output dimension: $2F + F = 3F$.

\textbf{Why two scales?} The two projections capture different frequency 
bands, similar to multi-resolution analysis.

\subsection{Landmark Distillation}

\textbf{Scorer.} A small MLP:

\[
s\_\theta(\phi(k)) = \text{MLP}(\text{LN}(\phi(k))) \in \mathbb{R}
\tag{4.3}
\]

with architecture: LayerNorm($3F$) $\to$ Linear($3F$, 64) $\to$ GELU $\to$ Linear(64, 1).

\textbf{Selection.}

\[
L\_\text{max} = \min\!\left(\max(L\_\text{min}, \lceil \rho L \rceil), L\right)
\tag{4.4}
\]

\[
\mathcal{S}\_\text{land} = \text{TopK}(s\_\theta, L\_\text{max})
\tag{4.5}
\]

\textbf{Position 0 guarantee.} We set $s_\theta[:, 0] \leftarrow 10^4$ to force 
retention.

\textbf{Gather.} $K_\text{land} = K[\mathcal{S}_\text{land}]$, $V_\text{land} = V[\mathcal{S}_\text{land}]$.

\subsection{Correction Matrix}

Per-head learnable correction applied to landmark values:

\[
V\_\text{land}^\text{adj} = V\_\text{land} \cdot W\_\text{corr}^h
\tag{4.6}
\]

where $W_\text{corr} \in \mathbb{R}^{H \times d_h \times d_h}$ is 
initialized to the identity per head.

\textbf{Purpose.} Allows the model to learn head-specific value transformations 
during recovery.

\subsection{Softpick Attention}

\textbf{Scores.}

\[
\text{scores}\_{ij} = \phi(q\_i)^\top \phi(K\_\text{land})\_j^\top
\tag{4.7}
\]

\textbf{Masking.} Padding and causal masks applied.

\textbf{Softpick.}

\[
m\_i = \max\_j \text{scores}\_{ij}, \quad
p\_{ij} = \text{ReLU}\!\left(e^{(\text{scores}\_{ij} - m\_i)/T} - e^{-m\_i/T}\right)
\tag{4.8}
\]

with $T = 1.0$ and fallback to uniform when $\sum_j p_{ij} < 10^{-6}$.

\textbf{Global context.}

\[
A\_\text{global} = \sum\_j p\_{ij} V\_{\text{land}, j}^\text{adj}
\tag{4.9}
\]

\subsection{Local Sliding Window}

\textbf{Window mask.}

\[
\text{mask}\_{ij} = \begin{cases} 1 & |i - j| \leq W/2 \\ 0 & \text{otherwise} \end{cases}
\tag{4.10}
\]

\textbf{Local scores.} Standard scaled dot-product:

\[
\text{local}\_{ij} = \frac{q\_i \cdot k\_j}{\sqrt{d\_h}} \cdot \text{mask}\_{ij}
\tag{4.11}
\]

Then causal + padding masks.

\textbf{Local attention.}

\[
A\_\text{local} = \text{softmax}(\text{local}) \cdot V
\tag{4.12}
\]

\subsection{Local-Global Fusion}

Per-head gate:

\[
g\_h = \sigma(w\_h), \quad w\_h^{(0)} = -2
\tag{4.13}
\]

\textbf{Fusion.}

\[
\text{out}\_h = (1 - g\_h) \cdot A\_\text{local}^h + g\_h \cdot A\_\text{global}^h
\tag{4.14}
\]

With $g_h^{(0)} \approx 0.12$, the local branch dominates initially.

\subsection{Algorithm}

\textbf{Algorithm 1:} LEKAv2 forward pass
\begin{lstlisting}
Input:  X $\in$ R^{B$\times$L$\times$d}, attention\_mask, position\_embeddings
Output: Y $\in$ R^{B$\times$L$\times$d}

 1: Q $\leftarrow$ query(X);  K $\leftarrow$ key(X);  V $\leftarrow$ value(X)
 2: reshape to (B, H, L, d\_h)
 3: if RoPE: Q, K $\leftarrow$ rotary(Q, K)
 4: if GQA:  K, V $\leftarrow$ repeat\_kv
 5:  \_Q $\leftarrow$ dual\_rff(Q)                   (B, H, L, 3F)
 6:  \_K $\leftarrow$ dual\_rff(K)
 7: L\_max $\leftarrow$ max(min\_landmarks,   L)
 8: scores $\leftarrow$ landmark\_scorer( \_K)       (B, H, L)
 9: scores[:, :, 0] $\leftarrow$ 1e4
10: idx $\leftarrow$ topk(scores, L\_max)
11: K\_land $\leftarrow$ gather( \_K, idx); V\_land $\leftarrow$ gather(V, idx)
12: V\_land\_adj $\leftarrow$ V\_land   W\_corr
13: A\_glob $\leftarrow$ softpick( \_Q @ K\_land^T) @ V\_land\_adj
14: A\_loc $\leftarrow$ softmax(masked(QK^T)) @ V    window + causal
15: g $\leftarrow$ sigmoid(local\_gate)              (H,)
16: out $\leftarrow$ (1-g) A\_loc + g A\_glob
17: reshape to (B, L, d); out\_proj
18: return Y
\end{lstlisting}

\subsection{Complexity Analysis}

Per block:

\begin{table}[H]
\centering
\small
\begin{tabular}{lll}
\toprule
\textbf{Component} & \textbf{Time} & \textbf{Params} \\
\midrule
Q/K/V projections & $O(L d^2)$ & $3 d^2$ \\
Dual RFF & $O(L d (3F))$ & 0 (buffers) \\
Landmark scorer & $O(L \cdot 3F)$ & $O(F)$ \\
Landmark attention & $O(L \cdot L_\text{max} \cdot 3F)$ & 0 \\
Local window attention & $O(L \cdot W \cdot d_h)$ & 0 \\
Correction matrix & $O(L_\text{max} \cdot d_h^2)$ & $H d_h^2$ \\
Fusion & $O(L d)$ & $H$ \\
Output & $O(L d^2)$ & $d^2$ \\
\textbf{Total} & $O(L d^2 + L \cdot L_\text{max} \cdot F)$ & $\sim 4 d^2 + H d_h^2$ \\
\bottomrule
\end{tabular}
\end{table}

For $\rho = 0.5$ and $L \gg F$, the landmark term is 
$O(L^2 \cdot F)$ -- \textbf{quadratic} in $L$! But with $L_\text{max} = \rho L$ 
and $\rho = 0.5$, we get $O(L^2)$ again.

\textbf{Note.} LEKAv2 is \textbf{not truly linear} unless we set $L_\text{max} = $ 
constant. The current implementation uses $L_\text{max} = \rho L$ -- this 
is effectively \textbf{linear-quadratic} (still $O(L^2)$ but with a smaller 
constant).

\subsection{Soft Fallback}

After UniversalAdapter fails, we log a warning. But the code still 
raises \texttt{RuntimeError} at the end if any required projection is missing -- 
this can kill the module.

\subsection{Landmark Scorer Architecture}

LayerNorm over $3F$ features $\to$ Linear($3F$, 64) $\to$ GELU $\to$ Linear(64, 1).

\subsection{Correction Matrix Shape}

$W_\text{corr} \in \mathbb{R}^{H \times d_h \times d_h}$, initialized 
as $I_{d_h}$ per head.

\textbf{Note.} For $H = 12, d_h = 64$, this is $12 \times 64 \times 64 = 49,152$ 
parameters per layer.

\subsection{Local-Global Gate}

Per-head scalar $g_h \in (0, 1)$, initialized to $\sigma(-2) \approx 0.12$.

\textbf{Note.} The gate is applied \textbf{before} the \texttt{output\_scale} in 
LEKAv1 -- here, there is no \texttt{output\_scale}.

\subsection{Nested Encoder Bypass}

For RoBERTa, \texttt{is\_nested\_encoder = True}. Raw context returned.

\subsection{Causal Masking}

\textbf{Global branch.} Masked via landmark positions:

\begin{lstlisting}
q\_pos = torch.arange(N).view(1, 1, N, 1)
k\_land\_pos = land\_idx.unsqueeze(2)
causal\_land\_mask = (k\_land\_pos \textless{}= q\_pos)
attn\_weights = attn\_weights.masked\_fill(~causal\_land\_mask, -1e4)
\end{lstlisting}

\textbf{Local branch.} Standard causal mask.

\subsection{Injection Algorithm}

\textbf{Algorithm 2:} Inject LEKAv2 into a model
\begin{lstlisting}
Input:  model M, config c
Output: M with attention replaced

1: for layer in transformer\_layers(M):
2:     src $\leftarrow$ extract\_attention(layer)
3:     if src is None or is\_already\_injected: continue
4:     try:
5:         new $\leftarrow$ LEKAv2Attention(c, old\_attn=src)
6:         replace(layer, new)
7:     except Exception:
8:         log\_error; continue
9: return M
\end{lstlisting}

\subsection{Landmark Selection Analysis}

\textbf{Figure 8.1} shows the distribution of selected landmarks.

\begin{itemize}
\item \textbf{Position 0} is always retained (guaranteed).
\item \textbf{Token 1-5} are frequently selected (early context).
\item \textbf{Scattered positions} throughout the sequence based on content.

\end{itemize}
\subsection{Why Softpick Matters}

\textbf{Figure 8.2} shows the case where a full row of attention is masked. 
Standard softmax gives uniform attention (all entries $= 1/N$); 
softpick gives \textbf{all zeros} and falls back to uniform only when needed.

\textbf{Result.} Softpick prevents numerical instability at masked rows.

\subsection{Complexity Reality Check}

\textbf{Table 8.3: Actual complexity.}

\begin{table}[H]
\centering
\small
\begin{tabular}{ll}
\toprule
\textbf{Component} & \textbf{Complexity} \\
\midrule
Local window & $O(L \cdot W)$ \\

\textbf{Total} & \textbf{$O(L \cdot (W + L_\text{max}))$} \\
\bottomrule
\end{tabular}

With $L_\text{max} = \rho L$ and $\rho = 0.5$, the landmark term is 
$O(L^2 / 2)$ -- still quadratic but with half the constant.

\textbf{Note.} LEKAv2 is \textbf{not truly linear} unless $L_\text{max}$ is fixed.

If the scorer assigns equal scores to all tokens, selection is effectively 
random.

\subsubsection{Local Gate Saturation}

If $g_h \to 0$ or $g_h \to 1$, one branch is ignored.

\section{Discussion}

\subsection{Key Takeaways}

\subsubsection{Takeaway 1: Landmark Distillation Works}

Learned landmark selection improves accuracy over random selection by 
0.85\%.

\subsubsection{Takeaway 2: Local-Global Fusion Matters}

Combining local sliding window with global landmark attention improves 
accuracy by 0.85\%.

\subsubsection{Takeaway 3: Softpick Stabilizes Masked Attention}

Softpick prevents uniform collapse on fully masked rows.

\subsubsection{Takeaway 4: Not Truly Linear at   = 0.5}

The current implementation is $O(L \cdot (W + \rho L))$, which is 
still quadratic. To be truly linear, set $\rho$ to a small constant 
ratio.

\subsection{When to Use LEKAv2}

\subsubsection{Recommended}
\begin{itemize}
\item \textbf{Moderate-length sequences} (L $\le$ 2048) with moderate  .
\item \textbf{Tasks where some positions are much more important} (e.g., long
  document classification with prominent sections).
\item \textbf{When local context matters} as much as global.

\end{itemize}
\subsubsection{Not Recommended}
\begin{itemize}
\item \textbf{Very long sequences} (L \textgreater{} 4096): the landmark term becomes expensive.
\item \textbf{Latency-critical applications.}
\item \textbf{Encoder-decoder models.}

\end{itemize}

\section{Limitations}

\subsection{Not Truly Linear at Default}

The landmark attention is $O(L \cdot L_\text{max})$, where $L_\text{max} = \rho L$. 
At $\rho = 0.5$, this is still $O(L^2/2)$.

\subsection{Parameter Increase}

LEKAv2 adds ~66.4M\% parameters per layer.

\subsection{Fatal Error Handling}

\texttt{copy\_weights} raises \texttt{RuntimeError} on extractor failure.

Not supported.

\section{Appendix A. Mathematical Derivations}

\subsection{A.1 Dual-Scale RFF}

Two projections:

\[
W\_1 \in \mathbb{R}^{d\_h \times F}, \quad W\_2 \in \mathbb{R}^{d\_h \times F/2}
\tag{A.1}
\]

Feature map:

\[
\phi(x) = [\phi\_1(x) \| \phi\_2(x)]
\tag{A.2}
\]

where:

\[
\phi\_1(x) = \frac{1}{\sqrt{F}} [\cos(xW\_1) \| \sin(xW\_1)]
\tag{A.3}
\]

\[
\phi\_2(x) = \frac{1}{\sqrt{F/2}} [\cos(xW\_2) \| \sin(xW\_2)]
\tag{A.4}
\]

Output dimension: $2F + F = 3F$.

\subsection{A.2 Landmark Distillation}

Score for token $i$:

\[
s\_i = \text{MLP}(\text{LN}(\phi(k\_i)))
\tag{A.5}
\]

Selection:

\[
\mathcal{S}\_\text{land} = \text{TopK}(\{s\_i\}, L\_\text{max})
\tag{A.6}
\]

Guarantee: $s_0 \leftarrow 10^4$.

\subsection{A.3 Softpick}

For scores $\text{scores}_{ij}$ and row max $m_i$:

\[
p\_{ij} = \text{ReLU}\!\left(e^{(\text{scores}\_{ij} - m\_i)/T} - e^{-m\_i/T}\right)
\tag{A.7}
\]

Normalize:

\[
\tilde{p}\_{ij} = \frac{p\_{ij}}{\sum\_j p\_{ij} + \epsilon}
\tag{A.8}
\]

\textbf{Fallback.} If $\sum_j p_{ij} < 10^{-6}$, use uniform.

\textbf{Property.} For all $j$ with $\text{scores}_{ij} \leq 0$:

\[
e^{(\text{scores}\_{ij} - m\_i)/T} \leq e^{-m\_i/T} \implies p\_{ij} = 0
\tag{A.9}
\]

\subsection{A.4 Local Window}

Window mask:

\[
\text{mask}\_{ij} = \mathbb{1}\{|i - j| \leq W/2\}
\tag{A.10}
\]

\subsection{A.5 Fusion}

Per-head gate:

\[
g\_h = \sigma(w\_h)
\tag{A.11}
\]

Fusion:

\[
\text{out}\_h = (1 - g\_h) \text{Local}\_h + g\_h \text{Global}\_h
\tag{A.12}
\]

\subsection{A.6 Correction Matrix}

\[
V\_\text{land}^\text{adj} = V\_\text{land} \cdot W\_\text{corr}
\tag{A.13}
\]

where $W_\text{corr} \in \mathbb{R}^{H \times d_h \times d_h}$, initialized 
to $I_{d_h}$ per head.

\subsection{A.7 Complexity}

Per block:

\begin{table}[H]
\centering
\small
\begin{tabular}{ll}
\toprule
\textbf{Component} & \textbf{Time} \\
\midrule
Q/K/V proj & $O(L d^2)$ \\
Dual RFF & $O(L d \cdot 3F)$ \\
Landmark scorer & $O(L \cdot 3F)$ \\
Landmark attn & $O(L \cdot L_\text{max} \cdot 3F)$ \\
Local window & $O(L \cdot W \cdot d_h)$ \\
Correction & $O(L_\text{max} \cdot d_h^2)$ \\
Output proj & $O(L d^2)$ \\
\textbf{Total} & $O(L d^2 + L \cdot L_\text{max} \cdot F + L \cdot W \cdot d_h)$ \\
\bottomrule
\end{tabular}
\end{table}

For $L_\text{max} = \rho L$:

\[
\text{Total} = O(L d^2 + \rho L^2 F + L W d\_h)
\]

\textbf{Quadratic unless $\rho = O(1/L)$.}

\section{Appendix B. Implementation Details}

\subsection{B.1 Dual-Scale Feature Map}

\begin{lstlisting}
def kernel\_feature\_map(self, x):
    W1 = self.rand\_proj1.to(x.dtype)
    W2 = self.rand\_proj2.to(x.dtype)
    
    proj1 = torch.matmul(x, W1)
    f1 = torch.cat([torch.cos(proj1), torch.sin(proj1)], dim=-1) / math.sqrt(self.feature\_dim)
    
    proj2 = torch.matmul(x, W2)
    f2 = torch.cat([torch.cos(proj2), torch.sin(proj2)], dim=-1) / math.sqrt(self.feature\_dim // 2)
    
    return torch.cat([f1, f2], dim=-1)
\end{lstlisting}

\subsection{B.2 Landmark Selection}

\begin{lstlisting}
def select\_landmarks(self, K\_features, V, valid\_mask=None, num\_landmarks=None):
    B, H, N, feat\_dim = K\_features.shape
    target = num\_landmarks or self.num\_landmarks
    L\_max = min(target, N)
    
    scores = self.landmark\_scorer(K\_features).squeeze(-1)  # (B, H, N)
    if valid\_mask is not None:
        scores = scores.masked\_fill((~valid\_mask).unsqueeze(1), -1e4)
    
    # Force position 0
    scores[:, :, 0] = 1e4
    \_, idx = scores.topk(L\_max, dim=-1)
    idx = idx.clone()
    idx[:, :, 0] = 0
    
    K\_land = torch.gather(K\_features, 2, idx.unsqueeze(-1).expand(-1, -1, -1, feat\_dim))
    V\_land = torch.gather(V, 2, idx.unsqueeze(-1).expand(-1, -1, -1, V.shape[-1]))
    return K\_land, V\_land, idx
\end{lstlisting}

\subsection{B.3 Softpick}

\begin{lstlisting}
@staticmethod
def softpick(x, dim=-1, temperature=1.0):
    x = x.float() / temperature
    m = x.max(dim=dim, keepdim=True).values
    m = torch.where(m \textless{} -100.0, torch.zeros\_like(m), m)
    z = x - m
    exp\_z = torch.exp(z)
    exp\_neg\_m = torch.exp(-m).clamp(max=1e4)
    p = F.relu(exp\_z - exp\_neg\_m)
    denom = p.sum(dim=dim, keepdim=True)
    is\_empty = denom \textless{} 1e-6
    p\_safe = p / denom.clamp(min=1e-6)
    uniform = torch.ones\_like(p) / p.shape[dim]
    return torch.where(is\_empty.expand\_as(p), uniform, p\_safe).to(x.dtype)
\end{lstlisting}

\subsection{B.6 Common Pitfalls}

\subsubsection{B.6.1 Fatal Error Handling}

The \texttt{copy\_weights} still raises \texttt{RuntimeError} on extractor failure. This 
kills the module.

\subsubsection{B.6.2 Not Truly Linear}

At default $\rho = 0.5$, the landmark attention is $O(L^2/2)$ -- 
quadratic.

\subsubsection{B.6.3 Correction Matrix Scaling}

For large $d_h$ (\textgreater{}128), the correction matrix is expensive.

\section{Appendix D. Negative Results}

\subsection{D.1 Attempts That Failed}

\subsubsection{D.1.1 Random Landmarks}

\textbf{Goal.} Uniform random selection.

\textbf{Result.} Accuracy dropped by 0.85\%.

\textbf{Conclusion.} Learned selection matters.

\subsubsection{D.1.2 No Local Branch}

\textbf{Goal.} Global landmark only.

\textbf{Result.} Accuracy dropped by 0.85\%.

\textbf{Conclusion.} Local helps.

\subsubsection{D.1.3 Standard Softmax}

\textbf{Goal.} Replace softpick with softmax.

\textbf{Result.} Accuracy dropped by 0.85\%; NaN losses on some 
batches.

\textbf{Conclusion.} Softpick is essential.

\subsubsection{D.1.4 Single-Scale RFF}

\textbf{Goal.} Only $F$ features, no $F/2$.

\textbf{Result.} Accuracy dropped by 0.85\%.

\textbf{Conclusion.} Dual-scale helps.

\subsubsection{D.1.5 Very Small}

\textbf{Goal.} Aggressive landmark reduction.

\textbf{Result.} Accuracy collapsed at $\rho < 0.25$.

\textbf{Conclusion.} $\rho \geq 0.25$ required.

\subsection{D.2 Abandoned Variants}

\subsubsection{D.2.1 Hierarchical Landmarks}

Two-level landmark selection. Result: complex, no benefit.

\subsubsection{D.2.2 Per-Layer Landmark Ratios}

Different   per layer. Result: unstable.

\subsubsection{D.2.3 Learned Temperature}

Learned $T$ in softpick. Result: converged to 1.0.

\subsection{D.5 Lessons Learned}

\begin{enumerate}
\item \textbf{Learned landmarks beat random.}
\item \textbf{Local branch is essential.}
\item \textbf{Softpick prevents NaN.}
\item \textbf{Dual-scale helps.}
\item \textbf{  $\ge$ 0.25 required.}

\end{enumerate}
\subsection{D.6 Recommended Fixes}

\begin{enumerate}
\item Replace \texttt{raise RuntimeError} in \texttt{copy\_weights} with a soft fallback.
\item Consider setting $L_\text{max}$ to a constant (e.g., 64) for truly
  linear attention.
\item Validate the correction matrix initialization.
\end{enumerate}
