
\section{Appendix B. Implementation Details}

\subsection{B.1 k-Means Implementation}

\begin{lstlisting}
def \_kmeans\_assign(self, feat, K, n\_iters):
    B, H, L, d = feat.shape
    step = max(1, L // K)
    init\_idx = torch.arange(0, L, step, device=feat.device)[:K]
    if init\_idx.size(0) \textless{} K:
        extra = torch.zeros(K - init\_idx.size(0), dtype=torch.long, device=feat.device)
        init\_idx = torch.cat([init\_idx, extra])

    centroids = feat[:, :, init\_idx, :].clone()
    cluster\_ids = torch.zeros(B, H, L, dtype=torch.long, device=feat.device)

    feat\_f = feat.float()
    for \_ in range(n\_iters):
        centroids\_f = centroids.float()
        dist = torch.cdist(feat\_f, centroids\_f)
        cluster\_ids = torch.argmin(dist, dim=-1)

        one\_hot = F.one\_hot(cluster\_ids, num\_classes=K).to(dtype=feat.dtype)
        counts = one\_hot.sum(dim=2).clamp(min=1.0).unsqueeze(-1)
        sum\_f = torch.einsum("bhlk,bhld-\textgreater{}bhkd", one\_hot, feat)
        centroids = sum\_f / counts

    return cluster\_ids
\end{lstlisting}

\textbf{Notes:}
\begin{itemize}
\item Deterministic init via stride.
\item \texttt{cdist} in FP32 for stability.
\item Empty clusters handled via \texttt{clamp(min=1.0)}.

\end{itemize}
\subsection{B.2 Attention Loop}

\begin{lstlisting}
outputs = torch.zeros\_like(Q)
for k in range(K\_num):
    mask\_k = (cluster\_ids == k) | (torch.arange(L, device=device).view(1, 1, -1) == 0)
    mask\_k = mask\_k & valid\_mask\_2d.unsqueeze(1)

    scores = torch.matmul(Q.float(), K.float().transpose(-2, -1)) * scale
    scores = scores.masked\_fill(~mask\_k.unsqueeze(2), -1e4)

    eye = torch.eye(L, device=device, dtype=torch.bool).unsqueeze(0).unsqueeze(0)
    scores = scores.masked\_fill(eye, 0.0)

    if is\_causal:
        causal\_mask = torch.triu(torch.full((L, L), True, device=device), diagonal=1).unsqueeze(0).unsqueeze(0)
        scores = scores.masked\_fill(causal\_mask & (~eye), -1e4)

    attn = F.softmax(scores, dim=-1)
    attn = torch.nan\_to\_num(attn, nan=0.0)
    attn = self.dropout(attn.to(dtype))

    out\_k = torch.matmul(attn, V)
    outputs = torch.where(mask\_k.unsqueeze(-1), out\_k, outputs)
\end{lstlisting}

\subsection{B.3 Common Pitfalls}

\subsubsection{B.3.1 Padding in k-Means}

Padding tokens are included in \texttt{joint\_feat}. To exclude:

\begin{lstlisting}
# Exclude padding
valid\_mask\_2d = get\_token\_valid\_mask(attention\_mask, B, L, device)
# Apply mask before clustering
joint\_feat = joint\_feat * valid\_mask\_2d.unsqueeze(1).unsqueeze(-1)
\end{lstlisting}

\subsubsection{B.3.2 FP32 for Numerical Stability}

The attention scores are computed in FP32:

\begin{lstlisting}
scores = torch.matmul(Q.float(), K.float().transpose(-2, -1)) * scale
\end{lstlisting}
