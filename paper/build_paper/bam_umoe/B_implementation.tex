\subsection{B.1 UnifiedMoE Module}

\begin{lstlisting}
class UnifiedMoE(nn.Module):
    def \_\_init\_\_(self, dim, num\_experts=8, top\_k=2, output\_dim=None,
                 noise\_std=0.1, expert\_multiplier=2):
        super().\_\_init\_\_()
        self.output\_scale = nn.Parameter(torch.ones(1))
        self.dim = dim
        self.num\_experts = max(1, num\_experts)
        self.top\_k = min(max(1, top\_k), self.num\_experts)
        self.out\_dim = output\_dim or dim
        self.noise\_std = noise\_std
        self.aux\_loss = torch.tensor(0.0)
        
        self.router\_ln = nn.LayerNorm(self.dim)
        self.router = nn.Linear(self.dim, self.num\_experts, bias=False)
        nn.init.uniform\_(self.router.weight, -0.01, 0.01)
        
        hidden\_dim = self.out\_dim * expert\_multiplier
        self.experts = nn.ModuleList([
            nn.Sequential(
                nn.Linear(dim, hidden\_dim),
                nn.GELU(),
                nn.Linear(hidden\_dim, self.out\_dim),
            )
            for \_ in range(self.num\_experts)
        ])
        
        # Step-0 preservation: final expert layer zero
        for exp in self.experts:
            nn.init.zeros\_(exp[2].weight)
            if exp[2].bias is not None:
                nn.init.zeros\_(exp[2].bias)
        
        self.gate\_bias = nn.Parameter(torch.tensor([-3.0]))
    
    def forward(self, x):
        B, L, D = x.shape
        x\_norm = self.router\_ln(x)
        logits = self.router(x\_norm)
        
        if self.training and self.noise\_std \textgreater{} 0:
            noise = torch.randn\_like(logits) * self.noise\_std
            routing\_logits = logits + noise
        else:
            routing\_logits = logits
        
        topk\_vals, topk\_idx = torch.topk(routing\_logits, self.top\_k, dim=-1)
        scores = F.softmax(topk\_vals, dim=-1)
        scores = torch.nan\_to\_num(scores, nan=0.0, posinf=1.0, neginf=0.0)
        
        # Auxiliary load balancing
        if self.training and self.num\_experts \textgreater{} 1:
            router\_probs = F.softmax(logits.float(), dim=-1).view(-1, self.num\_experts)
            expert\_mask = F.one\_hot(topk\_idx, num\_classes=self.num\_experts).float()
            density = expert\_mask.sum(dim=2).view(-1, self.num\_experts).mean(dim=0)
            prob\_mean = router\_probs.mean(dim=0)
            self.aux\_loss = self.num\_experts * torch.sum(density * prob\_mean)
        else:
            self.aux\_loss = torch.tensor(0.0, device=x.device, dtype=torch.float32)
        
        # Expert forward
        experts\_out = torch.stack([exp(x) for exp in self.experts], dim=2)
        topk\_out = torch.gather(
            experts\_out, 2,
            topk\_idx.unsqueeze(-1).expand(B, L, self.top\_k, self.out\_dim)
        )
        out = torch.sum(topk\_out * scores.unsqueeze(-1), dim=2)
        out = torch.clamp(torch.nan\_to\_num(out, nan=0.0), min=-65000.0, max=65000.0)
        
        gate = torch.sigmoid(self.gate\_bias)
        return out * self.output\_scale * gate
\end{lstlisting}

\subsection{B.2 Numerical Stability}

\begin{itemize}
\item \textbf{FP32 scores.} Attention scores in FP32 to avoid FP16 overflow.
\item \textbf{MoE clamp.} Expert outputs clamped to $\pm$65000 (FP16 max).
\item \textbf{NaN guards.} \texttt{torch.nan\_to\_num} on all intermediate outputs.
\item \textbf{Noise masking.} Training-time noise added only to logits, not expert
  outputs.

\end{itemize}
\subsection{B.3 Common Pitfalls}

\subsubsection{B.3.1 Forgetting Zero-Init}

Without zero-initialized final expert layers, step-0 forward pass differs 
from pretrained, causing catastrophic drift.

\subsubsection{B.3.2 Gate Bias Too High}

Gate bias \textgreater{} -1 gives fast ramp-up, destabilizing early training.

\subsubsection{B.3.3 Cross-Attention MoE}

Applying MoE to cross-attention disrupts encoder-decoder alignment. Skip it.

\subsubsection{B.3.4 Conv1D Transpose}

GPT-2 Conv1D stores \texttt{(in, out)}. Must transpose.

\subsubsection{B.3.5 Nested Encoder Bypass}

For RoBERTa, do NOT apply \texttt{out\_lin} -- the outer layer does it.
