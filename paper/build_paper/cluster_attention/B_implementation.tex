\subsection{B.1 K-Means Implementation}

\begin{lstlisting}
def \_kmeans\_cluster(self, Q, K\_num, iters=3, valid\_mask=None):
    B, H, L, d = Q.shape
    K = min(K\_num, L)
    
    # Padding-aware initialization
    if valid\_mask is not None:
        valid\_f = valid\_mask.float()
        num\_valid = valid\_f.sum(dim=-1, keepdim=True).clamp(min=1.0)
        step = (num\_valid / K).clamp(min=1.0)
        k\_idx = torch.arange(K, device=Q.device, dtype=torch.float32).unsqueeze(0)
        target\_ranks = (k\_idx * step).clamp(max=num\_valid - 1.0).long()
        
        ranks = (valid\_f.cumsum(dim=-1) - 1.0) * valid\_f
        is\_valid\_3d = valid\_mask.unsqueeze(1)
        rank\_diff = torch.where(
            is\_valid\_3d,
            (ranks.unsqueeze(1) - target\_ranks.unsqueeze(-1)).abs(),
            torch.tensor(1e6, device=Q.device),
        )
        chosen\_indices = torch.argmin(rank\_diff, dim=-1)
        idx\_expanded = chosen\_indices.unsqueeze(1).unsqueeze(-1).expand(B, H, K, d)
        centroids = torch.gather(Q, 2, idx\_expanded).clone()
    else:
        step = max(1, L // K)
        init\_indices = torch.arange(0, L, step, device=Q.device)[:K]
        centroids = Q[:, :, init\_indices, :].clone()
    
    # Lloyd iterations
    cluster\_ids = torch.zeros(B, H, L, dtype=torch.long, device=Q.device)
    counts = torch.ones(B, H, K, 1, dtype=Q.dtype, device=Q.device)
    Q\_f = Q.float()
    
    for \_ in range(iters):
        centroids\_f = centroids.float()
        dist = torch.cdist(Q\_f, centroids\_f)
        if valid\_mask is not None:
            dist = dist.masked\_fill(~valid\_mask.unsqueeze(1).unsqueeze(-1), 1e6)
        cluster\_ids = torch.argmin(dist, dim=-1)
        
        one\_hot = F.one\_hot(cluster\_ids, num\_classes=K).to(dtype=Q.dtype)
        if valid\_mask is not None:
            one\_hot = one\_hot * valid\_mask.unsqueeze(1).unsqueeze(-1).to(dtype=Q.dtype)
        
        counts = one\_hot.sum(dim=2).unsqueeze(-1)
        counts\_clamped = counts.clamp(min=1.0)
        sum\_q = torch.einsum("bhlk,bhld-\textgreater{}bhkd", one\_hot, Q)
        centroids = torch.where(counts \textgreater{} 0, sum\_q / (counts\_clamped + 1e-8), centroids)
        centroids = torch.nan\_to\_num(centroids, nan=0.0)
    
    return centroids, cluster\_ids, counts
\end{lstlisting}

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 Forgetting Padding Filter}

Without padding mask, K-means will pull centroids toward padding.

\subsubsection{B.5.2 Forgetting to Clamp Counts}

Empty clusters cause division by zero. Clamp counts to $\ge$1.

\subsubsection{B.5.3 Forgetting Temperature Clamp}

Unclamped $\tau$ can cause degenerate softmax (all mass on one cluster).

\subsubsection{B.5.4 Causal Decoder Clustering}

Do NOT apply clustering to causal decoders. Use exact attention instead.

\subsubsection{B.5.5 Conv1D Transpose}

GPT-2 Conv1D stores weights as \texttt{(in, out)}. Must transpose.

\subsubsection{B.5.6 Nested Encoder Bypass}

For RoBERTa, do not apply \texttt{out\_proj} -- the outer layer does.
