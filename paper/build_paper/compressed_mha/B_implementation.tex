\section{Appendix B. Implementation Details}

\subsection{B.1 Core Forward Pass}

\begin{lstlisting}
def forward(self, hidden\_states, attention\_mask=None, **kwargs):
    B, N, \_ = hidden\_states.shape
    
    # 1. Joint QKV projection (SVD + LoRA)
    proj = torch.matmul(hidden\_states, self.Vh\_qkv.t()) * self.S\_qkv.unsqueeze(0).unsqueeze(0)
    qkv\_comp = torch.matmul(proj, self.U\_qkv.t())
    if self.lora\_rank \textgreater{} 0:
        lora\_qkv = F.linear(F.linear(hidden\_states, self.lora\_A\_qkv), self.lora\_B\_qkv) * self.lora\_scaling
        qkv\_comp = qkv\_comp + lora\_qkv
    
    # 2. Split
    q\_raw, k\_raw, v\_raw = torch.split(qkv\_comp, 
                                       [self.all\_head\_size, self.kv\_head\_size, self.kv\_head\_size], dim=-1)
    q = q\_raw + self.bias\_q
    k = k\_raw + self.bias\_k
    v = v\_raw + self.bias\_v
    
    # 3. Reshape
    q = q.view(B, N, self.num\_heads, self.head\_dim).transpose(1, 2)
    k = k.view(B, N, self.num\_kv\_heads, self.head\_dim).transpose(1, 2)
    v = v.view(B, N, self.num\_kv\_heads, self.head\_dim).transpose(1, 2)
    
    # 4. RoPE
    if position\_embeddings is not None:
        q, k = apply\_rotary\_pos\_emb(q, k, cos, sin)
    
    # 5. GQA
    if self.kv\_repeat \textgreater{} 1:
        k = repeat\_kv(k, self.kv\_repeat)
        v = repeat\_kv(v, self.kv\_repeat)
    
    # 6. Attention
    scores = torch.matmul(q, k.transpose(-2, -1)) * (1.0 / math.sqrt(self.head\_dim))
    # masks
    if is\_decoder:
        causal = torch.tril(torch.ones(N, N))
        scores = scores.masked\_fill(~causal, -1e4)
    attn = F.softmax(scores.float(), dim=-1)
    context = torch.matmul(attn, v)
    context = context.transpose(1, 2).contiguous().view(B, N, self.all\_head\_size)
    
    # 7. Output
    if self.is\_nested\_encoder:
        return context, None
    out\_proj = torch.matmul(context, self.Vh\_out.t()) * self.S\_out.unsqueeze(0).unsqueeze(0)
    out = torch.matmul(out\_proj, self.U\_out.t()) + self.bias\_out
    if self.lora\_rank \textgreater{} 0:
        lora\_out = F.linear(F.linear(context, self.lora\_A\_out), self.lora\_B\_out) * self.lora\_scaling
        out = out + lora\_out
    return out, None
\end{lstlisting}

\subsection{B.2 Nested Detection Fix}

\begin{lstlisting}
# Priority 1: explicit flat output projections
if hasattr(src, "out\_lin") or hasattr(src, "c\_proj") or hasattr(src, "o\_proj"):
    self.is\_nested\_encoder = False
# Priority 2: nested structure
elif (hasattr(src, "self") and hasattr(src, "output") 
      and hasattr(src.output, "dense")):
    self.is\_nested\_encoder = True
# Priority 3: class name (last resort)
elif cls\_name in ("BertSelfAttention", "RobertaSelfAttention"):
    self.is\_nested\_encoder = True
else:
    self.is\_nested\_encoder = False
\end{lstlisting}

\textbf{Why the order matters:} \texttt{DistilBert.MultiHeadSelfAttention} ends with 
"SelfAttention" -- if we checked class name first, we'd incorrectly mark it 
as nested.

\begin{lstlisting}
rank\_val = rank or kwargs.get("rank\_qkv", rank\_qkv)
if rank is None and "rank\_qkv" not in kwargs:
    rank\_val = max(rank\_qkv, min(384, all\_head\_size + 2 * kv\_head\_size, hidden\_size))

\textbf{Important:} The default value is designed to be \textbf{safe but not compressive} 
(rank = 384 for $d=768$). For real compression, pass explicit ranks.

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 Wrong Nested Detection Order}

Checking class name before \texttt{out\_lin} marks DistilBERT as nested.

\subsubsection{B.5.2 Forgetting RoPE}

For Qwen models, RoPE must be applied before GQA expansion.

\subsubsection{B.5.3 Conv1D Orientation}

GPT-2's Conv1D stores weights as (in, out). Must transpose.

\subsubsection{B.5.4 Missing `is\_nested\_encoder` in Output}

If nested but you apply \texttt{out\_proj}, you get double projection.

\subsubsection{B.5.5 LoRA on Frozen SVD}

If LoRA isn't initialized to zero, the initial output differs from the 
pretrained model.
