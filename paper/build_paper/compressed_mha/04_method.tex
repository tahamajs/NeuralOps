\section{Method}

\subsection{Overview}

Compressed MHA has four stages:

\begin{enumerate}
\item \textbf{Weight extraction} ( 4.2): extract Q, K, V, O from pretrained attention.
\item \textbf{Joint QKV SVD} ( 4.3): stack and factorize Q, K, V.
\item \textbf{Separate O SVD} ( 4.4): factorize output projection.
\item \textbf{LoRA recovery} ( 4.5): add trainable adapters on both branches.

\end{enumerate}
\begin{lstlisting}
        Input X
           |
            
    +----------------------+
    | Joint QKV SVD        |
    |  Vh\_qkv: (r, d)      |  $\leftarrow$ shared input basis
    |  S\_qkv: (r,)         |
    |  U\_qkv: (3d, r)      |
    +-------+--------------+
            |
    +------- --------------+
    | QKV LoRA             |
    |  lora\_A\_qkv, lora\_B\_qkv |
    +-------+--------------+
            |
    +------- --------------+
    | Attention            |
    |  softmax(Q K^T) V    |
    +-------+--------------+
            |
    +------- --------------+
    | Separate O SVD       |
    |  Vh\_out, S\_out, U\_out|
    +-------+--------------+
            |
    +------- --------------+
    | O LoRA               |
    +-------+--------------+
            |
        Output
\end{lstlisting}

\subsection{Weight Extraction}

We extract weights via \texttt{copy\_weights}, handling four patterns:

\subsubsection{DistilBERT (Flat)}

\begin{lstlisting}
w\_q = src.q\_lin.weight
w\_k = src.k\_lin.weight
w\_v = src.v\_lin.weight
w\_o = src.out\_lin.weight
\end{lstlisting}

\textbf{Nested detection:} \texttt{out\_lin} exists $\to$ \texttt{is\_nested\_encoder = False}.

\subsubsection{RoBERTa (Nested)}

\begin{lstlisting}
if hasattr(src, "self") and hasattr(src, "output"):
    is\_nested\_encoder = True
    w\_q = src.self.query.weight
    w\_k = src.self.key.weight
    w\_v = src.self.value.weight
    w\_o = src.output.dense.weight
\end{lstlisting}

\textbf{Nested detection:} No \texttt{out\_lin}/\texttt{o\_proj}/\texttt{c\_proj} $\to$ \texttt{is\_nested\_encoder = True}.

\subsubsection{GPT-2 (Conv1D)}

\begin{lstlisting}
w = src.c\_attn.weight.t()  # Conv1D stored as (in, out)
q\_w, k\_w, v\_w = w.chunk(3, dim=0)
w\_o = src.c\_proj.weight.t()
\end{lstlisting}

\subsubsection{Qwen (GQA)}

\begin{lstlisting}
w\_q = src.q\_proj.weight
w\_k = src.k\_proj.weight
w\_v = src.v\_proj.weight
w\_o = src.o\_proj.weight
\end{lstlisting}

\subsection{Joint QKV SVD}

\subsubsection{Stacking}

We concatenate along the output dimension:

\[
W\_{QKV} = \begin{pmatrix} W\_Q \\ W\_K \\ W\_V \end{pmatrix}
\]

Shape: $[H d_h + 2 H_{kv} d_h, d]$.

\subsubsection{SVD}

Compute the full SVD:

\[
W\_{QKV} = U\_{QKV} \Sigma\_{QKV} V\_{QKV}^\top
\]

Truncate to rank $r_{qkv}$:

\[
r\_{qkv} = \min(\text{rank\_qkv}, H d\_h + 2 H\_{kv} d\_h, d)
\]

Store the factors:

\[
U\_{QKV, r} \in \mathbb{R}^{(H d\_h + 2 H\_{kv} d\_h) \times r\_{qkv}}
\]

\[
S\_{QKV, r} \in \mathbb{R}^{r\_{qkv}}, \quad
V\_{QKV, r} \in \mathbb{R}^{r\_{qkv} \times d}
\]

\subsubsection{Forward Pass}

Given input $X$:

\[
\text{Proj}\_1 = X V\_{QKV, r}^\top \in \mathbb{R}^{B \times N \times r\_{qkv}}
\]

\[
\text{Proj}\_2 = \text{Proj}\_1 \cdot \text{diag}(S\_{QKV, r}) \in \mathbb{R}^{B \times N \times r\_{qkv}}
\]

\[
\text{QKV} = \text{Proj}\_2 U\_{QKV, r}^\top \in \mathbb{R}^{B \times N \times (H d\_h + 2 H\_{kv} d\_h)}
\]

Then split into Q, K, V.

\subsubsection{Parameter Count}

\begin{table}[H]
\centering
\small
\begin{tabular}{ll}
\toprule
\textbf{Component} & \textbf{Parameters} \\
\midrule
$V_{QKV, r}$ & $r_{qkv} \cdot d$ \\
$S_{QKV, r}$ & $r_{qkv}$ \\
$U_{QKV, r}$ & $r_{qkv} \cdot (H d_h + 2 H_{kv} d_h)$ \\

\end{tabular}
\end{table}

For $H = H_{kv}$:
\[
\text{Total}\_{QKV} = r\_{qkv} (2d + 2 d\_h) = 2 r\_{qkv} (d + d\_h)
\]

\subsection{Separate O SVD}

The output projection $W_O \in \mathbb{R}^{d \times H d_h}$ gets its own SVD:

\[
W\_O = U\_O \Sigma\_O V\_O^\top
\]

Truncate to rank $r_{out}$:

\[
r\_{out} = \min(\text{rank\_out}, d, H d\_h)
\]

Forward:

\[
\text{Out}\_1 = \text{Context} V\_{O, r}^\top \in \mathbb{R}^{B \times N \times r\_{out}}
\]

\[
\text{Out}\_2 = \text{Out}\_1 \cdot \text{diag}(S\_{O, r}) \in \mathbb{R}^{B \times N \times r\_{out}}
\]

\[
\text{Out} = \text{Out}\_2 U\_{O, r}^\top + b\_O \in \mathbb{R}^{B \times N \times d}
\]

\textbf{Parameter count:}
\[
\text{Total}\_O = r\_{out} (d + H d\_h) + r\_{out}
\]

\subsection{LoRA Recovery}

Both branches have LoRA adapters:

\subsubsection{QKV LoRA}

\[
\text{QKV}\_\text{LoRA} = \frac{\alpha}{r\_L} B\_{QKV} A\_{QKV} \cdot X
\]

Added to the SVD output:

\[
\text{QKV}\_\text{final} = \text{QKV}\_\text{SVD} + \text{QKV}\_\text{LoRA}
\]

\subsubsection{Output LoRA}

\[
\text{Out}\_\text{LoRA} = \frac{\alpha}{r\_L} B\_O A\_O \cdot \text{Context}
\]

Added after SVD output.

\subsubsection{Trainable Parameters}

\subsection{The Full Forward Pass}

\textbf{Algorithm 1:} Compressed MHA forward
\begin{lstlisting}
Input:  X $\in$ R^{B$\times$N$\times$d}, attention\_mask, position\_embeddings
Output: Y $\in$ R^{B$\times$N$\times$d}

 1: Proj\_1 $\leftarrow$ X @ Vh\_qkv^T
 2: Proj\_2 $\leftarrow$ Proj\_1 * S\_qkv
 3: QKV $\leftarrow$ Proj\_2 @ U\_qkv^T
 4: if lora\_rank \textgreater{} 0:
 5:     QKV $\leftarrow$ QKV + lora\_B\_qkv(lora\_A\_qkv(X)) * scaling
 6: Q, K, V $\leftarrow$ split(QKV)
 7: Q $\leftarrow$ Q + bias\_q; K $\leftarrow$ K + bias\_k; V $\leftarrow$ V + bias\_v
 8: if RoPE: Q, K $\leftarrow$ RoPE(Q, K)
 9: if GQA: K, V $\leftarrow$ repeat\_kv(K, V)
10: scores $\leftarrow$ Q @ K^T / $\sqrt{}$d\_h
11: if causal: mask(scores)
12: if padding: mask(scores)
13: Attn $\leftarrow$ softmax(scores)
14: Context $\leftarrow$ Attn @ V
15: if is\_nested\_encoder:
16:     return Context  # outer layer applies W\_O
17: Out\_1 $\leftarrow$ Context @ Vh\_out^T
18: Out\_2 $\leftarrow$ Out\_1 * S\_out
19: Out $\leftarrow$ Out\_2 @ U\_out^T + bias\_out
20: if lora\_rank \textgreater{} 0:
21:     Out $\leftarrow$ Out + lora\_B\_out(lora\_A\_out(Context)) * scaling
22: return Out
\end{lstlisting}

\subsection{Nested Encoder Handling}

\textbf{Critical detail.} For nested encoders (RoBERTa), our module \textbf{must not} 
apply its own output projection:

\begin{lstlisting}
if self.is\_nested\_encoder:
    out = context
else:
    # Apply SVD output projection
    out = ...
\end{lstlisting}

The outer \texttt{BertSelfOutput.dense} will apply the projection.

\textbf{Detection rule (in \texttt{copy\_weights}):}

\begin{lstlisting}
# Check for flat output projections FIRST
if hasattr(src, "out\_lin") or hasattr(src, "c\_proj") or hasattr(src, "o\_proj"):
    self.is\_nested\_encoder = False
elif (hasattr(src, "self") and hasattr(src, "output")):
    self.is\_nested\_encoder = True
else:
    self.is\_nested\_encoder = False
\end{lstlisting}

\textbf{Why order matters.} DistilBERT's \texttt{MultiHeadSelfAttention} ends with 
"SelfAttention" and has \texttt{out\_lin}. An earlier implementation checked the 
class name \textbf{first}, mistakenly marking DistilBERT as nested $\to$ double 
projection $\to$ 50.92\% collapse. This version checks \texttt{out\_lin} first.

\subsection{Theoretical Notes}

\subsubsection{Joint SVD Optimality}

\textbf{Claim.} The joint SVD of stacked $[W_Q; W_K; W_V]$ gives the minimum 
Frobenius error for any rank-$r$ approximation of the stacked matrix.

\textbf{Proof.} Direct application of Eckart-Young to $W_{QKV}$. $\square$

\subsubsection{Shared Basis}

\textbf{Claim.} The shared basis $V_{QKV, r}$ captures the common structure of Q, 
K, V in the input space.

\textbf{Proof sketch.} If $W_Q \approx W_Q V V^\top$ for a shared subspace $V$, the 
joint SVD recovers $V$ as the dominant right singular vectors. $\square$

\subsubsection{Reconstruction Error}

For the joint SVD:

\[
\|W\_{QKV} - U\_{QKV, r} \Sigma\_r V\_{QKV, r}^\top\|\_F = \sqrt{\sum\_{i \textgreater{} r} \sigma\_i^2}
\]

\subsubsection{Causality}

\textbf{Claim.} The causal mask is preserved.

\textbf{Proof.} We apply a triangular mask before softmax, standard for causal 
attention. $\square$
