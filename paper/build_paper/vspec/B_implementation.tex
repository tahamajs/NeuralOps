\subsection{B.1 The DynamicLowRankProjector}

\begin{lstlisting}
class DynamicLowRankProjector(nn.Module):
    def \_\_init\_\_(self, rank, in\_dim, out\_dim, lora\_rank=16):
        super().\_\_init\_\_()
        self.U = nn.Parameter(torch.randn(out\_dim, rank) * 0.02)
        self.S = nn.Parameter(torch.ones(rank))
        self.V = nn.Parameter(torch.randn(in\_dim, rank) * 0.02)
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.empty(lora\_rank, in\_dim))
            self.lora\_B = nn.Parameter(torch.empty(out\_dim, lora\_rank))
            nn.init.kaiming\_uniform\_(self.lora\_A, a=math.sqrt(5))
            nn.init.zeros\_(self.lora\_B)
        self.freeze\_base\_weights()
    
    def set\_weight(self, weight, bias=None):
        U, S, Vh = torch.linalg.svd(weight.float(), full\_matrices=False)
        eff\_rank = min(self.rank, U.size(1), Vh.size(0))
        self.U.data.zero\_()
        self.S.data.zero\_()
        self.V.data.zero\_()
        self.U.data[:min(self.U.size(0), U.size(0)), :eff\_rank].copy\_(U[:min(self.U.size(0), U.size(0)), :eff\_rank])
        self.S.data[:eff\_rank].copy\_(S[:eff\_rank])
        self.V.data[:min(self.V.size(0), Vh.size(1)), :eff\_rank].copy\_(Vh[:eff\_rank, :min(self.V.size(0), Vh.size(1))].t())
        if bias is not None:
            self.bias.data[:min(...)].copy\_(bias[:min(...)])
    
    def forward(self, x, gate):
        z = torch.matmul(x, self.V)              # (B, k, r)
        z = z * self.S.unsqueeze(0).unsqueeze(0) # broadcast
        z = z * gate.unsqueeze(1)                # spectral gate
        out = torch.matmul(z, self.U.t())        # (B, k, out)
        if self.bias is not None:
            out = out + self.bias
        if self.lora\_A is not None:
            lora\_out = F.linear(F.linear(x, self.lora\_A), self.lora\_B) * self.scaling
            out = out + lora\_out
        return out
\end{lstlisting}

\subsection{B.2 The VariationalSpectralGate}

\begin{lstlisting}
class VariationalSpectralGate(nn.Module):
    def \_\_init\_\_(self, hidden\_dim, rank):
        super().\_\_init\_\_()
        self.net = nn.Sequential(
            nn.Linear(hidden\_dim, 64), nn.GELU(), nn.Linear(64, rank * 2)
        )
        nn.init.zeros\_(self.net[-1].weight)
        nn.init.constant\_(self.net[-1].bias[:rank], 3.0)
        nn.init.constant\_(self.net[-1].bias[rank:], -3.0)
    
    def forward(self, x, valid\_mask=None):
        pooled = \_masked\_mean(x, valid\_mask).float()
        stats = self.net(pooled)
        mu, log\_sigma = stats.chunk(2, dim=-1)
        log\_sigma = torch.clamp(log\_sigma, min=-10.0, max=2.0)
        if self.training:
            eps = torch.randn\_like(mu)
            z = mu + eps * torch.exp(log\_sigma)
        else:
            z = mu
        g = torch.sigmoid(z).to(x.dtype)
        return g, mu, log\_sigma
\end{lstlisting}

class HeadGate(nn.Module):
    def \_\_init\_\_(self, hidden\_dim, n\_heads):
        super().\_\_init\_\_()
        self.net = nn.Sequential(
            nn.Linear(hidden\_dim, 32), nn.GELU(), nn.Linear(32, n\_heads)
        )
        nn.init.zeros\_(self.net[-1].weight)
        nn.init.constant\_(self.net[-1].bias, 3.0)
    
    def forward(self, x, valid\_mask=None):
        pooled = \_masked\_mean(x, valid\_mask).float()
        return torch.sigmoid(self.net(pooled)).to(x.dtype)

\subsection{B.4 The TokenRouter}

\begin{lstlisting}
class TokenRouter(nn.Module):
    def \_\_init\_\_(self, hidden\_dim, keep\_ratio=0.75, min\_tokens=8):
        super().\_\_init\_\_()
        self.net = nn.Sequential(
            nn.Linear(hidden\_dim, 32), nn.GELU(), nn.Linear(32, 1)
        )
        self.keep\_ratio = keep\_ratio
        self.min\_tokens = min\_tokens
    
    def forward(self, x, valid\_mask=None, is\_decoder=False):
        B, N, D = x.shape
        scores = torch.sigmoid(self.net(x.float()).squeeze(-1))
        if valid\_mask is not None:
            scores = scores.masked\_fill(~valid\_mask, -10000.0)
        scores[:, 0] = 1e4
        keep = min(N, max(1, max(self.min\_tokens, int(N * self.keep\_ratio))))
        idx = scores.topk(keep, dim=1).indices
        idx, \_ = torch.sort(idx, dim=1)
        routed = torch.gather(x, 1, idx.unsqueeze(-1).expand(-1, -1, D))
        return routed, idx
\end{lstlisting}

\subsection{B.7 Common Pitfalls}

\subsubsection{B.7.1 Variational at Inference}

\textbf{Wrong:}
\begin{lstlisting}
g = torch.sigmoid(mu + torch.randn\_like(mu) * torch.exp(log\_sigma))
\end{lstlisting}

\textbf{Right:}
\begin{lstlisting}
g = torch.sigmoid(mu)  # only at inference
\end{lstlisting}

\subsubsection{B.7.2 Forgetting `s[:, 0] = 1e4`}

Without this, \texttt{[CLS]} may be dropped.

\subsubsection{B.7.3 RoPE Not Gathered}

If you apply RoPE to full-length Q, K without gathering, the positions 
are wrong.

\subsubsection{B.7.4 Nested Encoder Routing}

For RoBERTa/BERT, we skip routing entirely:

\begin{lstlisting}
if self.is\_nested\_encoder:
    routed = hidden\_states
    idx = torch.arange(N, device=device).unsqueeze(0).expand(B, -1)
\end{lstlisting}
