\subsection{B.3 SDPA Fallback}

\begin{lstlisting}
def \_sdpa\_fallback(self, q, k, v, attn\_mask, causal):
    if causal and attn\_mask is None:
        return F.scaled\_dot\_product\_attention(q, k, v, is\_causal=True)
    else:
        if causal:
            causal\_m = torch.triu(torch.full((q.shape[-2], k.shape[-2]), float("-inf")), diagonal=1)
            attn\_mask = causal\_m.unsqueeze(0).unsqueeze(0) if attn\_mask is None else attn\_mask + causal\_m
        return F.scaled\_dot\_product\_attention(q, k, v, attn\_mask=attn\_mask)
\end{lstlisting}

\subsection{B.5 Common Pitfalls}

\subsubsection{B.5.1 Forgetting import fallback}
\begin{lstlisting}
try:
    from flash\_attn import flash\_attn\_func
    FLASH\_ATTN\_AVAILABLE = True
except ImportError:
    FLASH\_ATTN\_AVAILABLE = False
\end{lstlisting}

\subsubsection{B.5.2 Wrong contiguity}
FlashAttention requires contiguous tensors in \texttt{(B, L, H, d\_h)} layout:
\begin{lstlisting}
q\_fa = q.transpose(1, 2).contiguous()
\end{lstlisting}

\subsubsection{B.5.3 FP8 without Hopper}
\texttt{torch.float8\_e4m3fn} exists on any CUDA, but tensor cores only on Hopper.

\subsubsection{B.5.4 Nested encoder double projection}
For RoBERTa: \texttt{is\_nested\_encoder = True} skips \texttt{out\_proj}.
