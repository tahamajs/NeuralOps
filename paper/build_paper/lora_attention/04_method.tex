\section{Method}

\subsection{Overview}

LoRA Attention replaces each of the four attention projections with a 
\textbf{LoRALinear} module:

\[
y = \underbrace{W\_\text{frozen} x}\_{\text{frozen, copied from pretrained}} + 
    \underbrace{\frac{\alpha}{r\_L} B A x}\_{\text{trainable LoRA}}
\tag{4.1}
\]

\textbf{Figure 4.1} shows the pipeline. The base weight is copied from the 
pretrained attention via \texttt{copy\_weights}, and LoRA adapters are initialized 
to zero ($B = 0$), so the initial forward pass exactly matches the baseline.

\begin{lstlisting}
        Input X
           |
            
    +----------------------+
    | Base W (frozen)      |
    | + LoRA (trainable)   |
    +-------+--------------+
            |
    +------- --------------+
    | Q, K, V projections  |
    +-------+--------------+
            |
    +------- --------------+
    | RoPE (if applicable) |
    +-------+--------------+
            |
    +------- --------------+
    | GQA repeat (if app.) |
    +-------+--------------+
            |
    +------- --------------+
    | Scaled dot-product   |
    | attention            |
    +-------+--------------+
            |
    +------- --------------+
    | Output projection    |
    | + LoRA               |
    +-------+--------------+
            |
        Output
\end{lstlisting}

\subsection{LoRALinear Module}

\begin{lstlisting}
class LoRALinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, bias=True, 
                 lora\_rank=16, lora\_alpha=32.0, freeze\_base=False):
        super().\_\_init\_\_()
        self.weight = nn.Parameter(torch.empty(out\_features, in\_features))
        if bias:
            self.bias = nn.Parameter(torch.empty(out\_features))
        
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.empty(lora\_rank, in\_features))
            self.lora\_B = nn.Parameter(torch.empty(out\_features, lora\_rank))
            # A: Kaiming, B: zeros
            nn.init.kaiming\_uniform\_(self.lora\_A)
            nn.init.zeros\_(self.lora\_B)
        
        if freeze\_base:
            self.weight.requires\_grad = False
            if self.bias is not None:
                self.bias.requires\_grad = False
        self.lora\_rank = lora\_rank
        self.lora\_alpha = lora\_alpha
        self.lora\_scaling = lora\_alpha / lora\_rank if lora\_rank \textgreater{} 0 else 1.0
    
    def forward(self, x):
        out = F.linear(x, self.weight, self.bias)
        if self.lora\_rank \textgreater{} 0:
            out = out + F.linear(F.linear(x, self.lora\_A), self.lora\_B) * self.lora\_scaling
        return out * self.output\_scale
\end{lstlisting}

\textbf{Note.} We add a trainable \texttt{output\_scale} scalar initialized to 1.0. This 
allows the model to globally re-scale the attention output if needed.

\subsection{Weight Transfer}

\texttt{copy\_weights} extracts base weights from the pretrained attention and 
copies them into the LoRALinear modules. It handles four patterns:

\subsubsection{Pattern 1: GPT-2 Conv1D}

\begin{lstlisting}
if hasattr(src, "c\_attn"):
    weight = src.c\_attn.weight.data
    if weight.shape[1] == 3 * self.hidden\_size:
        weight = weight.t()
    q\_w, k\_w, v\_w = weight.chunk(3, dim=0)
    extracted["q"] = (q\_w, ...)
    extracted["k"] = (k\_w, ...)
    extracted["v"] = (v\_w, ...)
\end{lstlisting}

\subsubsection{Pattern 2: RoBERTa Nested}

\begin{lstlisting}
if hasattr(src, "self"):
    s = src.self
    extracted["q"] = (s.query.weight, ...)
    extracted["k"] = (s.key.weight, ...)
    extracted["v"] = (s.value.weight, ...)
if hasattr(src, "output") and hasattr(src.output, "dense"):
    extracted["o"] = (src.output.dense.weight, ...)
\end{lstlisting}

\subsubsection{Pattern 3: Named Projections}

\begin{lstlisting}
named\_map = {
    "q": ["q\_proj", "q\_lin", "query"],
    "k": ["k\_proj", "k\_lin", "key"],
    "v": ["v\_proj", "v\_lin", "value"],
    "o": ["o\_proj", "out\_proj", "out\_lin", "dense"],
}
\end{lstlisting}

\subsubsection{Pattern 4: UniversalAdapter Fallback}

\begin{lstlisting}
for proj in ["q", "k", "v", "o"]:
    try:
        w, b = UniversalAdapter.get\_weight\_bias(src, proj)
        if w is not None:
            extracted[proj] = (w, b)
    except Exception:
        pass
\end{lstlisting}

\subsection{Nested Detection}

The critical detection logic:

\begin{lstlisting}
cls\_name = src.\_\_class\_\_.\_\_name\_\_
has\_out\_proj = any(hasattr(src, attr) for attr in 
                   ("dense", "out\_proj", "c\_proj", "out\_lin", "o\_proj", ...))

is\_nested = False
if hasattr(src, "self") and hasattr(getattr(src, "self"), "query"):
    is\_nested = True
elif (cls\_name in ("BertSelfAttention", "RobertaSelfAttention", ...)
      or (cls\_name.endswith("SelfAttention") and not has\_out\_proj)) and not has\_out\_proj:
    is\_nested = True
self.is\_nested\_encoder = bool(is\_nested)
\end{lstlisting}

\textbf{Rule:}
\begin{itemize}
\item If \texttt{self.query} exists $\to$ nested (RoBERTa).
\item If class name is \texttt{BertSelfAttention}/\texttt{RobertaSelfAttention} $\to$ nested.
\item If class name ends with "SelfAttention" \textbf{AND} lacks any output projection $\to$ nested.
\item Otherwise $\to$ flat.

\end{itemize}
\subsection{Forward Pass}

\textbf{Algorithm 1:} LoRA Attention forward
\begin{lstlisting}
Input:  X $\in$ R^{B$\times$L$\times$d}, attention\_mask, position\_embeddings
Output: Y $\in$ R^{B$\times$L$\times$d}

 1: Q $\leftarrow$ q\_proj(X)                         base + LoRA
 2: K $\leftarrow$ k\_proj(X)                         base + LoRA
 3: V $\leftarrow$ v\_proj(X)                         base + LoRA
 4: Reshape to (B, H, L, d\_h)
 5: if RoPE: Q, K $\leftarrow$ apply\_rotary(Q, K)
 6: if GQA: K, V $\leftarrow$ repeat\_kv(K, V)
 7: scores $\leftarrow$ Q K^T / $\sqrt{}$d\_h
 8: if causal: mask(scores, tril)
 9: if padding: mask(scores, valid)
10: attn $\leftarrow$ softmax(scores)
11: Context $\leftarrow$ attn V
12: Context $\leftarrow$ reshape(B, L, H d\_h)
13: if is\_nested\_encoder:
14:     return Context    outer layer applies out\_proj
15: Y $\leftarrow$ out\_proj(Context)
16: return Y
\end{lstlisting}

\[
\text{Trainable}\_\text{LoRA} = 16 \cdot (2 \cdot 768 + 2 \cdot 768) = 49,152
\tag{4.2}
\]

Per layer: ~49K trainable, compared to 2.36M frozen parameters.

\subsection{Mergeability}

After training, we can merge LoRA into the base weight:

\[
W\_\text{merged} = W\_\text{frozen} + \frac{\alpha}{r\_L} B A
\tag{4.3}
\]

Then:

\[
y = W\_\text{merged} x + b
\tag{4.4}
\]

\textbf{Zero inference overhead} -- the merged model is identical in structure to 
the original.

\subsection{Theoretical Notes}

\subsubsection{Zero Initialization}

\textbf{Claim.} At initialization, LoRA Attention is exactly equivalent to the 
pretrained attention.

\textbf{Proof.} $B = 0$, so $\Delta W = 0$. Thus $W_\text{eff} = W_\text{frozen}$. $\square$

\subsubsection{Mergeability Preserves Expressivity}

\textbf{Claim.} Any parameter set achievable by the LoRA branch is achievable 
by the merged weight.

\textbf{Proof.} $W_\text{merged} = W_\text{frozen} + BA$. Trivially achievable. $\square$

\subsubsection{Nested Encoder Correctness}

\textbf{Claim.} For nested encoders, skipping the output projection yields 
correct attention output.

\textbf{Proof.} The nested outer layer applies $W_O$. Our module returning raw 
context matches the original architecture's flow. $\square$
