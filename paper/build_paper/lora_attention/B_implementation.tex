class LoRALinear(nn.Module):
    def \_\_init\_\_(self, in\_features, out\_features, bias=True,
                 lora\_rank=16, lora\_alpha=32.0, freeze\_base=False):
        super().\_\_init\_\_()
        self.weight = nn.Parameter(torch.empty(out\_features, in\_features))
        if bias:
            self.bias = nn.Parameter(torch.empty(out\_features))
        
        if lora\_rank \textgreater{} 0:
            self.lora\_A = nn.Parameter(torch.empty(lora\_rank, in\_features))
            self.lora\_B = nn.Parameter(torch.empty(out\_features, lora\_rank))
        else:
            self.register\_parameter("lora\_A", None)
            self.register\_parameter("lora\_B", None)
        
        self.reset\_parameters()
        if freeze\_base:
            self.freeze\_base\_weights()
    
    def reset\_parameters(self):
        nn.init.kaiming\_uniform\_(self.weight, a=math.sqrt(5))
        if self.bias is not None:
            fan\_in, \_ = nn.init.\_calculate\_fan\_in\_and\_fan\_out(self.weight)
            bound = 1 / math.sqrt(fan\_in) if fan\_in \textgreater{} 0 else 0
            nn.init.uniform\_(self.bias, -bound, bound)
        
        if self.lora\_rank \textgreater{} 0:
            nn.init.kaiming\_uniform\_(self.lora\_A, a=math.sqrt(5))
            nn.init.zeros\_(self.lora\_B)
    
    def freeze\_base\_weights(self):
        self.weight.requires\_grad = False
        if self.bias is not None:
            self.bias.requires\_grad = False
        if self.lora\_rank \textgreater{} 0:
            self.lora\_A.requires\_grad = True
            self.lora\_B.requires\_grad = True
    
    def forward(self, x):
        out = F.linear(x, self.weight, self.bias)
        if self.lora\_rank \textgreater{} 0:
            out = out + F.linear(F.linear(x, self.lora\_A), self.lora\_B) * self.lora\_scaling
        return out * self.output\_scale
\end{lstlisting}

\subsection{B.3 Common Pitfalls}

\subsubsection{B.3.1 Silent Failure Corrupts Model}

If copy\_weights fails silently, the base weights stay at random init. 
LoRA Attention \textbf{raises exceptions} to prevent this.

\subsubsection{B.3.2 Wrong Nested Detection}

Checking class name before \texttt{self.query} misclassifies DistilBERT.

\subsubsection{B.3.3 Forgetting to Freeze Base}

If base weights train, you lose the "efficient" part of PEFT.

\subsubsection{B.3.4 Not Zero-Initializing B}

If $B \neq 0$ at start, the model differs from baseline.

\subsubsection{B.3.5 Merge Precision Loss}

FP16 merge introduces ~1e-3 error. For high-precision applications, merge 
in FP32.
