\subsubsection*{3.2 Mathematical Formulation}
Let $X \in \mathbb{R}^{B \times L \times d_{\text{in}}}$ denote the batch sequence representation tensor for batch size $B$, sequence length $L$, and channel dimension $d_{\text{in}}$. The exact forward mathematical transformation of \texttt{kmeans\_quantize} is formulated as:

\begin{align}
s &= \frac{\max(W) - \min(W)}{2^b - 1}, \quad z = \operatorname{round}\left(-\frac{\min(W)}{s}\right), \\
Q &= \operatorname{clip}\left(\operatorname{round}\left(\frac{W}{s}\right) + z, 0, 2^b - 1\right), \\
\hat{W} &= s \cdot (Q - z) \approx W.
\end{align}

\subsubsection*{3.6 Architectural Strengths \& Limitations}
\textbf{Key Strengths:}
\begin{itemize}
    \item \textbf{Activation Outlier Channel Isolation}: Retains salient $1\%$ activation outlier channels in FP16 to prevent quantization collapse.
    \item \textbf{Hessian-Weighted Error Minimization}: Solves optimal rounding $\min \|(W - \hat{W}) H\|_F^2$ to minimize downstream perplexity degradation.
    \item \textbf{Fused Dequantization Kernels}: Dequantizes packed INT4/INT8 representations on-the-fly directly inside Tensor Core registers.
\end{itemize}

\textbf{Potential Limitations:}
\begin{itemize}
    \item \textbf{Calibration Dataset Dependence}: Outlier detection and Hessian estimation require representative calibration batches.
    \item \textbf{Mixed-Precision Arithmetic Complexity}: Packing sub-byte integers requires specialized bit-shift instructions.
    \item \textbf{Non-Uniform Codebook Latency}: Highly non-linear codebooks require lookup table lookups on edge devices.
\end{itemize}
