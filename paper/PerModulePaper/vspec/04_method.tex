\section{Method}

\subsection{Overview}

VSpec++ replaces standard attention with:

\begin{enumerate}
\item \textbf{Token routing} -- select top-$k$ tokens.
\item \textbf{SVD projections} -- factorized Q/K/V/O.
\item \textbf{Spectral gating} -- variational weighting of singular values.
\item \textbf{Head gating} -- per-example head importance.
\item \textbf{LoRA recovery} -- trainable adapters on each projection.

\end{enumerate}
\textbf{Figure 4.1} shows the pipeline.

\begin{lstlisting}
    Input X $\in$ R^{B$$\times$L$$\times$d}
              |
               
      +---------------+
      | Token Router  |
      | (top-k + CLS) |
      +-------+-------+
              | X_routed (B$$\times$k$$\times$d)
       +------+------+
                      
    +------+ +------+ +------+
    |Pq(gq)| |Pk(gk)| |Pv(gv)|  $\leftarrow$ SVD + Spectral Gate
    +--+---+ +--+---+ +--+---+
       |        |        |
                          
       Q        K        V
       |        |        |
       +--------+--------+
                |
                 
        Attention $$\times$ HeadGate(h)
                |
                 
          Po(go)  $\leftarrow$ SVD + Spectral Gate
                |
                 
          Scatter back to L
                |
                 
             Output Y
\end{lstlisting}

\subsection{Token Router}

\textbf{Saliency network.}

\[
s_i = \sigma(f_\theta(x_i)), \quad f_\theta: \mathbb{R}^d \to [0, 1]
\tag{4.1}
\]

where $f_\theta$ is a 2-layer MLP: Linear($d$, 32) $\to$ GELU $\to$ Linear(32, 1).

\textbf{Masking.} Padding tokens get $s_i = -10^4$; \texttt{[CLS]} gets $s_0 = 10^4$.

\textbf{Selection.}

\[
k = \max(k_\text{min}, \lceil \rho L \rceil)
\tag{4.2}
\]

\[
\mathcal{S} = \text{Sort}(\text{TopK}(s, k))
\tag{4.3}
\]

\textbf{Gather.} $X_\text{routed} = \text{gather}(X, \mathcal{S})$.

\textbf{Nested encoder bypass.} For RoBERTa/BERT, routing is skipped (context 
must be preserved).

\subsection{SVD-Initialized Projector}

For each projection $W \in \mathbb{R}^{d_\text{out} \times d_\text{in}}$:

\textbf{Step 1: Truncated SVD.}

\[
W \approx U_r \Sigma_r V_r^\top
\tag{4.4}
\]

\textbf{Step 2: Store factors.}

$U_r \in \mathbb{R}^{d_\text{out} \times r}$, $S_r \in \mathbb{R}^r$, 
$V_r \in \mathbb{R}^{d_\text{in} \times r}$.

\textbf{Step 3: Forward with spectral gate.}

\[
z = (x V_r) \cdot (S_r \odot g)
\tag{4.5}
\]

\[
y = z U_r^\top + b + \text{LoRA}(x)
\tag{4.6}
\]

where $g \in [0, 1]^r$ is the spectral gate (see  4.4).

\subsection{Variational Spectral Gate}

\textbf{Pool.} Masked mean over routed tokens:

\[
\bar{x} = \frac{\sum_i m_i x_i}{\sum_i m_i + \epsilon}
\tag{4.7}
\]

\textbf{Network.}

\[
{[}\mu, \log \sigma] = f_\text{sg}(\bar{x}) \in \mathbb{R}^{2r}
\tag{4.8}
\]

where $f_\text{sg}$ is a 2-layer MLP: Linear($d$, 64) $\to$ GELU $\to$ Linear(64, $2r$).

\textbf{Clamp.} $\log \sigma \in [-10, 2]$.

\textbf{Sample (training).}

\[
\epsilon \sim \mathcal{N}(0, I), \quad z = \mu + \exp(\log \sigma) \odot \epsilon
\tag{4.9}
\]

\textbf{Gate.}

\[
g = \sigma(z) \in [0, 1]^r
\tag{4.10}
\]

\textbf{Inference.} $g = \sigma(\mu)$ (deterministic).

\textbf{Initialization.} We set last-layer weight to 0, bias to $[\mathbf{3}, \mathbf{-3}]$, 
giving $g \approx \sigma(3) \approx 0.95$ and low variance. This means the 
SVD representation is preserved at step 0.

\subsection{Head Gate}

\textbf{Pool.}

\[
\bar{x} = \text{masked-mean}(X_\text{routed})
\tag{4.11}
\]

\textbf{Network.}

\[
h = \sigma(f_h(\bar{x})) \in [0, 1]^H
\tag{4.12}
\]

where $f_h$ is Linear($d$, 32) $\to$ GELU $\to$ Linear(32, $H$).

\textbf{Application.} Multiply attention logits:

\[
\text{scores} \leftarrow \text{scores} \cdot h_i
\tag{4.13}
\]

where $h_i$ is the importance of head $i$.

\subsection{Combined Attention}

\[
Q = P_q(X_\text{routed}, g_q), \quad K = P_k(X_\text{routed}, g_k), \quad V = P_v(X_\text{routed}, g_v)
\tag{4.14}
\]

Reshape to multi-head, apply RoPE, GQA, and:

\[
\text{scores} = \frac{Q K^\top}{\sqrt{d_h}} \cdot h
\tag{4.15}
\]

\[
A = \text{softmax}(\text{scores} + \text{causal} + \text{pad})
\tag{4.16}
\]

\[
C = A V
\tag{4.17}
\]

\[
Y_\text{routed} = P_o(C, g_q)
\tag{4.18}
\]

\textbf{Scatter back.} For non-routed tokens, we set $Y = 0$ and rely on the 
residual connection to preserve them:

\[
Y = \text{scatter}(Y_\text{routed}, \mathcal{S})
\tag{4.19}
\]

\subsection{Algorithm}

\textbf{Algorithm 1:} VSpec++ forward pass
\begin{lstlisting}
Input:  X $\in$ R^{B$$\times$L$$\times$d}, attention_mask, position_embeddings
Output: Y $\in$ R^{B$$\times$L$$\times$d}

 1: if nested_encoder: S $\leftarrow$ arange(L); k $\leftarrow$ L
 2: else:              s $\leftarrow$ saliency(X)
 3:                    s[0] $\leftarrow$ 1e4; s[padding] $\leftarrow$ -1e4
 4:                    S $\leftarrow$ Sort(TopK(s, k))
 5: X_r $\leftarrow$ gather(X, S)
 6: x  $\leftarrow$ masked_mean(X_r)
 7: for proj in [q, k, v, o]:
 8:      _proj, log $\sigma$_proj $\leftarrow$ MLP(x )
 9:     z_proj $\leftarrow$  _proj + exp(log $\sigma$_proj)   $\epsilon$     training
10:     g_proj $\leftarrow$ $\sigma$(z_proj)
11: h $\leftarrow$ $\sigma$(MLP(x ))                                 head gate
12: Q $\leftarrow$ Pq(X_r, g_q);  K $\leftarrow$ Pk(X_r, g_k);  V $\leftarrow$ Pv(X_r, g_v)
13: Q, K $\leftarrow$ RoPE(Q, K, pos)
14: if GQA: K, V $\leftarrow$ repeat_kv
15: scores $\leftarrow$ Q K^T / $\sqrt{}$d_h $$\times$ h
16: A $\leftarrow$ softmax(scores + masks)
17: C $\leftarrow$ A V
18: Y_r $\leftarrow$ Po(C, g_q)
19: Y $\leftarrow$ scatter(Y_r, S)
20: return Y
\end{lstlisting}

\subsection{Complexity Analysis}

Per block:

\begin{table}[H]
\centering
\small
\begin{tabular}{lll}
\toprule
\textbf{Component} & \textbf{Time} & \textbf{Params} \\
\midrule
Router & $O(L d)$ & $O(d)$ \\
Spectral gates ($$\times$3) & $O(d \cdot 2r)$ & $O(dr)$ \\
Head gate & $O(d H)$ & $O(dH)$ \\
Q/K/V proj & $O(k d r)$ & $3 \cdot 2 d r$ \\
Attention & $O(k^2 d)$ & 0 \\
Out proj & $O(k d r)$ & $2 d r$ \\
LoRA & $O(k d r_L)$ & $4 d r_L$ \\
\textbf{Total} & $O(L d + k^2 d)$ & $\sim 8 r d + O(dH)$ \\
\bottomrule
\end{tabular}
\end{table}

For $d = 768$, $r = 128$, $H = 12$, $\rho = 0.75$:

\begin{table}[H]
\centering
\small
\begin{tabular}{ll}
\toprule
\textbf{Component} & \textbf{Params} \\
\midrule
SVD factors (Q/K/V/O) & $4 \times 2 \times 768 \times 128 = 786K$ \\
Gates & ~150K \\
LoRA & ~4 $$\times$ 16 $$\times$ 768 $$\times$ 2 = 98K \\
Router & ~25K \\
\textbf{Total} & ~1.06M \\
\bottomrule
\end{tabular}
\end{table}

vs baseline $4 d^2 = 2.36M$ $\to$ \textbf{2.2$$\times$ reduction}.

\subsection{Training}

Trainable:
\begin{itemize}
\item Router MLP.
\item Spectral gates (3 $$\times$ MLP).
\item Head gate MLP.
\item LoRA adapters ($A, B$ on each projection).

\end{itemize}
Frozen:
\begin{itemize}
\item SVD factors $U, S, V$.
\item Bias (optional).

\end{itemize}
\textbf{Variational penalty.} In practice, we add a small KL-like regularization:

\[
\mathcal{L}_\text{KL} = \beta \sum_i \left(\mu_i^2 + \sigma_i^2 - 1\right)
\tag{4.20}
\]

to encourage low-variance gates.

\subsection{Inference}

At inference, the variational sampling is replaced by $z = \mu$. This gives 
a \textbf{deterministic} forward pass, matching the training-time mean of the 
posterior.
