\section{Method}

\subsection{Overview}

TokenCompressedAttention (TCA) replaces standard attention by adding three 
steps before the attention computation:

\begin{enumerate}
\item \textbf{Score computation} ( 4.2): compute per-token importance.
\item \textbf{Token selection} ( 4.3): keep the top-$k$ tokens.
\item \textbf{Causal-aware attention} ( 4.4): apply attention with a mask over
  original positions.

\end{enumerate}
The overall pipeline:

\begin{lstlisting}
    Input X $\in$ R^{B$\times$L$\times$d}
              |
               
    +-----------------------+
    | Project Q/K/V         |  (frozen base weights + LoRA)
    +-----------+-----------+
                |
       +--------+--------+
                          
    +------+         +--------------+
    | Q    |         | K, V         |
    |      |         | Score K.V    |
    |      |         +------+-------+
    |      |                |
    |      |         +------ -------+
    |      |         | Top-k select |
    |      |         | [CLS] pinned |
    |      |         +------+-------+
    |      |                |
    +------+----------------+
              |
               
    +-----------------------+
    | Attention + causal    |
    | mask against original |
    | positions             |
    +-----------+-----------+
                |
                 
    +-----------------------+
    | Output projection     |
    +-----------------------+
\end{lstlisting}

\subsection{Token Importance Scoring}

For each token $i$, we compute:

\[
s_i = \underbrace{\|x_i\|_2}_{\text{energy}} + \underbrace{f_\theta(x_i)}_{\text{learned}}
\tag{4.1}
\]

where:

\begin{itemize}
\item $\|x_i\|_2$ is the $\ell_2$ norm of the hidden state.
\item $f_\theta$ is a small MLP: Linear($d$, 64) $\to$ GELU $\to$ Linear(64, 1).

\end{itemize}
\textbf{Why this combination?}

\begin{itemize}
\item The norm captures \textbf{activation energy} -- an established proxy for token
  importance (Lin et al., 2023).
\item The learned term captures \textbf{content-specific} importance that a
  norm-based heuristic misses.

\end{itemize}
\subsection{Token Selection}

We force the first token to be retained (attention sink):

\[
s_0 \leftarrow 10^4
\tag{4.2}
\]

We mask out padding tokens:

\[
s_i \leftarrow -10^4 \quad \text{for padding tokens}
\tag{4.3}
\]

Then we select the top-$k$ indices:

\[
\mathcal{S} = \text{TopK}(s, k), \quad k = \lceil \rho L \rceil
\tag{4.4}
\]

and sort them to preserve order:

\[
\mathcal{S} \leftarrow \text{Sort}(\mathcal{S})
\tag{4.5}
\]

Finally, we gather:

\[
K_c = K[\mathcal{S}], \quad V_c = V[\mathcal{S}]
\tag{4.6}
\]

\textbf{Short-sequence handling.} If $L \leq 2$, no compression is applied.

\subsection{Causal-Safe Attention}

Standard attention over the compressed keys:

\[
\text{scores}_{i, j'} = \frac{q_i \cdot K_{c, j'}^\top}{\sqrt{d_h}}
\tag{4.7}
\]

For decoders, we add a mask based on \textbf{original positions}:

\[
\text{mask}_{i, j'} = \begin{cases} 0 & \mathcal{S}_{j'} \leq i \\ -\infty & \mathcal{S}_{j'} > i \end{cases}
\tag{4.8}
\]

This ensures that query at position $i$ cannot attend to compressed keys 
whose original position exceeds $i$.

\textbf{Computing the mask:}

\begin{lstlisting}
q_pos = torch.arange(L).view(1, 1, L, 1)       # (1, 1, L, 1)
k_orig_pos = top_idx.unsqueeze(1).unsqueeze(2) # (B, 1, 1, k)
causal_mask = (k_orig_pos > q_pos)             # (B, 1, L, k)
scores = scores.masked_fill(causal_mask, -inf)
\end{lstlisting}

\subsection{Combined Algorithm}

\textbf{Algorithm 1:} TCA forward pass
\begin{lstlisting}
Input:  hidden_states X $\in$ R^{B$\times$L$\times$d}
        attention_mask $\in$ {0,1}^{B$\times$L}
        position_ids
Output: Y $\in$ R^{B$\times$L$\times$d}

 1: Q $\leftarrow$ X W_Q;  K $\leftarrow$ X W_K;  V $\leftarrow$ X W_V
 2: Q, K $\leftarrow$ RoPE(Q, K, position_ids)
 3: s $\leftarrow$  X _2 + score_net(X)          Per-token score
 4: s[0] $\leftarrow$ 1e4                         Pin [CLS]
 5: s[padding] $\leftarrow$ -1e4                  Mask padding
 6: S $\leftarrow$ TopK(s, k)                     Select top-k
 7: S $\leftarrow$ Sort(S)                        Order-preserve
 8: K_c $\leftarrow$ K[S];  V_c $\leftarrow$ V[S]            Gather
 9: K_c $\leftarrow$ repeat_kv(K_c, H/H_kv)       GQA expand
10: scores $\leftarrow$ Q @ K_c^T / $\sqrt{}$d_h
11: if is_decoder:
12:     mask $\leftarrow$ (S[None,None,None,:] > arange(L)[None,None,:,None])
13:     scores $\leftarrow$ scores.masked_fill(mask, -inf)
14: scores $\leftarrow$ mask_padding(scores, attention_mask)
15: A $\leftarrow$ softmax(scores)
16: Y $\leftarrow$ A @ V_c
17: Y $\leftarrow$ out_proj(Y)
18: return Y
\end{lstlisting}

\subsection{Memory and Compute Savings}

For a single attention layer:

\begin{table}[H]
\centering
\small
\begin{tabular}{llll}
\toprule
\textbf{Metric} & \textbf{Baseline} & \textbf{TCA} & \textbf{Savings} \\
\midrule
KV cache & $2 H_{kv} L d_h$ & $2 H_{kv} k d_h$ & $1 - \rho$ \\
Attention compute & $L^2$ & $L \cdot k$ & $1 - \rho$ \\
Score net compute & 0 & $2 L \cdot (d \cdot 64 + 64)$ & +small \\
\bottomrule
\end{tabular}
\end{table}

For $\rho = 0.5$:

\begin{itemize}
\item KV cache: \textbf{50\% reduction}
\item Attention compute: \textbf{50\% reduction}
\item Score net: \textbf{~0.5\% overhead}

\end{itemize}
