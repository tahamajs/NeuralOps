AdaSVD: Adaptive SVD with per-layer importance allocation and iterative ALS refinement.
Registered as 'adasvd'.
"""

from \_\_future\_\_ import annotations

import logging
from typing import Any, Dict, Optional, Tuple, Union

import torch
import torch.nn as nn

from framework.compression\_engine.aa\_svd\_compressor import SVDLinear
from framework.core.base import BaseCompressor
from framework.core.model\_utils import replace\_module
from framework.core.registry import COMPRESSOR\_REGISTRY

logger = logging.getLogger(\_\_name\_\_)


@COMPRESSOR\_REGISTRY.register("adasvd")
class AdaSVDCompressor(BaseCompressor):
    """
    Adaptive SVD Compressor (AdaSVD).

    Assigns SVD ranks dynamically across layers proportional to spectral importance
    (nuclear norm ||W||\_* = sum(sigma\_i)), initializes factorized matrices with
    symmetric square-root singular value distribution (A = U * sqrt(Sigma), B = sqrt(Sigma) * V^T),
    refines factors via Alternating Least Squares (ALS), and replaces dense nn.Linear layers
    with factorized SVDLinear modules.

    Args:
        model: PyTorch neural network to compress.
        rank: Optional baseline rank for all layers or global target rank.
        target\_param\_ratio: Target parameter retention ratio (e.g. 0.7 for 30% reduction).
        num\_iters: Number of alternating least squares refinement iterations.
        target\_modules: Optional list of module name substrings to target.
    """

    def \_\_init\_\_(
        self,
        model: nn.Module,
        rank: Optional[int] = None,
        target\_param\_ratio: float = 0.7,
        num\_iters: int = 5,
        target\_modules: Optional[list[str]] = None,
        **kwargs: Any,
    ) -> None:
        super().\_\_init\_\_(model, name="adasvd", **kwargs)
        cfg = kwargs.get("comp\_config", kwargs.get("cfg", None))
        if cfg is not None:
            rank = getattr(cfg, "adasvd\_rank", getattr(cfg, "rank", rank))
            target\_param\_ratio = getattr(
                cfg, "adasvd\_target\_ratio", getattr(cfg, "target\_param\_ratio", target\_param\_ratio)
            )
            num\_iters = getattr(cfg, "adasvd\_iters", getattr(cfg, "num\_iters", num\_iters))

        self.rank = rank
        self.target\_param\_ratio = float(target\_param\_ratio)
        self.num\_iters = int(num\_iters)
        self.target\_modules = target\_modules

    def \_is\_target\_module(self, name: str, module: nn.Module) -> bool:
        if not isinstance(module, nn.Linear):
            return False
        if isinstance(module, SVDLinear):
            return False
        if self.target\_modules is None:
            return True
        return any(t in name for t in self.target\_modules)

    @staticmethod
    def \_compute\_nuclear\_norm(W: torch.Tensor) -> float:
        """Compute nuclear norm ||W||\_* = sum(sigma\_i) in FP32."""
        with torch.no\_grad():
            s = torch.linalg.svdvals(W.float())
            return float(s.sum().item())

    def compress\_linear(
        self,
        layer: nn.Linear,
        rank: int,
    ) -> SVDLinear:
        """
        Compress a single nn.Linear layer using symmetric SVD and ALS refinement.
        """
        out\_f = layer.out\_features
        in\_f = layer.in\_features
        r = max(1, min(rank, out\_f, in\_f))
        device = layer.weight.device
        dtype = layer.weight.dtype

         Compute full SVD in FP32
        W = layer.weight.data.float()
        U, S, Vt = torch.linalg.svd(W, full\_matrices=False)

         Symmetric square-root singular value initialization
        sqrt\_S = torch.sqrt(torch.clamp(S[:r], min=1e-12))
        A = U[:, :r] * sqrt\_S.unsqueeze(0)        [out\_features, r]
        B = sqrt\_S.unsqueeze(1) * Vt[:r, :]       [r, in\_features]

         Alternating Least Squares (ALS) refinement
        eps\_eye = 1e-6 * torch.eye(r, device=device, dtype=torch.float32)
        for \_ in range(self.num\_iters):
             Update A: min ||W - A @ B||\_F^2 => A = W @ B^T @ (B @ B^T + eps*I)^-1
            BBt = B @ B.t() + eps\_eye
            A = W @ B.t() @ torch.linalg.inv(BBt)

             Update B: min ||W - A @ B||\_F^2 => B = (A^T @ A + eps*I)^-1 @ A^T @ W
            AtA = A.t() @ A + eps\_eye
            B = torch.linalg.inv(AtA) @ A.t() @ W

         Instantiate structural SVDLinear
        svd\_layer = SVDLinear(
            in\_features=in\_f,
            out\_features=out\_f,
            rank=r,
            bias=layer.bias is not None,
            dtype=dtype,
            device=device,
        )
        svd\_layer.set\_low\_rank\_factors(
            U\_factor=A.to(dtype=dtype),
            V\_factor=B.to(dtype=dtype),
        )
        if layer.bias is not None:
            svd\_layer.bias.data.copy\_(layer.bias.data)

        return svd\_layer

    def compress(self) -> Tuple[nn.Module, Dict[str, Any]]:
        """
        Executes adaptive rank assignment and ALS refinement across all linear layers.
        """
        target\_layers = [
            (name, mod)
            for name, mod in self.model.named\_modules()
            if self.\_is\_target\_module(name, mod)
        ]

        if not target\_layers:
            logger.warning("AdaSVD: No compressible linear layers found.")
            return self.model, self.get\_compression\_stats()

         Step 1: Compute spectral nuclear norms
        layer\_importances = {}
        total\_importance = 0.0
        for name, module in target\_layers:
            imp = self.\_compute\_nuclear\_norm(module.weight.data)
            layer\_importances[name] = max(imp, 1e-8)
            total\_importance += layer\_importances[name]

         Step 2: Assign ranks per layer
        for name, module in target\_layers:
            max\_r = min(module.in\_features, module.out\_features)
            if self.rank is not None:
                 Use specified rank or scale by normalized importance
                frac = layer\_importances[name] / (total\_importance / len(target\_layers))
                allocated\_rank = int(round(self.rank * frac))
            else:
                frac = layer\_importances[name] / total\_importance
                allocated\_rank = int(round(max\_r * self.target\_param\_ratio * frac * len(target\_layers)))

            allocated\_rank = max(1, min(allocated\_rank, max\_r))
            svd\_layer = self.compress\_linear(module, rank=allocated\_rank)
            replace\_module(self.model, name, svd\_layer)

        stats = self.get\_compression\_stats()
        return self.model, stats


def adasvd\_factory(model: Any, comp\_config: Any = None, **kwargs: Any) -> nn.Module:
    """AdaSVD factory function."""
    compressor = AdaSVDCompressor(model, comp\_config=comp\_config, **kwargs)
    compressed\_model, \_ = compressor.compress()
    return compressed\_model

        self,
        layer: nn.Linear,

        in_f = layer.in_features

        device = layer.weight.device
        dtype = layer.weight.dtype

        # Compute full SVD in FP32
        W = layer.weight.data.float()

        # Symmetric square-root singular value initialization
        sqrt_S = torch.sqrt(torch.clamp(S[:r], min=1e-12))
        A = U[:, :r] * sqrt_S.unsqueeze(0)       # [out_features, r]
        B = sqrt_S.unsqueeze(1) * Vt[:r, :]      # [r, in_features]

        # Alternating Least Squares (ALS) refinement
        eps_eye = 1e-6 * torch.eye(r, device=device, dtype=torch.float32)
        for _ in range(self.num_iters):
            # Update A: min ||W - A @ B||_F^2 => A = W @ B^T @ (B @ B^T + eps*I)^-1

            in_features=in_f,
            out_features=out_f,

            V_factor=B.to(dtype=dtype),
        )
        if layer.bias is not None:
            svd_layer.bias.data.copy_(layer.bias.data)

        return svd_layer

            (name, mod)
            for name, mod in self.model.named_modules()

            return self.model, self.get_compression_stats()

        layer_importances = {}
        total_importance = 0.0
        for name, module in target_layers:
            imp = self._compute_nuclear_norm(module.weight.data)
            layer_importances[name] = max(imp, 1e-8)
            total_importance += layer_importances[name]
