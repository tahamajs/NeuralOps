\subsubsection{Sparsity Ablation (Table 7.4)}
\begin{lstlisting}
for rho in 0.1 0.3 0.5 0.7 0.9; do

\end{lstlisting}

\subsubsection{Temperature Ablation (Table 7.5)}
\begin{lstlisting}

    python run_all_4_adaptive_sparse.py --temperature $T --output_suffix T$T
done
\end{lstlisting}
