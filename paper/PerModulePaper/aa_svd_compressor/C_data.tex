Activation-Aware Singular Value Decomposition (AASVD) Compressor.

Implements streaming Exponential Moving Average (EMA) activation covariance accumulation:
$$C\_X^{(t)} = \\beta C\_X^{(t-1)} + (1 - \\beta) \\frac{1}{B} \\sum\_{i=1}^B X\_i^\\top X\_i$$
Projects weights onto principal activation subspaces and replaces dense `nn.Linear` layers
with structural low-rank factorized `SVDLinear` modules ($y = (x V^\\top) U^\\top$) to guarantee
real physical parameter reduction and FLOP savings.
"""

class SVDLinear(nn.Module):
    """
    FIX: True Low-Rank Factorized Linear Layer executing as:
    y = (x @ V.T) @ U.T + b
    Achieving physical parameter reduction from d\_in * d\_out to r * (d\_in + d\_out).
    """

    def \_\_init\_\_(
        self,
        in\_features: int,
        out\_features: int,
        rank: int,
        bias: bool = True,
        use\_fused\_kernel: bool = False,
        dtype: torch.dtype = torch.float32,
        device: Optional[torch.device] = None,
    ) -> None:
        super().\_\_init\_\_()
        self.in\_features = in\_features
        self.out\_features = out\_features
        self.rank = rank
        self.use\_fused\_kernel = use\_fused\_kernel

         Factorized low-rank parameter tensors
        self.V = nn.Parameter(torch.empty(rank, in\_features, dtype=dtype, device=device))
        self.U = nn.Parameter(torch.empty(out\_features, rank, dtype=dtype, device=device))

        if bias:
            self.bias = nn.Parameter(torch.empty(out\_features, dtype=dtype, device=device))
        else:
            self.register\_parameter("bias", None)

        self.reset\_parameters()

    def reset\_parameters(self) -> None:
        with torch.no\_grad():
            if self.U.dtype in (torch.float16, torch.bfloat16):
                u\_fp32 = torch.empty\_like(self.U, dtype=torch.float32, device="cpu")
                nn.init.orthogonal\_(u\_fp32)
                self.U.copy\_(u\_fp32.to(dtype=self.U.dtype, device=self.U.device))
            else:
                nn.init.orthogonal\_(self.U)

            if self.V.dtype in (torch.float16, torch.bfloat16):
                v\_fp32 = torch.empty\_like(self.V, dtype=torch.float32, device="cpu")
                nn.init.orthogonal\_(v\_fp32)
                self.V.copy\_(v\_fp32.to(dtype=self.V.dtype, device=self.V.device))
            else:
                nn.init.orthogonal\_(self.V)

            if self.bias is not None:
                nn.init.zeros\_(self.bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        if self.use\_fused\_kernel:
             RESEARCH FIX: Accelerated fused kernel dispatch
            from framework.attentions.fused\_svd\_kernel import fused\_svd\_linear
            return fused\_svd\_linear(x=x, V=self.V, U=self.U, bias=self.bias, use\_rms\_norm=False)

         Sequential two-stage projection
        latent = F.linear(x, self.V)   [..., rank]
        out = F.linear(latent, self.U)   [..., out\_features]
        if self.bias is not None:
            out = out + self.bias
        return out

    def set\_low\_rank\_factors(self, U\_factor: torch.Tensor, V\_factor: torch.Tensor) -> None:
        with torch.no\_grad():
            self.U.copy\_(U\_factor)
            self.V.copy\_(V\_factor)


@COMPRESSOR\_REGISTRY.register("aasvd")
class AASVDCompressor(BaseCompressor):
    """
    Activation-Aware SVD Compressor with streaming EMA covariance accumulation
    and true module replacement.
    """

    def \_\_init\_\_(
        self,
        model: nn.Module,
        tokenizer: Optional[Any] = None,
        rank: int = 64,
        ema\_beta: float = 0.9,
        calibration\_texts: Optional[List[str]] = None,
        device: Optional[Union[str, torch.device]] = None,
        **kwargs: Any,
    ) -> None:
        super().\_\_init\_\_(model, name="aasvd", **kwargs)
        cfg = kwargs.get("comp\_config", kwargs.get("cfg", None))
        if cfg is not None:
            rank = getattr(cfg, "aasvd\_rank", getattr(cfg, "rank", rank))
            ema\_beta = getattr(cfg, "aasvd\_ema\_beta", ema\_beta)
            tokenizer = getattr(cfg, "tokenizer", tokenizer)
            calibration\_texts = getattr(cfg, "calibration\_texts", calibration\_texts)
        self.tokenizer = tokenizer
        self.rank = rank
        self.ema\_beta = ema\_beta
        self.calibration\_texts = calibration\_texts or DEFAULT\_CALIBRATION\_CORPUS
        self.device = (
            torch.device(device)
            if device is not None
            else next(model.parameters()).device
        )

    def compress\_linear(
        self,
        layer: nn.Linear,
        rank: Optional[int] = None,
        cov: Optional[torch.Tensor] = None,
    ) -> SVDLinear:
        """
        FIX: Factored linear compression implementing activation-aware subspace projection
        and symmetric singular value distribution.
        
        Args:
            layer: Source nn.Linear layer to compress
            rank: Target low-rank bottleneck dimension
            cov: Optional empirical activation covariance matrix (X^T @ X) / B
            
        Returns:
            Structural SVDLinear replacement module.
        """
        in\_feat = layer.in\_features
        out\_feat = layer.out\_features
        r = min(rank or self.rank, in\_feat, out\_feat)

        if cov is not None:
             Principal activation directions via SVD of covariance
            try:
                \_, \_, Vh\_act = torch.linalg.svd(cov.cpu().float(), full\_matrices=False)
                anchor = Vh\_act[:r, :].to(layer.weight.device)   [r, in\_feat]
            except Exception:
                anchor = torch.eye(in\_feat, device=layer.weight.device)[:r, :]

             Project weight W onto principal activation subspace
            W = layer.weight.data.float()   [out\_feat, in\_feat]
            W\_proj = torch.matmul(W, torch.matmul(anchor.T, anchor))   [out\_feat, in\_feat]
        else:
            W\_proj = layer.weight.data.float().cpu()

         Truncated SVD on projected weight
        U, S, Vt = torch.linalg.svd(W\_proj.cpu().float(), full\_matrices=False)

         FIX: Symmetrical Square-Root Singular Value Allocation U\_r = U @ sqrt(S), V\_r = sqrt(S) @ V^T
        sqrt\_S = torch.sqrt(torch.clamp(S[:r], min=1e-12))
        U\_r = (U[:, :r] * sqrt\_S.unsqueeze(0)).to(
            device=layer.weight.device, dtype=layer.weight.dtype
        )
        V\_r = (sqrt\_S.unsqueeze(1) * Vt[:r, :]).to(
            device=layer.weight.device, dtype=layer.weight.dtype
        )

         FIX: Instantiate structural SVDLinear replacement module (true parameter reduction)
        svd\_layer = SVDLinear(
            in\_features=in\_feat,
            out\_features=out\_feat,
            rank=r,
            bias=layer.bias is not None,
            dtype=layer.weight.dtype,
            device=layer.weight.device,
        )
        svd\_layer.set\_low\_rank\_factors(U\_r, V\_r)
        if layer.bias is not None:
            svd\_layer.bias.data.copy\_(layer.bias.data)

        return svd\_layer

    def compress(self) -> Tuple[nn.Module, Dict[str, Any]]:
        """
        Executes streaming covariance estimation over calibration batches,
        computes SVD in the activation-aligned coordinate frame, and replaces
        dense `nn.Linear` layers with `SVDLinear` factorized layers.
        """
        self.model.eval()

        if self.tokenizer is None:
            logger.warning(
                "AASVD: No tokenizer provided. Falling back to unweighted truncated SVD."
            )
            return self.\_compress\_unweighted\_svd()

        covariances: Dict[str, torch.Tensor] = {}
        hooks = []

        def make\_hook(name: str):
            def hook(module: nn.Module, inp: Tuple[torch.Tensor, ...], outp: Any):
                 Input shape: [batch\_size, seq\_len, in\_features]
                x = inp[0].detach().float()
                x\_flat = x.view(-1, x.size(-1))   [total\_tokens, in\_features]
                 FIX: Streaming Exponential Moving Average (EMA) Covariance Accumulation: (X^T @ X) / B
                batch\_cov = torch.matmul(x\_flat.T, x\_flat) / max(1, x\_flat.size(0))

                if name not in covariances:
                    covariances[name] = batch\_cov
                else:
                    covariances[name] = self.ema\_beta * covariances[name] + (1.0 - self.ema\_beta) * batch\_cov

            return hook

         1. Register forward hooks on all linear layers
        for name, module in self.model.named\_modules():
            if isinstance(module, nn.Linear):
                hooks.append(module.register\_forward\_hook(make\_hook(name)))

         2. Stream calibration batches through the model
        try:
            inputs = self.tokenizer(
                self.calibration\_texts,
                return\_tensors="pt",
                padding=True,
                truncation=True,
                max\_length=512,
            )
            inputs = {k: v.to(self.device) for k, v in inputs.items()}
            with torch.no\_grad():
                self.model(**inputs)
        except Exception as e:
            logger.warning(
                "AASVD: Calibration forward failed (%s). Falling back to unweighted truncated SVD.",
                e,
            )
            for h in hooks:
                h.remove()
            return self.\_compress\_unweighted\_svd()

         3. Clean up forward hooks
        for h in hooks:
            h.remove()

         4. Replace each Linear layer with an SVDLinear module
        replaced\_count = 0
        for name, module in list(self.model.named\_modules()):
            if isinstance(module, nn.Linear) and name in covariances:
                svd\_layer = self.compress\_linear(module, rank=self.rank, cov=covariances[name])
                replace\_module(self.model, name, svd\_layer)
                replaced\_count += 1

        logger.info("AASVD: Structurally factorized %d linear layers into SVDLinear.", replaced\_count)
        return self.model, self.get\_compression\_stats()

    def \_compress\_unweighted\_svd(self) -> Tuple[nn.Module, Dict[str, Any]]:
        """Fallback when no tokenizer is present."""
        replaced\_count = 0
        for name, module in list(self.model.named\_modules()):
            if isinstance(module, nn.Linear):
                svd\_layer = self.compress\_linear(module, rank=self.rank, cov=None)
                replace\_module(self.model, name, svd\_layer)
                replaced\_count += 1

        logger.info("AASVD (unweighted): Structurally factorized %d linear layers into SVDLinear.", replaced\_count)
        return self.model, self.get\_compression\_stats()


def aasvd\_factory(model: Any, comp\_config: Any = None, **kwargs: Any) -> nn.Module:
    """AASVD factory function."""
    compressor = AASVDCompressor(model, comp\_config=comp\_config, **kwargs)
    compressed\_model, \_ = compressor.compress()
    return compressed\_model
