ADMM Pruner: Alternating Direction Method of Multipliers for Structured \& Semi-Structured Pruning with STE \& Dynamic Rho.
Registered in COMPRESSOR\_REGISTRY as 'admm\_pruner'.

Mathematical Formulation:
$$\\min\_W \\frac{1}{2} \\|W - W\_0\\|\_F^2 + g(Z) \\quad \\text{s.t.} \\quad W = Z$$

ADMM Updates with Dynamic Rho Scheduling:
1. $Z^{(t+1)} = \\Pi\_{\\mathcal{S}}(W^{(t)} + U^{(t)})$
2. $W^{(t+1)} = \\frac{1}{1 + \\rho^{(t)}} \\left( W\_0 + \\rho^{(t)} (Z^{(t+1)} - U^{(t)}) \\right)$
3. $U^{(t+1)} = U^{(t)} + W^{(t+1)} - Z^{(t+1)}$
4. Primal residual $r^{(t+1)} = \\|W^{(t+1)} - Z^{(t+1)}\\|\_F$, Dual residual $s^{(t+1)} = \\rho^{(t)} \\|Z^{(t+1)} - Z^{(t)}\\|\_F$

class ADMMMaskedLinear(nn.Module):
    """
    Linear layer maintaining continuous weights and an ADMM-derived binary mask with STE gradient backprop.
    """

    def \_\_init\_\_(
        self,
        in\_features: int,
        out\_features: int,
        bias: bool = True,
        dtype: torch.dtype = torch.float32,
        device: Optional[torch.device] = None,
    ) -> None:
        super().\_\_init\_\_()
        self.in\_features = in\_features
        self.out\_features = out\_features
        self.weight = nn.Parameter(torch.empty(out\_features, in\_features, dtype=dtype, device=device))
        self.register\_buffer("mask", torch.ones(out\_features, in\_features, dtype=dtype, device=device))

        if bias:
            self.bias = nn.Parameter(torch.empty(out\_features, dtype=dtype, device=device))
        else:
            self.register\_parameter("bias", None)

    @classmethod
    def from\_linear(cls, linear: nn.Linear, mask: torch.Tensor) -> ADMMMaskedLinear:
        layer = cls(
            in\_features=linear.in\_features,
            out\_features=linear.out\_features,
            bias=linear.bias is not None,
            dtype=linear.weight.dtype,
            device=linear.weight.device,
        )
        layer.weight.data.copy\_(linear.weight.data)
        layer.mask.copy\_(mask.to(dtype=linear.weight.dtype, device=linear.weight.device))
        if linear.bias is not None:
            layer.bias.data.copy\_(linear.bias.data)
        return layer

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        w\_masked = ADMMSTEPrune.apply(self.weight, self.mask)
        return F.linear(x, w\_masked, self.bias)

    """
    ADMM Pruner supporting unstructured sparsity, block-wise structured sparsity,
    and 2:4 semi-structured tensor core acceleration with STE gradients and dynamic rho scheduling.
    """

        self.block\_size = block\_size
        self.dynamic\_rho = dynamic\_rho

        self.tokenizer = tokenizer or getattr(model, "tokenizer", None)
        self.calibration\_loader = calibration\_loader
        self.calibration\_texts = calibration\_texts

            torch.device(device)
            if device is not None
            else (next(model.parameters(), torch.tensor(0.0)).device)
        )
        self.stats: Dict[str, Any] = {}

    def \_project\_sparse(self, tensor: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """Projects continuous weights onto sparse constraint set and returns (pruned\_tensor, binary\_mask)."""
        if self.structured\_24:
            pruned, mask = apply\_24\_structured\_mask(tensor, saliency=torch.abs(tensor), dim=1)
            return pruned, mask
        elif self.block\_size != (1, 1):
            pruned, mask = apply\_block\_mask(tensor, block\_size=self.block\_size, sparsity=self.sparsity)
            return pruned, mask
        else:
            flat = tensor.abs().reshape(-1)
            k = max(1, int(flat.numel() * (1.0 - self.sparsity)))
            thr = torch.topk(flat, k, largest=True).values[-1]
            mask = (tensor.abs() >= thr).float()
            return tensor * mask, mask

    def compress(self, **kwargs: Any) -> Tuple[nn.Module, Dict[str, Any]]:
        """Executes ADMM pruning with dynamic rho scheduling, STE integration, and LayerNorm recalibration."""
        pruned\_layers\_count = 0

        for name, module in list(self.model.named\_modules()):
            if isinstance(module, nn.Linear):
                W0 = module.weight.data.clone().float()
                device = W0.device
                dtype = module.weight.dtype

                W = W0.clone()
                Z, mask = self.\_project\_sparse(W0)
                U = torch.zeros\_like(W0)

                rho = self.initial\_rho
                mu = 10.0
                tau\_incr = 1.5
                tau\_decr = 1.5
                rho\_min, rho\_max = 1e-4, 100.0

                for it in range(self.max\_iter):
                    Z\_prev = Z.clone()

                     1. Update Z (Sparse projection)
                    Z, mask = self.\_project\_sparse(W + U)

                     2. Update W
                    W = (W0 + rho * (Z - U)) / (1.0 + rho)

                     3. Update Dual variable U
                    U = U + W - Z

                     4. Dynamic Rho Scheduling based on primal/dual residuals
                    if self.dynamic\_rho:
                        primal\_res = torch.norm(W - Z).item()
                        dual\_res = (rho * torch.norm(Z - Z\_prev)).item()
                        if primal\_res > mu * (dual\_res + 1e-8):
                            rho = min(rho * tau\_incr, rho\_max)
                        elif dual\_res > mu * (primal\_res + 1e-8):
                            rho = max(rho / tau\_decr, rho\_min)

                 Final assignment to projection Z and STE Module conversion
                if self.use\_ste:
                    admm\_layer = ADMMMaskedLinear.from\_linear(module, mask=mask)
                    replace\_module(self.model, name, admm\_layer)
                else:
                    module.weight.data = Z.to(dtype=dtype, device=device)

                pruned\_layers\_count += 1

         LayerNorm Recalibration
        if self.recalibrate\_ln:
            recalibrate\_layernorms(
                self.model,
                tokenizer=self.tokenizer,
                calibration\_loader=self.calibration\_loader,
                calibration\_texts=self.calibration\_texts,
                device=self.target\_device,
            )

        self.\_compressed\_params = self.\_count\_params(self.model)
        stats = self.get\_compression\_stats()
        stats.update({
            "pruned\_layers": pruned\_layers\_count,
            "sparsity\_target": self.sparsity,
            "structured\_24": self.structured\_24,
            "admm\_iterations": self.max\_iter,
            "dynamic\_rho": self.dynamic\_rho,
            "use\_ste": self.use\_ste,
            "recalibrate\_ln": self.recalibrate\_ln,
            "method": "admm\_pruner\_ste\_dynamic\_rho",
        })
        self.stats = stats
