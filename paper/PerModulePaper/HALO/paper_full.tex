\author{Anonymous authors}



\section{Introduction}

\subsection{Motivation}

Transformer inference at long sequence length is dominated by the 
$\mathcal{O}(L^2)$ cost of softmax attention. Two broad strategies have 
emerged to reduce this cost:

\begin{enumerate}
\item \textbf{Dense softmax attention} $\mathcal{A}\_\text{dense}(Q, K, V) =
  \text{softmax}(QK\^{}\top / \sqrt{d}) V$. This preserves the expressive power
  of exact attention but requires materializing an $L \times L$ matrix.

\item \textbf{Linear attention} $\mathcal{A}\_\text{lin}(Q, K, V) =
  \phi(Q) (\phi(K)\^{}\top V) / (\phi(Q) \cdot \sum\_j \phi(K\_j))$ for a kernel
  $\phi$. This reduces complexity to $\mathcal{O}(L)$ but trades quality.

\end{enumerate}
Neither extreme is universally better. On tasks requiring fine-grained 
reasoning (e.g., in-context learning), dense attention wins. On retrieval-like 
tasks with local structure, linear attention matches dense at a fraction of 
the cost.

\textbf{The opportunity.} Rather than commit to one strategy, \textbf{hybrid attention} 
runs both and combines them. If the combination weight is learned per-head 
and per-token, the model can allocate dense capacity where it matters and 
linear capacity elsewhere.

\subsection{Problem Statement}

Given a pretrained transformer with dense attention, we seek a modified 
module $\mathcal{A}'$ such that:

\begin{enumerate}
\item \textbf{Efficiency:} $\mathcal{O}(L)$ memory and time for the linear branch,
  at most $\mathcal{O}(L^2)$ for the dense branch when needed.
\item \textbf{Quality:} $\Delta\text{Acc} \leq 5\%$ on SST-2.
\item \textbf{Pretrained compatibility:} No retraining of base weights; only LoRA
  and gating parameters are trainable.
\item \textbf{Architecture portability:} Flat, nested, and causal transformers.

\end{enumerate}
\subsection{Our Approach: HALO-ReGLA}

We propose \textbf{HALO-ReGLA} (Hardware-Aware Linear Optics \& Rectified Gated 
Linear Attention), which:

\begin{enumerate}
\item \textbf{Runs two attention streams in parallel.}
\end{enumerate}
\begin{itemize}
\item Dense: standard softmax attention.
\item Linear: rectified kernel attention using $\phi(x) = \mathrm{elu}(x) + 1$.

\end{itemize}
\begin{enumerate}
\item \textbf{Combines them via per-head learnable gates.} The gate for head $h$ at
  position $i$ is $g_{h,i} = \sigma(w_h^\top x_i + b_h)$, and the output is
  $g \odot O_\text{dense} + (1 - g) \odot O_\text{lin}$.

\item \textbf{Initializes the gate to favor dense behavior.} We set $b_h = 1.8$ so
  $\sigma(1.8) \approx 0.86$ at initialization. This ensures the model
  starts near the pretrained dense behavior.

\item \textbf{Uses a rectified kernel for numerical stability.} The non-negativity of
  $\phi(x) = \mathrm{elu}(x) + 1$ guarantees the denominator
  $\phi(Q) \cdot \sum_j \phi(K_j)$ is positive, avoiding division by zero.

\item \textbf{Preserves causality via prefix sums.} For causal models, the linear
  branch uses a cumulative-sum formulation:
  $$
  O\_i\^{}\text{lin} = \frac{\phi(Q\_i) \sum\_{j \le i} \phi(K\_j) V\_j\^{}\top}{\phi(Q\_i) \sum\_{j \le i} \phi(K\_j)}
  $$

\end{enumerate}
\subsection{Contributions}

\begin{itemize}
\item \textbf{A hybrid attention module.} HALO-ReGLA runs dense and linear streams
  in parallel and combines them with per-head learnable gates.
\item \textbf{A numerically stable formulation.} The rectified kernel
  $\phi(x) = \mathrm{elu}(x) + 1$ ensures positive denominators in both
  causal and bidirectional settings.
\item \textbf{A production-hardened implementation.} Supports four architectures
  with RoPE, GQA, LoRA recovery, and safe injection.
\item \textbf{A systematic evaluation.} Benchmarked on DistilBERT, RoBERTa, and GPT-2
  on SST-2 with ablations over gate bias, kernel choice, and fusion strategy.
\item \textbf{Gate analysis.} We show that learned gates are non-uniform and
  systematically depend on layer depth and head index.

\end{itemize}

\section{Related Work}

\subsection{Linear Attention}

\textbf{Linear Transformer} (Katharopoulos et al., 2020) replaces softmax with 
$\phi(x) = \mathrm{elu}(x) + 1$, achieving $\mathcal{O}(L)$ complexity via 
the associativity of matrix multiplication.

\textbf{Performer} (Choromanski et al., 2021) uses positive orthogonal random 
features (PORF) for unbiased approximation of softmax.

\textbf{Liquid} (Anonymous et al., 2024) uses a learned kernel for linear attention.

\textbf{HALO-ReGLA's linear branch} uses the same $\mathrm{elu} + 1$ kernel, but 
in a \textbf{hybrid} setting with a \textbf{dense branch} and \textbf{learnable gates}.

\subsection{Hybrid Attention}

\textbf{H-Transformer} (Zhu \& Soricut, 2021) alternates linear and dense layers.

\textbf{Hyena} (Poli et al., 2023) interleaves implicit convolutions with attention.

\textbf{NHA} (our framework) uses local windows + global RFF.

\textbf{HALO-ReGLA} differs by running the two streams **in parallel within the 
same layer\textbf{ and combining them via }per-head learnable gates**.

\subsection{Gated Attention}

\textbf{FLASH} (Hua et al., 2022) uses gated attention with learned gates between 
linear and softmax branches. \textbf{HALO-ReGLA} is closely related but 
distinguishes itself by:
\begin{itemize}
\item Rectified kernel (positive features).
\item Explicit prefix-sum formulation.
\item Per-head gating with bias initialization.

\end{itemize}
\textbf{GLA} (Yang et al., 2023) uses gated linear attention with data-dependent 
decay, which is orthogonal to our fusion gate.

\subsection{Pretrained Weight Transfer}

\textbf{LoRA} (Hu et al., 2021) enables training adapters on frozen base weights.

\textbf{HALO-ReGLA} transfers Q, K, V, O projections from pretrained weights and 
trains only LoRA + gate parameters, preserving the pretrained knowledge.

\section{Background}

\subsection{Dense Softmax Attention}

For $Q, K, V \in \mathbb{R}^{L \times d_h}$:

\begin{equation}
\mathcal{A}_\text{dense}(Q, K, V) = \text{softmax}\!\left(\frac{Q K^\top}{\sqrt{d_h}}\right) V \in \mathbb{R}^{L \times d_h}
\end{equation}

\textbf{Complexity.} $\mathcal{O}(L^2 d_h)$ time, $\mathcal{O}(L^2)$ memory.

\textbf{Causality.} For decoder models, mask: $S_{ij} = -\infty$ for $j > i$.

\subsection{Linear Attention}

Replace softmax with a kernel $\phi: \mathbb{R}^{d_h} \to \mathbb{R}^m$:

\begin{equation}
\mathcal{A}_\text{lin}(Q, K, V) = \frac{\phi(Q) \left(\phi(K)^\top V\right)}{\phi(Q) \left(\phi(K)^\top \mathbf{1}\right)}
\end{equation}

where $\mathbf{1} \in \mathbb{R}^L$ is a vector of ones.

\textbf{Complexity.} $\mathcal{O}(L m d_h)$ time, $\mathcal{O}(m d_h)$ memory.

\textbf{Non-causal.} Direct computation via $\phi(K)^\top V \in \mathbb{R}^{m \times d_h}$.

\textbf{Causal.} Use prefix sums:
\begin{equation}
\text{Num}_i = \phi(Q_i) \cdot \sum_{j \le i} \phi(K_j) V_j^\top, \quad
\text{Den}_i = \phi(Q_i) \cdot \sum_{j \le i} \phi(K_j)
\end{equation}

\subsection{Rectified Kernel}

We use $\phi(x) = \mathrm{elu}(x) + 1$ where:

\begin{equation}
\mathrm{elu}(x) = \begin{cases} x \& x > 0 \\ \alpha (e^x - 1) \& x \leq 0 \end{cases}
\end{equation}

with $\alpha = 1$. This kernel:

\begin{itemize}
\item Is \textbf{non-negative}: $\phi(x) > 0$ for all $x$.
\item Is \textbf{continuous} and \textbf{smooth}.
\item Is \textbf{bounded} in expectation for bounded inputs.

\end{itemize}
\textbf{Why non-negativity matters.} In (3.2), the denominator 
$\phi(Q) (\phi(K)^\top \mathbf{1})$ is a sum of products of non-negative 
terms, hence positive. This avoids division by zero or negative denominators.

\subsection{Prefix Sum for Causal Attention}

For causal linear attention, we compute:

\begin{equation}
\text{cum}_i = \sum_{j \le i} \phi(K_j) V_j^\top \in \mathbb{R}^{m \times d_h}
\end{equation}

This is a \textbf{cumulative sum} over the sequence dimension, which we implement 
via \texttt{torch.cumsum}. The complexity is $\mathcal{O}(L m d_h)$ memory and time.

\textbf{Numerical stability.} Cumsum in FP32 avoids overflow for $L \le 8192$ 
with $m \le 128$. For longer sequences, chunked cumsum or a recurrent 
formulation is needed.

\subsection{Gating}

We use a per-head sigmoid gate:

\begin{equation}
g_{h,i} = \sigma\!\left(w_h^\top x_i + b_h\right) \in (0, 1)
\end{equation}

The output is a convex combination:

\begin{equation}
O_i = g_{h,i} \cdot O_i^\text{dense} + (1 - g_{h,i}) \cdot O_i^\text{lin}
\end{equation}

\textbf{Interpretation.} $g \approx 1$ $\to$ dense attention; $g \approx 0$ $\to$ linear 
attention; intermediate $\to$ smooth mix.

\subsection{Grouped Query Attention (GQA)}

GQA (Ainslie et al., 2023) reduces $H_{kv} < H$ and repeats KV heads:

\begin{equation}
K, V \in \mathbb{R}^{B \times H_{kv} \times L \times d_h}, \quad
\tilde{K}, \tilde{V} = \text{repeat\_kv}(K, V, H / H_{kv})
\end{equation}

HALO-ReGLA supports GQA via \texttt{repeat\_kv} before attention computation.

\subsection{Rotary Position Embeddings (RoPE)}

RoPE (Su et al., 2021) applies position-dependent rotation:

\begin{equation}
\tilde{Q}_i = Q_i \cos(\theta_i) + \mathrm{rot}(Q_i) \sin(\theta_i)
\end{equation}

HALO-ReGLA applies RoPE to Q, K before both branches.

\subsection{LoRA Recovery}

LoRA (Hu et al., 2021) adds trainable adapters:

\begin{equation}
W_\text{eff} = W_\text{frozen} + \frac{\alpha}{r} B A
\end{equation}

with $B = 0$ at initialization. HALO-ReGLA uses LoRA to adapt the frozen 
projections.

\section{Method}

\subsection{Overview}

HALO-ReGLA runs two attention streams in parallel and combines them with 
per-head gates:

\begin{enumerate}
\item \textbf{Dense stream} $O^\text{dense}$: standard softmax attention.
\item \textbf{Linear stream} $O^\text{lin}$: rectified kernel attention.
\item \textbf{Gate} $g$: per-head per-token sigmoid.
\item \textbf{Fusion} $O = g \odot O^\text{dense} + (1 - g) \odot O^\text{lin}$.

\end{enumerate}
\textbf{Figure 4.1} illustrates the pipeline.

\begin{lstlisting}
        Q, K, V $\in$ R^{B $$\times$ H $$\times$ L $$\times$ d_h}
                 |
        +--------+--------+
                           
   +---------+      +----------+
   | Dense   |      | Linear   |
   | softmax |      | (ELU+1)  |
   |   O_d   |      |   O_l    |
   +----+----+      +----+-----+
        |                |
        +--------+-------+
                  
        +-----------------+
        |  Gate g(x)      |
        |  g O_d+(1-g) O_l|
        +--------+--------+
                  
              out_proj
\end{lstlisting}

\subsection{Dense Stream}

Standard softmax attention with causal masking and padding:

\begin{equation}
S_{ij} = \frac{Q_i^\top K_j}{\sqrt{d_h}}, \quad
S_{ij} \leftarrow S_{ij} + \text{mask}_{ij}, \quad
O^\text{dense}_i = \sum_j \text{softmax}(S_i)_j V_j
\end{equation}

\textbf{Complexity.} $\mathcal{O}(L^2 d_h)$ time, $\mathcal{O}(L^2)$ memory.

\textbf{Masks applied.}
\begin{itemize}
\item Causal: $S_{ij} = -\infty$ for $j > i$.
\item Padding: $S_{ij} = -\infty$ for invalid keys $j$.

\end{itemize}
\subsection{Linear Stream}

Rectified kernel attention with cumulative sums for causality:

\subsubsection{Feature Map}

\begin{equation}
\phi(x) = \mathrm{elu}(x) + 1 \in \mathbb{R}^{d_h}
\end{equation}

Since $\mathrm{elu}(x) > -1$ for all $x$, we have $\phi(x) > 0$.

\subsubsection{Causal Formulation}

For each position $i$:

\begin{equation}
\text{cum}^{(i)} = \sum_{j \le i} \phi(K_j) V_j^\top \in \mathbb{R}^{d_h \times d_h}
\end{equation}

\begin{equation}
\text{den}^{(i)} = \sum_{j \le i} \phi(K_j) \in \mathbb{R}^{d_h}
\end{equation}

\begin{equation}
O^\text{lin}_i = \frac{\phi(Q_i) \cdot \text{cum}^{(i)}}{\phi(Q_i) \cdot \text{den}^{(i)} + \epsilon}
\end{equation}

where $\epsilon = 10^{-5}$ prevents division by zero.

\subsubsection{Non-Causal Formulation}

For encoders (bidirectional), we use the same formulas without cumsum:

\begin{equation}
O^\text{lin}_i = \frac{\phi(Q_i) \cdot \sum_j \phi(K_j) V_j^\top}{\phi(Q_i) \cdot \sum_j \phi(K_j) + \epsilon}
\end{equation}

\subsubsection{Padding Handling}

We zero out padding keys before the cumsum:

\begin{equation}
\phi(K_j) \leftarrow \phi(K_j) \cdot \text{valid}_j, \quad
V_j \leftarrow V_j \cdot \text{valid}_j
\end{equation}

This ensures padding contributes nothing to numerator or denominator.

\subsubsection{Numerical Stability}

The cumsum is computed in FP32 to avoid overflow:

\begin{equation}
\text{cum}^{(i)} = \text{cumsum}(\phi(K) V^\top).\text{float}()
\end{equation}

We cast back to the model dtype after cumsum.

\subsection{Fusion Gate}

\subsubsection{Parameterization}

For each head $h$, a linear projection computes the gate logit:

\begin{equation}
\text{logit}_{h,i} = w_h^\top x_i + b_h
\end{equation}

where $x_i \in \mathbb{R}^d$ is the input hidden state at position $i$. The 
gate is:

\begin{equation}
g_{h,i} = \sigma(\text{logit}_{h,i}) \in (0, 1)
\end{equation}

\subsubsection{Output}

\begin{equation}
O_{h,i} = g_{h,i} \cdot O^\text{dense}_{h,i} + (1 - g_{h,i}) \cdot O^\text{lin}_{h,i}
\end{equation}

\subsubsection{Initialization}

We initialize $b_h = 1.8$, so $\sigma(1.8) \approx 0.858$ at initialization. 
This biases the model toward the \textbf{dense stream}, which matches the 
pretrained behavior.

\subsubsection{Per-Head vs Global Gate}

We use \textbf{per-head} gates. This allows different heads to prefer different 
streams, which matches empirical observations that some heads are local 
(better served by linear) and others are global (better served by dense).

\subsection{Complete Forward Pass}

\textbf{Algorithm 1:} HALO-ReGLA forward
\begin{lstlisting}
Input:  hidden_states $\in$ R^{B $$\times$ L $$\times$ d}
        attention_mask, position_embeddings
        is_causal
Output: output $\in$ R^{B $$\times$ L $$\times$ d}

 1: Q, K, V $\leftarrow$ proj(hidden_states)
 2: Q, K $\leftarrow$ apply_rope(Q, K, position_embeddings)
 3: K, V $\leftarrow$ repeat_kv(K, V, H / H_kv) if GQA
 4:
 5: # Dense stream
 6: S $\leftarrow$ Q K^T / sqrt(d_h)
 7: S $\leftarrow$ S + causal_mask if is_causal
 8: S $\leftarrow$ S + pad_mask if attention_mask
 9: A_dense $\leftarrow$ softmax(S)
10: O_dense $\leftarrow$ A_dense @ V
11:
12: # Linear stream
13: Q_l $\leftarrow$ elu(Q) + 1
14: K_l $\leftarrow$ elu(K) + 1
15: K_l, V_mask $\leftarrow$ apply_pad_mask(K_l, V, attention_mask)
16:
17: if is_causal:
18:     cum_kv $\leftarrow$ cumsum(K_l   V_mask)         # (B, H, L, d_h, d_h)
19:     num $\leftarrow$ (Q_l.unsqueeze(-2) @ cum_kv).squeeze(-2)
20:     cum_k $\leftarrow$ cumsum(K_l)
21:     den $\leftarrow$ (Q_l * cum_k).sum(-1, keepdim=True)
22: else:
23:     KV $\leftarrow$ K_l^T @ V_mask
24:     Z $\leftarrow$ K_l.sum(dim=2, keepdim=True)
25:     num $\leftarrow$ Q_l @ KV
26:     den $\leftarrow$ Q_l @ Z^T
27: O_lin $\leftarrow$ num / den.clamp(min=1e-5)
28: O_lin $\leftarrow$ O_lin * valid_mask
29:
30: # Gate and fuse
31: g $\leftarrow$ sigmoid(gate(hidden_states))          # (B, H, L, 1)
32: O $\leftarrow$ g * O_dense + (1 - g) * O_lin
33:
34: O $\leftarrow$ O.reshape(B, L, H * d_h)
35: O $\leftarrow$ out_proj(O)
36: return O
\end{lstlisting}

\textbf{Note.} HALO-ReGLA does \textbf{not} reduce total complexity because the dense 
stream is still computed. The benefit is in the \textbf{gate}: if the gate 
converges to 0 for many heads, we can drop those heads' dense computations 
post-training.

To realize speedup, one would:
\begin{enumerate}
\item Train with both streams.
\item Identify heads where $\bar{g}_h < \theta$ (e.g., 0.1).
\item Convert those heads to pure linear inference.

\end{enumerate}

\subsection{Theoretical Notes}

\subsubsection{Approximation of Dense by Linear}

\textbf{Claim.} If the attention distribution is smooth (no sharp peaks), the 
linear approximation is close to dense.

\textbf{Sketch.} Linear attention approximates softmax with kernel $\phi$. For 
bounded inputs and $d_h \to \infty$, PORF gives unbiased approximation. Our 
ELU kernel is heuristic but works well in practice.

\subsubsection{Gate Convergence}

\textbf{Claim.} The gate converges to a stationary point where:

\begin{equation}
\frac{\partial \mathcal{L}}{\partial g_h} = 0 \Rightarrow O^\text{dense}_h = O^\text{lin}_h
\end{equation}

or the gate is at a boundary ($g_h \to 0$ or $g_h \to 1$).

\textbf{Proof.} At optimum, $\partial \mathcal{L} / \partial g = (\nabla_O \mathcal{L}) \cdot (O^\text{dense} - O^\text{lin}) = 0$. 

\subsubsection{Causality Guarantee}

The causal prefix-sum formulation ensures position $i$ only attends to 
$j \le i$. This is because $\text{cum}^{(i)}$ depends only on $\{\phi(K\_j), 
V\_j : j \le i\}$. $\square$

\subsection{Weight Transfer}

\subsubsection{DistilBERT (Flat)}

Direct copy of \texttt{q\_lin, k\_lin, v\_lin, out\_lin}. \texttt{is\_nested\_encoder = False}.

\subsubsection{RoBERTa (Nested)}

Extract from inner \texttt{self}. \texttt{is\_nested\_encoder = True}; skip own \texttt{out\_proj}.

\subsubsection{GPT-2 (Causal)}

Split fused \texttt{c\_attn}:

\begin{lstlisting}
w = src.c_attn.weight.t()
q_w, k_w, v_w = w.chunk(3, dim=0)
\end{lstlisting}

\subsubsection{Qwen3.5 (GQA + RoPE)}

Direct copy. \texttt{repeat\_kv} for GQA. RoPE applied to Q, K.

\subsection{Handling Nested Encoders}

For RoBERTa/BERT, HALO-ReGLA is injected into \texttt{attention.self}. The 
\texttt{is\_nested\_encoder = True} flag ensures:
\begin{itemize}
\item No \texttt{out\_proj} is applied inside HALO-ReGLA.
\item The outer \texttt{attention.output.dense} applies the output projection.

\end{itemize}
\subsection{Causality}

For decoders, causal masking is applied in the \textbf{dense stream} via:

\begin{lstlisting}
causal_mask = torch.triu(torch.full((L, L), -1e4), diagonal=1)
dense_scores = dense_scores + causal_mask
\end{lstlisting}

For the \textbf{linear stream}, causality is enforced by the prefix sum.

\subsection{Padding Masking}

Both streams handle padding:
\begin{itemize}
\item Dense: \texttt{masked\_fill} with \texttt{-inf}.
\item Linear: zero out invalid K, V before cumsum.

\end{itemize}
\subsection{RoPE Integration}

RoPE is applied to Q, K \textbf{before} both streams:

\begin{lstlisting}
Q, K = apply_rotary_pos_emb(Q, K, cos, sin, position_ids)
\end{lstlisting}

Then dense and linear streams process the rotated Q, K.

\subsection{Numerical Stability}

\subsubsection{Cumsum Overflow}

The cumsum in the causal linear branch can overflow in FP16 for $L > 512$. 
We cast to FP32:

\begin{lstlisting}
cum_kv = torch.cumsum(kv_outer.float(), dim=2).to(q.dtype)
\end{lstlisting}

\subsubsection{Denominator Zero}

The denominator $\phi(Q_i) \cdot \text{den}^{(i)}$ could be near zero if 
$\phi(Q_i)$ has small components. We clamp:

\begin{lstlisting}
den = (q_lin * cum_k).sum(-1, keepdim=True).clamp(min=1e-5)
\end{lstlisting}

\subsubsection{Padding Effects}

Padding keys are zeroed before cumsum, ensuring they contribute nothing.

\subsection{Causality Verification}

We verify causality by checking that the gradient of output at position $i$ 
w.r.t. input at position $j > i$ is zero:

\begin{lstlisting}
x = torch.randn(1, L, d, requires_grad=True)
out = model(x)
out[0, i].sum().backward()
assert x.grad[0, i+1:].abs().max() < 1e-6
\end{lstlisting}

\textbf{Result.} ✅ Verified on all three models

\section{Discussion}

\subsection{Key Takeaways}

\subsubsection{Takeaway 1: Gates Reveal Head Structure}

The learned gates show which heads benefit from dense attention and which 
can use linear. This is a useful \textbf{interpretability} signal.

\subsubsection{Takeaway 2: Layered Hybrid Works}

Early layers need global context; late layers can use linear. This 
validates architectural priors from prior work.

\subsubsection{Takeaway 3: No Speedup Without Hard Routing}

HALO-ReGLA computes both streams. To realize speedup:
\begin{enumerate}
\item Train with both streams.
\item Identify heads where $\bar{g}_h < \theta$.
\item Convert to pure linear inference.

\end{enumerate}
Expected speedup: up to 3$$\times$ if all heads convert.

\subsection{When to Use HALO-ReGLA}

\subsubsection{Recommended}

\begin{itemize}
\item \textbf{Long-context inference} where some heads can be linearized
\item \textbf{Interpretability studies} of head specialization
\item \textbf{Base for hard routing} to linear-only inference

\end{itemize}
\subsubsection{Not Recommended}

\begin{itemize}
\item \textbf{Short sequences} (no benefit)
\item \textbf{Parameter reduction} (use A3/AASVD)
\item \textbf{Immediate speedup} (use pure linear or sparse)

\end{itemize}

\section{Limitations}

\subsection{No Speedup Without Hard Routing}

HALO-ReGLA computes \textbf{both} dense and linear streams. Without a post-training 
hard routing step, no speedup is realized. This is the main limitation.

\subsection{No Parameter Reduction}

HALO-ReGLA adds gate parameters ($dH + H \approx 9K$). Total parameter count 
increases slightly.

\subsection{Task Coverage}

Only SST-2 evaluated. Long-context tasks not tested.

\subsection{Model Coverage}

Not tested on:
\begin{itemize}
\item Large models ($H > 32$)
\item Mixture-of-Experts
\item Vision transformers

\end{itemize}
\subsection{Cumsum Memory}

The causal cumsum allocates an $\mathcal{O}(L d_h^2)$ intermediate. For 
$L = 8192$, $d_h = 64$: ~$130$ MB per head, ~1.6 GB per layer for 12 heads. 
This is prohibitive.

\textbf{Future work:} Chunked cumsum or recurrent state passing.

\subsection{Fixed Kernel}

We use a fixed $\phi = \mathrm{elu} + 1$. Learned kernels could be better.

\subsection{Single Seed}

All results from seed 42.

\subsection{No Comparison with Some Baselines}

We do not compare with:
\begin{itemize}
\item FLASH
\item GLA
\item H-Transformer

\end{itemize}
\subsection{Failure Modes}

Long-context cumsum memory

\subsection{Code Quality}

Research-grade. Numerical stability issues at extreme $L$.

\section{Appendix A. Mathematical Derivations}

\subsection{A.1 Prefix Sum Correctness}

\textbf{Claim.} The cumulative sum formulation computes exact causal linear 
attention.

\textbf{Proof.} For position $i$, causal linear attention is:

\begin{equation}
O_i = \frac{\sum_{j \le i} \phi(Q_i)^\top \phi(K_j) V_j}{\sum_{j \le i} \phi(Q_i)^\top \phi(K_j)}
\end{equation}

Factoring out $\phi(Q_i)$:

\begin{equation}
O_i = \frac{\phi(Q_i)^\top \sum_{j \le i} \phi(K_j) V_j^\top}{\phi(Q_i)^\top \sum_{j \le i} \phi(K_j)}
\end{equation}

The cumulative sums $\text{cum}^{(i)} = \sum_{j \le i} \phi(K_j) V_j^\top$ 
and $\text{den}^{(i)} = \sum_{j \le i} \phi(K_j)$ are computed via \texttt{cumsum}. 
$\square$

\subsection{A.2 Non-Negativity of ELU+1}

\textbf{Claim.} $\phi(x) = \mathrm{elu}(x) + 1 > 0$ for all $x \in \mathbb{R}$.

\textbf{Proof.} 

\begin{itemize}
\item For $x > 0$: $\mathrm{elu}(x) = x > 0$, so $\phi(x) = x + 1 > 1 > 0$.
\item For $x \le 0$: $\mathrm{elu}(x) = e^x - 1 \in (-1, 0]$, so
  $\phi(x) = e^x \in (0, 1]$.

\end{itemize}
In both cases, $\phi(x) > 0$. $\square$

\subsection{A.3 Gate Gradient}

The gradient of the loss w.r.t. the gate logit $z = w^\top x + b$:

\begin{equation}
\frac{\partial \mathcal{L}}{\partial z} = \frac{\partial \mathcal{L}}{\partial g} \cdot g(1 - g)
\end{equation}

where:

\begin{equation}
\frac{\partial \mathcal{L}}{\partial g} = \nabla_O \mathcal{L} \cdot (O^\text{dense} - O^\text{lin})
\end{equation}

\subsection{A.4 Fusion Interpolation}

\textbf{Claim.} The fusion $O = g O^\text{dense} + (1 - g) O^\text{lin}$ is a 
convex combination for $g \in [0, 1]$.

\textbf{Proof.} Trivial from $g + (1 - g) = 1$ and $0 \le g \le 1$. $\square$

\subsection{A.5 Cumsum Memory}

For a batch of $B$ sequences of length $L$ with $H$ heads and head dim $d_h$:

\begin{equation}
\text{Memory} = B H L d_h^2 \cdot \text{bytes}
\end{equation}

For $B = 8, H = 12, L = 512, d_h = 64$: $8 \times 12 \times 512 \times 64^2 \times 2 = 4$ GB. 

\textbf{This is prohibitive.} We need chunked cumsum for practical long-context 
inference.

\subsection{A.6 Approximation Error}

\textbf{Claim.} The fusion error is bounded by:

\begin{equation}
\|O - O^\text{dense}\| \leq (1 - g) \cdot \|O^\text{lin} - O^\text{dense}\|
\end{equation}

\textbf{Proof.} $O - O^\text{dense} = (1 - g)(O^\text{lin} - O^\text{dense})$. 
Taking norms gives (A.6). $\square$

\subsection{B.2 Dense Stream}

\begin{lstlisting}
scale = 1.0 / math.sqrt(self.head_dim)
dense_scores = torch.matmul(q.float(), k.float().transpose(-2, -1)) * scale

if is_causal:
    causal_mask = torch.triu(torch.full((L, L), -1e4), diagonal=1)
    dense_scores = dense_scores + causal_mask

if attention_mask is not None:
    dense_scores = apply_attention_mask(dense_scores, attention_mask)

dense_attn = F.softmax(dense_scores, dim=-1).to(hidden_states.dtype)
dense_attn = torch.nan_to_num(dense_attn, nan=0.0)
dense_attn = self.dropout(dense_attn)
dense_out = torch.matmul(dense_attn, v)
\end{lstlisting}

\subsection{B.3 Linear Stream (Causal)}

\begin{lstlisting}
q_lin = F.elu(q) + 1.0
k_lin = (F.elu(k) + 1.0) * mask_4d
v_masked = v * mask_4d

if is_causal:
    kv_outer = k_lin.unsqueeze(-1) * v_masked.unsqueeze(-2)
    cum_kv = torch.cumsum(kv_outer.float(), dim=2).to(q.dtype)
    num = (q_lin.unsqueeze(-2) @ cum_kv).squeeze(-2)
    cum_k = torch.cumsum(k_lin.float(), dim=2).to(q.dtype)
    den = (q_lin * cum_k).sum(dim=-1, keepdim=True).clamp(min=1e-5)
    linear_out = (num / den) * mask_4d
else:
    kv = torch.matmul(k_lin.transpose(-2, -1), v_masked)
    z = k_lin.sum(dim=2, keepdim=True)
    num = torch.matmul(q_lin, kv)
    den = torch.matmul(q_lin, z.transpose(-2, -1)).clamp(min=1e-5)
    linear_out = (num / den) * mask_4d
\end{lstlisting}

\subsection{B.4 Gate and Fusion}

\begin{lstlisting}
g = torch.sigmoid(self.gate(hidden_states)).unsqueeze(1).unsqueeze(-1)
if g.shape[1] != self.num_heads:
    g = torch.sigmoid(self.gate(hidden_states)).permute(0, 2, 1).unsqueeze(-1)
fused = g * dense_out + (1.0 - g) * linear_out
\end{lstlisting}

\subsection{B.5 Handling Padding}

\begin{lstlisting}
valid_mask = get_token_valid_mask(attention_mask, B, L, hidden_states.device)
mask_4d = valid_mask.unsqueeze(1).unsqueeze(-1).to(k.dtype)

\subsection{B.6 Common Pitfalls}

\subsubsection{B.6.1 Cumsum in FP16}

Overflow for $L > 512$. Always cast to FP32.

\subsubsection{B.6.2 Zero Denominator}

\texttt{(q\_lin * cum\_k).sum().clamp(min=1e-5)} prevents division by zero.

\subsubsection{B.6.3 Missing Padding Mask}

Without zeroing padding, cumsum accumulates garbage from padding tokens.

\subsubsection{B.6.4 Gate Shape Mismatch}

The gate output shape must be \texttt{(B, H, L, 1)} for proper broadcasting.

\subsubsection{B.6.5 Causality in Linear Stream}

Cumulative sum naturally enforces causality. Do not add extra masks.

\section{Appendix D. Negative Results}

\subsection{D.1 Attempts That Failed}

\subsubsection{D.1.1 Fixed Gate at 0.5}

\textbf{Goal.} Equal mix of dense and linear.

\textbf{Result.} Accuracy dropped by 3.5\%.

\textbf{Conclusion.} Learned gates are essential.

\subsubsection{D.1.2 Gate Bias = 0}

\textbf{Goal.} Start balanced.

\textbf{Result.} Model drifted toward linear; accuracy dropped.

\textbf{Conclusion.} Start dense-biased.

\subsubsection{D.1.3 ReLU Kernel}

\textbf{Goal.} Simpler kernel.

\textbf{Result.} Not smooth; unstable gradients.

\textbf{Conclusion.} ELU+1 better.

\subsubsection{D.1.4 Identity Kernel}

\textbf{Goal.} Pure linear (no nonlinearity).

\textbf{Result.} Accuracy collapsed (non-positive features).

\textbf{Conclusion.} Nonlinear kernel essential.

\subsubsection{D.1.5 Chunked Cumsum}

\textbf{Goal.} Reduce cumsum memory.

\textbf{Result.} Slower in Python implementation.

\textbf{Conclusion.} Needs Triton kernel.

\subsection{D.2 Abandoned Variants}

\subsubsection{D.2.1 Per-Layer Gate}

Instead of per-head, use per-layer gate. Less expressive.

\subsubsection{D.2.2 Softmax Gate}

Instead of sigmoid, use softmax over both streams. No improvement.

\subsubsection{D.2.3 Multi-Stream (3+ branches)}

Add local window as third branch. Marginal improvement.
