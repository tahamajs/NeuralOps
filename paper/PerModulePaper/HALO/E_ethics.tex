
\section{Appendix E. Ethical Considerations}

\subsection{E.1 Broader Impact}

\subsubsection{Positive}
\begin{itemize}
\item \textbf{Long-context inference:} Enables more capable applications.
\item \textbf{Interpretability:} Reveals head structure.
\item \textbf{Foundation for efficiency:} Basis for hard routing.

\end{itemize}
\subsubsection{Negative}
\begin{itemize}
\item \textbf{Dual-use:} Efficiency gains benefit malicious applications.
\item \textbf{Concentration:} Favors large labs.

\end{itemize}
\subsection{E.2 Environmental Impact}

\begin{itemize}
\item \textbf{Total GPU-hours:} ~15
\item \textbf{Estimated CO$_2$:} 1.0 kg CO$_2$eq

\end{itemize}
