
\section{Appendix D. Negative Results}

\subsection{D.1 Attempts That Failed}

\subsubsection{D.1.1 Fixed Gate at 0.5}

\textbf{Goal.} Equal mix of dense and linear.

\textbf{Result.} Accuracy dropped by 3.5\%.

\textbf{Conclusion.} Learned gates are essential.

\subsubsection{D.1.2 Gate Bias = 0}

\textbf{Goal.} Start balanced.

\textbf{Result.} Model drifted toward linear; accuracy dropped.

\textbf{Conclusion.} Start dense-biased.

\subsubsection{D.1.3 ReLU Kernel}

\textbf{Goal.} Simpler kernel.

\textbf{Result.} Not smooth; unstable gradients.

\textbf{Conclusion.} ELU+1 better.

\subsubsection{D.1.4 Identity Kernel}

\textbf{Goal.} Pure linear (no nonlinearity).

\textbf{Result.} Accuracy collapsed (non-positive features).

\textbf{Conclusion.} Nonlinear kernel essential.

\subsubsection{D.1.5 Chunked Cumsum}

\textbf{Goal.} Reduce cumsum memory.

\textbf{Result.} Slower in Python implementation.

\textbf{Conclusion.} Needs Triton kernel.

\subsection{D.2 Abandoned Variants}

\subsubsection{D.2.1 Per-Layer Gate}

Instead of per-head, use per-layer gate. Less expressive.

\subsubsection{D.2.2 Softmax Gate}

Instead of sigmoid, use softmax over both streams. No improvement.

\subsubsection{D.2.3 Multi-Stream (3+ branches)}

Add local window as third branch. Marginal improvement.

\subsection{D.6 Lessons Learned}

\begin{enumerate}
\item \textbf{Learned gates beat fixed.}
\item \textbf{Dense bias at init is essential.}
\item \textbf{ELU+1 is the best kernel.}
\item \textbf{Cumsum needs FP32.}
\item \textbf{No speedup without hard routing.}
\end{enumerate}
