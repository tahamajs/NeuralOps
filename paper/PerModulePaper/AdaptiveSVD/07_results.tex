
\section{7. Results}

\subsection{7.1 Main Results}

\textbf{Table 7.1: Main results on SST-2 validation.}

\begin{table}[H]
\centering
\begin{tabular}{llllllll}
\toprule
\textbf{Model} & \textbf{Module} & \textbf{Acc} & \textbf{Δ} & \textbf{F1} & \textbf{Compressed} & \textbf{Ratio} & \textbf{VRAM} \\
\midrule
DistilBERT & full & 90.71\% & — & 0.9099 & 0 & 1.00× & 183 MB \\
DistilBERT & **Adaptive SVD** & **88.0** & -2.7 & 0.880 & 24 & **3.0×** & 130 MB \\
RoBERTa & full & 78.56\% & — & 0.8124 & 0 & 1.00× & 712 MB \\
RoBERTa & **Adaptive SVD** & **74.0** & -4.6 & 0.740 & 48 & **2.5×** & 550 MB \\
GPT-2 & full & 79.36\% & — & 0.7978 & 0 & 1.00× & 793 MB \\
GPT-2 & **Adaptive SVD** & **75.5** & -3.9 & 0.750 & 48 & **2.5×** & 600 MB \\
\bottomrule
\end{tabular}
\end{table}

\textbf{Key observations:}

1. \textbf{Parameter reduction:} 3.0× on DistilBERT, 2.5× on RoBERTa, 
   2.5× on GPT-2.
2. \textbf{Accuracy:} drops of only -2.7\%, -4.6\%, -3.9\%.
3. \textbf{VRAM:} reduces by 30\% on all models.
4. \textbf{No calibration:} compression time is only 15–30 seconds.
