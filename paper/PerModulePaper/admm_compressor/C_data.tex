"""
ADMM Compressor: Alternating Direction Method of Multipliers for Joint Pruning and Quantization.
Registered in COMPRESSOR\_REGISTRY as 'admm'.

Mathematical Formulation:
$$\\min\_{W, Z, Q} \\frac{1}{2} \\|W - W\_0\\|\_F^2 + g(Z) + h(Q) \\quad \\text{s.t.} \\quad W = Z, \\; W = Q$$

Augmented Lagrangian:
$$\\mathcal{L}(W, Z, Q, U\_1, U\_2) = \\frac{1}{2} \\|W - W\_0\\|\_F^2 + g(Z) + h(Q) + \\frac{\\rho\_1}{2} \\|W - Z + U\_1\\|\_F^2 + \\frac{\\rho\_2}{2} \\|W - Q + U\_2\\|\_F^2$$

ADMM Iterations:
1. $W^{(t+1)} = \\frac{1}{1 + \\rho\_1 + \\rho\_2} \\left( W\_0 + \\rho\_1 (Z^{(t)} - U\_1^{(t)}) + \\rho\_2 (Q^{(t)} - U\_2^{(t)}) \\right)$
2. $Z^{(t+1)} = \\Pi\_{\\mathcal{S}}(W^{(t+1)} + U\_1^{(t)})$ (Structured 2:4 or magnitude/Hessian pruning projection)
3. $Q^{(t+1)} = \\Pi\_{\\mathcal{Q}}(W^{(t+1)} + U\_2^{(t)})$ ($b$-bit Quantization projection)
4. $U\_1^{(t+1)} = U\_1^{(t)} + W^{(t+1)} - Z^{(t+1)}$
5. $U\_2^{(t+1)} = U\_2^{(t)} + W^{(t+1)} - Q^{(t+1)}$
"""

from \_\_future\_\_ import annotations

from typing import Any, Dict, List, Optional, Tuple, Union
import logging
import torch
import torch.nn as nn

from framework.core.base import BaseCompressor
from framework.core.registry import COMPRESSOR\_REGISTRY
from framework.compression\_engine.pruning\_utils import (
    apply\_24\_structured\_mask,
    apply\_block\_mask,
)

logger = logging.getLogger(\_\_name\_\_)


class ADMMCompressor(BaseCompressor):
    """
    ADMM-based Joint Structured Pruning and Non-Uniform / Uniform Quantization Engine.
    """

    def \_\_init\_\_(
        self,
        model: nn.Module,
        sparsity: float = 0.5,
        quant\_bits: int = 4,
        rho1: float = 0.1,
        rho2: float = 0.1,
        max\_iter: int = 8,
        structured\_24: bool = True,
        block\_size: Tuple[int, int] = (4, 4),
        **kwargs: Any,
    ) -> None:
        super().\_\_init\_\_(model, name="admm", **kwargs)
        self.sparsity = max(0.0, min(float(sparsity), 0.95))
        self.quant\_bits = int(quant\_bits)
        self.rho1 = rho1
        self.rho2 = rho2
        self.max\_iter = max\_iter
        self.structured\_24 = structured\_24
        self.block\_size = block\_size
        self.stats: Dict[str, Any] = {}

    def \_quantize\_projection(self, tensor: torch.Tensor) -> torch.Tensor:
        """Projects continuous tensor onto b-bit uniform quantization grid."""
        if self.quant\_bits >= 32:
            return tensor
        scale = 2.0 ** (self.quant\_bits - 1) - 1.0
        max\_val = torch.max(torch.abs(tensor)).clamp(min=1e-6)
        normalized = torch.clamp(tensor / max\_val, -1.0, 1.0)
        quantized = torch.round(normalized * scale) / scale
        return quantized * max\_val

    def \_prune\_projection(self, tensor: torch.Tensor) -> torch.Tensor:
        """Projects continuous tensor onto target sparse constraint set S."""
        if self.structured\_24:
            pruned, \_ = apply\_24\_structured\_mask(tensor, saliency=torch.abs(tensor), dim=1)
            return pruned
        elif self.block\_size != (1, 1):
            pruned, \_ = apply\_block\_mask(tensor, block\_size=self.block\_size, sparsity=self.sparsity)
            return pruned
        else:
            flat = tensor.abs().reshape(-1)
            k = max(1, int(flat.numel() * (1.0 - self.sparsity)))
            thr = torch.topk(flat, k, largest=True).values[-1]
            mask = (tensor.abs() >= thr).float()
            return tensor * mask

    def compress(self, **kwargs: Any) -> Tuple[nn.Module, Dict[str, Any]]:
        """Executes ADMM joint optimization across all Linear modules."""
        compressed\_layers\_count = 0

        for name, module in self.model.named\_modules():
            if isinstance(module, nn.Linear):
                W0 = module.weight.data.clone().float()
                device = W0.device
                dtype = module.weight.dtype

                 Initialize variables
                W = W0.clone()
                Z = self.\_prune\_projection(W0)
                Q = self.\_quantize\_projection(W0)
                U1 = torch.zeros\_like(W0)
                U2 = torch.zeros\_like(W0)

                for it in range(self.max\_iter):
                     1. Update W
                    denom = 1.0 + self.rho1 + self.rho2
                    W = (W0 + self.rho1 * (Z - U1) + self.rho2 * (Q - U2)) / denom

                     2. Update Z (Pruning projection)
                    Z = self.\_prune\_projection(W + U1)

                     3. Update Q (Quantization projection)
                    Q = self.\_quantize\_projection(W + U2)

                     4. Update Dual Multipliers
                    U1 = U1 + W - Z
                    U2 = U2 + W - Q

                 Final combined representation: quantized pruned weight
                W\_final = self.\_quantize\_projection(Z)
                module.weight.data = W\_final.to(dtype=dtype, device=device)
                compressed\_layers\_count += 1

        self.\_compressed\_params = self.\_count\_params(self.model)
        stats = self.get\_compression\_stats()
        stats.update({
            "compressed\_layers": compressed\_layers\_count,
            "sparsity\_target": self.sparsity,
            "quant\_bits": self.quant\_bits,
            "structured\_24": self.structured\_24,
            "admm\_iterations": self.max\_iter,
            "method": "admm\_joint\_prune\_quant",
        })
        self.stats = stats
        return self.model, stats
