\section{Method}

\subsection{Overview}

MoE-DeltaKV-Paged has four stages:

\begin{enumerate}
\item \textbf{Projection + RoPE} ( 4.2): standard Q/K/V projection with optional RoPE.
\item \textbf{Delta-KV compression} ( 4.3): temporal deltas $\to$ SVD $\to$ 4-bit quantization $\to$ reconstruction.
\item \textbf{Paged attention with MoBA routing} ( 4.4): top-$k$ page selection per query page.
\item \textbf{Gated fusion + output projection} ( 4.5): mixture weights $$\times$ page contexts + output.

\end{enumerate}
\begin{lstlisting}
        Input X
           |
            
    +----------------------+
    | Q/K/V projection     |
    | + RoPE               |
    +-------+--------------+
            |
    +------- --------------+
    | Delta-KV compression |
    |   $\Delta$K = K - shift(K)  |
    |   SVD $\to$ 4-bit quant  |
    |   cumsum reconstruct |
    +-------+--------------+
            |
    +------- --------------+
    | Page partition       |
    |   P tokens per page  |
    +-------+--------------+
            |
    +------- --------------+
    | MoBA Router          |
    |    _ij = q _i   k _j    |
    |   TopK pages         |
    |   Page-0 boost       |
    +-------+--------------+
            |
    +------- --------------+
    | Block attention      |
    |   over selected pages|
    +-------+--------------+
            |
    +------- --------------+
    | Gated fusion         |
    |   $\Sigma$ g_j   out_j      |
    +-------+--------------+
            |
    +------- --------------+
    | out_proj             |
    +-------+--------------+
            |
        Output
\end{lstlisting}

\subsection{Projection + RoPE}

Standard LoRALinear projections:

\begin{lstlisting}
q = self.query(hidden_states)
k = self.key(hidden_states)
v = self.value(hidden_states)
\end{lstlisting}

Reshape to $[B, H, N, d_h]$ and apply RoPE if \texttt{position\_embeddings} is provided.

\subsection{Delta-KV Compression}

\subsubsection{Delta Computation}

\begin{lstlisting}
delta_K = K[:, :, 1:, :] - K[:, :, :-1, :]  # [B, H, N-1, d]
delta_V = V[:, :, 1:, :] - V[:, :, :-1, :]
\end{lstlisting}

\subsubsection{Low-Rank SVD + 4-bit Quantization}

\begin{lstlisting}
def low_rank_svd_quantize(delta, rank):
    B, H, N, d = delta.shape
    delta_flat = delta.reshape(-1, d)
    
    U, S, Vh = torch.linalg.svd(delta_flat.float(), full_matrices=False)
    U, S, Vh = U[:, :rank], S[:rank], Vh[:rank, :]
    
    delta_approx = ((U * S.unsqueeze(0)) @ Vh).to(delta.dtype)
    delta_approx = delta_approx.reshape(B, H, N, d)
    
    return ste_quantize_4bit(delta_approx)
\end{lstlisting}

\textbf{Why SVD first?} Temporal deltas have low-rank structure (natural language is 
smooth), so truncating to rank $r$ removes noise without losing signal. The 
4-bit quantization then simulates low-precision storage.

\subsubsection{Reconstruction}

\begin{lstlisting}
K_hat[:, :, 0] = K[:, :, 0]
K_hat[:, :, 1:] = K[:, :, 0:1] + torch.cumsum(delta_K_hat, dim=2)
\end{lstlisting}

\subsubsection{Caveat: This is Simulation}

The output $K_\text{hat}$ has the \textbf{same shape} as $K$. No memory is 
saved. The delta + SVD + quantization pipeline is a **training-time 
regularization** that makes the model robust to future real compression.

\subsection{MoBA Router}

\subsubsection{Page Partition}

Partition $q, k_\text{hat}, v_\text{hat}$ into $N_p = \lceil N / P \rceil$ 
pages of size $P$ (padding with zeros if needed):

\begin{lstlisting}
num_pages = (N + P - 1) // P
pad_len = num_pages * P - N
q_padded = F.pad(q, (0, 0, 0, pad_len))
k_padded = F.pad(k_hat, (0, 0, 0, pad_len))
\end{lstlisting}

\subsubsection{Page-Mean Projections}

\begin{lstlisting}
q_bar = (q_blocks * valid_b.unsqueeze(-1)).sum(dim=3) / valid_b.sum(dim=3, keepdim=True)
k_bar = (k_blocks * valid_b.unsqueeze(-1)).sum(dim=3) / valid_b.sum(dim=3, keepdim=True)
\end{lstlisting}

\subsubsection{Router Logits}

\[
\ell_{ij} = \frac{(W_\text{rg} \bar{q}_i) \cdot \bar{k}_j}{\sqrt{d_h}}
\tag{4.1}
\]

$W_\text{rg}$ is initialized to identity.

\subsubsection{Causal Masking}

For decoders:

\[
\ell_{ij} \leftarrow -\infty \quad \text{if } j > i
\tag{4.2}
\]

\subsubsection{CLS Boost}

\[
\ell_{i, 0} \leftarrow \ell_{i, 0} + 10^4
\tag{4.3}
\]

\subsubsection{Top-K Selection}

\begin{lstlisting}
k_experts = min(self.k_experts, num_pages)
_, topk_indices = torch.topk(router_logits_biased, k_experts, dim=-1)
topk_logits = torch.gather(router_logits, -1, topk_indices)
g_j = F.softmax(topk_logits, dim=-1)
\end{lstlisting}

\subsection{Block Attention}

\subsubsection{Gather Selected Pages}

\begin{lstlisting}
expanded_indices = topk_indices.unsqueeze(-1).unsqueeze(-1).expand(-1, -1, -1, -1, P, d)
k_blocks_exp = k_blocks.unsqueeze(2).expand(-1, -1, num_pages, -1, -1, -1)
k_gathered = torch.gather(k_blocks_exp, dim=3, index=expanded_indices)
\end{lstlisting}

\subsubsection{Attention Per Expert}

\begin{lstlisting}
attn_scores = torch.matmul(q_expanded, k_gathered.transpose(-1, -2)) / math.sqrt(d)
attn_scores = attn_scores.masked_fill(~k_valid.unsqueeze(-2), -10000.0)
attn_probs = F.softmax(attn_scores, dim=-1)
context_per_expert = torch.matmul(attn_probs, v_gathered)
\end{lstlisting}

\subsubsection{Gated Fusion}

\begin{lstlisting}
g_j_exp = g_j.unsqueeze(-1).unsqueeze(-1)
context = (context_per_expert * g_j_exp).sum(dim=3)
\end{lstlisting}

\subsection{Full Algorithm}

\textbf{Algorithm 1:} MoE-DeltaKV-Paged forward
\begin{lstlisting}
Input:  X $\in$ R^{B$$\times$N$$\times$d}, attention_mask, position_embeddings
Output: Y $\in$ R^{B$$\times$N$$\times$d}

 1: Q $\leftarrow$ query(X); K $\leftarrow$ key(X); V $\leftarrow$ value(X)
 2: if RoPE: (Q, K) $\leftarrow$ apply_rotary(Q, K)
 3: (K_hat, V_hat) $\leftarrow$ simulate_paged_delta_kv(K, V, rank)    Delta-KV
 4: Partition into N_p pages of size P
 5: Compute q_bar, k_bar (page means)
 6:   $\leftarrow$ (W_rg q_bar)   k_bar^T / $\sqrt{}$d_h                      Router
 7: if causal:   $\leftarrow$ mask_triu( )
 8:  [:, :, :, 0] += 1e4                                     CLS boost
 9: topk_idx $\leftarrow$ TopK( , k_experts)
10: g $\leftarrow$ softmax(gather( , topk_idx))
11: Gather K, V from selected pages
12: scores $\leftarrow$ q @ k_gathered^T / $\sqrt{}$d_h
13: attn $\leftarrow$ softmax(scores)
14: context $\leftarrow$ $\Sigma$ g_j   (attn @ v_j)                          Gated fusion
15: Y $\leftarrow$ out_proj(context)
16: return Y
\end{lstlisting}

\subsection{Complexity Analysis}

\subsubsection{Per-Query-Page Complexity}

For query page $i$:
\begin{itemize}
\item Router: $\mathcal{O}(N_p \cdot d_h)$
\item Attention over selected pages: $\mathcal{O}(k_\text{exp} \cdot P^2 \cdot d_h)$

\end{itemize}
Total: $\mathcal{O}(N_p \cdot k_\text{exp} \cdot P^2 \cdot d_h)$.

\textbf{Compare to standard attention:} $\mathcal{O}(N^2 d_h) = \mathcal{O}(N_p^2 P^2 d_h)$.

\textbf{Speedup:} $\frac{N_p^2 P^2}{N_p \cdot k_\text{exp} \cdot P^2} = \frac{N_p}{k_\text{exp}}$.

For $N_p \gg k_\text{exp}$: substantial speedup.

\subsubsection{Memory}

Router logits: $\mathcal{O}(N_p^2)$.

Gathered K/V: $\mathcal{O}(N_p \cdot k_\text{exp} \cdot P \cdot d_h)$.

\subsubsection{Delta-KV Cost}

The Delta-KV step adds:
\begin{itemize}
\item Delta computation: $\mathcal{O}(N d_h)$
\item SVD: $\mathcal{O}(\min((BH(N-1)) d_h^2, (BH(N-1))^2 d_h))$
\item Quantization: $\mathcal{O}(N d_h)$

\end{itemize}
SVD dominates for large $N$.

\subsection{Theoretical Notes}

\subsubsection{Delta Sparsity}

\textbf{Claim.} Natural language exhibits temporal smoothness: 
$\|\Delta k_t\| \ll \|k_t\|$.

\textbf{Evidence.} See Section 8.5 (empirical).

\subsubsection{STE Correctness}

\textbf{Claim.} STE preserves gradient flow through quantization.

\textbf{Proof.} $\frac{\partial \text{STE}(x)}{\partial x} = 1$ by construction 
(the detach stops gradient on the correction term). $\square$

\subsubsection{CLS Preservation}

\textbf{Claim.} With the $+10^4$ boost, page 0 is always in the top-$k$.

\textbf{Proof.} For any $k \geq 1$ and any logits $\ell_{ij}$ with $|\ell| < 10^3$, 
the boosted $\ell_{i, 0} > 10^3$ exceeds all others. $\square$

\subsubsection{Causality}

\textbf{Claim.} For decoders, causality is preserved.

\textbf{Proof.} Router logits mask $j > i$ to $-\infty$; page-level causal mask 
enforces block-causality. Within a page, no mask is needed because a page 
contains consecutive tokens. $\square$
