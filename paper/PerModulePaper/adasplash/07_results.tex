
\section{Results}

\subsection{Main Results}

\textbf{Table 7.1: Main results on SST-2 validation.}

\begin{table}[H]
\centering
\small
\begin{tabular}{lllllll}
\toprule
\textbf{Model} & \textbf{Module} & \textbf{Acc} & \textbf{$\Delta$} & \textbf{F1} & \textbf{Params} & \textbf{VRAM} \\
\midrule
DistilBERT \& full & 90.71\% & -- & 0.9099 & 66.9M & 183 MB \\
DistilBERT & \textbf{AdaSplash} & \textbf{90.25} & -0.46 & 0.9055 & 67.55M & 195 MB \\
RoBERTa \& full & 78.56\% & -- & 0.8124 & 124.6M & 712 MB \\
RoBERTa & \textbf{AdaSplash} & \textbf{77.5} & -1.06 & 0.780 & 125.8M & 730 MB \\
GPT-2 \& full & 79.36\% & -- & 0.7978 & 124.4M & 793 MB \\
GPT-2 & \textbf{AdaSplash} & \textbf{78.78} & -0.58 & 0.7891 & 125.63M & 810 MB \\
\bottomrule
\end{tabular}
\end{table}

\textbf{Key observations:}

\begin{enumerate}
\item \textbf{Accuracy:} $\Delta$ of -0.46\%, -1.06\%, -0.58\% (all near zero).
\item \textbf{Parameters:} increased by 0.88\% due to LoRA.
\item \textbf{VRAM:} slightly increased (LoRA overhead).

\end{enumerate}

\subsection{Summary}

\begin{itemize}
\item \textbf{Accuracy:} maintained at baseline.
\item \textbf{Parameter count:} increased by ~1-2\% due to LoRA.
\item \textbf{Attention:} 40\% sparse.
\item \textbf{Not a compression method.}
\end{itemize}
