\section{Introduction}

\subsection{Motivation}

Linear attention (Katharopoulos et al., 2020) computes attention in 
$\mathcal{O}(L)$ time by replacing softmax with a positive kernel feature 
map $\phi$:

\[
O_t = \frac{\phi(q_t)^\top \sum_{j \leq t} \phi(k_j) v_j^\top}{\phi(q_t)^\top \sum_{j \leq t} \phi(k_j)}
\tag{1.1}
\]

The state $S_t = \sum_{j \leq t} \phi(k_j) v_j^\top \in \mathbb{R}^{F \times d_h}$ 
is updated incrementally, giving constant memory per step.

\textbf{Two practical problems remain:}

\begin{enumerate}
\item \textbf{Uniform contribution of past keys.} In (1.1), the coefficient of
  $v_j$ in $O_t$ is $\phi(q_t)^\top \phi(k_j) / Z_t$ -- it does \textbf{not}
  decay with $t - j$. Long-range dependencies and recent context are
  treated identically. In practice, transformer attention has been shown
  to concentrate on nearby tokens (attention sinks aside).
\item \textbf{Causal implementations often use cumsum.} Naive causal linear attention
  materializes $S_t$ for every $t$, giving an $\mathcal{O}(L \cdot F \cdot d_h)$
  intermediate. This defeats the memory advantage.

\end{enumerate}
\textbf{DecayLinearAttention} addresses both:

\begin{enumerate}
\item \textbf{Data-dependent decay.} We learn a per-head, per-token decay
  $\lambda_{h,t} \in (0, 1)$ and update the state with exponential decay:
  
  $$
  S\_t = \lambda\_{h,t} S\_{t-1} + \phi(k\_t) v\_t\^{}\top
  \tag{1.2}
  $$
  
\item \textbf{Chunked scan with $\mathcal{O}(1)$ state.} We carry only
  $S \in \mathbb{R}^{d_h \times d_h}$ between chunks, processing each
  chunk in parallel.

\end{enumerate}
\subsection{Problem Statement}

Given a pretrained transformer with softmax attention $\mathcal{A}$, we seek 
a linear attention $\mathcal{A}'$ such that:

\begin{enumerate}
\item \textbf{Linear complexity:} $\mathcal{A}'$ runs in $\mathcal{O}(L)$ time.
\item \textbf{O(1) memory during causal decoding.} No $\mathcal{O}(L)$ intermediate.
\item \textbf{Data-dependent decay.} The decay rate depends on the input.
\item \textbf{Bidirectional \& causal support.}
\item \textbf{Cross-attention support.}
\item \textbf{Pretrained compatibility} via LoRA recovery.

\end{enumerate}
\subsection{Our Approach}

Three components:

\subsubsection{Data-Dependent Exponential Decay}

\[
\lambda_{h,t} = \sigma\!\left(W_g x_t + b_g + \ell_h\right), \quad
\ell_h \in [{{LOG_DECAY_MIN}}, {{LOG_DECAY_MAX}}]
\tag{1.3}
\]

The per-head bias $\ell_h$ is initialized to linearly spaced values, so 
different heads start with different context horizons.

\subsubsection{Chunked Scan}

For a chunk of length $C$, we compute:

\begin{itemize}
\item \textbf{Intra-chunk:} Dense causal attention within the chunk.
\item \textbf{Inter-chunk:} State-passing from previous chunks via $S$.

\end{itemize}
Total memory: $\mathcal{O}(C^2 + d_h^2)$ per chunk, \textbf{independent of $L$}.

\subsubsection{ELU+1 Feature Map}

\[
\phi(x) = \text{elu}(x) + 1
\tag{1.4}
\]

This is positive (needed for stable linear attention) and cheap to compute.

\subsection{Contributions}

\begin{itemize}
\item \textbf{Data-dependent decay.} A per-head, per-token decay rate learned from
  the input.
\item \textbf{Chunked scan with O(1) state.} Memory independent of sequence length.
\item \textbf{Bidirectional and causal support.} Symmetric average of forward and
  backward scans for encoders.
\item \textbf{Cross-attention support.} All four projections factorized.
\item \textbf{Multi-architecture.} DistilBERT, RoBERTa, GPT-2, Qwen3.5.
\item \textbf{Honest memory reporting.} We document the exact memory footprint.
\item \textbf{Full reproducibility.} Code, weights, and scripts released.

\end{itemize}
