\subsection{Evaluation Metrics}

\subsubsection{Quality}
\begin{itemize}
\item Accuracy, F1, precision, recall, loss.

\end{itemize}
\subsubsection{Efficiency}
\begin{itemize}
\item \textbf{State memory} (constant in L).
\item \textbf{Peak VRAM}.
\item Inference time at multiple L.

\end{itemize}
\subsubsection{Decay Analysis}
\begin{itemize}
\item Learned decay rates per head.
\item Effective horizon $1 / (1 - \lambda)$ per head.

\end{itemize}
