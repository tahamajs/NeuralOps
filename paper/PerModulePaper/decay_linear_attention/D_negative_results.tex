\section{Appendix D. Negative Results}

\subsection{D.1 Attempts That Failed}

\subsubsection{D.1.1 No Decay ($\lambda$ = 1)}

\textbf{Goal.} Standard linear attention without decay.

\textbf{Result.} Accuracy dropped by 4.5\%.

\textbf{Conclusion.} Decay is essential.

\subsubsection{D.1.2 Very Small $\lambda$ ([-5, -3])}

\textbf{Goal.} Very fast decay (only recent tokens matter).

\textbf{Result.} Loss convergence failed; accuracy collapsed.

\textbf{Conclusion.} $\lambda$ must stay near 1 for stable training.

\subsubsection{D.1.3 No Chunking (single-pass cumsum)}

\textbf{Goal.} Use cumsum over the full sequence.

\textbf{Result.} OOM at L > 512.

\textbf{Conclusion.} Chunking is essential for O(1) memory.

\subsubsection{D.1.4 Learned Feature Map}

\textbf{Goal.} Use a small MLP as feature map.

\textbf{Result.} No improvement over ELU+1, slower.

\textbf{Conclusion.} ELU+1 is sufficient.

\subsubsection{D.1.5 Per-Token Chunk Size}

\textbf{Goal.} Adaptive chunk size per token.

\textbf{Result.} Complex implementation, no benefit.

\textbf{Conclusion.} Fixed chunk size works.

\subsection{D.2 Abandoned Variants}

\subsubsection{D.2.1 Non-Exponential Decay}

Tested polynomial decay $1/t^k$. Result: worse accuracy.

\subsubsection{D.2.2 Learned $\lambda$ Temperature}

Tested learnable $\sigma$ temperature for $\lambda$. Result: no improvement.

\subsubsection{D.2.3 Different Bidirectional Weights}

Tested learnable weight for $O^\text{fwd}$ vs $O^\text{bwd}$. Result: 
converged to 0.5 anyway.

\subsection{D.5 Lessons Learned}

\begin{enumerate}
\item \textbf{Decay is essential.}
\item \textbf{$\lambda$ must be initialized near 1.}
\item \textbf{Chunking enables O(1) memory.}
\item \textbf{ELU+1 is sufficient.}
\item \textbf{Bidirectional average at 0.5 is fine.}

\end{enumerate}
