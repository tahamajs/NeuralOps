
\section{Appendix D. Negative Results}

\subsection{D.1 Attempts That Failed}

\subsubsection{D.1.1 FP16 SVD}

\textbf{Goal.} Run SVD in FP16 to save memory.

\textbf{Result.} Occasional NaN for ill-conditioned matrices.

\textbf{Conclusion.} Compute SVD in FP32, cast back to FP16.

\subsubsection{D.1.2 Random Calibration}

\textbf{Goal.} Use random inputs instead of real text.

\textbf{Result.} Fisher weighting accuracy degraded to \textbf{86.4\%\%}.

\textbf{Conclusion.} Real calibration data is required.

\subsubsection{D.1.3 Full Fisher}

\textbf{Goal.} Compute the true Fisher diagonal via backprop.

\textbf{Result.} 
\begin{itemize}
\item Memory: O(batch $\times$ d) instead of O(d)
\item Time: 3$\times$ slower
\item Accuracy improvement: <0.5\%

\end{itemize}
\textbf{Conclusion.} Cheap $\mathbb{E}[x^2]$ proxy is sufficient.

\subsubsection{D.1.4 Per-Layer Rank Search}

\textbf{Goal.} Grid search optimal rank per layer.

\textbf{Result.}
\begin{itemize}
\item Time: 20$\times$ slower than adaptive escalation
\item Accuracy improvement: <1\%

\end{itemize}
\textbf{Conclusion.} Adaptive escalation is a good approximation.

\subsubsection{D.1.5 No Fallback}

\textbf{Goal.} Force compression even if error exceeds threshold.

\textbf{Result.} Accuracy collapsed to \textbf{86.4\%\%}.

\textbf{Conclusion.} Fallback is essential.

\subsection{D.2 Abandoned Variants}

\subsubsection{D.2.1 Per-Channel LoRA}

We tested assigning different LoRA ranks to different channels based on 
Fisher. Result: no improvement over uniform LoRA.

\subsubsection{D.2.2 Iterative Refinement}

We tried iteratively reweighting after the first SVD pass. Result: no 
significant improvement, and 3$\times$ slower.

\subsubsection{D.2.3 Non-Diagonal Fisher}

We tried using the full Fisher matrix instead of just diagonal. Result: 
5$\times$ memory, 2$\times$ slower, <0.3\% improvement.

because A3 compresses weights, not activations.

\subsection{D.7 Lessons Learned}

\begin{enumerate}
\item \textbf{FP32 SVD is necessary} for stability.
\item \textbf{Real calibration data matters.}
\item \textbf{Cheap Fisher proxy is sufficient.}
\item \textbf{Adaptive rank is better than grid search.}
\item \textbf{Fallback is essential} for safety.

\end{enumerate}
\subsection{D.8 Recommendations}

\begin{enumerate}
\item \textbf{Use default hyperparameters} unless you have evidence otherwise.
\item \textbf{Start with rank 64} and adjust based on error.
\item \textbf{Always include calibration data} -- even 100 examples help.
\item \textbf{Enable fallback} in production.
\end{enumerate}
