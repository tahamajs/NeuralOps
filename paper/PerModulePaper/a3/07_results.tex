\textbf{Key observations:}

\begin{enumerate}
\item Parameter reduction: \textbf{5.0$\times$} (DistilBERT), \textbf{4.5$\times$} (RoBERTa), \textbf{4.5$\times$} (GPT-2).
\item Accuracy drop: \textbf{-2.2\%}, \textbf{-2.8\%}, \textbf{-2.9\%}.
\item Peak VRAM reduced by \textbf{40\%} on all models.

\end{enumerate}

\begin{itemize}

\item Accuracy drop is \textbf{-2.2\%---2.9\%}.

\item Fisher weighting contributes \textbf{35\%}.
\item LayerNorm recalibration contributes \textbf{25\%}.
\end{itemize}
