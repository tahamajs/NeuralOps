
\section{Appendix D. Negative Results}

\subsection{D.1 The `copy\_weights` Raise Bug}

\begin{lstlisting}
except Exception as e:
    raise RuntimeError(f"copy_weights failed: {e}")
\end{lstlisting}

\textbf{Impact.} Any UniversalAdapter failure aborts injection.

\textbf{Fallback.} \texttt{inject\_orion} has its own try/except $\to$ returns None.

\subsection{D.2 The `rff\_proj` Persistence Bug}

\begin{lstlisting}
self.register_buffer("rff_proj", ..., persistent=False)
\end{lstlisting}

\textbf{Impact.} Train/test mismatch on model save/reload.

\textbf{Verification.}

\begin{lstlisting}
sd = model1.state_dict()
assert 'rff_proj' not in sd  # Confirms bug
\end{lstlisting}

\subsection{D.3 Softpick Masked Row Collapse}

\begin{lstlisting}
p = F.relu(torch.exp(x_f) - 1.0)
return p / (p.sum(dim=dim, keepdim=True) + 1e-8)
\end{lstlisting}

If all keys in a row are masked, \texttt{p.sum() = 0} $\to$ output is zero.

\textbf{Impact.} For fully-masked query rows, attention is undefined.

\subsection{D.6 Lessons Learned}

\begin{enumerate}
\item \textbf{\texttt{copy\_weights} robustness matters.} Raise-on-failure is dangerous.
\item \textbf{Persistent buffers matter.} \texttt{persistent=False} breaks save/reload.
\item \textbf{Softpick needs fallback.} Masked rows otherwise collapse.
\item \textbf{Head bifurcation works.} 2/3 : 1/3 split is a good default.

\end{enumerate}
\subsection{D.7 Recommendations}

\begin{enumerate}
\item \textbf{Fix the \texttt{copy\_weights} raise.}
\item \textbf{Set \texttt{persistent=True}} for \texttt{rff\_proj}.
\item \textbf{Add softpick fallback.}
\item \textbf{Test per-head rank selection.}

\end{enumerate}
\subsection{D.8 What Would Make This Work?}

\begin{enumerate}

\item \textbf{Fused kernel} -- faster factorized projections.
\item \textbf{Learned head split} -- adaptive bifurcation.
\end{enumerate}
