class AATC_XKVCache(_HFDynamicCache if _HAS_HF_CACHE else object):
    """
    Drop-in replacement for HF DynamicCache that transparently compresses K/V
    via AATC-XKV.

    Storage per (layer, batch, head, token):
      - q_k: int8 coefficients (r_k,) + fp32 scale vector (r_k,)
      - q_v: int8 coefficients (r_v,) + fp32 scale vector (r_v,)
      - sink_k/sink_v (fp32) for the first `sink_tokens` tokens of the sequence
      - last_k/last_v (fp32) only during PREFILL (T_new > 1) when
        `preserve_last_token=True`

    Reconstruction:
      - `_reconstruction_cache[layer_idx]` holds the fully-reconstructed
        fp16 K/V up to the last decompressed entry, so decoding only pays
        O(new_tokens · r · D) per step.
    """

    def __init__(
        self,
        subspaces: dict[int, SharedSubspace],
        layer_to_group: dict[int, int],
        sink_tokens: int = 1,
        preserve_last_token: bool = True,
    ):
        super().__init__()
        self.subspaces = subspaces
        self.layer_to_group = layer_to_group
        self.sink_tokens = max(0, int(sink_tokens))
        self.preserve_last_token = bool(preserve_last_token)
        self.enabled = True

        self._compressed: dict[int, list[dict[str, torch.Tensor]]] = {}
        # Reconstruction cache: layer_idx -> (full_k, full_v, n_entries_seen)
        self._reconstruction_cache: dict[int, tuple[torch.Tensor, torch.Tensor, int]] = {}

        self._total_original_bytes = 0
        self._total_compressed_bytes = 0

    # ------------------------------------------------------------------
    # HF API
    # ------------------------------------------------------------------
    def __len__(self) -> int:
        # AXKV-FIX-04: sum over all entries
        if not self._compressed:
            return 0
        total = 0
        for layer_idx, entries in self._compressed.items():
            n = 0
            for e in entries:
                if e.get("sink_k") is not None:
                    n += e["sink_k"].shape[2]
                n += e["q_k"].shape[2]
                if e.get("last_k") is not None:
                    n += e["last_k"].shape[2]
            total = max(total, n)
        return total

    def __getitem__(self, layer_idx: int):
        if layer_idx not in self._compressed:
            raise KeyError(f"AATC_XKVCache: no cache for layer {layer_idx}")
        return self._reconstruct_full(layer_idx)

    def __iter__(self):
        for layer_idx in sorted(self._compressed.keys()):
            yield self._reconstruct_full(layer_idx)

    def __contains__(self, layer_idx: int) -> bool:
        return layer_idx in self._compressed

    def update(
        self,
        key_states: torch.Tensor,
        value_states: torch.Tensor,
        layer_idx: int,
        cache_kwargs: dict[str, Any] | None = None,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """Append new K/V, return the full K/V needed for attention."""
        if not self.enabled or layer_idx not in self.layer_to_group:
            return super().update(key_states, value_states, layer_idx, cache_kwargs)

        sub = self.subspaces.get(self.layer_to_group[layer_idx])
        if sub is None:
            return super().update(key_states, value_states, layer_idx, cache_kwargs)

        entry = self._new_entry(key_states, value_states, sub, layer_idx)
        self._compressed.setdefault(layer_idx, []).append(entry)

        # AXKV-FIX-03: use reconstruction cache
        cached = self._reconstruction_cache.get(layer_idx)
        n_entries = len(self._compressed[layer_idx])

        if cached is not None and cached[2] == n_entries - 1:
            full_k, full_v, _ = cached
            new_k, new_v = self._reconstruct_entry(entry, sub, key_states.dtype)
            full_k = torch.cat([full_k, new_k], dim=2)
            full_v = torch.cat([full_v, new_v], dim=2)
        else:
            full_k, full_v = self._reconstruct_full(layer_idx)

        self._reconstruction_cache[layer_idx] = (full_k, full_v, n_entries)
        return full_k, full_v

    def get_seq_length(self, layer_idx: int = 0) -> int:
        entries = self._compressed.get(layer_idx, [])
        n = 0
        for e in entries:
            if e["sink_k"] is not None:
                n += e["sink_k"].shape[2]
            n += e["q_k"].shape[2]
            if e["last_k"] is not None:
                n += e["last_k"].shape[2]
        return n

    def get_max_length(self) -> int | None:
        return None

    def get_usable_length(self, new_seq_length: int, layer_idx: int = 0) -> int:
        return self.get_seq_length(layer_idx)

    def get_max_cache_shape(self) -> int:
        return -1

    def reset(self) -> None:
        self._compressed.clear()
        self._reconstruction_cache.clear()
        self._total_original_bytes = 0
        self._total_compressed_bytes = 0

    def reorder_cache(self, beam_idx: torch.LongTensor) -> None:
        """
        AXKV-FIX-05: correct beam-search reorder. Rebuilds the reconstruction
        cache lazily after reordering the compressed entries.
        """
        with torch.no_grad():
            for layer_idx, entries in self._compressed.items():
                for e in entries:
                    for k in ("sink_k", "sink_v", "last_k", "last_v"):
                        if e[k] is not None:
                            e[k] = e[k].index_select(0, beam_idx.to(e[k].device))
                    for k in ("q_k", "q_v"):
                        e[k] = e[k].index_select(0, beam_idx.to(e[k].device))
            # Invalidate reconstruction cache
            self._reconstruction_cache.clear()

    # ------------------------------------------------------------------
    # Entry creation
    # ------------------------------------------------------------------
    def _new_entry(
        self,
        k: torch.Tensor,
        v: torch.Tensor,
        sub: SharedSubspace,
        layer_idx: int,
    ) -> dict[str, torch.Tensor]:
        B, H_kv, T_new, D = k.shape

        # AXKV-FIX-12: sinks are prepended only once — the first time we
        # ever see this layer's cache.
        is_first_chunk = len(self._compressed.get(layer_idx, [])) == 0
        sink_prefix = min(self.sink_tokens, T_new) if is_first_chunk else 0

        if sink_prefix > 0:
            sink_k = k[:, :, :sink_prefix, :].to(torch.float32).contiguous()
            sink_v = v[:, :, :sink_prefix, :].to(torch.float32).contiguous()
            rest_k = k[:, :, sink_prefix:, :]
            rest_v = v[:, :, sink_prefix:, :]
        else:
            sink_k = None
            sink_v = None
            rest_k = k
            rest_v = v

        # AXKV-FIX-02: preserve_last_token only during prefill (T_new > 1).
        last_k = None
        last_v = None
        if self.preserve_last_token and T_new > 1 and rest_k.size(2) > 0:
            last_k = rest_k[:, :, -1:, :].to(torch.float32).contiguous()
            last_v = rest_v[:, :, -1:, :].to(torch.float32).contiguous()
            rest_k = rest_k[:, :, :-1, :]
            rest_v = rest_v[:, :, :-1, :]

        V_k = sub.V_k.to(k.device, torch.float32)
        V_v = sub.V_v.to(v.device, torch.float32)
        mean_k = sub.mean_k.to(k.device, torch.float32)
        mean_v = sub.mean_v.to(v.device, torch.float32)

        if rest_k.size(2) == 0:
            r_k = V_k.size(0)
            r_v = V_v.size(0)
            q_k = torch.zeros((B, H_kv, 0, r_k), dtype=torch.int8, device=k.device)
            q_v = torch.zeros((B, H_kv, 0, r_v), dtype=torch.int8, device=v.device)
            s_k = torch.ones(r_k, dtype=torch.float32, device=k.device)
            s_v = torch.ones(r_v, dtype=torch.float32, device=v.device)
        else:
            rest_k_f = rest_k.to(torch.float32) - mean_k.view(1, 1, 1, -1)
            rest_v_f = rest_v.to(torch.float32) - mean_v.view(1, 1, 1, -1)
            U_k = torch.matmul(rest_k_f, V_k.t())
            U_v = torch.matmul(rest_v_f, V_v.t())
            q_k, s_k = quantize_coefficients(U_k, sub.bit_alloc_k)
            q_v, s_v = quantize_coefficients(U_v, sub.bit_alloc_v)

        entry = {
            "q_k": q_k,
            "s_k": s_k,
            "q_v": q_v,
            "s_v": s_v,
            "sink_k": sink_k,
            "sink_v": sink_v,
            "last_k": last_k,
            "last_v": last_v,
        }

        # AXKV-FIX-09: byte accounting
        with torch.no_grad():
            orig_b = (k.numel() + v.numel()) * k.element_size()
            comp_b = (
                q_k.numel() * q_k.element_size()
                + q_v.numel() * q_v.element_size()
                + s_k.numel() * s_k.element_size()
                + s_v.numel() * s_v.element_size()
                + (sink_k.numel() * sink_k.element_size() if sink_k is not None else 0)
                + (sink_v.numel() * sink_v.element_size() if sink_v is not None else 0)
                + (last_k.numel() * last_k.element_size() if last_k is not None else 0)
                + (last_v.numel() * last_v.element_size() if last_v is not None else 0)
            )
            self._total_original_bytes += orig_b
            self._total_compressed_bytes += comp_b

        return entry

    def _reconstruct_entry(
        self,
        e: dict[str, torch.Tensor],
        sub: SharedSubspace,
        out_dtype: torch.dtype,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """AXKV-FIX-11: reconstruct one entry in fp32, cast at return."""
        V_k = sub.V_k.to(torch.float32)
        V_v = sub.V_v.to(torch.float32)
        mean_k = sub.mean_k.to(torch.float32)
        mean_v = sub.mean_v.to(torch.float32)

        parts_k, parts_v = [], []

        if e["sink_k"] is not None:
            parts_k.append(e["sink_k"].to(out_dtype))
            parts_v.append(e["sink_v"].to(out_dtype))

        if e["q_k"].size(2) > 0:
            U_k = dequantize_coefficients(e["q_k"], e["s_k"])
            U_v = dequantize_coefficients(e["q_v"], e["s_v"])
            device = U_k.device
            rec_k = torch.matmul(U_k, V_k.to(device)) + mean_k.to(device).view(1, 1, 1, -1)
            rec_v = torch.matmul(U_v, V_v.to(device)) + mean_v.to(device).view(1, 1, 1, -1)
            parts_k.append(rec_k.to(out_dtype))
            parts_v.append(rec_v.to(out_dtype))

        if e["last_k"] is not None:
            parts_k.append(e["last_k"].to(out_dtype))
            parts_v.append(e["last_v"].to(out_dtype))

        if not parts_k:
            return torch.empty(0), torch.empty(0)

        k_entry = torch.cat(parts_k, dim=2) if len(parts_k) > 1 else parts_k[0]
        v_entry = torch.cat(parts_v, dim=2) if len(parts_v) > 1 else parts_v[0]
        return k_entry, v_entry

    def _reconstruct_full(
        self,
        layer_idx: int,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        entries = self._compressed.get(layer_idx, [])
        if not entries:
            raise RuntimeError("AATC_XKVCache: no entries for layer.")

        sub = self.subspaces[self.layer_to_group[layer_idx]]

        # Infer dtype from the first entry's sink or quantized buffer
        ref = entries[0].get("sink_k")
        if ref is None:
            ref = entries[0].get("last_k")
        if ref is None:
            # Fall back to the dtype of the shared subspace
            out_dtype = sub.V_k.dtype
        else:
            out_dtype = ref.dtype

        k_list, v_list = [], []
        for e in entries:
            k_i, v_i = self._reconstruct_entry(e, sub, out_dtype)
            if k_i.numel() > 0:
                k_list.append(k_i)
                v_list.append(v_i)

        if not k_list:
            raise RuntimeError("AATC_XKVCache: empty reconstruction.")

        full_k = torch.cat(k_list, dim=2) if len(k_list) > 1 else k_list[0]
        full_v = torch.cat(v_list, dim=2) if len(v_list) > 1 else v_list[0]
        return full_k, full_v

    # Public metrics
    # ------------------------------------------------------------------
    def memory_savings_ratio(self) -> float:
        if self._total_compressed_bytes == 0:
            return 1.0
        return self._total_original_bytes / self._total_compressed_bytes

    def memory_bytes(self) -> dict[str, int]:
        return {
            "original": self._total_original_bytes,
            "compressed": self._total_compressed_bytes,
        }

    def _find_attention_modules(self) -> list[tuple[int, str, nn.Module]]:
        """
        AXKV-FIX-08: identify attention submodules with K/V projections.
        Avoids `value` attribute collisions by requiring explicit projection
        names or a fused GPT-2 `c_attn`.
        """
        attn_modules = []
        for name, mod in self.model.named_modules():
            name_lower = name.lower()
            if not any(k in name_lower for k in ("self_attn", "attention", "attn")):
                continue

            # Fused GPT-2 c_attn
            if hasattr(mod, "c_attn"):
                is_attn = True
            else:
                k_names = ("k_proj", "k_lin", "key", "W_k")
                v_names = ("v_proj", "v_lin", "value", "W_v")
                has_k = any(hasattr(mod, a) for a in k_names)
                has_v = any(hasattr(mod, a) for a in v_names)
                # Reject if the attribute exists but is a method (not a module)
                if has_k:
                    has_k = any(
                        isinstance(getattr(mod, a, None), nn.Module)
                        for a in k_names
                        if hasattr(mod, a)
                    )
                if has_v:
                    has_v = any(
                        isinstance(getattr(mod, a, None), nn.Module)
                        for a in v_names
                        if hasattr(mod, a)
                    )
                is_attn = has_k and has_v

            if not is_attn:
                continue

            idx = len(attn_modules)
            for part in name.split("."):
                if part.isdigit():
                    idx = int(part)
                    break
            attn_modules.append((idx, name, mod))

        attn_modules.sort(key=lambda x: x[0])
        return attn_modules

    def _build_subspaces(
        self,
        calib: dict[str, dict[int, torch.Tensor]],
        attn_modules: list[tuple[int, str, nn.Module]],
    ) -> None:
        layer_indices = [idx for idx, _, _ in attn_modules]
        G = max(1, self.group_size)
        groups = [layer_indices[i : i + G] for i in range(0, len(layer_indices), G)]

        for g_id, g_layers in enumerate(groups):
            for l_idx in g_layers:
                self.layer_to_group[l_idx] = g_id

            k_list = [calib["k"][l] for l in g_layers if calib["k"][l].numel() > 0]
            v_list = [calib["v"][l] for l in g_layers if calib["v"][l].numel() > 0]
            gk_list = [calib["gk"][l] for l in g_layers if calib["gk"][l].numel() > 0]
            gv_list = [calib["gv"][l] for l in g_layers if calib["gv"][l].numel() > 0]

            if not k_list or not v_list:
                continue

            K_stack = torch.cat(k_list, 0)
            V_stack = torch.cat(v_list, 0)
            gK_stack = torch.cat(gk_list, 0) if gk_list else torch.zeros_like(K_stack)
            gV_stack = torch.cat(gv_list, 0) if gv_list else torch.zeros_like(V_stack)

            D = K_stack.size(-1)
            rank_cap = max(4, D // 4)

            V_k, S_k, mean_k = _joint_svd_shared_basis(K_stack, rank_cap, self.energy_threshold)
            V_v, S_v, mean_v = _joint_svd_shared_basis(V_stack, rank_cap, self.energy_threshold)

            r_k = V_k.size(0)
            r_v = V_v.size(0)

            proj_gk = gK_stack @ V_k.t()
            proj_k = K_stack @ V_k.t()
            sens_k = proj_gk.pow(2).mean(0) + proj_k.pow(2).mean(0)

            proj_gv = gV_stack @ V_v.t()
            proj_v = V_stack @ V_v.t()
            sens_v = proj_gv.pow(2).mean(0) + proj_v.pow(2).mean(0)

            total_bits_k = int(round(r_k * self.target_bits))
            total_bits_v = int(round(r_v * self.target_bits))
            alloc_k = reverse_waterfilling(S_k.pow(2), sens_k, total_bits_k, 0, 8)
            alloc_v = reverse_waterfilling(S_v.pow(2), sens_v, total_bits_v, 0, 8)

            self.subspaces[g_id] = SharedSubspace(
                V_k=V_k,
                V_v=V_v,
                mean_k=mean_k,
                mean_v=mean_v,
                bit_alloc_k=alloc_k,
                bit_alloc_v=alloc_v,
                group_id=g_id,
                layer_indices=g_layers,
            )

    # ------------------------------------------------------------------

        attn_modules = self._find_attention_modules()

        if not attn_modules:

            self.stats = {"status": "NO_ATTN_MODULES", "num_compressed_layers": 0}

        try:

            from packaging import version

            if version.parse(transformers.__version__) >= version.parse("4.36.0"):
                self.model._supports_cache_class = True
        except Exception:

        # AXKV-FIX-09: single, honest bit-accounting pass
        total_orig_bits = 0
        total_comp_bits = 0
        for g_id, sub in self.subspaces.items():
            D = sub.V_k.size(1)

            total_orig_bits += 16 * (2 * D) * n_layers
            # Compressed: bit_alloc per K component + per V component,

            "total_params_before": self._original_params,
            "total_params_after": self._original_params,
            "weights_modified": False,
        }

            len(attn_modules),

class CPTGenerator(nn.Module):
    """
    Learnable Task-Aware Compression Planning Token (CPT) Generator.
    Maps arbitrary task descriptions to continuous per-feature importance vectors for weight SVD
    and KV cache compression across arbitrary dimension sizes d.
    """

    def __init__(self, hidden_dim: int = 128, num_freqs: int = 8, vocab_size: int = 5000):
        super().__init__()
        self.hidden_dim = hidden_dim
        self.num_freqs = num_freqs
        self.vocab_size = vocab_size

        self.text_embedding = nn.EmbeddingBag(vocab_size, hidden_dim, mode="mean")
        self.encoder = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.LayerNorm(hidden_dim),
            nn.GELU(),
            nn.Linear(hidden_dim, hidden_dim),
        )

        in_dim = hidden_dim + num_freqs * 2
        self.projector = nn.Sequential(
            nn.Linear(in_dim, hidden_dim),
            nn.GELU(),
            nn.Linear(hidden_dim, 1),
            nn.Sigmoid(),
        )

    def _tokenize_text(self, text: str, device: torch.device) -> torch.Tensor:
        """Deterministic hash-based tokenization without external library dependencies."""
        words = (
            str(text)
            .lower()
            .replace(",", " ")
            .replace(".", " ")
            .replace(":", " ")
            .replace("-", " ")
            .split()
        )
        if not words:
            return torch.tensor([0], dtype=torch.long, device=device)
        indices = []
        for w in words:
            h = int(hashlib.md5(w.encode("utf-8")).hexdigest(), 16)
            indices.append(h % (self.vocab_size - 1) + 1)
        return torch.tensor(indices, dtype=torch.long, device=device)

    def forward(self, task: str, d: int, device: torch.device | None = None) -> torch.Tensor:
        """
        Generate a per-feature importance modulation vector of size d in [0.1, 2.0].
        """
        target_device = device if device is not None else next(self.parameters()).device
        tokens = self._tokenize_text(task, target_device)
        h = self.text_embedding(tokens.unsqueeze(0)).squeeze(0)  # [hidden_dim]
        h = self.encoder(h)  # [hidden_dim]

        # Continuous NeRF-style positional coordinate encoding
        coords = torch.linspace(0.0, 1.0, d, device=target_device)
        freqs = (2.0 ** torch.arange(self.num_freqs, device=target_device)) * math.pi
        args = coords.unsqueeze(1) * freqs.unsqueeze(0)  # [d, num_freqs]
        enc = torch.cat([torch.sin(args), torch.cos(args)], dim=-1)  # [d, 2 * num_freqs]

        h_exp = h.unsqueeze(0).expand(d, -1)  # [d, hidden_dim]
        inp = torch.cat([h_exp, enc], dim=-1)  # [d, in_dim]

        # Importance modulation strictly in [0.1, 2.0]
        importance = self.projector(inp).squeeze(-1) * 1.9 + 0.1
        return importance

                "question answering: extract concise answer from supporting context passage",
                "code generation: implement efficient python function and classes",
                "summarization: produce concise abstractive summary of input document",
                "mathematical reasoning: solve multi-step theorem and calculus equations",
                "sentiment classification: predict positive or negative customer review sentiment",
            ]

        if sample_layers is None:
            sample_layers = [
                (name, mod)
                for name, mod in self.model.named_modules()
                if self._is_target_module(name, mod)
            ][:4]  # train on up to 4 sample layers for efficiency

        if not sample_layers:
            logger.warning("No sample layers found to train CPT generator.")
            return 0.0

        self.cpt_generator.train()
        for p in self.cpt_generator.parameters():
            p.requires_grad = True

        optimizer = torch.optim.AdamW(self.cpt_generator.parameters(), lr=lr)

        total_loss = 0.0
        steps = 0

        for epoch in range(epochs):
            for task in task_descriptions:
                for name, module in sample_layers:
                    target_mod = getattr(module, "base_layer", getattr(module, "linear", module))
                    cls_name = module.__class__.__name__.lower()
                    is_conv1d = (
                        "conv1d" in cls_name or hasattr(module, "nf") or hasattr(target_mod, "nf")
                    )
                    raw_w = target_mod.weight.data.float().to(self.target_device)
                    W = raw_w.t().contiguous() if is_conv1d else raw_w
                    out_f, in_f = W.shape

                    # Generate learned importance
                    importance = self.cpt_generator(task, out_f, device=W.device)
                    W_cpt = W * importance.unsqueeze(1)

                    r = max(
                        1,
                        min(int(min(out_f, in_f) * self.keep_ratio), min(out_f, in_f)),
                    )
                    try:
                        U, S, Vt = torch.linalg.svd(W_cpt, full_matrices=False)
                        W_lowrank = torch.matmul(U[:, :r] * S[:r].unsqueeze(0), Vt[:r, :])
                        W_approx = W_lowrank / importance.unsqueeze(1)
                        loss = torch.norm(W - W_approx, p="fro") / max(
                            torch.norm(W, p="fro").item(), 1e-6
                        )

                        optimizer.zero_grad()
                        loss.backward()
                        torch.nn.utils.clip_grad_norm_(
                            self.cpt_generator.parameters(), max_norm=1.0
                        )
                        optimizer.step()

                        total_loss += loss.item()
                        steps += 1
                    except Exception as e:
                        logger.debug("CPT generator training step exception: %s", e)

        self.cpt_generator.eval()
        self.cpt_generator_trained = True
        for p in self.cpt_generator.parameters():
            p.requires_grad = False

        avg_loss = total_loss / max(1, steps)
        logger.info(
            "CPT generator trained successfully over %d steps with avg loss %.4f",
            steps,
            avg_loss,
        )
        return avg_loss

    def save_cpt_generator(self, path: str | Path) -> None:
        """Save CPT generator state dict to file."""
        path = Path(path)
        path.parent.mkdir(parents=True, exist_ok=True)
        torch.save(self.cpt_generator.state_dict(), path)
        logger.info("Saved CPT generator checkpoint to %s", path)

    def load_cpt_generator(self, path: str | Path) -> bool:
        """Load CPT generator state dict from file."""
        path = Path(path)
        if path.exists():
            try:
                state_dict = torch.load(path, map_location=self.target_device)
                self.cpt_generator.load_state_dict(state_dict)
                self.cpt_generator_trained = True
                self.cpt_generator.eval()
                for p in self.cpt_generator.parameters():
                    p.requires_grad = False
                logger.info("Loaded CPT generator checkpoint from %s", path)
                return True
            except Exception as e:
                logger.warning("Failed to load CPT generator from %s: %s", path, e)
        return False

    def _is_gqa_asymmetric(self, name: str, module: nn.Module) -> bool:
        if not self.skip_attention_qkv:
            return False
        if self.target_attention_only:
            return False
        cfg = getattr(self.model, "config", None)
        num_heads = getattr(cfg, "num_attention_heads", None)
        num_kv_heads = getattr(cfg, "num_key_value_heads", None)
        if num_heads is not None and num_kv_heads is not None and num_kv_heads < num_heads:
            name_lower = name.lower()
            if any(
                p in name_lower
                for p in [
                    "q_proj",
                    "k_proj",
                    "v_proj",
                    "query",
                    "key",
                    "value",
                    "c_attn",
                ]
            ):
                return True
        return False

    def _is_target_module(self, name: str, module: nn.Module) -> bool:
        if isinstance(module, CacheNotesLoRALinear):
            return False

        # Do not target inner submodules of LoRA or wrappers directly
        if any(
            name.endswith(sfx)
            for sfx in [".base_layer", ".lora_A", ".lora_B", ".lora_dropout", ".linear"]
        ):
            return False

        target_mod = getattr(module, "base_layer", getattr(module, "linear", module))
        cls_name = module.__class__.__name__.lower()
        target_cls = target_mod.__class__.__name__.lower()

        is_linear_like = (
            isinstance(module, nn.Linear)
            or isinstance(target_mod, nn.Linear)
            or (
                Conv1D is not None
                and (isinstance(module, Conv1D) or isinstance(target_mod, Conv1D))
            )
            or "conv1d" in cls_name
            or "conv1d" in target_cls
            or hasattr(module, "nf")
            or hasattr(target_mod, "nf")
        )
        if not is_linear_like:
            return False

        name_lower = name.lower()
        # Protect classification / output heads
        if any(
            skip in name_lower
            for skip in [
                "classifier",
                "score",
                "lm_head",
                "pooler",
                "embed",
                "logits_processor",
                "head",
                "pre_classifier",
            ]
        ):
            return False

        if self._is_gqa_asymmetric(name, module):
            return False

        if self.target_attention_only:
            # Skip FFN / MLP layers
            ffn_patterns = [
                "ffn",
                "mlp",
                "lin1",
                "lin2",
                "intermediate",
                "gate_proj",
                "up_proj",
                "down_proj",
                "c_fc",
            ]
            if any(p in name_lower for p in ffn_patterns):
                return False
            # In RoBERTa, layer.X.output.dense is FFN, while attention.output.dense is attention
            if "output.dense" in name_lower and "attention" not in name_lower:
                return False
            # Match attention projections
            attn_patterns = [
                "attn",
                "attention",
                "q_proj",
                "k_proj",
                "v_proj",
                "o_proj",
                "out_lin",
                "q_lin",
                "k_lin",
                "v_lin",
                "c_attn",
                "c_proj",
                "query",
                "key",
                "value",
                "in_proj_qkv",
                "out_proj",
            ]
            if not any(p in name_lower for p in attn_patterns):
                return False
        else:
            if self.protect_critical and not should_compress_linear(self.model, name):
                return False

        if self.target_modules is not None:
            return any(t in name for t in self.target_modules)
        return True

    def compress(self, **kwargs: Any) -> tuple[nn.Module, dict[str, Any]]:
        """Compress linear layers using task-guided CPT-weighted low-rank SVD."""
        # Auto-train learnable CPT generator if active and untrained
        if self.use_learned_cpt and not self.cpt_generator_trained:
            logger.info(
                "CPT generator not yet trained. Running initial task-aware training pass..."
            )
            self.train_cpt_generator(epochs=3, lr=1e-4)

        self.compressed_layers = []
        self.fallback_layers = []
        self.skipped_layers = []
        self.layer_errors = {}
        self.actual_ranks = {}
        seen_weights: set[int] = set()

        for name, module in list(self.model.named_modules()):
            if not self._is_target_module(name, module):
                continue

            target_mod = getattr(module, "base_layer", getattr(module, "linear", module))
            module.__class__.__name__.lower()
            is_conv1d = (
                (
                    Conv1D is not None
                    and (isinstance(module, Conv1D) or isinstance(target_mod, Conv1D))
                )
                or hasattr(module, "nf")
                or hasattr(target_mod, "nf")
            )
            has_lora = (hasattr(module, "lora_A") and module.lora_A is not None) or hasattr(
                module, "base_layer"
            )

            w_id = id(target_mod.weight)
            if w_id in seen_weights:
                self.skipped_layers.append(name)
                continue
            seen_weights.add(w_id)

            raw_w = target_mod.weight.data
            device = raw_w.device
            dtype = raw_w.dtype

            # Standardize shape to [out_features, in_features] in FP32
            W = raw_w.t().contiguous().clone().float() if is_conv1d else raw_w.clone().float()
            out_f, in_f = W.shape

            # Retrieve CPT importance modulation per output feature (learned or static)
            importance = self.get_importance(out_f, device=device)

            # Weight rows by CPT importance
            W_cpt = W * importance.unsqueeze(1)

            # Compute SVD in CPT-weighted space
            r = max(1, min(int(min(out_f, in_f) * self.keep_ratio), min(out_f, in_f)))
            self.actual_ranks[name] = r
            try:
                U, S, Vt = torch.linalg.svd(W_cpt, full_matrices=False)
                # Low-rank approximation
                W_lowrank = torch.matmul(U[:, :r] * S[:r].unsqueeze(0), Vt[:r, :])
                # Unweight back to physical space: all channels remain active and non-zero
                W_approx = W_lowrank / importance.unsqueeze(1)
                if torch.isnan(W_approx).any() or torch.isinf(W_approx).any():
                    W_approx = W
            except Exception as e:
                logger.warning("SVD failed in CacheNotes on %s: %s", name, e)
                W_approx = W

            w_norm = torch.norm(W, p="fro").item()
            err = float(torch.norm(W - W_approx, p="fro").item() / max(w_norm, 1e-12))
            self.layer_errors[name] = err

            if torch.isnan(W_approx).any() or torch.isinf(W_approx).any() or err > self.max_error:
                logger.warning(
                    "CacheNotes reconstruction error %.4f on %s exceeded threshold %.4f.",
                    err,
                    name,
                    self.max_error,
                )
                if self.fallback_to_original:
                    logger.info(
                        "Skipping CacheNotes compression for %s (fallback to original).",
                        name,
                    )
                    self.fallback_layers.append(name)
                    # Freeze fallback layers
                    for p in module.parameters():
                        p.requires_grad = False
                    continue

            # Update weights / module
            if has_lora:
                # Layer is already a LoRALinear wrapper: update base layer weights in-place
                if is_conv1d:
                    target_mod.weight.data.copy_(
                        W_approx.t().contiguous().to(dtype=dtype, device=device)
                    )
                else:
                    target_mod.weight.data.copy_(
                        W_approx.contiguous().to(dtype=dtype, device=device)
                    )
                # Keep base weights frozen, but keep LoRA adapters trainable
                target_mod.weight.requires_grad = False
                if getattr(target_mod, "bias", None) is not None and target_mod.bias is not None:
                    target_mod.bias.requires_grad = False
                for p_name, p in module.named_parameters():
                    if any(t in p_name.lower() for t in ("lora_a", "lora_b", "adapter")):
                        p.requires_grad = True
            elif self.lora_r > 0:
                # Not LoRALinear yet, and lora_r > 0: wrap in CacheNotesLoRALinear
                cn_layer = CacheNotesLoRALinear.from_linear(
                    target_mod,
                    weight=W_approx,
                    lora_r=self.lora_r,
                    lora_alpha=self.lora_alpha,
                )
                if name == "":
                    self.model = cn_layer
                else:
                    replace_module(self.model, name, cn_layer)
            else:
                # Pure static compression (lora_r == 0)
                if is_conv1d:
                    target_mod.weight.data.copy_(
                        W_approx.t().contiguous().to(dtype=dtype, device=device)
                    )
                else:
                    target_mod.weight.data.copy_(
                        W_approx.contiguous().to(dtype=dtype, device=device)
                    )

            self.compressed_layers.append(name)

        # LayerNorm Recalibration
        if self.recalibrate_ln and (
            self.calibration_loader is not None or self.calibration_texts is not None
        ):
            try:
                recalibrate_layernorms(
                    self.model,
                    tokenizer=self.tokenizer,
                    calibration_loader=self.calibration_loader,
                    calibration_texts=self.calibration_texts,
                    device=self.target_device,
                )
            except Exception as e:
                logger.warning("CacheNotes LayerNorm recalibration skipped: %s", e)

        # Post-compression forward sanity check
        forward_check_success = False
        try:
            self.model.eval()
            with torch.no_grad():
                d = next(self.model.parameters()).device
                dummy = torch.randint(0, 100, (1, 16), device=d)
                self.model(dummy)
            forward_check_success = True
        except Exception as e:
            logger.debug("CacheNotes post-compression forward check exception: %s", e)

        self._compressed_params = self._count_params(self.model)
        avg_err = (
            sum(self.layer_errors[k] for k in self.compressed_layers)
            / max(1, len(self.compressed_layers))
            if self.compressed_layers
            else 0.0
        )
        avg_rank = (
            float(
                sum(self.actual_ranks[k] for k in self.compressed_layers)
                / max(1, len(self.compressed_layers))
            )
            if self.compressed_layers
            else 0.0
        )
        stats = self.get_compression_stats()
        stats.update(
            {
                "compressed_layers": self.compressed_layers,
                "fallback_layers": self.fallback_layers,
                "skipped_layers": self.skipped_layers,
                "num_compressed": len(self.compressed_layers),
                "num_fallback": len(self.fallback_layers),
                "num_skipped": len(self.skipped_layers),
                "layer_errors": self.layer_errors,
                "actual_ranks": self.actual_ranks,
                "avg_rank": avg_rank,
                "avg_reconstruction_error": avg_err,
                "task_type": self.task_type,
                "task_description": self.task_description,
                "use_learned_cpt": self.use_learned_cpt,
                "cpt_generator_trained": self.cpt_generator_trained,
                "keep_ratio": self.keep_ratio,
                "max_error": self.max_error,
                "fallback_to_original": self.fallback_to_original,
                "target_attention_only": self.target_attention_only,
                "lora_r": self.lora_r,
                "lora_alpha": self.lora_alpha,
                "forward_check_success": forward_check_success,
                "recalibrate_ln": self.recalibrate_ln,
                "method": "cachenotes_learned_cpt_svd",
                "kv_cache_budget_saving": f"{(1.0 - self.keep_ratio) * 100:.1f}% Footprint Reduction",
            }
        )
        self.stats = stats
        self.model.compression_stats = stats
        return self.model, stats

def get_importance(self, d: int, device: torch.device) -> torch.Tensor:
    """
    Retrieves feature importance vector using learned CPT generator if trained/active,
    otherwise falling back to static heuristic.
    """

def train_cpt_generator(
    self,
    task_descriptions: list[str] | None = None,
    sample_layers: list[tuple[str, nn.Module]] | None = None,
    epochs: int = 5,
    lr: float = 1e-4,
) -> float:
    """
    Train CPT generator on sample layers across diverse task descriptions,
    minimizing reconstruction error after CPT-weighted SVD compression.
    """
    if task_descriptions is None:
        task_descriptions = [
            "question answering: extract concise answer from supporting context passage",
            "code generation: implement efficient python function and classes",
            "summarization: produce concise abstractive summary of input document",
            "mathematical reasoning: solve multi-step theorem and calculus equations",
            "sentiment classification: predict positive or negative customer review sentiment",
        ]

    if sample_layers is None:
        sample_layers = [
            (name, mod)
            for name, mod in self.model.named_modules()
            if self._is_target_module(name, mod)
        ][:4]  # train on up to 4 sample layers for efficiency

    if not sample_layers:
        logger.warning("No sample layers found to train CPT generator.")
        return 0.0

    self.cpt_generator.train()
    for p in self.cpt_generator.parameters():
        p.requires_grad = True

    optimizer = torch.optim.AdamW(self.cpt_generator.parameters(), lr=lr)

    total_loss = 0.0
    steps = 0

    for epoch in range(epochs):
        for task in task_descriptions:
            for name, module in sample_layers:
                target_mod = getattr(module, "base_layer", getattr(module, "linear", module))
                cls_name = module.__class__.__name__.lower()
                is_conv1d = (
                    "conv1d" in cls_name or hasattr(module, "nf") or hasattr(target_mod, "nf")
                )
                raw_w = target_mod.weight.data.float().to(self.target_device)
                W = raw_w.t().contiguous() if is_conv1d else raw_w
                out_f, in_f = W.shape

                # Generate learned importance
                importance = self.cpt_generator(task, out_f, device=W.device)
                W_cpt = W * importance.unsqueeze(1)

                r = max(
                    1,
                    min(int(min(out_f, in_f) * self.keep_ratio), min(out_f, in_f)),
                )
                try:
                    U, S, Vt = torch.linalg.svd(W_cpt, full_matrices=False)
                    W_lowrank = torch.matmul(U[:, :r] * S[:r].unsqueeze(0), Vt[:r, :])
                    W_approx = W_lowrank / importance.unsqueeze(1)
                    loss = torch.norm(W - W_approx, p="fro") / max(
                        torch.norm(W, p="fro").item(), 1e-6
                    )

                    optimizer.zero_grad()
                    loss.backward()
                    torch.nn.utils.clip_grad_norm_(
                        self.cpt_generator.parameters(), max_norm=1.0
                    )
                    optimizer.step()

                    total_loss += loss.item()
                    steps += 1
                except Exception as e:
                    logger.debug("CPT generator training step exception: %s", e)

    self.cpt_generator.eval()
    self.cpt_generator_trained = True
    for p in self.cpt_generator.parameters():
        p.requires_grad = False

    avg_loss = total_loss / max(1, steps)
    logger.info(
        "CPT generator trained successfully over %d steps with avg loss %.4f",
        steps,
        avg_loss,
    )
    return avg_loss

    """
    Runtime task-guided dynamic KV cache compressor.
    Evaluates token task-relevance dynamically and prunes non-essential context tokens,
    supporting variable sequence lengths during prefill and autoregressive decoding.
    """

    def __init__(
        self,
        max_cache_len: int = 128,
        task_type: str = "qa",
        keep_ratio: float = 0.3,
        num_sinks: int = 4,
        cpt_generator: CPTGenerator | None = None,
    ) -> None:
        self.max_cache_len = int(max_cache_len)
        self.task_type = task_type
        self.keep_ratio = float(keep_ratio)
        self.num_sinks = int(num_sinks)
        self.cpt_generator = cpt_generator
        self.k_cache: torch.Tensor | None = None
        self.v_cache: torch.Tensor | None = None

    def reset(self) -> None:
        """Clear cache state to prevent memory leaks across sequences."""
        self.k_cache = None
        self.v_cache = None

    @torch.no_grad()
    def update_and_compress(
        self,
        new_k: torch.Tensor,
        new_v: torch.Tensor,
        is_new_sequence: bool = False,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Append new KV tokens and apply task-aware retention when exceeding capacity.
        Shapes: (B, H, N, D).
        Automatically resets cache on new multi-token prompt prefill (N > 1) or when is_new_sequence=True (H6).
        """
        k_4d = new_k
        v_4d = new_v
        if new_k.ndim == 2:
            k_4d = new_k.unsqueeze(0).unsqueeze(0)
            v_4d = new_v.unsqueeze(0).unsqueeze(0)
        elif new_k.ndim == 3:
            k_4d = new_k.unsqueeze(1)
            v_4d = new_v.unsqueeze(1)

        B, H, N, D = k_4d.shape

        # H6: Prevent cross-sequence cache pollution on prompt prefill (N > 1) or explicit flag
        if is_new_sequence or N > 1:
            self.reset()

        if self.k_cache is None:
            self.k_cache = k_4d
            self.v_cache = v_4d
        else:
            if self.k_cache.shape[0] != B:
                self.k_cache = self.k_cache[:1].expand(B, -1, -1, -1).clone()
                self.v_cache = self.v_cache[:1].expand(B, -1, -1, -1).clone()
            self.k_cache = torch.cat([self.k_cache, k_4d], dim=2)
            self.v_cache = torch.cat([self.v_cache, v_4d], dim=2)

        cur_len = self.k_cache.size(2)
        if cur_len <= self.max_cache_len:
            return self.k_cache, self.v_cache

        sinks = min(self.num_sinks, cur_len)
        retained_cap = max(sinks + 1, self.max_cache_len)

        # Task importance weighting across tokens
        if self.cpt_generator is not None:
            cpt_importance = self.cpt_generator(self.task_type, D, device=self.k_cache.device)
        else:
            cpt_importance = CompressionPlanningTokens.generate_importance(
                self.task_type, D, device=self.k_cache.device
            )

        weighted_k = self.k_cache * cpt_importance.view(1, 1, 1, D)
        token_scores = torch.norm(weighted_k.float(), dim=-1)  # (B, H, cur_len)

        # Always preserve initial sink tokens
        token_scores[:, :, :sinks] += 1e6

        _, top_indices = torch.topk(token_scores, retained_cap, dim=-1)
        top_indices, _ = torch.sort(top_indices, dim=-1)

        idx_exp = top_indices.unsqueeze(-1).expand(-1, -1, -1, D)
        self.k_cache = self.k_cache.gather(2, idx_exp).contiguous()
        self.v_cache = self.v_cache.gather(2, idx_exp).contiguous()

        return self.k_cache, self.v_cache

            _cfg_get(
                cfg,
                "skip_should_compress_filter",
                skip_should_compress_filter,
            )

        self.fallback_layers: list[str] = []
        self.skipped_layers: list[str] = []

        self._compressed_params: int = 0

    # Target check (DKV-FIX-07)
    # ------------------------------------------------------------------
    def _is_target_module(self, name: str, module: nn.Module, _debug: bool = False) -> bool:
        name_lower = name.lower()
        if any(part in name.split(".") for part in ("lora_A", "lora_B", "base_layer")):
            if _debug:
                logger.info(f"  [skip] {name}: lora/base_layer part")
            return False

        # Skip already compressed (DKV-FIX-05 companion)
        cls_name = module.__class__.__name__
        if cls_name in (
            "DeltaKVLinear",
            "LSCPrunedLoRALinear",
            "DynamicSVDLinear",
            "SVDLinear",
            "FactorisedLinear",
        ):
            if _debug:
                logger.info(f"  [skip] {name}: already compressed ({cls_name})")
            return False

        if any(
            skip in name_lower
            for skip in [
                "classifier",
                "score",
                "lm_head",
                "pooler",
                "embedding",
                "router",
                "gate",
            ]
        ):
            if _debug:
                logger.info(f"  [skip] {name}: skip keyword")
            return False

        target_mod = getattr(module, "base_layer", module)
        is_linear = isinstance(target_mod, nn.Linear)
        is_conv1d = _is_conv1d(target_mod)
        if not (is_linear or is_conv1d):
            if _debug:
                logger.info(f"  [skip] {name}: not Linear/Conv1D")
            return False

        # DKV-FIX-06: bypass framework filter if requested.
        if not self.skip_should_compress_filter:
            try:
                from neuralops.utils.compat import should_compress_linear

                if not should_compress_linear(self.model, name):
                    if _debug:
                        logger.info(f"  [skip] {name}: should_compress_linear=False")
                    return False
            except Exception as e:
                if _debug:
                    logger.info(f"  [skip] {name}: should_compress_linear raised {e}")
                return False

        w = getattr(target_mod, "weight", None)
        if w is None:
            if _debug:
                logger.info(f"  [skip] {name}: weight None")
            return False
        # DKV-FIX-11: min_layer_dim filter.
        if min(w.shape[0], w.shape[1]) < self.min_layer_dim:
            if _debug:
                logger.info(f"  [skip] {name}: dim too small {tuple(w.shape)}")
            return False

        if self.target_attention_only:
            last_part = name.split(".")[-1]
            is_attn = (last_part in _ATTN_NAMES) or any(
                k in name_lower for k in ("attn", "attention")
            )
            if not is_attn:
                if _debug:
                    logger.info(f"  [skip] {name}: target_attention_only filter")
                return False

        if _debug:
            logger.info(f"  [TARGET] {name}: shape={tuple(w.shape)}")
        return True

    # Main compress
    # ------------------------------------------------------------------
    def compress(self) -> tuple[nn.Module, dict[str, Any]]:
        # DKV-FIX-09: idempotence
        if getattr(self.model, "_delta_kv_compressed", False):
            logger.warning("DeltaKV: model already compressed; skipping.")
            try:
                stats = self.get_compression_stats()
            except Exception:
                stats = {}
            return self.model, stats

        self.compressed_layers = []
        self.fallback_layers = []
        self.skipped_layers = []
        self.layer_errors = {}
        self.layer_ranks = {}

        total_orig_params = 0
        total_comp_params = 0

        # Diagnostic breakdown if no targets.
        target_list = [
            (name, module)
            for name, module in self.model.named_modules()
            if self._is_target_module(name, module)
        ]
        if not target_list:
            logger.warning("DeltaKV: no compressible layers found. Running diagnostic...")
            n_scanned = 0
            for name, module in self.model.named_modules():
                n_scanned += 1
                self._is_target_module(name, module, _debug=True)
            logger.warning("DeltaKV diagnostic: %d modules scanned, 0 targets.", n_scanned)
            try:
                stats = self.get_compression_stats()
            except Exception:
                stats = {}
            return self.model, stats

        logger.info("DeltaKV: %d target layers identified.", len(target_list))

        # Freeze non-target attention/linear layers when target_attention_only.
        if self.target_attention_only:
            for name, module in self.model.named_modules():
                if any(part in name.split(".") for part in ("lora_A", "lora_B", "base_layer")):
                    continue
                target_mod = getattr(module, "base_layer", module)
                if isinstance(target_mod, nn.Linear) or _is_conv1d(target_mod):
                    if not self._is_target_module(name, module):
                        for p in module.parameters():
                            p.requires_grad = False
                        self.skipped_layers.append(name)

        for name, module in target_list:
            target_mod = getattr(module, "base_layer", module)
            raw_w = target_mod.weight.data
            is_conv1d = _is_conv1d(target_mod)

            if is_conv1d:
                in_f, out_f = raw_w.shape[0], raw_w.shape[1]
                W = raw_w.t().contiguous().float()
            else:
                out_f, in_f = raw_w.shape[0], raw_w.shape[1]
                W = raw_w.float()

            orig_layer_params = out_f * in_f
            total_orig_params += orig_layer_params
            norm_w = torch.norm(W, p="fro").item()

            try:
                U, S, Vt = torch.linalg.svd(W, full_matrices=False)
                # DKV-FIX-03: correct min over 2D shape.
                r = max(1, min(self.rank, out_f, in_f, len(S)))
                L = (U[:, :r] * S[:r].unsqueeze(0)) @ Vt[:r, :]

                residual = W - L
                flat_r = residual.contiguous().abs().reshape(-1)
                keep_n = max(1, min(int(flat_r.numel() * self.sparsity), flat_r.numel()))
                threshold = torch.topk(flat_r, keep_n).values[-1]
                mask = (residual.abs() >= threshold).float()
                residual_sparse = residual * mask

                W_compressed = L + residual_sparse
                err = float(torch.norm(W - W_compressed, p="fro").item() / max(norm_w, 1e-12))
                self.layer_errors[name] = err
                self.layer_ranks[name] = r

                if err > self.max_error and self.fallback_to_original:
                    logger.warning(
                        "DeltaKV: error %.4f on %s > %.4f; keeping original.",
                        err,
                        name,
                        self.max_error,
                    )
                    self.fallback_layers.append(name)
                    for p in module.parameters():
                        p.requires_grad = False
                    total_comp_params += orig_layer_params
                    continue

                # Write back in original orientation and dtype.
                if is_conv1d:
                    target_mod.weight.data.copy_(
                        W_compressed.t().contiguous().to(dtype=target_mod.weight.dtype)
                    )
                else:
                    target_mod.weight.data.copy_(W_compressed.to(dtype=target_mod.weight.dtype))

                # DKV-FIX-10: requires_grad handled at module level.
                for p in module.parameters():
                    p.requires_grad = True

                self.compressed_layers.append(name)
                # DKV-FIX-12: honest parameter accounting.
                # Note: the dense weight is retained in-place (no reduction);
                # we report the theoretical storage ratio here for reference.
                comp_layer_params = r * (in_f + out_f) + keep_n
                total_comp_params += comp_layer_params

            except Exception as e:
                logger.warning("DeltaKV: decomposition failed on %s: %s", name, e)
                self.fallback_layers.append(name)
                for p in module.parameters():
                    p.requires_grad = False
                total_comp_params += orig_layer_params

        # DKV-FIX-08: forward sanity check.
        forward_check_success = True
        try:
            with torch.no_grad():
                first_p = next(iter(self.model.parameters()), None)
                dev = first_p.device if first_p is not None else torch.device("cpu")
                dummy = torch.randint(0, 100, (1, 8), device=dev, dtype=torch.long)
                try:

                except Exception:
                    try:
                        self.model(dummy)
                    except Exception as e2:
                        logger.warning("DeltaKV: forward check failed: %s", e2)
                        forward_check_success = False
        except Exception as e:
            logger.warning("DeltaKV: forward check raised: %s", e)
            forward_check_success = False

        try:
            self._compressed_params = self._count_params(self.model)
        except Exception:
            self._compressed_params = sum(p.numel() for p in self.model.parameters())

        effective_ratio = total_comp_params / total_orig_params if total_orig_params > 0 else 1.0
        avg_err = (
            sum(self.layer_errors.values()) / max(1, len(self.layer_errors))
            if self.layer_errors
            else 0.0
        )

        try:
            stats = self.get_compression_stats()
        except Exception:
            stats = {}

        stats.update(
            {
                "rank": self.rank,
                "sparsity": self.sparsity,
                "max_error": self.max_error,
                "target_attention_only": self.target_attention_only,
                "fallback_to_original": self.fallback_to_original,
                "compressed_layers": self.compressed_layers,
                "compressed_layer_count": len(self.compressed_layers),
                "fallback_layers": self.fallback_layers,
                "fallback_layer_count": len(self.fallback_layers),
                "skipped_layers": self.skipped_layers,
                "skipped_layer_count": len(self.skipped_layers),
                "layer_errors": self.layer_errors,
                "layer_ranks": self.layer_ranks,
                "avg_reconstruction_error": avg_err,
                "effective_parameter_ratio": round(effective_ratio, 4),
                "kv_cache_storage_ratio": f"{self.sparsity * 100:.1f}% of full KV cache",
                "forward_check_success": forward_check_success,
                "method": "delta_kv",
                "note": (

                    "The stored weight retains its full (out, in) shape; framework-level "
                    "compression_rate reads 1.0. Use effective_parameter_ratio for the "
                    "theoretical storage metric."
                ),
            }

        try:

                    "num_fallback": len(self.fallback_layers),

                }

            pass

            len(self.fallback_layers),
            len(self.skipped_layers),
            avg_err,
            effective_ratio,
        )

class DeltaKVStore:
    """
    Optional runtime cache that stores delta residuals relative to retrieved
    historical K/V references. Not used by the weight-only compressor path.
    """

    def __init__(
        self,
        max_refs: int = 128,
        topk: int = 4,
        compression_ratio: float = 0.29,
        residual_threshold: float = 1e-3,
    ) -> None:
        self.max_refs = int(max_refs)
        self.topk = int(topk)
        self.compression_ratio = float(compression_ratio)
        self.residual_threshold = float(residual_threshold)
        self.ref_k: torch.Tensor | None = None
        self.ref_v: torch.Tensor | None = None

    def reset(self) -> None:
        self.ref_k = None
        self.ref_v = None

    @torch.no_grad()
    def update(self, k: torch.Tensor, v: torch.Tensor) -> None:
        orig_k = k.detach()
        orig_v = v.detach()
        if orig_k.ndim == 2:
            orig_k = orig_k.unsqueeze(0).unsqueeze(0)
            orig_v = orig_v.unsqueeze(0).unsqueeze(0)
        elif orig_k.ndim == 3:
            orig_k = orig_k.unsqueeze(1)
            orig_v = orig_v.unsqueeze(1)

        B, H, N, D = orig_k.shape
        if N == 0 or B == 0:
            return

        if self.ref_k is None:
            self.ref_k = orig_k
            self.ref_v = orig_v
        else:
            if self.ref_k.device != orig_k.device or self.ref_k.dtype != orig_k.dtype:
                self.ref_k = self.ref_k.to(device=orig_k.device, dtype=orig_k.dtype)
                self.ref_v = self.ref_v.to(device=orig_v.device, dtype=orig_v.dtype)

            if self.ref_k.shape[0] != B:
                self.ref_k = self.ref_k[:1].expand(B, -1, -1, -1).clone()
                self.ref_v = self.ref_v[:1].expand(B, -1, -1, -1).clone()

            self.ref_k = torch.cat([self.ref_k, orig_k], dim=2)
            self.ref_v = torch.cat([self.ref_v, orig_v], dim=2)

        if self.ref_k.size(2) > self.max_refs:
            self.ref_k = self.ref_k[:, :, -self.max_refs :, :].contiguous()
            self.ref_v = self.ref_v[:, :, -self.max_refs :, :].contiguous()

    @torch.no_grad()
    def compress(
        self,
        k: torch.Tensor,
        v: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        orig_shape = k.shape
        if self.ref_k is None or self.ref_k.size(2) == 0:
            self.update(k, v)
            return k, v

        k_4d = k
        v_4d = v
        if k.ndim == 2:
            k_4d = k.unsqueeze(0).unsqueeze(0)
            v_4d = v.unsqueeze(0).unsqueeze(0)
        elif k.ndim == 3:
            k_4d = k.unsqueeze(1)
            v_4d = v.unsqueeze(1)

        B, H, N, D = k_4d.shape
        ref_b, ref_h, ref_n, ref_d = self.ref_k.shape

        ref_k_cur = self.ref_k.to(device=k.device, dtype=k.dtype)
        ref_v_cur = self.ref_v.to(device=v.device, dtype=v.dtype)

        ref_k_b = ref_k_cur if ref_b == B else ref_k_cur[:1].expand(B, -1, -1, -1)
        ref_v_b = ref_v_cur if ref_b == B else ref_v_cur[:1].expand(B, -1, -1, -1)

        tk = min(self.topk, ref_n)
        if tk == 0:
            self.update(k, v)
            return k, v

        sim = torch.matmul(k_4d.float(), ref_k_b.float().transpose(-2, -1).contiguous()) / (D**0.5)
        _, top_idx = torch.topk(sim, tk, dim=-1)

        top_idx_exp = top_idx.unsqueeze(-1).expand(B, H, N, tk, D)
        ref_k_exp = ref_k_b.unsqueeze(2).expand(B, H, N, ref_n, D)
        ref_v_exp = ref_v_b.unsqueeze(2).expand(B, H, N, ref_n, D)

        rk = ref_k_exp.gather(3, top_idx_exp).mean(dim=3).to(dtype=k_4d.dtype)
        rv = ref_v_exp.gather(3, top_idx_exp).mean(dim=3).to(dtype=v_4d.dtype)

        k_residual = k_4d - rk
        v_residual = v_4d - rv

        k_norm = torch.norm(k_residual.float(), dim=-1, keepdim=True)
        v_norm = torch.norm(v_residual.float(), dim=-1, keepdim=True)

        k_mask = (k_norm > self.residual_threshold).to(dtype=k_4d.dtype)
        v_mask = (v_norm > self.residual_threshold).to(dtype=v_4d.dtype)

        k_hat = rk + k_residual * k_mask
        v_hat = rv + v_residual * v_mask

        self.update(k, v)
        return k_hat.reshape(orig_shape), v_hat.reshape(orig_shape)

class HybridKVCache:
    """
    Dynamic KV Cache Manager for HybridKV.
    Maintains separate retention policies for static vs dynamic attention heads:
    - Static heads: pinned attention sinks (first S tokens) + global heavy hitters
    - Dynamic heads: recent local sliding window (last W tokens) + chunked key retention
    Preserves all model stationary weights and strictly compresses dynamic KV cache tensors.
    Supports variable sequence lengths across prefill and token-by-token decoding.
    """

    def __init__(
        self,
        max_cache_len: int = 128,
        num_sinks: int = 4,
        recent_window: int = 32,
        budget: float = 0.3,
        static_ratio: float = 0.7,
        static_heads_mask: torch.Tensor | None = None,
    ) -> None:
        self.budget = float(budget)
        self.max_cache_len = (
            max(8, int(max_cache_len * self.budget)) if self.budget > 0 else int(max_cache_len)
        )
        self.num_sinks = int(num_sinks)
        self.recent_window = int(recent_window)
        self.static_ratio = float(static_ratio)
        self.static_heads_mask = static_heads_mask
        self.k_cache: torch.Tensor | None = None
        self.v_cache: torch.Tensor | None = None

    def reset(self) -> None:
        """Clear cache state to prevent memory leaks."""
        self.k_cache = None
        self.v_cache = None

    @torch.no_grad()
    def update_and_compress(
        self,
        new_k: torch.Tensor,
        new_v: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Append new KV representations and apply hybrid eviction if cache exceeds capacity.
        Shapes: (B, H, N, D) or (B, N, D) or (N, D).
        """
        orig_ndim = new_k.ndim
        k_4d = new_k
        v_4d = new_v
        if orig_ndim == 2:
            k_4d = new_k.unsqueeze(0).unsqueeze(0)
            v_4d = new_v.unsqueeze(0).unsqueeze(0)
        elif orig_ndim == 3:
            k_4d = new_k.unsqueeze(1)
            v_4d = new_v.unsqueeze(1)

                self.v_cache = self.v_cache.to(device=v_4d.device, dtype=v_4d.dtype)

                return out_k.squeeze(1), out_v.squeeze(1)
            return out_k, out_v

        target_len = self.max_cache_len

        head_v_list: list[torch.Tensor] = []

        is_static = (
            self.static_heads_mask
            if self.static_heads_mask is not None
            else torch.ones(H, dtype=torch.bool, device=self.k_cache.device)
        )

            k_h = self.k_cache[:, h_idx, :, :]  # (B, cur_len, D)
            v_h = self.v_cache[:, h_idx, :, :]

                # Static head policy: Sink tokens + Heavy Hitters via L2 Key norm
                sinks_k = k_h[:, : self.num_sinks, :]
                sinks_v = v_h[:, : self.num_sinks, :]

                remaining_v = v_h[:, self.num_sinks :, :]

                evicted_v = v_h[:, -window_budget:, :]

            head_v_list.append(evicted_v.unsqueeze(1))

    def _classify_head(self, weight: torch.Tensor) -> str:
        """Classify attention projection weight into static vs dynamic based on variance."""
        var = torch.var(weight.float(), dim=-1).mean().item()
        return "static" if var < 0.1 else "dynamic"

    def _is_attention_proj(self, name: str, module: nn.Module) -> bool:
        """Check if module corresponds to attention projection."""
        name_lower = name.lower()
        if any(part in name.split(".") for part in ["lora_A", "lora_B"]):
            return False
        if any(
            skip in name_lower
            for skip in [
                "classifier",
                "score",
                "lm_head",
                "pooler",
                "embedding",
                "router",
                "gate",
            ]
        ):
            return False

        target_mod = getattr(module, "base_layer", module)
        cls_name = target_mod.__class__.__name__.lower()
        is_linear = isinstance(target_mod, nn.Linear)
        is_conv1d = "conv1d" in cls_name or hasattr(target_mod, "nf")
        if not (is_linear or is_conv1d):
            return False

        if self.target_attention_only:
            return any(kw in name_lower for kw in _ATTN_NAMES)
        return True

    def compress(self) -> tuple[nn.Module, dict[str, Any]]:
        """Classify heads and configure dynamic KV caches without mutating model weights."""
        head_stats: dict[str, Any] = {}
        total_heads = 0
        static_heads_count = 0
        dynamic_heads_count = 0

        for name, module in self.model.named_modules():
            if self._is_attention_proj(name, module):
                target_mod = getattr(module, "base_layer", module)
                W = target_mod.weight.data
                cls_name = target_mod.__class__.__name__.lower()
                is_conv1d = "conv1d" in cls_name or hasattr(target_mod, "nf")
                out_f = W.shape[1] if is_conv1d else W.shape[0]

                head_dim = 64 if out_f % 64 == 0 else (128 if out_f % 128 == 0 else 32)
                num_heads = max(1, out_f // head_dim)
                head_type = self._classify_head(W)

                num_static = max(1, int(num_heads * self.static_ratio))
                num_dynamic = max(0, num_heads - num_static)
                static_mask = torch.zeros(num_heads, dtype=torch.bool, device=W.device)
                static_mask[:num_static] = True

                kv_cache = HybridKVCache(
                    max_cache_len=self.max_cache_len,
                    num_sinks=self.num_sinks,
                    recent_window=self.recent_window,
                    budget=self.budget,
                    static_ratio=self.static_ratio,
                    static_heads_mask=static_mask,
                )
                # Attach KV cache manager without mutating stationary weights
                module.hybrid_kv_cache = kv_cache
                if hasattr(module, "base_layer"):
                    module.base_layer.hybrid_kv_cache = kv_cache

                head_stats[name] = {
                    "head_type": head_type,
                    "num_heads": num_heads,
                    "static_heads": num_static,
                    "dynamic_heads": num_dynamic,
                }
                total_heads += num_heads
                static_heads_count += num_static
                dynamic_heads_count += num_dynamic
            elif hasattr(module, "weight") or list(module.parameters(recurse=False)):
                # Freeze fallback/skipped layers to prevent unintended gradient flow
                for p in module.parameters(recurse=False):
                    p.requires_grad = False

        # Forward sanity check
        forward_check_success = True
        try:
            with torch.no_grad():
                dev = next(self.model.parameters(), torch.tensor(0.0)).device

        self._compressed_params = self._count_params(self.model)

        stats["static_ratio"] = self.static_ratio
        stats["max_cache_len"] = self.max_cache_len

        stats["total_attention_heads"] = total_heads

    @staticmethod
    def _percentile_clip(W: torch.Tensor, low: float = 0.001, high: float = 0.999) -> torch.Tensor:
        """Clip extreme outliers from weight tensor before computing quantization scale."""
        if W.numel() == 0:
            return W
        q_low = torch.quantile(W.float(), low)
        q_high = torch.quantile(W.float(), high)

    def _quantize_to_int(
        self, W: torch.Tensor, bits: int
    ) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
        """
        Quantize continuous tensor to true integer format (torch.int8 or torch.int16).
        Uses signed symmetric quantization around 0:
            int_w in [-2^(bits-1), 2^(bits-1) - 1]
            scale = max(|W|) / (2^(bits-1) - 1)
            W_dequant ≈ int_w * scale + zero_point (zero_point = 0.0)
        """
        qmax = (2 ** (bits - 1)) - 1
        qmin = -(2 ** (bits - 1))

        max_abs = torch.clamp(W.abs().max(), min=1e-6)
        scale = (max_abs / float(qmax)).item()
        zero_point = 0.0

        q_float = torch.round(W / scale)
        q_clamped = torch.clamp(q_float, qmin, qmax)

        scale_t = torch.tensor(scale, dtype=torch.float32, device=W.device)
        zp_t = torch.tensor(zero_point, dtype=torch.float32, device=W.device)

        if bits <= 8:
            int_w = q_clamped.to(torch.int8)
        else:
            int_w = q_clamped.to(torch.int16)

        return int_w, scale_t, zp_t

    def _dequantize_from_int(
        self, int_w: torch.Tensor, scale: torch.Tensor, zero_point: torch.Tensor
    ) -> torch.Tensor:
        """Dequantize integer weight back to floating point."""
        return int_w.to(dtype=scale.dtype) * scale + zero_point

    def _estimate_actual_bytes(self, model: nn.Module) -> int:
        """
        Estimate physical memory storage in bytes across all parameters and buffers,
        taking into account actual tensor element size (int8=1 byte, fp16=2 bytes, fp32=4 bytes).
        Uses tensor storage data_ptr() to avoid duplicate counting of shared weights.
        """
        total_bytes = 0
        seen_ptrs: set[int] = set()

        for p in model.parameters():
            ptr = p.data_ptr()
            if ptr not in seen_ptrs:
                seen_ptrs.add(ptr)
                total_bytes += p.numel() * p.element_size()

        for b in model.buffers():
            ptr = b.data_ptr()
            if ptr not in seen_ptrs:
                seen_ptrs.add(ptr)
                total_bytes += b.numel() * b.element_size()

        return total_bytes

    def _is_gqa_asymmetric(self, name: str, module: nn.Module) -> bool:
        if not self.skip_attention_qkv:
            return False
        cfg = getattr(self.model, "config", None)
        num_heads = getattr(cfg, "num_attention_heads", None)
        num_kv_heads = getattr(cfg, "num_key_value_heads", None)
        if num_heads is not None and num_kv_heads is not None and num_kv_heads < num_heads:
            name_lower = name.lower()
            if any(
                p in name_lower
                for p in [
                    "q_proj",
                    "k_proj",
                    "v_proj",
                    "query",
                    "key",
                    "value",
                    "c_attn",
                ]
            ):
                return True
        return False

    def _is_target_module(self, name: str, module: nn.Module) -> bool:
        name_lower = name.lower()
        if any(part in name.split(".") for part in ["lora_A", "lora_B", "base_layer"]):
            return False
        if any(
            skip in name_lower
            for skip in [
                "classifier",
                "score",
                "lm_head",
                "pooler",
                "embedding",
                "router",
                "gate",
            ]
        ):
            return False

        if self.target_attention_only:
            if not any(attn_kw in name_lower for attn_kw in _ATTN_NAMES):
                return False

        if self._is_gqa_asymmetric(name, module):
            return False

        target_mod = getattr(module, "base_layer", module)
        cls_name = target_mod.__class__.__name__.lower()
        is_linear_like = (
            isinstance(target_mod, nn.Linear) or "conv1d" in cls_name or hasattr(target_mod, "nf")
        )
        if not is_linear_like:
            return False
        if isinstance(target_mod, KVTCQuantizedLoRALinear):
            return False
        if not should_compress_linear(self.model, name):
            return False
        if self.target_modules is None:
            return True
        return any(t in name for t in self.target_modules)

    def compress(self, **kwargs: Any) -> tuple[nn.Module, dict[str, Any]]:
        """Compress linear layers using PCA decorrelation and true integer quantization."""
        total_entropy = 0.0
        self.compressed_layers = []
        self.fallback_layers = []
        self.skipped_layers = []
        self.layer_errors = {}
        layer_details: dict[str, Any] = {}
        seen_weights: set[int] = set()

        total_orig_params = 0
        total_comp_params = 0

        # Physical byte tracking before compression
        initial_actual_bytes = self._estimate_actual_bytes(self.model)

        # Freeze non-target linear/conv1d layers when target_attention_only is True
        if self.target_attention_only:
            for name, module in self.model.named_modules():
                target_mod = getattr(module, "base_layer", module)
                cls_name = target_mod.__class__.__name__.lower()
                is_linear = isinstance(target_mod, nn.Linear)
                is_conv1d = "conv1d" in cls_name or hasattr(target_mod, "nf")
                if (is_linear or is_conv1d) and not self._is_target_module(name, module):
                    for p in module.parameters():
                        p.requires_grad = False
                    self.skipped_layers.append(name)

        for name, module in list(self.model.named_modules()):
            if not self._is_target_module(name, module):
                continue

            target_mod = getattr(module, "base_layer", module)
            if not hasattr(target_mod, "weight") or target_mod.weight is None:
                continue

            w_id = id(target_mod.weight)
            if w_id in seen_weights:
                self.skipped_layers.append(name)
                continue
            seen_weights.add(w_id)

            raw_w = target_mod.weight.data
            orig_params = raw_w.numel()
            if getattr(target_mod, "bias", None) is not None:
                orig_params += target_mod.bias.numel()
            total_orig_params += orig_params

            cls_name = target_mod.__class__.__name__.lower()
            is_conv1d = "conv1d" in cls_name or hasattr(target_mod, "nf")
            dtype = raw_w.dtype

            W = raw_w.t().contiguous().clone().float() if is_conv1d else raw_w.clone().float()
            d_out, d_in = W.shape
            min_dim = min(d_out, d_in)

            try:
                # Step 1: PCA decorrelation via SVD in FP32
                U, S, Vt = torch.linalg.svd(W, full_matrices=False)
                total_energy = (S**2).sum()
                if total_energy > 0:
                    cum_energy = torch.cumsum(S**2, dim=0) / total_energy
                    idx = torch.where(cum_energy >= self.pca_energy)[0]
                    rank = int(idx[0].item()) + 1 if len(idx) > 0 else min_dim
                else:
                    rank = min_dim

                effective_min_rank = min(self.min_rank, min_dim)
                rank = max(effective_min_rank, min(rank, min_dim))

                # Reconstruct low-rank approximation
                U_r = U[:, :rank]
                S_r = S[:rank]
                Vt_r = Vt[:rank, :]
                W_lowrank = (U_r * S_r.unsqueeze(0)) @ Vt_r

                # Step 2: Percentile outlier clipping before quantization
                W_clipped = self._percentile_clip(W_lowrank, low=0.001, high=0.999)

                # Step 3: Integer quantization
                int_w, scale, zp = self._quantize_to_int(W_clipped, bits=self.bits)
                W_quant = self._dequantize_from_int(int_w, scale, zp)
                W_quant = torch.nan_to_num(W_quant, nan=0.0)

                # Reconstruction error check against original W
                norm_orig = torch.norm(W, p="fro").item()
                rel_err = (torch.norm(W - W_quant, p="fro") / max(norm_orig, 1e-12)).item()
                self.layer_errors[name] = float(rel_err)

                if not math.isfinite(rel_err) or rel_err > self.max_error:
                    logger.warning(
                        "KVTC reconstruction error %.4f on %s exceeded max_error %.4f%s",
                        rel_err,
                        name,
                        self.max_error,
                        (
                            ". Keeping original layer."
                            if self.fallback_to_original
                            else ". Proceeding with quantized weight."
                        ),
                    )
                    if self.fallback_to_original:
                        self.fallback_layers.append(name)
                        for p in module.parameters():
                            p.requires_grad = False
                        total_comp_params += orig_params
                        continue

                # Step 4: Empirical Shannon entropy calculation on discrete integer symbols
                entropy = SimpleEntropyCoder.compute_entropy(int_w)
                total_entropy += entropy

                new_layer = KVTCQuantizedLoRALinear.from_linear(
                    module,
                    int_weight=int_w,
                    scale=scale,
                    zero_point=zp,
                    lora_r=self.lora_r,
                    lora_alpha=self.lora_alpha,
                    compute_dtype=dtype,
                )

                # Effective compressed parameters count for reporting
                quant_equiv_params = int(orig_params * (self.bits / 16.0))
                lora_params = (
                    (new_layer.lora_A.numel() + new_layer.lora_B.numel())
                    if (new_layer.lora_A is not None and new_layer.lora_B is not None)
                    else 0
                )
                total_comp_params += quant_equiv_params + lora_params

                replace_module(self.model, name, new_layer)
                self.compressed_layers.append(name)
                layer_details[name] = {
                    "rank": rank,
                    "rel_error": rel_err,
                    "entropy": entropy,
                    "shape": tuple(int_w.shape),
                }
            except Exception as e:
                logger.warning(
                    "KVTC compression exception on %s: %s. Keeping original layer.",
                    name,
                    e,
                )
                self.fallback_layers.append(name)
                for p in module.parameters():
                    p.requires_grad = False
                total_comp_params += orig_params
                self.layer_errors[name] = 1.0

        # Physical byte tracking after compression
        final_actual_bytes = self._estimate_actual_bytes(self.model)

        # Real post-compression forward sanity check
        forward_check_success = False
        forward_check_error = None
        try:
            p0 = next(self.model.parameters(), None)
            if p0 is None:
                p0 = next(self.model.buffers(), None)
            dev = p0.device if p0 is not None else torch.device("cpu")
            self.model.eval()

            # Try text classification / sequence classification format: input_ids (1, 8)
            with torch.no_grad():
                test_passed = False
                try:
                    dummy_ids = torch.randint(0, 100, (1, 8), device=dev)
                    _ = self.model(dummy_ids)
                    test_passed = True
                except Exception:
                    pass

                if not test_passed:
                    # Try continuous hidden tensor (1, 8, d_model)
                    dummy_x = torch.randn(1, 8, 64, device=dev)
                    _ = self.model(dummy_x)
                    test_passed = True

            forward_check_success = test_passed
        except Exception as e:
            forward_check_error = str(e)
            logger.error("KVTC post-compression forward pass failed: %s", e)
            if not kwargs.get("ignore_forward_check", False):
                raise RuntimeError(
                    f"KVTC produced a model that fails the forward pass sanity check: {e}"
                ) from e

        self._compressed_params = self._count_params(self.model)
        stats = self.get_compression_stats()
        avg_entropy = total_entropy / max(len(self.compressed_layers), 1)
        avg_err = (
            sum(self.layer_errors[k] for k in self.compressed_layers)
            / max(1, len(self.compressed_layers))
            if self.compressed_layers
            else 0.0
        )
        param_ratio = (
            round(total_comp_params / max(total_orig_params, 1), 4)
            if total_orig_params > 0
            else 1.0
        )

        bytes_reduction_pct = (
            (initial_actual_bytes - final_actual_bytes) / float(initial_actual_bytes) * 100.0
            if initial_actual_bytes > 0
            else 0.0
        )

        stats.update(
            {
                "compressed_layers": self.compressed_layers,
                "fallback_layers": self.fallback_layers,
                "skipped_layers": self.skipped_layers,
                "num_compressed": len(self.compressed_layers),
                "num_fallback": len(self.fallback_layers),
                "num_skipped": len(self.skipped_layers),
                "bits": self.bits,
                "pca_energy": self.pca_energy,
                "max_error": self.max_error,
                "fallback_to_original": self.fallback_to_original,
                "target_attention_only": self.target_attention_only,
                "skip_attention_qkv": self.skip_attention_qkv,
                "avg_entropy_bits": round(avg_entropy, 2),
                "entropy_lower_bound_bits": round(avg_entropy, 2),
                "entropy_coding_note": "Theoretical lower bound; actual bitstream not implemented.",
                "avg_reconstruction_error": avg_err,
                "effective_parameter_ratio": param_ratio,
                "initial_actual_bytes": initial_actual_bytes,
                "final_actual_bytes": final_actual_bytes,
                "actual_bytes_reduction_pct": round(bytes_reduction_pct, 2),
                "theoretical_entropy_compression_ratio": round(16.0 / max(avg_entropy, 1e-4), 2),
                "theoretical_vram_saving": f"{(1.0 - self.bits / 16.0) * 100:.1f}% vs FP16",
                "layer_errors": self.layer_errors,
                "layer_details": layer_details,
                "lora_r": self.lora_r,
                "lora_alpha": self.lora_alpha,
                "forward_check_success": forward_check_success,
                "forward_check_error": forward_check_error,
                "method": "kvtc_pca_quant",
            }
        )
        self.stats = stats
        return self.model, stats

class KVTCQuantizedLoRALinear(nn.Module):
    """
    Linear layer containing true integer quantized base weights (int8/int16 buffer)
    with scale and zero_point buffers, dequantized on the fly during forward(),
    plus optional trainable LoRA adapters (lora_A, lora_B).
    """

    def __init__(
        self,
        in_features: int,
        out_features: int,
        int_weight: torch.Tensor,
        scale: torch.Tensor,
        zero_point: torch.Tensor,
        bias: torch.Tensor | None = None,
        lora_r: int = 0,
        lora_alpha: float = 1.0,
        device: torch.device | None = None,
        compute_dtype: torch.dtype = torch.float32,
    ) -> None:
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.nf = out_features  # Conv1D compatibility
        self.compute_dtype = compute_dtype
        self.lora_r = int(lora_r)
        self.lora_alpha = float(lora_alpha)
        self.scaling = (self.lora_alpha / self.lora_r) if self.lora_r > 0 else 1.0

        target_device = device if device is not None else int_weight.device

        # Store true integer weights as buffer (not float Parameter)
        self.register_buffer("int_weight", int_weight.to(device=target_device))
        self.register_buffer("scale", scale.to(device=target_device, dtype=torch.float32))
        self.register_buffer("zero_point", zero_point.to(device=target_device, dtype=torch.float32))

        if bias is not None:
            self.bias = nn.Parameter(
                bias.to(device=target_device, dtype=compute_dtype),
                requires_grad=False,
            )
        else:
            self.register_parameter("bias", None)

        if self.lora_r > 0:
            self.lora_A = nn.Parameter(
                torch.empty(
                    (self.lora_r, in_features),
                    dtype=compute_dtype,
                    device=target_device,
                )
            )
            self.lora_B = nn.Parameter(
                torch.zeros(
                    (out_features, self.lora_r),
                    dtype=compute_dtype,
                    device=target_device,
                )
            )
            nn.init.kaiming_uniform_(self.lora_A, a=math.sqrt(5))
        else:
            self.register_parameter("lora_A", None)
            self.register_parameter("lora_B", None)

    @property
    def weight(self) -> torch.Tensor:
        """Backward-compatibility property returning on-the-fly dequantized weight."""
        return self.int_weight.to(dtype=self.compute_dtype) * self.scale + self.zero_point

    @classmethod
    def from_linear(
        cls,
        linear: nn.Module,
        int_weight: torch.Tensor,
        scale: torch.Tensor,
        zero_point: torch.Tensor,
        lora_r: int = 0,
        lora_alpha: float = 1.0,
        compute_dtype: torch.dtype | None = None,
    ) -> KVTCQuantizedLoRALinear:
        target_mod = getattr(linear, "base_layer", linear)
        bias = (
            target_mod.bias.detach()
            if hasattr(target_mod, "bias") and target_mod.bias is not None
            else None
        )
        in_features = getattr(target_mod, "in_features", int_weight.shape[1])
        out_features = getattr(target_mod, "out_features", int_weight.shape[0])

        existing_lora_A = getattr(linear, "lora_A", None)
        existing_lora_B = getattr(linear, "lora_B", None)
        effective_lora_r = lora_r
        effective_lora_alpha = lora_alpha
        if existing_lora_A is not None and existing_lora_B is not None:
            w_A = getattr(existing_lora_A, "weight", existing_lora_A)
            effective_lora_r = w_A.shape[0]
            if hasattr(linear, "scaling") and linear.scaling:
                effective_lora_alpha = linear.scaling * effective_lora_r
            elif hasattr(linear, "lora_alpha"):
                effective_lora_alpha = linear.lora_alpha

        if compute_dtype is None:
            if hasattr(target_mod, "weight") and target_mod.weight is not None:
                compute_dtype = target_mod.weight.dtype
            else:
                compute_dtype = torch.float32

        new_layer = cls(
            in_features=in_features,
            out_features=out_features,
            int_weight=int_weight,
            scale=scale,
            zero_point=zero_point,
            bias=bias,
            lora_r=effective_lora_r,
            lora_alpha=effective_lora_alpha,
            device=int_weight.device,
            compute_dtype=compute_dtype,
        )

        if existing_lora_A is not None and existing_lora_B is not None:
            w_A = getattr(existing_lora_A, "weight", existing_lora_A)
            w_B = getattr(existing_lora_B, "weight", existing_lora_B)
            if (
                hasattr(w_A, "data")
                and hasattr(w_B, "data")
                and new_layer.lora_A is not None
                and new_layer.lora_B is not None
            ):
                new_layer.lora_A.data.copy_(
                    w_A.data.to(dtype=new_layer.lora_A.dtype, device=new_layer.lora_A.device)
                )
                new_layer.lora_B.data.copy_(
                    w_B.data.to(dtype=new_layer.lora_B.dtype, device=new_layer.lora_B.device)
                )
                new_layer.lora_A.requires_grad = True
                new_layer.lora_B.requires_grad = True

        return new_layer

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        w_eff = self.int_weight.to(dtype=x.dtype) * self.scale.to(
            dtype=x.dtype
        ) + self.zero_point.to(dtype=x.dtype)
        bias = self.bias.to(dtype=x.dtype) if self.bias is not None else None
        out = F.linear(x, w_eff, bias)
        if self.lora_r > 0 and self.lora_A is not None and self.lora_B is not None:
            lora_out = F.linear(
                F.linear(x.to(dtype=self.lora_A.dtype), self.lora_A), self.lora_B
            ).to(dtype=out.dtype)
            out = out + self.scaling * lora_out
        return out

class KVTCStore:
    is_custom_attention: bool = True
    """
    Runtime KV Cache Transform Coding engine.
    Applies PCA decorrelation across head feature dimensions and adaptive quantization,
    supporting dynamic sequence lengths across prefill and decoding phases.
    """

    def __init__(self, bits: int = 4, pca_energy: float = 0.95) -> None:
        self.bits = int(bits)
        self.pca_energy = float(pca_energy)
        self.pca_basis_k: torch.Tensor | None = None
        self.pca_basis_v: torch.Tensor | None = None

    def reset(self) -> None:
        """Reset PCA basis to prevent cross-sequence memory leaks."""
        self.pca_basis_k = None
        self.pca_basis_v = None

    @torch.no_grad()
    def calibrate_basis(self, k: torch.Tensor, v: torch.Tensor) -> None:
        """Calibrate PCA projection basis from sample K/V tensor (B, H, N, D)."""
        k_flat = k.detach().float().reshape(-1, k.shape[-1])
        v_flat = v.detach().float().reshape(-1, v.shape[-1])

        # SVD on centered channels in FP32
        _, _, Vt_k = torch.linalg.svd(
            k_flat - k_flat.mean(dim=0, keepdim=True), full_matrices=False
        )
        _, _, Vt_v = torch.linalg.svd(
            v_flat - v_flat.mean(dim=0, keepdim=True), full_matrices=False
        )

        self.pca_basis_k = Vt_k.t().contiguous().to(device=k.device, dtype=k.dtype)
        self.pca_basis_v = Vt_v.t().contiguous().to(device=v.device, dtype=v.dtype)

    @torch.no_grad()
    def compress_kv_cache(
        self, k: torch.Tensor, v: torch.Tensor
    ) -> tuple[torch.Tensor, torch.Tensor, dict[str, Any]]:
        """
        Transform and quantize dynamic KV cache tensors.
        Returns reconstructed (k_hat, v_hat) and coding statistics.
        """
        orig_shape = k.shape
        D = orig_shape[-1]

        if self.pca_basis_k is None or self.pca_basis_v is None:
            self.calibrate_basis(k, v)

        k_flat = k.reshape(-1, D)
        v_flat = v.reshape(-1, D)

        k_trans = torch.matmul(k_flat, self.pca_basis_k)
        v_trans = torch.matmul(v_flat, self.pca_basis_v)

        qmax = (2 ** (self.bits - 1)) - 1
        scale_k = torch.clamp(k_trans.abs().max() / float(qmax), min=1e-6)
        scale_v = torch.clamp(v_trans.abs().max() / float(qmax), min=1e-6)

        q_k = torch.clamp(torch.round(k_trans / scale_k), -qmax, qmax)
        q_v = torch.clamp(torch.round(v_trans / scale_v), -qmax, qmax)

        h_k = SimpleEntropyCoder.compute_entropy(q_k)
        h_v = SimpleEntropyCoder.compute_entropy(q_v)

        k_hat = torch.matmul(q_k * scale_k, self.pca_basis_k.t().contiguous()).reshape(orig_shape)
        v_hat = torch.matmul(q_v * scale_v, self.pca_basis_v.t().contiguous()).reshape(orig_shape)

        mean_h = (h_k + h_v) / 2.0
        stats: dict[str, Any] = {
            "entropy_k_bits": h_k,
            "entropy_v_bits": h_v,
            "mean_entropy_bits": mean_h,
            "entropy_lower_bound_bits": mean_h,
            "note": "Theoretical lower bound; actual bitstream not implemented.",
            "theoretical_compression_ratio": 16.0 / max(mean_h, 1e-4),
        }
        return k_hat, v_hat, stats

    """Return an absolute version of this path
    No normalization or symlink resolution is performed.

    Use resolve() to resolve symlinks and remove '..' segments.
    """

        return self

        drive = os.path.splitroot(os.getcwd())[0]
        return self._from_parsed_parts(drive, self.root, self._tail)

        # There is a CWD on each drive-letter drive.
        cwd = os.path.abspath(self.drive)

    kind, including directories) matching the given relative pattern.

class SimpleEntropyCoder:
    """Lightweight entropy coder using empirical symbol frequency estimation."""

    @staticmethod
    def compute_entropy(tensor: torch.Tensor) -> float:
        """Compute empirical Shannon entropy in bits per symbol."""
        if tensor.numel() == 0:
            return 0.0
        unique_vals, counts = torch.unique(tensor, return_counts=True)
        probs = counts.float() / float(tensor.numel())
        entropy = -torch.sum(probs * torch.log2(torch.clamp(probs, min=1e-12))).item()
        return float(entropy)

    @staticmethod
    def compute_compression_ratio(tensor: torch.Tensor, uncompressed_bits: int = 16) -> float:
        """Estimate theoretical compression ratio relative to FP16/FP32 baseline."""
        entropy = SimpleEntropyCoder.compute_entropy(tensor)
        return float(uncompressed_bits / max(entropy, 1e-4))

class ZeroKVNeoCache:
    """
    Dynamic KV Cache Manager for ZeroKV-Neo.
    Maintains bounded O(1) cache memory by partitioning budget:
    - Pinned attention sinks (first N_sinks prompt tokens)
    - Pinned local sliding window (last N_window recent tokens)
    - Dynamic heavy hitters in the middle sequence retained via L2 norm / cumulative score.
    """

    def __init__(
        self,
        max_cache_len: int = 128,
        num_attention_sinks: int = 4,
        recent_window: int = 32,
    ) -> None:
        self.max_cache_len = int(max_cache_len)
        self.num_attention_sinks = int(num_attention_sinks)
        self.recent_window = int(recent_window)
        self.k_cache: torch.Tensor | None = None
        self.v_cache: torch.Tensor | None = None

    def reset(self) -> None:
        """Clear cache tensors to avoid memory leaks."""
        self.k_cache = None
        self.v_cache = None

    @torch.no_grad()
    def update_and_compress(
        self,
        new_k: torch.Tensor,
        new_v: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Append new KV tokens and evict non-essential tokens when exceeding capacity.
        Shapes: (B, H, N, D).
        """
        k_4d = new_k
        v_4d = new_v
        if new_k.ndim == 2:
            k_4d = new_k.unsqueeze(0).unsqueeze(0)
            v_4d = new_v.unsqueeze(0).unsqueeze(0)
        elif new_k.ndim == 3:
            k_4d = new_k.unsqueeze(1)
            v_4d = new_v.unsqueeze(1)

        B, H, N, D = k_4d.shape

        if self.k_cache is None:
            self.k_cache = k_4d
            self.v_cache = v_4d
        else:
            if self.k_cache.shape[0] != B:
                self.k_cache = self.k_cache[:1].expand(B, -1, -1, -1).clone()
                self.v_cache = self.v_cache[:1].expand(B, -1, -1, -1).clone()
            self.k_cache = torch.cat([self.k_cache, k_4d], dim=2)
            self.v_cache = torch.cat([self.v_cache, v_4d], dim=2)

        cur_len = self.k_cache.size(2)
        if cur_len <= self.max_cache_len:
            return self.k_cache, self.v_cache

        sinks = min(self.num_attention_sinks, cur_len)
        window = min(self.recent_window, cur_len - sinks)
        heavy_budget = max(0, self.max_cache_len - sinks - window)

        sink_k = self.k_cache[:, :, :sinks, :]
        sink_v = self.v_cache[:, :, :sinks, :]
        win_k = self.k_cache[:, :, -window:, :] if window > 0 else self.k_cache[:, :, :0, :]
        win_v = self.v_cache[:, :, -window:, :] if window > 0 else self.v_cache[:, :, :0, :]

        if heavy_budget > 0 and cur_len > (sinks + window):
            mid_k = self.k_cache[:, :, sinks : cur_len - window, :]
            mid_v = self.v_cache[:, :, sinks : cur_len - window, :]
            mid_norms = torch.norm(mid_k.float(), dim=-1)  # (B, H, mid_len)
            _, top_idx = torch.topk(mid_norms, min(heavy_budget, mid_norms.size(-1)), dim=-1)
            top_idx, _ = torch.sort(top_idx, dim=-1)

            idx_exp = top_idx.unsqueeze(-1).expand(-1, -1, -1, D)
            sel_mid_k = mid_k.gather(2, idx_exp)
            sel_mid_v = mid_v.gather(2, idx_exp)

            self.k_cache = torch.cat([sink_k, sel_mid_k, win_k], dim=2).contiguous()
            self.v_cache = torch.cat([sink_v, sel_mid_v, win_v], dim=2).contiguous()
        else:
            self.k_cache = torch.cat([sink_k, win_k], dim=2).contiguous() if window > 0 else sink_k
            self.v_cache = torch.cat([sink_v, win_v], dim=2).contiguous() if window > 0 else sink_v

        return self.k_cache, self.v_cache

    def _is_target_module(self, name: str, module: nn.Module) -> bool:
        # Protect critical classification heads, poolers, lm_heads, and embeddings
        if any(
            k in name.lower() for k in ["classifier", "score", "lm_head", "pooler", "embedding"]
        ):
            return False

        base_layer = getattr(module, "base_layer", module)
        cls_name = base_layer.__class__.__name__
        is_linear_like = (
            isinstance(base_layer, nn.Linear)
            or "conv1d" in cls_name.lower()
            or hasattr(base_layer, "nf")
            or (
                hasattr(base_layer, "weight")
                and hasattr(base_layer.weight, "ndim")
                and base_layer.weight.ndim == 2
                and not isinstance(base_layer, (nn.Embedding, nn.LayerNorm, nn.RMSNorm))
            )
        )
        if not is_linear_like:
            return False
        if not should_compress_linear(self.model, name):
            return False
        if self.target_attention_only:
            return self._is_attention_proj(name)
        return True

    def compress(self) -> tuple[nn.Module, dict[str, Any]]:
        """Compress linear projection weights preserving attention sinks and heavy hitters."""
        self.compressed_layers = []
        self.fallback_layers = []
        self.layer_errors = {}
        total_error = 0.0

        for name, module in list(self.model.named_modules()):
            if not self._is_target_module(name, module):
                continue

            base_layer = getattr(module, "base_layer", module)
            raw_w = base_layer.weight.data
            orig_dtype = raw_w.dtype
            orig_device = raw_w.device
            cls_name = base_layer.__class__.__name__

            is_conv1d = "conv1d" in cls_name.lower() or hasattr(base_layer, "nf")

            if is_conv1d:
                W = raw_w.t().float().contiguous()
                in_f = W.shape[1]
                out_f = W.shape[0]
            else:
                W = raw_w.float().contiguous()

    @staticmethod
    def _is_attention_proj(name: str) -> bool:
        """Check if module name corresponds to a QKV attention projection (not output projection).

        Explicitly excludes output projections first, then checks for QKV attention keywords.
        This prevents generic keywords like 'attn'/'attention' from matching out_proj, c_proj, etc.
        """
        name_lower = name.lower()

        # Exclusion list: output projections must never be treated as QKV targets
        output_proj_keywords = [
            "out_lin",
            "out_proj",
            "o_proj",
            "c_proj",
            "output.dense",
            "self.output",
            ".output",
        ]
        if any(kw in name_lower for kw in output_proj_keywords):
            return False

        # Inclusion list: only match true QKV projections
        qkv_keywords = [
            "q_lin",
            "k_lin",
            "v_lin",
            "q_proj",
            "k_proj",
            "v_proj",
            "query",
            "key",
            "value",
            "c_attn",
            "in_proj_qkv",
        ]
        return any(kw in name_lower for kw in qkv_keywords)
