    def _is_target_module(self, name: str, module: nn.Module) -> bool:
        if any(
            name.endswith(sfx)
            for sfx in [".base_layer", ".lora_A", ".lora_B", ".lora_dropout", ".linear"]
        ):
            return False

        target_mod = getattr(module, "base_layer", getattr(module, "linear", module))
        cls_name = module.__class__.__name__.lower()

        weight_tensor = getattr(target_mod, "weight", getattr(module, "weight", None))
        if weight_tensor is None or not hasattr(weight_tensor, "dim") or weight_tensor.dim() != 2:
            return False

        is_linear_like = (
            isinstance(module, nn.Linear)
            or isinstance(target_mod, nn.Linear)
            or (
                Conv1D is not None
                and (isinstance(module, Conv1D) or isinstance(target_mod, Conv1D))
            )
            or hasattr(module, "nf")
            or hasattr(target_mod, "nf")
            or "loralinear" in cls_name
            or hasattr(module, "base_layer")
        )
        if not is_linear_like:
            return False

        name_lower = name.lower()
        if any(
            skip in name_lower
            for skip in [
                "lm_head",
                "classifier",
                "head",
                "output_layer",
                "score",
                "pooler",
                "pre_classifier",
            ]
        ):
            return False

        if self.target_attention_only:
            ffn_patterns = [
                "ffn",
                "mlp",
                "lin1",
                "lin2",
                "intermediate",
                "gate_proj",
                "up_proj",
                "down_proj",
                "c_fc",
            ]
            if any(p in name_lower for p in ffn_patterns):
                return False
            if "output.dense" in name_lower and "attention" not in name_lower:
                return False
            attn_patterns = [
                "attn",
                "attention",
                "q_proj",
                "k_proj",
                "v_proj",
                "o_proj",
                "out_lin",
                "q_lin",
                "k_lin",
                "v_lin",
                "c_attn",
                "c_proj",
                "query",
                "key",
                "value",
                "in_proj_qkv",
                "in_proj_z",
                "in_proj_b",
                "out_proj",
            ]
            if not any(p in name_lower for p in attn_patterns):
                return False
        else:
            if not should_compress_linear(self.model, name):
                return False

        if self.target_modules is not None:
            return any(t in name for t in self.target_modules)

    # AC-FIX-02/10: per-channel quantization with outlier robustness.
    # ------------------------------------------------------------------
    def _quantize_projection(self, tensor: torch.Tensor) -> torch.Tensor:
        """
        Project continuous tensor onto a b-bit uniform quantization grid (FP32).
        When per-channel: each output row is scaled independently.
        """
        if self.quant_bits >= 32:
            return tensor
        scale_levels = float(2 ** (self.quant_bits - 1) - 1)
        if scale_levels <= 0:
            return tensor

        if self.quant_per_channel and tensor.dim() == 2:
            max_val = tensor.abs().amax(dim=1, keepdim=True).clamp(min=1e-6)
        else:
            max_val = tensor.abs().amax().clamp(min=1e-6)

        normalized = torch.clamp(tensor / max_val, -1.0, 1.0)
        quantized = torch.round(normalized * scale_levels) / scale_levels
        return quantized * max_val

    # ------------------------------------------------------------------
    # AC-FIX-03: no_grad projection helper.
    # ------------------------------------------------------------------
    def _prune_projection(self, tensor: torch.Tensor) -> torch.Tensor:
        with torch.no_grad():
            if self.structured_24:
                pruned, _ = apply_24_structured_mask(tensor, saliency=tensor.abs(), dim=1)
                return pruned
            elif self.block_size != (1, 1):
                pruned, _ = apply_block_mask(
                    tensor, block_size=self.block_size, sparsity=self.sparsity
                )
                return pruned
            else:
                flat = tensor.abs().reshape(-1)
                k = max(1, int(flat.numel() * (1.0 - self.sparsity)))
                thr = torch.topk(flat, k, largest=True).values[-1]
                mask = (tensor.abs() >= thr).float()
                return tensor * mask

    def compress(self, **kwargs: Any) -> tuple[nn.Module, dict[str, Any]]:
        self.compressed_layers = []
        self.fallback_layers = []
        self.layer_errors = {}
        self.layer_sparsities = {}
        self.layer_actual_sparsities = {}

        targets = [
            (name, mod)
            for name, mod in self.model.named_modules()
            if self._is_target_module(name, mod)
        ]

        if not targets:
            logger.warning("ADMM: No compressible layers found.")
            return self.model, self.get_compression_stats()

        for name, module in targets:
            target_mod = getattr(module, "base_layer", getattr(module, "linear", module))
            target_cls = target_mod.__class__.__name__.lower()
            is_conv1d = "conv1d" in target_cls

            raw_w = target_mod.weight.data
            orig_shape = raw_w.shape
            device = raw_w.device
            dtype = raw_w.dtype

            W0 = raw_w.t().float() if is_conv1d else raw_w.float()
            # AC-FIX-07: safe norm.
            norm_w0 = float(torch.norm(W0, p="fro").item())
            norm_w0 = max(norm_w0, 1e-12)

            best_W_final: torch.Tensor | None = None
            best_err = float("inf")

            try:
                with torch.no_grad():
                    # Baseline candidate.
                    Z0 = self._prune_projection(W0)
                    # AC-FIX-06: mask excludes non-finite.
                    finite_mask = torch.isfinite(Z0).float()
                    mask0 = (Z0 != 0.0).float() * finite_mask
                    W_cand = self._quantize_projection(Z0) * mask0
                    err0 = float(torch.norm(W0 - W_cand, p="fro").item() / norm_w0)
                    best_W_final = W_cand
                    best_err = err0

                    # ADMM state in FP32.
                    W = W0.clone()
                    Z = Z0.clone()
                    Q = self._quantize_projection(W0)
                    U1 = torch.zeros_like(W0)
                    U2 = torch.zeros_like(W0)

                    denom = 1.0 + self.rho1 + self.rho2

                    for it in range(self.max_iter):
                        # W-update.
                        W = (W0 + self.rho1 * (Z - U1) + self.rho2 * (Q - U2)) / denom

                        # Z-update (sparse).
                        Z = self._prune_projection(W + U1)

                        # Q-update (quantized).
                        Q = self._quantize_projection(W + U2)

                        # Dual updates.
                        U1 = U1 + (W - Z)
                        U2 = U2 + (W - Q)

                        # Non-finite guard.
                        if (
                            not torch.isfinite(W).all()
                            or not torch.isfinite(Z).all()
                            or not torch.isfinite(Q).all()
                        ):
                            logger.warning(
                                "ADMM: non-finite state at iter %d on %s",
                                it,
                                name,
                            )
                            break

                        # AC-FIX-04: track best iterate using ADMM state W.
                        finite_mask_curr = torch.isfinite(W).float()
                        mask_curr = (W != 0.0).float() * finite_mask_curr
                        W_curr = self._quantize_projection(W) * mask_curr
                        curr_err = float(torch.norm(W0 - W_curr, p="fro").item() / norm_w0)
                        if curr_err < best_err:
                            best_err = curr_err
                            best_W_final = W_curr

            except Exception as e:
                logger.warning("ADMM optimization failed on %s: %s", name, e)
                best_W_final = None
                best_err = 1.0

            W_final = best_W_final
            final_err = best_err

            self.layer_errors[name] = final_err
            self.layer_sparsities[name] = 0.5 if self.structured_24 else self.sparsity

            if W_final is None or not torch.isfinite(W_final).all() or final_err > self.max_error:
                logger.warning(
                    "ADMM reconstruction error %.4f on %s exceeded threshold %.4f.",
                    final_err,
                    name,
                    self.max_error,
                )
                if self.fallback_to_original:
                    logger.info("ADMM: fallback to original on %s.", name)
                    self.fallback_layers.append(name)
                    for p in module.parameters():
                        p.requires_grad = False
                    continue

            if is_conv1d:
                res_w = W_final.t().contiguous().to(dtype=dtype, device=device)
            else:
                res_w = W_final.contiguous().to(dtype=dtype, device=device)

            if res_w.shape != orig_shape:
                logger.warning(
                    "ADMM shape mismatch on %s: %s vs %s. Skipping.",
                    name,
                    tuple(res_w.shape),
                    tuple(orig_shape),
                )
                self.fallback_layers.append(name)
                for p in module.parameters():
                    p.requires_grad = False
                continue

            # AC-FIX-09: mutation under no_grad.
            with torch.no_grad():
                target_mod.weight.data.copy_(res_w)

            actual_zero_frac = float((res_w == 0).sum().item() / max(1, res_w.numel()))
            self.layer_actual_sparsities[name] = actual_zero_frac

            # AC-FIX-05: LoRA always trainable (recovery).
            has_lora = (hasattr(module, "lora_A") and module.lora_A is not None) or hasattr(
                module, "base_layer"
            )
            if has_lora:
                if hasattr(target_mod, "weight") and target_mod.weight is not None:
                    target_mod.weight.requires_grad = False
                if hasattr(target_mod, "bias") and target_mod.bias is not None:
                    target_mod.bias.requires_grad = False

                if hasattr(module, "lora_A") and module.lora_A is not None:
                    for p in module.lora_A.parameters():
                        p.requires_grad = True
                if hasattr(module, "lora_B") and module.lora_B is not None:
                    for p in module.lora_B.parameters():
                        p.requires_grad = True

            self.compressed_layers.append(name)

        try:
            self._compressed_params = self._count_params(self.model)
        except Exception:
            self._compressed_params = sum(p.numel() for p in self.model.parameters())

        try:
            stats = self.get_compression_stats()
        except Exception:
            stats = {}

        avg_err = (
            sum(self.layer_errors[k] for k in self.compressed_layers)
            / max(1, len(self.compressed_layers))
            if self.compressed_layers
            else 0.0
        )
        avg_actual_sparsity = (
            sum(self.layer_actual_sparsities.values()) / max(1, len(self.layer_actual_sparsities))
            if self.layer_actual_sparsities
            else 0.0
        )

        # AC-FIX-01: effective compression rate combines quantization and sparsity.
        quant_factor = 16.0 / max(1, self.quant_bits)
        sparse_factor = 1.0 / max(1e-6, 1.0 - avg_actual_sparsity)
        effective_storage_comp_rate = round(quant_factor * sparse_factor, 2)

        stats.update(
            {
                "compressed_layers": self.compressed_layers,
                "fallback_layers": self.fallback_layers,
                "layer_errors": self.layer_errors,
                "layer_sparsities": self.layer_sparsities,
                "layer_actual_sparsities": self.layer_actual_sparsities,
                "actual_sparsity": avg_actual_sparsity,
                "sparsity_target": self.sparsity,
                "quant_bits": self.quant_bits,
                "quant_per_channel": self.quant_per_channel,
                "structured_24": self.structured_24,
                "quant_factor": round(quant_factor, 2),
                "sparse_factor": round(sparse_factor, 2),
                "effective_compression_rate": effective_storage_comp_rate,
                "admm_iterations": self.max_iter,
                "max_error": self.max_error,
                "fallback_to_original": self.fallback_to_original,
                "target_attention_only": self.target_attention_only,
                "avg_reconstruction_error": avg_err,
                "num_compressed": len(self.compressed_layers),
                "num_fallback": len(self.fallback_layers),
                "method": "admm_joint_prune_quant",
            }
        )
        logger.info(
            "ADMM compression complete: %d compressed, %d fallback. "
            "Avg error: %.4f, Actual sparsity: %.2f%%, Effective comp rate: %.2fx",
            len(self.compressed_layers),
            len(self.fallback_layers),
            avg_err,
            avg_actual_sparsity * 100.0,
            effective_storage_comp_rate,
        )
        return self.model, stats

class ADMMMaskedLinear(nn.Module):
    """
    Linear layer with continuous weight + binary mask and STE gradient.
    """

    def __init__(
        self,
        in_features: int,
        out_features: int,
        bias: bool = True,
        dtype: torch.dtype = torch.float32,
        device: torch.device | None = None,
    ) -> None:
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.weight = nn.Parameter(
            torch.empty(out_features, in_features, dtype=dtype, device=device)
        )
        # AD-FIX-07: non-persistent mask.
        self.register_buffer(
            "mask",
            torch.ones(out_features, in_features, dtype=dtype, device=device),
            persistent=False,
        )

        if bias:
            self.bias = nn.Parameter(torch.empty(out_features, dtype=dtype, device=device))
        else:
            self.register_parameter("bias", None)

    @classmethod
    def from_linear(cls, linear: nn.Module, mask: torch.Tensor) -> ADMMMaskedLinear:
        # AD-FIX-05: handle Conv1D (nn.Module with .nf) as well as nn.Linear.
        raw_w = linear.weight.data
        is_conv1d = "conv1d" in linear.__class__.__name__.lower() or hasattr(linear, "nf")
        if is_conv1d:
            in_f, out_f = raw_w.shape[0], raw_w.shape[1]
            w_data = raw_w.t().contiguous()
        else:
            out_f, in_f = raw_w.shape
            w_data = raw_w

        layer = cls(
            in_features=in_f,
            out_features=out_f,
            bias=getattr(linear, "bias", None) is not None and linear.bias is not None,
            dtype=raw_w.dtype,
            device=raw_w.device,
        )
        with torch.no_grad():
            layer.weight.data.copy_(w_data)
            layer.mask.copy_(mask.to(dtype=raw_w.dtype, device=raw_w.device))
            if linear.bias is not None and layer.bias is not None:
                layer.bias.data.copy_(linear.bias.data)
        return layer

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        w_masked = ADMMSTEPrune.apply(self.weight, self.mask)
        return F.linear(x, w_masked, self.bias)

    def _is_target_module(self, name: str, module: nn.Module) -> bool:
        if isinstance(module, ADMMMaskedLinear):
            return False
        if any(
            name.endswith(sfx)
            for sfx in [".base_layer", ".lora_A", ".lora_B", ".lora_dropout", ".linear"]
        ):
            return False

        target_mod = getattr(module, "base_layer", getattr(module, "linear", module))
        cls_name = module.__class__.__name__.lower()

        weight_tensor = getattr(target_mod, "weight", getattr(module, "weight", None))
        if weight_tensor is None or not hasattr(weight_tensor, "dim") or weight_tensor.dim() != 2:
            return False

        target_cls = target_mod.__class__.__name__.lower()
        is_conv1d = "conv1d" in target_cls
        is_linear_like = (
            isinstance(module, nn.Linear)
            or isinstance(target_mod, nn.Linear)
            or is_conv1d
            or (
                Conv1D is not None
                and (isinstance(module, Conv1D) or isinstance(target_mod, Conv1D))
            )
            or "loralinear" in cls_name
            or hasattr(module, "base_layer")
        )
        if not is_linear_like:
            return False

        name_lower = name.lower()
        if any(
            skip in name_lower
            for skip in [
                "lm_head",
                "classifier",
                "head",
                "output_layer",
                "score",
                "pooler",
                "pre_classifier",
            ]
        ):
            return False

        if self.target_attention_only:
            ffn_patterns = [
                "ffn",
                "mlp",
                "lin1",
                "lin2",
                "intermediate",
                "gate_proj",
                "up_proj",
                "down_proj",
                "c_fc",
            ]
            if any(p in name_lower for p in ffn_patterns):
                return False
            if "output.dense" in name_lower and "attention" not in name_lower:
                return False
            attn_patterns = [
                "attn",
                "attention",
                "q_proj",
                "k_proj",
                "v_proj",
                "o_proj",
                "out_lin",
                "q_lin",
                "k_lin",
                "v_lin",
                "c_attn",
                "c_proj",
                "query",
                "key",
                "value",
                "in_proj_qkv",
                "in_proj_z",
                "in_proj_b",
                "out_proj",
            ]
            if not any(p in name_lower for p in attn_patterns):
                return False
        else:
            if not should_compress_linear(self.model, name):
                return False

        if self.target_modules is not None:
            return any(t in name for t in self.target_modules)
        return True

    def _collect_calibration_inputs(self, target_names: list[str]) -> dict[str, torch.Tensor]:
        collected: dict[str, list[torch.Tensor]] = {name: [] for name in target_names}

        if self.calibration_inputs:
            for name in target_names:
                if name in self.calibration_inputs:
                    collected[name].append(self.calibration_inputs[name])

        has_pre_collected = any(len(v) > 0 for v in collected.values())
        if has_pre_collected:
            return {k: torch.cat(v, dim=0) for k, v in collected.items() if v}

        if self.calibration_loader is None and self.calibration_texts is None:
            return {}

        hooks = []
        name_to_mod = dict(self.model.named_modules())

        def make_hook(layer_name: str):
            def hook_fn(mod, inp, out):
                if not inp or inp[0] is None:
                    return
                cur_total = sum(t.shape[0] for t in collected[layer_name])
                # AD-FIX-09: budget check before slicing.
                rem = self.distill_batch_size - cur_total
                if rem <= 0:
                    return
                x = inp[0].detach()
                if x.ndim >= 2:
                    x_flat = x.reshape(-1, x.shape[-1]).float().cpu()
                    collected[layer_name].append(x_flat[:rem])

            return hook_fn

        for name in target_names:
            if name in name_to_mod:
                target_mod = getattr(name_to_mod[name], "base_layer", name_to_mod[name])
                hooks.append(target_mod.register_forward_hook(make_hook(name)))

        device = self.target_device
        was_training = self.model.training
        self.model.eval()

        try:
            with torch.no_grad():
                if self.calibration_loader is not None:
                    for batch in self.calibration_loader:
                        if isinstance(batch, dict):
                            batch = {
                                k: v.to(device) if isinstance(v, torch.Tensor) else v
                                for k, v in batch.items()
                            }
                            self.model(**batch)
                        elif isinstance(batch, (list, tuple)):
                            batch = [
                                v.to(device) if isinstance(v, torch.Tensor) else v for v in batch
                            ]
                            self.model(*batch)
                        elif isinstance(batch, torch.Tensor):
                            self.model(batch.to(device))
                        if all(
                            sum(t.shape[0] for t in collected[n]) >= self.distill_batch_size
                            for n in target_names
                        ):
                            break
                elif self.calibration_texts is not None and self.tokenizer is not None:
                    for text in self.calibration_texts[:4]:
                        toks = self.tokenizer(
                            text,
                            return_tensors="pt",
                            truncation=True,
                            max_length=128,
                        )
                        toks = {k: v.to(device) for k, v in toks.items()}
                        self.model(**toks)
        except Exception as e:
            logger.warning("ADMM calibration input collection failed: %s", e)
        finally:
            for h in hooks:
                h.remove()
            if was_training:
                self.model.train()

        result: dict[str, torch.Tensor] = {}
        for name, tensor_list in collected.items():
            if tensor_list:
                result[name] = torch.cat(tensor_list, dim=0)
        return result

    def _project_sparse(self, candidate: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Project candidate (typically W + U) onto the sparse constraint set.
        Returns (pruned_tensor, binary_mask).
        AD-FIX-11: saliency is computed from the *candidate*, not from W.
        """
        if self.structured_24:
            saliency = candidate.abs()
            pruned, raw_mask = apply_24_structured_mask(candidate, saliency=saliency, dim=1)
            if raw_mask is None:
                mask = (pruned.abs() > 1e-12).float()
            elif raw_mask.dtype == torch.bool:
                mask = raw_mask.float()
            else:
                mask = (raw_mask > 0.5).float()
            return pruned, mask
        elif self.block_size != (1, 1):
            pruned, raw_mask = apply_block_mask(
                candidate, block_size=self.block_size, sparsity=self.sparsity
            )
            if raw_mask is None:
                mask = (pruned.abs() > 1e-12).float()
            elif raw_mask.dtype == torch.bool:
                mask = raw_mask.float()
            else:
                mask = (raw_mask > 0.5).float()
            return pruned, mask
        else:
            flat = candidate.abs().reshape(-1)
            k = max(1, int(flat.numel() * (1.0 - self.sparsity)))
            thr = torch.topk(flat, k, largest=True).values[-1]
            mask = (candidate.abs() >= thr).float()
            return candidate * mask, mask

    def compress(self, **kwargs: Any) -> tuple[nn.Module, dict[str, Any]]:
        self.compressed_layers = []
        self.fallback_layers = []
        self.layer_errors = {}
        self.layer_sparsities = {}
        self.layer_actual_sparsities = {}
        self.layer_distill_losses = {}

        targets = [
            (name, mod)
            for name, mod in self.model.named_modules()
            if self._is_target_module(name, mod)
        ]

        if not targets:
            logger.warning("ADMMPruner: No compressible target modules found.")
            stats = self.get_compression_stats()
            self.stats = stats
            self.model.compression_stats = stats
            return self.model, stats

        target_names = [name for name, _ in targets]
        calib_inputs = (
            self._collect_calibration_inputs(target_names) if self.lambda_distill > 0 else {}
        )

        for name, module in targets:
            target_mod = getattr(module, "base_layer", getattr(module, "linear", module))
            target_cls = target_mod.__class__.__name__.lower()
            is_conv1d = "conv1d" in target_cls

            raw_w = target_mod.weight.data
            orig_shape = raw_w.shape
            device = raw_w.device
            dtype = raw_w.dtype

            W0 = raw_w.t().float() if is_conv1d else raw_w.float()
            out_features, in_features = W0.shape
            w_norm = torch.norm(W0, p="fro").item()

            X_calib = calib_inputs.get(name, None)
            if self.lambda_distill > 0:
                if X_calib is None or X_calib.numel() == 0:
                    torch.manual_seed(hash(name) % (2**31 - 1))
                    X_calib = torch.randn(
                        min(32, self.distill_batch_size),
                        in_features,
                        device=device,
                    ) / math.sqrt(in_features)
                else:
                    # AD-FIX-08: cast X_calib to match W0 dtype.
                    X_calib = X_calib.to(device=device, dtype=torch.float32)
                    if X_calib.ndim > 2:
                        X_calib = X_calib.reshape(-1, in_features)
                    if X_calib.shape[0] > self.distill_batch_size:
                        X_calib = X_calib[: self.distill_batch_size]
                with torch.no_grad():
                    Y0 = X_calib @ W0.t()
            else:
                X_calib = None
                Y0 = None

            # Baseline projection.
            Z0, mask0 = self._project_sparse(W0)
            W_sparse_0 = W0 * mask0
            best_W_sparse = W_sparse_0
            best_mask = mask0.clone()  # AD-FIX-04
            base_w_err = float(torch.norm(W0 - W_sparse_0, p="fro").item() / max(w_norm, 1e-12))
            best_err = base_w_err
            best_score = base_w_err
            if self.lambda_distill > 0 and X_calib is not None and Y0 is not None:
                y0_norm = max(torch.norm(Y0, p="fro").item(), 1e-12)
                base_y_err = float(
                    torch.norm((X_calib @ W_sparse_0.t()) - Y0, p="fro").item() / y0_norm
                )
                best_score = base_w_err + self.lambda_distill * base_y_err

            # ADMM state.
            W = W0.clone()
            Z = Z0.clone()
            mask = mask0.clone()
            U = torch.zeros_like(W0)

            rho = self.initial_rho
            mu = 10.0
            tau_incr = 1.5
            tau_decr = 1.5
            rho_min, rho_max = 1e-4, 100.0
            final_distill_loss = 0.0

            try:
                for it in range(self.max_iter):
                    with torch.no_grad():
                        Z_prev = Z.clone()

                        # 1. Sparse projection.
                        Z, mask = self._project_sparse(W + U)

                        # 2. Closed-form W update.
                        W = (W0 + rho * (Z - U)) / (1.0 + rho)

                        # 3. Layer-Wise Reconstruction Distillation.
                        # AD-FIX-02: scheduled, bounded distill step.
                        if self.lambda_distill > 0 and X_calib is not None and Y0 is not None:
                            for _ in range(self.distill_steps):
                                if self.use_ste:
                                    W_eff = ADMMSTEPrune.apply(W, mask)
                                else:
                                    W_eff = W * mask

                                Y_pruned = X_calib @ W_eff.t()
                                E = Y_pruned - Y0
                                N_pts, d_out = E.shape
                                grad_W = ((2.0 / max(1, N_pts * d_out)) * (E.t() @ X_calib)) * mask

                                grad_norm = torch.norm(grad_W)
                                if grad_norm > 1e-8:
                                    # AD-FIX-02: no artificial scaling distortion;
                                    # lr already encodes lambda_distill influence.
                                    lr_eff = self.distill_lr
                                    direction = grad_W / grad_norm.clamp(min=1e-12)
                                    step_size = lr_eff * min(1.0, float(grad_norm))
                                    W = W - step_size * direction
                                else:
                                    break

                        # 4. Dual update.
                        U = U + W - Z

                        # 5. Dynamic rho scheduling.
                        if self.dynamic_rho:
                            primal_res = torch.norm(W - Z).item()
                            dual_res = (rho * torch.norm(Z - Z_prev)).item()
                            if primal_res > mu * (dual_res + 1e-8):
                                rho = min(rho * tau_incr, rho_max)
                            elif dual_res > mu * (primal_res + 1e-8):
                                rho = max(rho / tau_decr, rho_min)

                        if not torch.isfinite(W).all():
                            logger.warning(
                                "ADMMPruner: non-finite W at iter %d on %s",
                                it,
                                name,
                            )
                            break

                        # Monotonic best-iterate tracking.
                        W_sparse_curr = W * mask
                        curr_w_err = float(
                            torch.norm(W0 - W_sparse_curr, p="fro").item() / max(w_norm, 1e-12)
                        )
                        curr_score = curr_w_err
                        if self.lambda_distill > 0 and X_calib is not None and Y0 is not None:
                            curr_y_err = float(
                                torch.norm((X_calib @ W_sparse_curr.t()) - Y0, p="fro").item()
                                / y0_norm
                            )
                            curr_score = curr_w_err + self.lambda_distill * curr_y_err

                        if curr_score < best_score:
                            best_score = curr_score
                            best_err = curr_w_err
                            best_W_sparse = W_sparse_curr.clone()
                            best_mask = mask.clone()  # AD-FIX-04

                # Final distill loss on best sparse candidate.
                if self.lambda_distill > 0 and X_calib is not None and Y0 is not None:
                    with torch.no_grad():
                        final_distill_loss = float(
                            torch.mean(((X_calib @ best_W_sparse.t()) - Y0) ** 2).item()
                        )

            except Exception as e:
                logger.warning("ADMMPruner optimization failed on %s: %s", name, e)
                best_W_sparse = None
                best_err = 1.0

            final_err = best_err
            W_final = best_W_sparse
            self.layer_errors[name] = final_err
            self.layer_sparsities[name] = 0.5 if self.structured_24 else self.sparsity
            self.layer_distill_losses[name] = final_distill_loss

            if W_final is None or not torch.isfinite(W_final).all() or final_err > self.max_error:
                logger.warning(
                    "ADMMPruner reconstruction error %.4f on %s exceeded threshold %.4f.",
                    final_err,
                    name,
                    self.max_error,
                )
                if self.fallback_to_original:
                    logger.info("ADMMPruner: fallback to original on %s.", name)
                    self.fallback_layers.append(name)
                    for p in module.parameters():
                        p.requires_grad = False
                    continue

            if is_conv1d:
                res_w = W_final.t().contiguous().to(dtype=dtype, device=device)
            else:
                res_w = W_final.contiguous().to(dtype=dtype, device=device)

            if res_w.shape != orig_shape:
                logger.warning(
                    "ADMMPruner shape mismatch on %s: %s vs %s. Skipping.",
                    name,
                    tuple(res_w.shape),
                    tuple(orig_shape),
                )
                self.fallback_layers.append(name)
                for p in module.parameters():
                    p.requires_grad = False
                continue

            has_lora = (hasattr(module, "lora_A") and module.lora_A is not None) or hasattr(
                module, "base_layer"
            )

            # AD-FIX-05: ADMMMaskedLinear applies whenever use_ste is on and no LoRA / no Conv1D.
            if self.use_ste and not has_lora and not is_conv1d and isinstance(module, nn.Linear):
                admm_layer = ADMMMaskedLinear.from_linear(module, mask=best_mask)
                replace_module(self.model, name, admm_layer)
            else:
                with torch.no_grad():
                    target_mod.weight.data.copy_(res_w)

            actual_zero_frac = float((res_w == 0).sum().item() / max(1, res_w.numel()))
            self.layer_actual_sparsities[name] = actual_zero_frac

            if has_lora:
                if hasattr(target_mod, "weight") and target_mod.weight is not None:
                    target_mod.weight.requires_grad = False
                if hasattr(target_mod, "bias") and target_mod.bias is not None:
                    target_mod.bias.requires_grad = False
                for p_name, p in module.named_parameters():
                    if any(t in p_name.lower() for t in ("lora_a", "lora_b", "adapter")):
                        p.requires_grad = True

            self.compressed_layers.append(name)

        # LayerNorm recalibration.
        if self.recalibrate_ln and (
            self.calibration_loader is not None or self.calibration_texts is not None
        ):
            try:
                recalibrate_layernorms(
                    self.model,
                    tokenizer=self.tokenizer,
                    calibration_loader=self.calibration_loader,
                    calibration_texts=self.calibration_texts,
                    device=self.target_device,
                )
            except Exception as e:
                logger.warning("ADMM LayerNorm recalibration skipped: %s", e)

        try:
            self._compressed_params = self._count_params(self.model)
        except Exception:
            self._compressed_params = sum(p.numel() for p in self.model.parameters())

        avg_err = (
            sum(self.layer_errors[k] for k in self.compressed_layers)
            / max(1, len(self.compressed_layers))
            if self.compressed_layers
            else 0.0
        )
        avg_actual_sparsity = (
            sum(self.layer_actual_sparsities[k] for k in self.compressed_layers)
            / max(1, len(self.compressed_layers))
            if self.compressed_layers
            else (0.5 if self.structured_24 else self.sparsity)
        )
        avg_distill_loss = (
            sum(self.layer_distill_losses[k] for k in self.compressed_layers)
            / max(1, len(self.compressed_layers))
            if self.compressed_layers
            else 0.0
        )

        try:
            stats = self.get_compression_stats()
        except Exception:
            stats = {}
        stats.update(
            {
                "pruned_layers": len(self.compressed_layers),
                "compressed_layers": self.compressed_layers,
                "fallback_layers": self.fallback_layers,
                "num_compressed": len(self.compressed_layers),
                "num_fallback": len(self.fallback_layers),
                "layer_errors": self.layer_errors,
                "layer_distill_losses": self.layer_distill_losses,
                "avg_reconstruction_error": avg_err,
                "avg_distill_loss": avg_distill_loss,
                "sparsity_target": self.sparsity,
                "actual_sparsity": avg_actual_sparsity,
                "layer_actual_sparsities": self.layer_actual_sparsities,
                "structured_24": self.structured_24,
                "admm_iterations": self.max_iter,
                "dynamic_rho": self.dynamic_rho,
                "use_ste": self.use_ste,
                "recalibrate_ln": self.recalibrate_ln,
                "max_error": self.max_error,
                "fallback_to_original": self.fallback_to_original,
                "target_attention_only": self.target_attention_only,
                "lambda_distill": self.lambda_distill,
                "distill_batch_size": self.distill_batch_size,
                "distill_use_activation": self.distill_use_activation,
                "method": "admm_pruner_distill_ste_dynamic_rho",
            }
        )
        self.stats = stats
        self.model.compression_stats = stats
        return self.model, stats

class ARMORLinear(nn.Module):
    """
    Decoupled sparse + low-rank residual layer.

    Forward:
        y = x @ (sparse_weight ⊙ sparse_mask)^T
            + (scaling) * (x @ lora_A^T) @ lora_B^T
            + bias

    Invariants:
      - `sparse_weight` holds the ORIGINAL weight values (not pre-masked).
      - `sparse_mask` (non-persistent) holds a 0/1 mask applied once per forward.
      - `lora_alpha == rank` so `scaling == 1.0`, so the SVD residual is exact.
    """

    is_custom_attention: bool = True

    def __init__(
        self,
        in_features: int,
        out_features: int,
        rank: int,
        bias: bool = True,
        lora_alpha: float | None = None,
        dtype: torch.dtype = torch.float32,
        device: torch.device | None = None,
    ) -> None:
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.nf = out_features
        self.rank = max(1, min(rank, in_features, out_features))

        # AR-FIX-01: default alpha to rank so scaling is 1.0 by construction.
        self.lora_alpha = float(lora_alpha) if lora_alpha is not None else float(self.rank)
        self.scaling = (self.lora_alpha / self.rank) if self.rank > 0 else 1.0

        # AR-FIX-11: store raw (un-masked) weight.
        self.sparse_weight = nn.Parameter(
            torch.empty(out_features, in_features, dtype=dtype, device=device),
            requires_grad=False,
        )
        # AR-FIX-02: non-persistent mask.
        self.register_buffer(
            "sparse_mask",
            torch.ones(out_features, in_features, dtype=dtype, device=device),
            persistent=False,
        )

        self.lora_A = nn.Parameter(torch.empty(self.rank, in_features, dtype=dtype, device=device))
        self.lora_B = nn.Parameter(torch.empty(out_features, self.rank, dtype=dtype, device=device))

        if bias:
            self.bias = nn.Parameter(torch.empty(out_features, dtype=dtype, device=device))
        else:
            self.register_parameter("bias", None)

    @classmethod
    def from_linear(
        cls,
        linear: nn.Linear | Any,
        mask: torch.Tensor,
        lora_A: torch.Tensor,
        lora_B: torch.Tensor,
        lora_alpha: float | None = None,
    ) -> ARMORLinear:
        cls_name = linear.__class__.__name__.lower()
        is_conv1d = "conv1d" in cls_name or hasattr(linear, "nf")
        raw_w = linear.weight.data
        if is_conv1d:
            in_f, out_f = raw_w.shape[0], raw_w.shape[1]
            w_data = raw_w.t().contiguous()
        else:
            out_f, in_f = raw_w.shape
            w_data = raw_w

        # AR-FIX-10: clamp rank to actual dims.
        requested_rank = int(lora_A.shape[0])
        actual_rank = max(1, min(requested_rank, in_f, out_f))

        # AR-FIX-01: ensure scaling == 1.0 by default.
        effective_alpha = float(lora_alpha) if lora_alpha is not None else float(actual_rank)

        layer = cls(
            in_features=in_f,
            out_features=out_f,
            rank=actual_rank,
            bias=(getattr(linear, "bias", None) is not None and linear.bias is not None),
            lora_alpha=effective_alpha,
            dtype=raw_w.dtype,
            device=raw_w.device,
        )
        mask_dev = mask.to(dtype=w_data.dtype, device=w_data.device)
        with torch.no_grad():
            # AR-FIX-11: store raw weight, do not pre-mask.
            layer.sparse_weight.data.copy_(w_data)
            layer.sparse_mask.copy_(mask_dev)
            layer.lora_A.data.copy_(lora_A[:actual_rank].to(dtype=raw_w.dtype, device=raw_w.device))
            layer.lora_B.data.copy_(
                lora_B[:, :actual_rank].to(dtype=raw_w.dtype, device=raw_w.device)
            )
            if (
                getattr(linear, "bias", None) is not None
                and linear.bias is not None
                and layer.bias is not None
            ):
                layer.bias.data.copy_(linear.bias.data)
        return layer

    @property
    def weight(self) -> torch.Tensor:
        # AR-FIX-09: compute in fp32, cast back to sparse_weight dtype.
        with torch.no_grad():
            base = (self.sparse_weight.float() * self.sparse_mask.float()).to(
                self.sparse_weight.dtype
            )
            if self.lora_A is not None and self.lora_B is not None and self.rank > 0:
                lora_w = torch.matmul(self.lora_B.float(), self.lora_A.float()) * self.scaling
                return base + lora_w.to(base.dtype)
            return base

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        orig_shape = x.shape
        x_flat = x.reshape(-1, self.in_features)

        # 1. Sparse core. Mask applied once.
        w_eff = self.sparse_weight * self.sparse_mask
        out_sparse = F.linear(x_flat, w_eff)

        # 2. Low-rank residual.
        lr_dtype = self.lora_A.dtype
        x_lr = x_flat.to(lr_dtype) if x_flat.dtype != lr_dtype else x_flat
        out_lr = F.linear(F.linear(x_lr, self.lora_A), self.lora_B)
        # AR-FIX-03: explicit dtype of scaling.
        out_lr = (out_lr * self.scaling).to(out_sparse.dtype)

        out = out_sparse + out_lr
        if self.bias is not None:
            out = out + self.bias.to(out.dtype)
        return out.view(*orig_shape[:-1], self.out_features)

    def _collect_activations(self) -> dict[str, torch.Tensor]:
        """
        AR-FIX-12: Collect per-channel mean |x| instead of full activation tensors.
        Returns {layer_name: (in_features,) float32 tensor}.
        """
        self.model.eval()
        activations: dict[str, list[torch.Tensor]] = {}
        hooks = []

        def make_hook(layer_name: str):
            def hook(module, inp, outp):
                if inp and isinstance(inp[0], torch.Tensor):
                    x = inp[0].detach()
                    # Per-channel mean absolute value; collapse all dims but last.
                    if x.dim() >= 2:
                        col_mean = x.abs().mean(dim=tuple(range(x.dim() - 1)))
                    else:
                        col_mean = x.abs()
                    activations.setdefault(layer_name, []).append(col_mean.cpu())

            return hook

        for name, module in self.model.named_modules():
            if self._is_target_module(name, module):
                target_mod = getattr(module, "base_layer", getattr(module, "linear", module))
                hooks.append(target_mod.register_forward_hook(make_hook(name)))

                                }
                                self.model(**b_dev)
                            elif torch.is_tensor(batch):

                            elif isinstance(batch, (list, tuple)):
                                b_dev = [
                                    (v.to(self.target_device) if torch.is_tensor(v) else v)
                                    for v in batch
                                ]
                                self.model(*b_dev)
                        except Exception as b_err:
                            logger.debug("ARMOR calibration batch failed: %s", b_err)
                else:
                    calib_inputs = get_calibration_tensors(
                        self.model,
                        tokenizer=self.tokenizer,
                        calibration_loader=None,
                        calibration_texts=self.calibration_texts,
                        device=self.target_device,
                    )
                    try:
                        if isinstance(calib_inputs, dict):
                            self.model(**calib_inputs)
                        elif isinstance(calib_inputs, torch.Tensor):

                        elif isinstance(calib_inputs, (list, tuple)):
                            self.model(*calib_inputs)

                        logger.debug("ARMOR calibration pass exception: %s", e)
        finally:
            for h in hooks:
                h.remove()

        aggregated: dict[str, torch.Tensor] = {}

            device = raw_w.device

            W = raw_w.t().contiguous().clone().float() if is_conv1d else raw_w.clone().float()
            out_features, in_features = W.shape
            grad = getattr(weight_tensor, "grad", None)
            act = act_map.get(name, None)

            try:
                saliency = compute_resilient_taylor_saliency(W, grad=grad, activations=act)

                # AR-FIX-04: robust mask handling.
                if self.structured_24:
                    W_sparse, raw_mask = apply_24_structured_mask(W, saliency=saliency, dim=1)
                    if raw_mask is None:
                        mask = (W_sparse.abs() > 1e-12).float()
                    else:
                        if raw_mask.dtype == torch.bool:
                            mask = raw_mask.float()
                        else:
                            mask = (raw_mask > 0.5).float()
                else:
                    flat = saliency.reshape(-1)
                    k = max(1, int(flat.numel() * (1.0 - self.sparsity)))
                    thr = torch.topk(flat, k, largest=True).values[-1]
                    mask = (saliency >= thr).float()
                    W_sparse = W * mask

                # Residual low-rank.
                residual = W - W_sparse
                r = min(self.rank, residual.shape[0], residual.shape[1])
                try:
                    U, S, Vh = torch.linalg.svd(residual, full_matrices=False)
                    sqrt_S = torch.sqrt(torch.clamp(S[:r], min=1e-12))
                    lora_B = (U[:, :r] * sqrt_S.unsqueeze(0)).to(dtype=dtype, device=device)
                    lora_A = (sqrt_S.unsqueeze(1) * Vh[:r, :]).to(dtype=dtype, device=device)
                    correction = torch.matmul(lora_B.float(), lora_A.float())
                    total_correction_energy += torch.norm(correction).item()
                except Exception as e:
                    logger.debug("ARMOR residual SVD failed on %s: %s", name, e)
                    lora_A = torch.zeros(r, in_features, dtype=dtype, device=device)
                    lora_B = torch.zeros(out_features, r, dtype=dtype, device=device)

                # AR-FIX-01: alpha defaults to rank so scaling = 1.0.
                effective_alpha = self.lora_alpha if self.lora_alpha is not None else float(r)
                scaling = effective_alpha / r if r > 0 else 1.0
                W_recon = W_sparse + scaling * torch.matmul(lora_B.float(), lora_A.float())

                # AR-FIX-06: fast finite check.
                if not torch.isfinite(W_recon).all():
                    raise RuntimeError("non-finite reconstruction")

                w_norm = torch.norm(W, p="fro").clamp_min(1e-8)
                err = float(torch.norm(W - W_recon, p="fro").item() / w_norm)
                self.layer_errors[name] = err

                if err > self.max_error and self.fallback_to_original:
                    logger.warning(
                        "ARMOR reconstruction error %.4f on %s exceeded threshold %.4f.",
                        err,
                        name,
                        self.max_error,
                    )
                    self.fallback_layers.append(name)
                    for p in module.parameters():
                        p.requires_grad = False
                    continue

                orig_count = int(W.numel())
                base_effective = int((mask > 0).sum().item())
                corr_count = int(lora_A.numel() + lora_B.numel()) if self.decouple else 0

                total_base_sparse_effective += base_effective
                total_base_sparse_storage += orig_count
                total_correction += corr_count
                total_orig += orig_count

                actual_zero_frac = float((mask == 0).sum().item() / max(1, orig_count))
                eff_ratio = float((base_effective + corr_count) / max(1, orig_count))
                self.layer_actual_sparsities[name] = actual_zero_frac
                self.layer_effective_ratios[name] = eff_ratio

                if self.decouple:
                    armor_layer = ARMORLinear.from_linear(
                        target_mod,
                        mask=mask,
                        lora_A=lora_A,
                        lora_B=lora_B,
                        lora_alpha=effective_alpha,
                    )
                    if has_lora:
                        module.base_layer = armor_layer
                        for p_name, p in module.named_parameters():
                            if any(t in p_name.lower() for t in ("lora_a", "lora_b", "adapter")):
                                if not p_name.startswith("base_layer"):
                                    with torch.no_grad():
                                        p.data.zero_()
                                    p.requires_grad = False
                    elif name == "":
                        self.model = armor_layer
                    else:
                        replace_module(self.model, name, armor_layer)
                else:
                    # AR-FIX: mutation under no_grad. NOTE: this branch does NOT
                    # reduce parameter count; it is a numerical approximation only.
                    with torch.no_grad():
                        if is_conv1d:
                            weight_tensor.data.copy_(
                                W_recon.t().contiguous().to(dtype=dtype, device=device)
                            )
                        else:
                            weight_tensor.data.copy_(W_recon.to(dtype=dtype, device=device))

                self.compressed_layers.append(name)

            except Exception as e:
                logger.warning("ARMOR pruning failed on %s: %s", name, e)
                self.fallback_layers.append(name)
                for p in module.parameters():
                    p.requires_grad = False

        # AR-FIX-08: LayerNorm recalibration only when data available.
        if self.recalibrate_ln and (
            self.calibration_loader is not None or self.calibration_texts is not None
        ):
            try:
                recalibrate_layernorms(
                    self.model,
                    tokenizer=self.tokenizer,
                    calibration_loader=self.calibration_loader,
                    calibration_texts=self.calibration_texts,
                    device=self.target_device,
                )
            except Exception as e:
                logger.warning("ARMOR LayerNorm recalibration skipped: %s", e)

        try:
            self.model.eval()
            with torch.no_grad():
                first_param = next(self.model.parameters(), torch.tensor(0.0))
                dev = first_param.device

                    first_param.dtype if first_param.dtype.is_floating_point else torch.float32

                hidden_size = getattr(cfg, "hidden_size", None)
                try:
                    dummy_ids = torch.zeros((1, 8), dtype=torch.long, device=dev)
                    self.model(
                        input_ids=dummy_ids,
                        attention_mask=torch.ones_like(dummy_ids),
                    )
                except Exception:
                    if hidden_size is not None:
                        dummy_embeds = torch.randn(1, 8, hidden_size, dtype=p_dtype, device=dev)
                    else:
                        first_linear = next(
                            (
                                m
                                for m in self.model.modules()
                                if isinstance(m, (nn.Linear, ARMORLinear))
                            ),

            if self.layer_actual_sparsities
            else (0.5 if self.structured_24 else self.sparsity)
        )

        effective_param_ratio = (
            float((total_base_sparse_effective + total_correction) / max(1, total_orig))
            if total_orig > 0
            else 1.0
        )
        storage_param_ratio = (

        )

class ASTPrunedLoRALinear(nn.Module):
    """
    Linear layer with trainable base weights and dynamic 2:4 structured sparsity
    via Soft Straight-Through Estimator (STE) + optional trainable LoRA adapters.

    The base weight is stored in original coordinate basis with a true 2:4
    sparsity pattern enforced in the forward pass, while receiving dense
    unfrozen gradients in the backward pass.
    """

    def __init__(
        self,
        in_features: int,
        out_features: int,
        weight: torch.Tensor,
        bias: torch.Tensor | None = None,
        mask: torch.Tensor | None = None,
        lora_r: int = 0,
        lora_alpha: float = 1.0,
        device: torch.device | None = None,
        dtype: torch.dtype | None = None,
    ) -> None:
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.nf = out_features  # Conv1D compat
        self.lora_r = int(lora_r)
        self.lora_alpha = float(lora_alpha)
        self.scaling = (self.lora_alpha / self.lora_r) if self.lora_r > 0 else 1.0

        target_device = device if device is not None else weight.device
        target_dtype = dtype if dtype is not None else weight.dtype

        # FIXED: Ensure base weight is trainable with requires_grad=True so STE passes dense gradients back
        self.weight = nn.Parameter(
            weight.to(device=target_device, dtype=target_dtype),
            requires_grad=True,
        )

        if mask is not None:
            self.register_buffer("mask", mask.to(device=target_device, dtype=target_dtype))
        else:
            self.register_buffer("mask", torch.ones_like(self.weight))

        if bias is not None:
            self.bias = nn.Parameter(
                bias.to(device=target_device, dtype=target_dtype),
                requires_grad=True,
            )
        else:
            self.register_parameter("bias", None)

        if self.lora_r > 0:
            self.lora_A = nn.Parameter(
                torch.empty(
                    (self.lora_r, in_features),
                    dtype=torch.float32,
                    device=target_device,
                )
            )
            self.lora_B = nn.Parameter(
                torch.zeros(
                    (out_features, self.lora_r),
                    dtype=torch.float32,
                    device=target_device,
                )
            )
            nn.init.kaiming_uniform_(self.lora_A, a=math.sqrt(5))
        else:
            self.register_parameter("lora_A", None)
            self.register_parameter("lora_B", None)

    @classmethod
    def from_linear(
        cls,
        linear: nn.Module,
        weight: torch.Tensor,
        mask: torch.Tensor | None = None,
        lora_r: int = 0,
        lora_alpha: float = 1.0,
    ) -> ASTPrunedLoRALinear:
        bias = None
        if hasattr(linear, "bias") and linear.bias is not None:
            bias = linear.bias.detach()
        in_features = getattr(linear, "in_features", weight.shape[1])
        out_features = getattr(linear, "out_features", weight.shape[0])
        return cls(
            in_features=in_features,
            out_features=out_features,
            weight=weight,
            bias=bias,
            mask=mask,
            lora_r=lora_r,
            lora_alpha=lora_alpha,
            device=weight.device,
            dtype=weight.dtype,
        )

    def merge(self) -> None:
        """Merge LoRA into base weight in place."""
        if self.lora_r > 0 and self.lora_A is not None and self.lora_B is not None:
            with torch.no_grad():
                delta = (self.lora_B.float() @ self.lora_A.float()) * self.scaling
                self.weight.data = (self.weight.data.float() + delta).to(self.weight.dtype)
                self.lora_A = None
                self.lora_B = None
                self.lora_r = 0

    def merge_and_unload(self) -> None:
        self.merge()

    def unmerge(self) -> None:
        pass

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        orig_shape = x.shape
        x_flat = x.reshape(-1, self.in_features)

        # FIXED: Soft Straight-Through Estimator (STE) for 2:4 structured sparsity.
        # W_forward has strict 2:4 sparsity in forward pass, but d(W_forward)/d(W) = 1 in backward pass.
        if self.mask is not None:
            w_pruned = self.weight * self.mask
            w_forward = w_pruned + self.weight - self.weight.detach()
        else:
            w_forward = self.weight

        out = F.linear(x_flat, w_forward, self.bias)

        if self.lora_r > 0 and self.lora_A is not None and self.lora_B is not None:
            x_f = x_flat.float()
            lora_out = F.linear(F.linear(x_f, self.lora_A), self.lora_B)
            out = out + (self.scaling * lora_out).to(out.dtype)

        return out.reshape(*orig_shape[:-1], self.out_features)

        if self._is_gqa_asymmetric(name, module):
            return False
        if self.target_attention_only and not any(
            k in name.lower()
            for k in [
                "attn",
                "attention",
                "q_proj",
                "k_proj",
                "v_proj",
                "out_proj",
                "o_proj",
                "q_lin",
                "k_lin",
                "v_lin",
                "out_lin",
                "query",
                "key",
                "value",
                "c_attn",
                "c_proj",
            ]
        ):
            return False
        if self.target_modules is not None:
            return any(t in name for t in self.target_modules)
        return True

    def _collect_jacobian_and_activations(
        self,
    ) -> tuple[dict[str, torch.Tensor], dict[str, torch.Tensor]]:
        activations: dict[str, list[torch.Tensor]] = {}
        fisher_info: dict[str, torch.Tensor] = {}
        hooks = []

        def make_hook(name: str):
            def hook(module, inp, outp):
                if inp and isinstance(inp[0], torch.Tensor):
                    activations.setdefault(name, []).append(inp[0].detach().cpu())

            return hook

        for name, module in self.model.named_modules():
            if self._is_target_module(name, module):
                hooks.append(module.register_forward_hook(make_hook(name)))

        batches: list[Any] = []
        if self.calibration_loader is not None:
            for b in self.calibration_loader:
                batches.append(b)
                if len(batches) >= 4:
                    break
        if not batches:
            batches = [
                get_calibration_tensors(
                    self.model,
                    tokenizer=self.tokenizer,
                    calibration_loader=self.calibration_loader,
                    calibration_texts=self.calibration_texts,
                    device=self.target_device,
                )
            ]

        self.model.eval()
        n_batches_run = 0
        try:
            with torch.enable_grad():
                for batch_input in batches:
                    out = None
                    if isinstance(batch_input, dict):
                        moved = {
                            k: v.to(self.target_device) if hasattr(v, "to") else v
                            for k, v in batch_input.items()
                        }
                        out = self.model(**moved)
                    elif isinstance(batch_input, torch.Tensor):
                        out = self.model(batch_input.to(self.target_device))
                    elif isinstance(batch_input, (list, tuple)):
                        out = self.model(
                            *[
                                b.to(self.target_device) if hasattr(b, "to") else b
                                for b in batch_input
                            ]
                        )

                    if self.use_fisher and out is not None:
                        logits = None
                        if hasattr(out, "logits"):
                            logits = out.logits
                        elif isinstance(out, torch.Tensor):
                            logits = out
                        elif (
                            isinstance(out, (tuple, list))
                            and out
                            and isinstance(out[0], torch.Tensor)
                        ):
                            logits = out[0]

                        if logits is not None and logits.requires_grad:
                            loss = 0.5 * torch.sum(logits.pow(2))
                            self.model.zero_grad(set_to_none=True)
                            loss.backward()
                            for name, module in self.model.named_modules():
                                if not self._is_target_module(name, module):
                                    continue
                                target_w = getattr(module, "weight", None)
                                base = getattr(module, "base_layer", module)
                                if target_w is None:
                                    target_w = getattr(base, "weight", None)
                                if target_w is None or target_w.grad is None:
                                    continue
                                g = target_w.grad.detach()
                                g_sq = g.pow(2).cpu()
                                if name in fisher_info:
                                    fisher_info[name] = fisher_info[name] + g_sq
                                else:
                                    fisher_info[name] = g_sq
                            self.model.zero_grad(set_to_none=True)
                            n_batches_run += 1
        except Exception as e:
            logger.warning("AST Jacobian calibration failed: %s", e)
        finally:
            for h in hooks:
                h.remove()

        if n_batches_run >= 1:
            inv = 1.0 / float(n_batches_run)
            for k in list(fisher_info.keys()):
                fisher_info[k] = fisher_info[k] * inv
        else:
            logger.warning(
                "AST: no successful backward passes; saliency will fall back to |W| only."
            )

        cov_map: dict[str, torch.Tensor] = {}
        for name, act_list in activations.items():
            if not act_list:
                continue
            act_concat = torch.cat([a.reshape(-1, a.shape[-1]) for a in act_list], dim=0).float()
            N, D = act_concat.shape
            if N < 2:
                continue
            mean = torch.mean(act_concat, dim=0, keepdim=True)
            centered = act_concat - mean
            cov = torch.matmul(centered.T, centered) / float(N - 1)
            cov = (cov + cov.T) * 0.5
            cov_map[name] = cov
            del act_concat, centered, mean
            if name not in fisher_info:
                fisher_info[name] = torch.var(torch.zeros(1, D), dim=0, keepdim=True).expand(1, D)
        activations.clear()

        return cov_map, fisher_info

    @staticmethod
    def _pca_basis(cov: torch.Tensor, d_in: int, device: torch.device) -> torch.Tensor:
        cov = cov.to(device).float()
        if cov.shape != (d_in, d_in):
            return torch.eye(d_in, device=device)

        diag_mean = float(cov.diagonal().mean().item())
        eps = max(1e-8, 1e-6 * diag_mean)
        cov = cov + eps * torch.eye(d_in, device=device)

        try:
            eigvals, Q = torch.linalg.eigh(cov)
            idx = torch.argsort(eigvals, descending=True)
            Q = Q[:, idx]
            if torch.isnan(Q).any() or torch.isinf(Q).any():
                logger.warning("AST: PCA basis contained NaN/Inf; using identity.")
                return torch.eye(d_in, device=device)
            return Q.contiguous()
        except Exception as e:
            logger.warning("AST: eigh failed (%s); using identity.", e)
            return torch.eye(d_in, device=device)

    def compress(self, **kwargs: Any) -> tuple[nn.Module, dict[str, Any]]:
        from neuralops.core.model_utils import replace_module

        cov_map, fisher_map = self._collect_jacobian_and_activations()
        self.compressed_layers = []
        self.fallback_layers = []
        self.skipped_layers = []
        self.layer_errors = {}
        seen_weights: set = set()

        for name, module in list(self.model.named_modules()):
            if not self._is_target_module(name, module):
                continue
            if not hasattr(module, "weight") or module.weight is None:
                continue

            w_id = id(module.weight)
            if w_id in seen_weights:
                for p in module.parameters():
                    p.requires_grad = False
                self.skipped_layers.append(name)
                continue
            seen_weights.add(w_id)

            target_mod = getattr(module, "base_layer", module)
            raw_w = target_mod.weight.data
            cls_name = module.__class__.__name__.lower()
            is_conv1d = "conv1d" in cls_name or hasattr(module, "nf")

            # W: (d_out, d_in)
            W = raw_w.t().clone().float() if is_conv1d else raw_w.clone().float()
            d_out, d_in = W.shape
            device = W.device
            dtype = target_mod.weight.dtype

            # 1. Compute saliency in ORIGINAL basis
            fisher = fisher_map.get(name, None)
            if fisher is not None and self.use_fisher:
                F_mat = fisher.to(device).float()
                F_mat = torch.nan_to_num(F_mat, nan=0.0, posinf=0.0, neginf=0.0)

                if is_conv1d and F_mat.dim() == 2 and F_mat.shape == raw_w.shape:
                    F_mat = F_mat.t().contiguous()

                if F_mat.dim() == 2 and F_mat.shape == W.shape:
                    if self.use_pca and name in cov_map:
                        Q = self._pca_basis(cov_map[name], d_in, device)
                        F_rot = F_mat @ (Q * Q)
                        F_proj = F_rot @ Q.T
                    else:
                        F_proj = F_mat
                elif F_mat.shape[-1] == d_in and F_mat.dim() == 1:
                    F_proj = F_mat.reshape(1, d_in).expand(d_out, d_in)
                else:
                    F_proj = torch.ones_like(W)
                F_proj = torch.clamp(F_proj, min=1e-8)
                saliency = torch.abs(W) * torch.sqrt(F_proj)
            else:
                saliency = torch.abs(W)

            saliency = torch.nan_to_num(saliency, nan=0.0, posinf=1e4, neginf=0.0)

            # 2. Compute 2:4 structured mask on ORIGINAL weights
            W_pruned, mask_24 = apply_24_structured_mask(W, saliency=saliency, dim=self.dim)
            W_final = torch.nan_to_num(W_pruned, nan=0.0, posinf=0.0, neginf=0.0)
            mask_final = torch.nan_to_num(mask_24, nan=1.0, posinf=1.0, neginf=1.0)

            # 3. Error check
            w_norm = torch.linalg.norm(W).item()
            err = float(torch.linalg.norm(W - W_final).item() / max(w_norm, 1e-12))
            self.layer_errors[name] = err

            if err > self.max_error:
                logger.warning(
                    "AST: reconstruction error %.4f on %s exceeds %.4f.",
                    err,
                    name,
                    self.max_error,
                )
                if self.fallback_to_original:
                    logger.info("AST: falling back to original for %s.", name)
                    # Keep fallback layers trainable
                    for pn, p in module.named_parameters(recurse=False):
                        p.requires_grad = True
                    self.fallback_layers.append(name)
                    continue

            # 4. FIXED: Replace with ASTPrunedLoRALinear implementing Soft STE
            # Store dense trainable W in weight parameter and mask in buffer.
            # Base weights remain trainable (requires_grad=True) to absorb KD gradients.
            new_layer = ASTPrunedLoRALinear.from_linear(
                target_mod,
                weight=W.contiguous().to(dtype=dtype, device=device),
                mask=mask_final.contiguous().to(dtype=dtype, device=device),
                lora_r=self.lora_r,
                lora_alpha=self.lora_alpha,
            )
            # Ensure base weight is trainable
            new_layer.weight.requires_grad = True

            if hasattr(module, "base_layer"):
                if hasattr(module, "lora_A") and module.lora_A is not None:
                    src_a = getattr(module.lora_A, "weight", module.lora_A)
                    if (
                        isinstance(src_a, torch.Tensor)
                        and src_a.shape == new_layer.lora_A.shape
                    ):
                        new_layer.lora_A.data.copy_(src_a.to(new_layer.lora_A.dtype))
                if hasattr(module, "lora_B") and module.lora_B is not None:
                    src_b = getattr(module.lora_B, "weight", module.lora_B)
                    if (
                        isinstance(src_b, torch.Tensor)
                        and src_b.shape == new_layer.lora_B.shape
                    ):
                        new_layer.lora_B.data.copy_(src_b.to(new_layer.lora_B.dtype))

                module.base_layer = new_layer
            else:
                if name != "":
                    replace_module(self.model, name, new_layer)
                else:
                    self.model = new_layer

            self.compressed_layers.append(name)

        if self.recalibrate_ln and (
            self.calibration_loader is not None or self.calibration_texts is not None
        ):
            try:
                recalibrate_layernorms(
                    self.model,
                    tokenizer=self.tokenizer,
                    calibration_loader=self.calibration_loader,
                    calibration_texts=self.calibration_texts,
                    device=self.target_device,
                )
            except Exception as e:
                logger.warning("AST: LayerNorm recalibration skipped: %s", e)

        forward_check_success = False
        try:
            self.model.eval()
            with torch.no_grad():
                d = next(self.model.parameters()).device
                dummy = torch.randint(0, 100, (1, 16), device=d)
                mask = torch.ones_like(dummy)
                try:
                    self.model(input_ids=dummy, attention_mask=mask)
                except TypeError:
                    self.model(dummy)
            forward_check_success = True
        except Exception as e:
            logger.warning("AST: post-compression forward check failed: %s", e)

        if self.skipped_layers:
            logger.info(
                "AST: skipped %d layers (dup weights / target rules).",
                len(self.skipped_layers),
            )

        self._compressed_params = self._count_params(self.model)
        avg_err = (
            sum(self.layer_errors[k] for k in self.compressed_layers)
            / max(1, len(self.compressed_layers))
            if self.compressed_layers
            else 0.0
        )
        stats = self.get_compression_stats()
        stats.update(
            {
                "pruned_layers": len(self.compressed_layers),
                "compressed_layers": self.compressed_layers,
                "fallback_layers": self.fallback_layers,
                "skipped_layers": self.skipped_layers,
                "num_compressed": len(self.compressed_layers),
                "num_fallback": len(self.fallback_layers),
                "num_skipped": len(self.skipped_layers),
                "layer_errors": self.layer_errors,
                "avg_reconstruction_error": avg_err,
                "block_size": self.block_size,
                "n_keep": self.n_keep,
                "pattern": f"{self.n_keep}:{self.block_size}",
                "use_pca": self.use_pca,
                "use_fisher": self.use_fisher,
                "recalibrate_ln": self.recalibrate_ln,
                "max_error": self.max_error,
                "fallback_to_original": self.fallback_to_original,
                "lora_r": self.lora_r,
                "lora_alpha": self.lora_alpha,
                "forward_check_success": forward_check_success,
                "method": "ast_jacobian_pca_2_4_pruning_with_soft_ste",
            }
        )
        self.stats = stats
        logger.info(
            "Compressor ast finished: %d compressed, %d fallback, avg_err=%.4f, max_err_threshold=%.4f",
            len(self.compressed_layers),
            len(self.fallback_layers),
            avg_err,
            self.max_error,
        )
        return self.model, stats

class BaseCompressor:
    """Base class for all compressors. Not a nn.Module to avoid parameter storage."""

    def __init__(self, model: nn.Module, name: str = "base", **kwargs) -> None:
        self.model = model
        self.name = name
        self.kwargs = kwargs
        self._original_params = self._count_params(model)

    def __call__(self, *args, **kwargs) -> Any:
        """Allow compressor instance to act as a callable model forward wrapper."""
        if not getattr(self, "_compressed_once", False):
            self.compress()
            self._compressed_once = True
        return self.model(*args, **kwargs)

    @abstractmethod
    def compress(self) -> tuple[nn.Module, dict[str, Any]]:
        """Return (compressed_model, statistics)."""
        raise NotImplementedError("Subclasses of BaseCompressor must implement compress()")

    def _count_params(self, model: nn.Module) -> int:
        """count params."""
        return sum(p.numel() for p in model.parameters())

    def get_compression_stats(self) -> dict[str, Any]:
        """Return the compression stats."""
        compressed = self._count_params(self.model)
        orig = self._original_params
        return {
            "compressor_name": self.name,
            "original_parameters": orig,
            "compressed_parameters": compressed,
            "saved_parameters": orig - compressed,
            "compression_ratio": compressed / orig if orig > 0 else 1.0,
            "reduction_percentage": ((orig - compressed) / orig * 100 if orig > 0 else 0.0),
        }

    def summary(self) -> str:
        """Summary."""
        s = self.get_compression_stats()
        return (

class BonsaiPrunedLoRALinear(nn.Module):
    """
    Linear layer with progressive sparsity scheduling:
    - Base weight starts dense and trainable (requires_grad=True).
    - Mask smoothly increases sparsity from 0.0 to target_sp via cubic warmup over target_steps.
    - Eliminates catastrophic pruning damage at step 0.
    """

    is_custom_attention: bool = True

    def __init__(
        self,
        in_features: int,
        out_features: int,
        bias: bool = True,
        target_sp: float = 0.5,
        target_steps: int = 200,
        structured_24: bool = False,
        block_size: tuple[int, int] = (1, 1),
        lora_r: int = 0,
        lora_alpha: float = 1.0,
        freeze_base: bool = False,
        dtype: torch.dtype | None = None,
        device: torch.device | None = None,
    ) -> None:
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.nf = out_features  # Conv1D compatibility
        self.target_sp = float(target_sp)
        self.target_steps = max(1, int(target_steps))
        self.structured_24 = structured_24
        self.block_size = block_size
        self.current_step = 0

        self.lora_r = max(0, int(lora_r))
        self.lora_alpha = float(lora_alpha)
        self.scaling = (self.lora_alpha / max(1, self.lora_r)) if self.lora_r > 0 else 1.0

        weight_dtype = dtype if dtype is not None else torch.float32

        # FIXED: Base weights remain trainable so real KD/self-training backprop updates weights!
        self.weight = nn.Parameter(
            torch.empty(out_features, in_features, dtype=weight_dtype, device=device),
            requires_grad=not freeze_base,
        )

        # FIXED: Mask starts at all ones (0% initial sparsity = 0 initial damage)
        self.register_buffer(
            "mask",
            torch.ones(out_features, in_features, dtype=weight_dtype, device=device),
            persistent=True,
        )

        if bias:
            self.bias = nn.Parameter(torch.zeros(out_features, dtype=weight_dtype, device=device))
        else:
            self.register_parameter("bias", None)

        if self.lora_r > 0:
            self.lora_A = nn.Parameter(
                torch.empty(self.lora_r, in_features, dtype=weight_dtype, device=device)
            )
            self.lora_B = nn.Parameter(
                torch.empty(out_features, self.lora_r, dtype=weight_dtype, device=device)
            )
            with torch.no_grad():
                tmp_A = self.lora_A.data.float()
                nn.init.kaiming_uniform_(tmp_A, a=math.sqrt(5))
                self.lora_A.data.copy_(tmp_A.to(weight_dtype))
                self.lora_B.data.zero_()
        else:
            self.register_parameter("lora_A", None)
            self.register_parameter("lora_B", None)

    @property
    def effective_weight(self) -> torch.Tensor:
        W_eff = self.weight * self.mask
        if self.lora_r > 0 and self.lora_A is not None and self.lora_B is not None:
            W_eff = W_eff + (torch.matmul(self.lora_B, self.lora_A) * self.scaling)
        return W_eff

        orig_shape = x.shape
        x_flat = x.reshape(-1, self.in_features)

        # FIXED: Progressive Sparsity Scheduler during training forward passes
        # Gradually ramp sparsity from 0.0 to target_sp via cubic warmup schedule.
        if self.training and self.current_step < self.target_steps:
            with torch.no_grad():
                t_ratio = float(self.current_step + 1) / float(self.target_steps)
                current_sp = self.target_sp * (t_ratio ** 3)

                    _, new_mask = apply_24_structured_mask(self.weight, dim=1)
                elif self.block_size != (1, 1) and current_sp > 0.01:
                    _, new_mask = apply_block_mask(
                        self.weight,
                        block_size=self.block_size,
                        sparsity=current_sp,
                    )
                elif current_sp > 0.001:

                    k = max(1, int(flat.numel() * (1.0 - current_sp)))
                    thr = torch.topk(flat, k, largest=True).values[-1]
                    new_mask = (torch.abs(self.weight) >= thr).to(self.weight.dtype)
                else:
                    new_mask = torch.ones_like(self.mask)

                self.mask.copy_(new_mask)
                self.current_step += 1

        out = F.linear(x_flat, W_sparse, self.bias)

            lora_out = F.linear(F.linear(x_flat, self.lora_A), self.lora_B) * self.scaling
            out = out + lora_out

        return out.view(*orig_shape[:-1], self.out_features)

            bias=bias_present,

            lora_alpha=lora_alpha,

                layer.mask.copy_(mask.to(layer.mask.dtype))
            if bias_present and layer.bias is not None and linear.bias is not None:
                layer.bias.data.copy_(linear.bias.data)
        return layer

        """
        Progressive Sparsity Initialization:
        Replaces target layers with BonsaiPrunedLoRALinear modules that smoothly
        increase sparsity over training steps via cubic warmup, avoiding step-0 shock.
        """
        self.pruned_layers = []
        self.protected_layers = []
        self.skipped_layers = []
        self.fallback_layers = []
        self.layer_errors = {}
        self.layer_sparsities = {}

        for name, module in list(self.model.named_modules()):
            cls_name = module.__class__.__name__
            is_linear_like = (
                isinstance(module, nn.Linear)
                or "conv1d" in cls_name.lower()
                or hasattr(module, "nf")
            )
            if not is_linear_like or isinstance(module, BonsaiPrunedLoRALinear):
                continue

            if self._is_protected_layer(name, module):
                for p in module.parameters():
                    p.requires_grad = True
                self.protected_layers.append(name)
                continue

            target_mod = getattr(module, "base_layer", module)
            raw_w = target_mod.weight.data
            dtype = raw_w.dtype
            device = raw_w.device
            is_conv1d = "conv1d" in cls_name.lower() or hasattr(module, "nf")

            W0 = raw_w.t().float().contiguous() if is_conv1d else raw_w.float().contiguous()
            out_f, in_f = W0.shape

            if (out_f * in_f) < self.min_numel:
                for p in module.parameters():
                    p.requires_grad = True
                self.skipped_layers.append(name)
                continue

            target_sp = self.sparsity

            # FIXED: Deploy Progressive BonsaiPrunedLoRALinear.
            # Base weights start 100% dense (0% damage) and remain trainable (requires_grad=True).
            # Sparsity increases smoothly over self.target_steps in forward passes.
            pruned_layer = BonsaiPrunedLoRALinear.from_linear(
                linear=target_mod,
                weight=W0.to(dtype=dtype, device=device),
                mask=torch.ones(out_f, in_f, dtype=dtype, device=device),
                target_sp=target_sp,
                target_steps=self.target_steps,
                structured_24=self.structured_24,
                block_size=self.block_size,
                lora_r=self.lora_r,
                lora_alpha=self.lora_alpha,
                freeze_base=False,
            )

            # Ensure all weights remain trainable
            pruned_layer.weight.requires_grad = True

            if hasattr(module, "base_layer"):
                if (
                    hasattr(module, "lora_A")
                    and module.lora_A is not None
                    and pruned_layer.lora_A is not None
                ):
                    src_A = getattr(module.lora_A, "weight", module.lora_A)
                    pruned_layer.lora_A.data.copy_(src_A.data)
                if (
                    hasattr(module, "lora_B")
                    and module.lora_B is not None
                    and pruned_layer.lora_B is not None

                    src_B = getattr(module.lora_B, "weight", module.lora_B)
                    pruned_layer.lora_B.data.copy_(src_B.data)

                pruned_layer = module

            self.layer_errors[name] = 0.0  # 0 initial reconstruction error
            self.layer_sparsities[name] = target_sp

                recalibrate_layernorms(
                    self.model,
                    tokenizer=self.tokenizer,
                    calibration_loader=self.calibration_loader,
                    calibration_texts=self.calibration_texts,

                "lora_r": self.lora_r,

class Conv1D(nn.Module):
    """
    1D-convolutional layer as defined by Radford et al. for OpenAI GPT (and also used in GPT-2).

    Basically works like a linear layer but the weights are transposed.

    Args:
        nf (`int`): The number of output features.
        nx (`int`): The number of input features.
    """

    def __init__(self, nf, nx):
        super().__init__()
        self.nf = nf
        self.nx = nx
        self.weight = nn.Parameter(torch.empty(nx, nf))
        self.bias = nn.Parameter(torch.zeros(nf))
        nn.init.normal_(self.weight, std=0.02)

    def __repr__(self) -> str:
        return "Conv1D(nf={nf}, nx={nx})".format(**self.__dict__)

    def forward(self, x):
        size_out = x.size()[:-1] + (self.nf,)
        x = torch.addmm(self.bias, x.view(-1, x.size(-1)), self.weight)
        x = x.view(size_out)
        return x

    def _normalize_calibration_batches(self, calib_inputs: Any) -> list[Any]:

    def _collect_layer_activations(self) -> dict[str, torch.Tensor]:
        """
        Capture per-feature activation variance across calibration batches.
        Returns: {name: variance_tensor of shape (in_features,)}
        """
        self.model.eval()
        variance_sum: dict[str, torch.Tensor] = {}
        batch_count: dict[str, int] = {}
        hooks = []

        def make_hook(name: str):
            def hook(module, inputs, output):
                if not inputs or not isinstance(inputs[0], torch.Tensor):
                    return
                inp = inputs[0].detach().float()
                target_mod = getattr(module, "base_layer", module)
                cls_name = target_mod.__class__.__name__.lower()
                is_conv1d = "conv1d" in cls_name or hasattr(target_mod, "nf")
                in_f = target_mod.weight.shape[0] if is_conv1d else target_mod.weight.shape[1]
                flat = inp.reshape(-1, in_f).cpu()
                if flat.size(0) >= 2:
                    var = flat.var(dim=0)
                else:
                    var = torch.zeros(in_f)
                if name not in variance_sum:
                    variance_sum[name] = var
                    batch_count[name] = 1
                else:
                    variance_sum[name] += var
                    batch_count[name] += 1

            return hook

        for name, module in self.model.named_modules():
            if self._is_target_module(name, module):
                hooks.append(module.register_forward_hook(make_hook(name)))

        try:
            calib_inputs = get_calibration_tensors(
                self.model,
                tokenizer=self.tokenizer,
                calibration_loader=self.calibration_loader,
                calibration_texts=self.calibration_texts,
                device=self.target_device,
            )
            batches = self._normalize_calibration_batches(calib_inputs)
            n = min(len(batches), self.num_calibration_batches)

            with torch.no_grad():
                for i in range(n):
                    b = batches[i]
                    try:
                        if isinstance(b, dict):
                            self.model(**b)
                        elif isinstance(b, torch.Tensor):
                            self.model(b)
                        elif isinstance(b, (list, tuple)):
                            self.model(*b)
                    except Exception as e:
                        logger.debug("FLAP calibration batch %d failed: %s", i, e)
        finally:
            for h in hooks:
                h.remove()

        aggregated: dict[str, torch.Tensor] = {}
        for name, vsum in variance_sum.items():
            aggregated[name] = (vsum / max(1, batch_count[name])).to(self.target_device)
        return aggregated

    def _compute_fluctuation_metric(
        self,
        weight: torch.Tensor,
        activations: torch.Tensor | None,
    ) -> torch.Tensor:
        """Per-column fluctuation saliency score in fp32."""
        d_out, d_in = weight.shape
        device = weight.device

        if activations is not None and activations.numel() > 0:
            inp_var = activations.reshape(-1).float().to(device)
            if inp_var.shape[0] != d_in:
                inp_var = torch.ones(d_in, device=device)
            else:
                inp_var = inp_var.clamp(min=1e-8)
        else:
            inp_var = torch.ones(d_in, device=device)

        W = weight.float()
        if self.metric_type == "IFV":
            scores = inp_var
        elif self.metric_type == "WIFV":
            scores = inp_var * torch.sum(W.pow(2), dim=0)
        elif self.metric_type == "WIFN":
            scores = torch.sum(torch.abs(W), dim=0) * torch.sqrt(inp_var)
        else:
            scores = inp_var * torch.sum(W.pow(2), dim=0)

        return scores

    # Target check (MS-FIX-16: cached)

    def _is_target_module(self, name: str, module: nn.Module, _debug: bool = False) -> bool:
        name_lower = name.lower()
        if any(part in name.split(".") for part in ("lora_A", "lora_B", "base_layer")):
            if _debug:
                logger.info(f"  [skip] {name}: lora/base_layer part")
            return False

        cls_name = module.__class__.__name__
        if cls_name in (
            "DynamicSVDLinear",
            "SVDLinear",
            "SVDLinearLayer",
            "FactorisedLinear",
            "FactorizedLinear",
            "LowRankPlusSparseLinear",
            "MambaShedderPrunedLinear",
        ):
            if _debug:
                logger.info(f"  [skip] {name}: already compressed ({cls_name})")
            return False

        if not self.skip_should_compress_filter:
            try:
                if not should_compress_linear(self.model, name):
                    if _debug:
                        logger.info(f"  [skip] {name}: should_compress_linear=False")
                    return False
            except Exception as e:
                if _debug:
                    logger.info(f"  [skip] {name}: should_compress_linear raised {e}")
                return False

        base_mod = getattr(module, "base_layer", module)
        cls_lower = base_mod.__class__.__name__.lower()
        is_linear_like = (
            isinstance(base_mod, nn.Linear) or "conv1d" in cls_lower or hasattr(base_mod, "nf")
        )
        if not is_linear_like:
            if _debug:
                logger.info(f"  [skip] {name}: not linear-like")
            return False

        w = getattr(base_mod, "weight", None)
        if w is None:
            if _debug:
                logger.info(f"  [skip] {name}: weight None")
            return False
        if min(w.shape[0], w.shape[1]) < self.min_layer_dim:
            if _debug:
                logger.info(f"  [skip] {name}: dim too small {tuple(w.shape)}")
            return False

        if self.target_attention_only and not any(k in name_lower for k in _ATTN_NAMES):
            if _debug:
                logger.info(f"  [skip] {name}: not attention")
            return False

        if self.target_modules is not None:
            ok = any(t in name for t in self.target_modules)
            if not ok and _debug:
                logger.info(f"  [skip] {name}: not in target_modules")
            return ok

        if _debug:
            logger.info(f"  [TARGET] {name}: shape={tuple(w.shape)}")
        return True

    # ------------------------------------------------------------------
    # Activation collection
    # ------------------------------------------------------------------
    def _collect_activations(self, target_names: list[str]) -> dict[str, torch.Tensor]:
        self.model.eval()
        accum: dict[str, list[torch.Tensor]] = {}
        hooks = []

        def make_hook(layer_name: str):
            def hook(module, inputs, output):
                if inputs and isinstance(inputs[0], torch.Tensor):
                    inp = inputs[0].detach()
                    if inp.dim() >= 2 and inp.shape[0] * inp.shape[1] > 32768:
                        inp = inp.reshape(-1, inp.shape[-1])[:32768].unsqueeze(0)
                    accum.setdefault(layer_name, []).append(inp.cpu())

            return hook

        target_set = set(target_names)
        for name, module in self.model.named_modules():
            if name in target_set:
                hooks.append(module.register_forward_hook(make_hook(name)))

        try:
            calib_inputs = get_calibration_tensors(
                self.model,
                tokenizer=self.tokenizer,
                calibration_loader=self.calibration_loader,
                calibration_texts=self.calibration_texts,
                device=self.target_device,
            )
            batches = self._normalize_calibration_batches(calib_inputs)
            n_batches = min(len(batches), self.max_calibration_batches)

            with torch.no_grad():
                for i in range(n_batches):
                    batch = batches[i]
                    try:
                        if isinstance(batch, dict):
                            self.model(**batch)
                        elif isinstance(batch, torch.Tensor):
                            self.model(batch)
                        elif isinstance(batch, (list, tuple)):
                            self.model(*batch)
                    except Exception as e:
                        logger.debug("Calibration batch %d failed: %s", i, e)
        finally:
            for h in hooks:
                h.remove()

        aggregated: dict[str, torch.Tensor] = {}
        for name, act_list in accum.items():
            if act_list:
                try:
                    aggregated[name] = torch.cat(act_list, dim=0)
                except Exception as e:
                    logger.debug("Activation concat failed for %s: %s", name, e)
        return aggregated

    @staticmethod
    def _normalize_calibration_batches(calib_inputs: Any) -> list[Any]:
        if calib_inputs is None:
            return []
        if isinstance(calib_inputs, (list, tuple)):
            if len(calib_inputs) == 0:
                return []
            first = calib_inputs[0]
            if isinstance(first, (dict, torch.Tensor)):
                return list(calib_inputs)
            if isinstance(first, (list, tuple)):
                return [calib_inputs]
            return [calib_inputs]
        if isinstance(calib_inputs, (dict, torch.Tensor)):
            return [calib_inputs]
        return [calib_inputs]

    # ------------------------------------------------------------------
    # Layer shedding
    # ------------------------------------------------------------------
    def _apply_layer_shedding(self) -> int:
        if self.shed_layers <= 0:
            return 0
        if getattr(self.model, "_mamba_shed_done", False):
            logger.info("MambaShedder: layer shedding already applied; skipping.")
            return 0

        target_container = None
        container_parent = None
        container_attr = None

        candidates = [
            "backbone.layers",
            "transformer.layer",
            "transformer.h",
            "model.layers",
            "encoder.layer",
            "layers",
            "h",
        ]

        def _resolve(path: str):
            parts = path.split(".")
            parent = self.model
            for p in parts[:-1]:
                parent = getattr(parent, p, None)
                if parent is None:
                    return None, None
            container = getattr(parent, parts[-1], None)
            if isinstance(container, (nn.ModuleList, list)):
                return parent, parts[-1]
            return None, None

        for path in candidates:
            parent, attr = _resolve(path)
            if parent is None:
                continue
            container = getattr(parent, attr)
            if len(container) > self.shed_layers:
                target_container = container
                container_parent = parent
                container_attr = attr
                break

        if target_container is None:
            # Generic search: longest module list.
            best_len = 0
            for name, mod in self.model.named_modules():
                if not isinstance(mod, nn.ModuleList):
                    continue
                if len(mod) <= self.shed_layers or len(mod) <= best_len:
                    continue
                parent, attr = _resolve(name)
                if parent is None:
                    continue
                target_container = mod
                container_parent = parent
                container_attr = attr
                best_len = len(mod)

        if target_container is None or len(target_container) <= self.shed_layers:
            logger.warning("MambaShedder: no suitable ModuleList found.")
            return 0

        orig_len = len(target_container)
        keep_len = orig_len - self.shed_layers

        # Always keep first and last (per Mamba-Shedder paper).
        keep_set = {0, orig_len - 1}
        n_middle = keep_len - len(keep_set)
        if n_middle > 0 and orig_len > 2:
            middle = list(range(1, orig_len - 1))
            rng = random.Random(42)
            middle_kept = sorted(rng.sample(middle, min(n_middle, len(middle))))
            keep_set.update(middle_kept)
        else:
            keep_set = set(range(keep_len))

        kept_indices = sorted(keep_set)[:keep_len]
        old_container = getattr(container_parent, container_attr)
        new_layers = nn.ModuleList([target_container[i] for i in kept_indices])
        setattr(container_parent, container_attr, new_layers)

        # Validation with revert
        if self.validate_after_shed:
            try:
                self.model.eval()
                dummy = torch.randint(0, 100, (1, 8), device=self.target_device)
                with torch.no_grad():
                    try:
                        self.model(input_ids=dummy)
                    except Exception:
                        try:
                            self.model(dummy)
                        except Exception:
                            cfg = getattr(self.model, "config", None)
                            hidden = getattr(cfg, "hidden_size", None) if cfg else None
                            if hidden is None:
                                first_lin = next(
                                    (m for m in self.model.modules() if isinstance(m, nn.Linear)),
                                    None,
                                )
                                hidden = getattr(first_lin, "in_features", 8)
                            dt = next(self.model.parameters()).dtype
                            self.model(
                                torch.randn(1, 8, hidden, device=self.target_device, dtype=dt)
                            )
            except Exception as e:
                logger.warning("MambaShedder: shedding validation failed (%s); reverting.", e)
                setattr(container_parent, container_attr, old_container)
                return 0

        shed_done = orig_len - len(new_layers)
        logger.info(
            "MambaShedder: shed %d layers (%d → %d).",
            shed_done,
            orig_len,
            len(new_layers),
        )
        self.model._mamba_shed_done = True
        return shed_done

    # ------------------------------------------------------------------
    # Main compress
    # ------------------------------------------------------------------
    def compress(self, **kwargs: Any) -> tuple[nn.Module, dict[str, Any]]:
        from neuralops.core.model_utils import replace_module

        shed_done = self._apply_layer_shedding()

        # Single-pass target discovery (MS-FIX-16)
        targets = [
            (name, module)
            for name, module in self.model.named_modules()
            if self._is_target_module(name, module)
        ]

        if not targets:
            logger.warning("MambaShedder: no target layers found.")
            n = 0
            for name, module in self.model.named_modules():
                n += 1
                self._is_target_module(name, module, _debug=True)
            logger.warning("MambaShedder diagnostic: %d modules scanned, 0 targets.", n)

        activations_map = self._collect_activations([n for n, _ in targets])

        pruned_layers_count = 0
        self.layer_errors = {}
        layer_details: dict[str, Any] = {}

        total_orig_bytes = 0
        total_comp_bytes = 0
        seen_weight_ids: set[int] = set()

        for name, module in targets:
            base_mod = getattr(module, "base_layer", module)
            w = getattr(base_mod, "weight", None)
            if w is None:
                continue
            w_id = id(w)
            if w_id in seen_weight_ids:
                continue
            seen_weight_ids.add(w_id)

            raw_w = w.data
            device = raw_w.device
            dtype = raw_w.dtype
            cls_name = base_mod.__class__.__name__.lower()
            is_conv1d = "conv1d" in cls_name or hasattr(base_mod, "nf")

            W = raw_w.t().clone().float() if is_conv1d else raw_w.clone().float()
            d_out, d_in = W.shape

            orig_bytes = raw_w.numel() * raw_w.element_size()
            if getattr(base_mod, "bias", None) is not None:
                orig_bytes += base_mod.bias.numel() * base_mod.bias.element_size()
            total_orig_bytes += orig_bytes

            # Sensitivity: Fisher or weight-norm.
            act = activations_map.get(name, None)
            used_act_sensitivity = False
            if act is not None and act.numel() > 0:
                try:
                    act_flat = act.reshape(-1, act.shape[-1]).float().to(device)
                    if act_flat.shape[-1] == d_in:
                        fisher = (act_flat**2).mean(dim=0).clamp(min=1e-8)
                        sens = fisher.sqrt()
                        used_act_sensitivity = True
                    else:
                        sens = torch.norm(W, dim=0)
                except Exception as e:
                    logger.debug("Fisher computation failed for %s: %s", name, e)
                    sens = torch.norm(W, dim=0)
            else:
                sens = torch.norm(W, dim=0)

            # Decide kept output channels (or 2:4 mask).
            if self.structured_24:
                # 2:4 → no storage savings (dense storage kept). Fallback to
                # in-place weight modification.
                saliency = sens.unsqueeze(0).expand_as(W)
                try:
                    W_pruned, _ = apply_24_structured_mask(W, saliency=saliency, dim=1)
                except Exception as e:
                    logger.warning("2:4 pruning failed for %s: %s", name, e)
                    total_comp_bytes += orig_bytes
                    continue

                norm_orig = torch.norm(W, p="fro").clamp_min(1e-12)
                rel_err = (torch.norm(W - W_pruned, p="fro") / norm_orig).item()
                self.layer_errors[name] = rel_err

                if rel_err > self.max_error and self.fallback_to_original:
                    total_comp_bytes += orig_bytes
                    continue

                final_w = W_pruned.t().contiguous() if is_conv1d else W_pruned
                if final_w.shape != raw_w.shape:
                    total_comp_bytes += orig_bytes
                    continue

                with torch.no_grad():
                    base_mod.weight.data.copy_(final_w.to(dtype=dtype, device=device))
                    base_mod.weight.requires_grad = False
                    if getattr(base_mod, "bias", None) is not None:
                        base_mod.bias.requires_grad = False

                pruned_layers_count += 1
                total_comp_bytes += orig_bytes  # no real storage change
                layer_details[name] = {
                    "rel_error": rel_err,
                    "used_act_sensitivity": used_act_sensitivity,
                    "storage_bytes_before": orig_bytes,
                    "storage_bytes_after": orig_bytes,
                    "note": "2:4 sparsity: no storage savings",
                }
            else:
                # Channel pruning → real storage savings.
                n_keep = max(1, int(d_out * (1.0 - self.sparsity)))
                _, keep_idx = torch.topk(sens, k=n_keep, largest=True)
                keep_idx, _ = torch.sort(keep_idx)
                keep_idx = keep_idx.to(device)

                W_kept = W.index_select(0, keep_idx).contiguous()
                W_pruned_full = torch.zeros_like(W)
                W_pruned_full.index_copy_(0, keep_idx, W_kept)

                norm_orig = torch.norm(W, p="fro").clamp_min(1e-12)
                rel_err = (torch.norm(W - W_pruned_full, p="fro") / norm_orig).item()
                self.layer_errors[name] = rel_err

                if rel_err > self.max_error and self.fallback_to_original:
                    logger.warning(
                        "MambaShedder: err %.4f on %s > %.4f; keeping original.",
                        rel_err,
                        name,
                        self.max_error,
                    )
                    total_comp_bytes += orig_bytes
                    continue

                bias_kept = None
                if getattr(base_mod, "bias", None) is not None:
                    b_full = base_mod.bias.data.float()
                    bias_kept = (
                        b_full.index_select(0, keep_idx).contiguous().to(dtype=dtype, device=device)
                    )

                # Existing LoRA
                lora_A_src = getattr(module, "lora_A", None)
                lora_B_src = getattr(module, "lora_B", None)
                eff_lora_r = self.lora_r
                if lora_A_src is not None and hasattr(lora_A_src, "weight"):
                    eff_lora_r = int(lora_A_src.weight.shape[0])

                new_layer = MambaShedderPrunedLinear(
                    in_features=d_in,
                    original_out_features=d_out,
                    W_kept=W_kept.to(dtype=dtype, device=device),
                    keep_idx=keep_idx.cpu(),
                    bias_kept=bias_kept,
                    lora_r=eff_lora_r,
                    lora_alpha=self.lora_alpha,
                )

                # Preserve LoRA weights if shapes match.
                if (
                    lora_A_src is not None
                    and lora_B_src is not None
                    and new_layer.lora_A is not None
                ):
                    with torch.no_grad():
                        src_A = (
                            lora_A_src.weight.data
                            if hasattr(lora_A_src, "weight")
                            else lora_A_src.data
                        )
                        src_B = (
                            lora_B_src.weight.data
                            if hasattr(lora_B_src, "weight")
                            else lora_B_src.data
                        )
                        if (
                            src_A.shape == new_layer.lora_A.shape
                            and src_B.shape == new_layer.lora_B.shape
                        ):
                            new_layer.lora_A.copy_(
                                src_A.to(
                                    new_layer.lora_A.device,
                                    dtype=new_layer.lora_A.dtype,
                                )
                            )
                            new_layer.lora_B.copy_(
                                src_B.to(
                                    new_layer.lora_B.device,
                                    dtype=new_layer.lora_B.dtype,
                                )
                            )

                # Splice with wrapper preservation (MS-FIX-18)
                if hasattr(module, "base_layer"):
                    module.base_layer = new_layer
                else:
                    replace_module(self.model, name, new_layer)

                pruned_layers_count += 1
                total_comp_bytes += new_layer.storage_bytes()
                layer_details[name] = {
                    "rel_error": rel_err,
                    "used_act_sensitivity": used_act_sensitivity,
                    "storage_bytes_before": orig_bytes,
                    "storage_bytes_after": new_layer.storage_bytes(),
                    "keep_ratio": n_keep / d_out,
                }

        # ------------------------------------------------------------------
        # Forward sanity check (MS-FIX-17)
        # ------------------------------------------------------------------
        forward_check_success = True
        try:
            with torch.no_grad():
                dev = next(self.model.parameters(), torch.tensor(0.0)).device
                dummy = torch.randint(0, 100, (1, 8), device=dev, dtype=torch.long)
                try:
                    self.model(input_ids=dummy)
                except Exception:
                    try:
                        self.model(dummy)
                    except Exception:
                        cfg = getattr(self.model, "config", None)
                        hidden = getattr(cfg, "hidden_size", None) if cfg else None
                        if hidden is None:
                            first_lin = next(
                                (m for m in self.model.modules() if isinstance(m, nn.Linear)),
                                None,
                            )
                            hidden = getattr(first_lin, "in_features", 8)
                        dt = next(self.model.parameters()).dtype
                        self.model(torch.randn(1, 8, hidden, device=dev, dtype=dt))
        except Exception as e:
            logger.warning("MambaShedder forward sanity check failed: %s", e)
            forward_check_success = False

        # ------------------------------------------------------------------
        # Stats
        # ------------------------------------------------------------------
        try:
            self._compressed_params = self._count_params(self.model)
        except Exception:
            self._compressed_params = sum(p.numel() for p in self.model.parameters())

        avg_err = (
            sum(self.layer_errors.values()) / max(1, len(self.layer_errors))
            if self.layer_errors
            else 0.0
        )

        storage_ratio = total_orig_bytes / max(1, total_comp_bytes) if total_comp_bytes > 0 else 1.0
        effective_ratio = (
            total_comp_bytes / max(1, total_orig_bytes) if total_orig_bytes > 0 else 1.0
        )

        try:
            stats = self.get_compression_stats()
        except Exception:
            stats = {}

        stats.update(
            {
                "pruned_layers": pruned_layers_count,
                "shed_layers": shed_done,
                "sparsity_target": self.sparsity,
                "structured_24": self.structured_24,
                "max_error": self.max_error,
                "fallback_to_original": self.fallback_to_original,
                "avg_reconstruction_error": avg_err,
                "layer_errors": self.layer_errors,
                "layer_details": layer_details,
                "orig_storage_bytes": total_orig_bytes,
                "comp_storage_bytes": total_comp_bytes,
                "storage_compression_ratio": round(storage_ratio, 3),
                "storage_reduction_pct": round(100.0 * (1.0 - effective_ratio), 2),
                "effective_parameter_ratio": round(effective_ratio, 4),
                "forward_check_success": forward_check_success,
                "method": "mamba_shedder_sensitivity",
                "note": (
                    "Channel pruning now replaces target layers with "
                    "MambaShedderPrunedLinear (real storage reduction). "
                    "2:4 sparsity stores dense weights, so its storage "
                    "compression ratio stays at 1.0."
                ),
            }
        )
        self.stats = stats

        logger.info(
            "MambaShedder: %d pruned, %d shed, avg_err=%.4f, storage=%.2fx.",
            pruned_layers_count,
            shed_done,
            avg_err,
            storage_ratio,
        )
        return self.model, stats

class MambaShedderPrunedLinear(nn.Module):
    """
    Linear layer with pruned output channels.

    Stores only the kept rows of W:
      - W_kept:    (k, in_features)
      - bias_kept: (k,) or None
      - keep_idx:  (k,) int64 (output positions of kept rows)
      - original_out_features: int

    Forward:
      h = F.linear(x, W_kept, bias_kept)              # (..., k)
      out = zeros(..., original_out) then scatter to keep_idx
      + LoRA(x) if present.

    Storage: k * in instead of out * in.
    """

    is_custom_attention: bool = False

    def __init__(
        self,
        in_features: int,
        original_out_features: int,
        W_kept: torch.Tensor,
        keep_idx: torch.Tensor,
        bias_kept: torch.Tensor | None = None,
        lora_r: int = 0,
        lora_alpha: float = 1.0,
    ) -> None:
        super().__init__()
        self.in_features = int(in_features)
        self.original_out_features = int(original_out_features)
        self.out_features = self.original_out_features
        self.nf = self.out_features
        self.num_kept = int(W_kept.shape[0])
        self.lora_r = int(lora_r)
        self.lora_alpha = float(lora_alpha)
        self.scaling = (self.lora_alpha / self.lora_r) if self.lora_r > 0 else 1.0

        self.register_buffer("keep_idx", keep_idx.long(), persistent=True)

            self.bias_kept = nn.Parameter(bias_kept, requires_grad=False)
        else:
            self.register_parameter("bias_kept", None)

                torch.empty(
                    (self.lora_r, self.in_features),
                    dtype=torch.float32,
                    device=W_kept.device,
                )
            )

                torch.zeros(
                    (self.original_out_features, self.lora_r),
                    dtype=torch.float32,
                    device=W_kept.device,
                )
            )
            nn.init.kaiming_uniform_(self.lora_A, a=math.sqrt(5))
        else:
            self.register_parameter("lora_A", None)
            self.register_parameter("lora_B", None)

    @property
    def weight(self) -> torch.Tensor:
        """Reconstruct (original_out, in) weight with zeros in pruned rows."""
        W = self.W_kept.new_zeros(self.original_out_features, self.in_features)
        W.index_copy_(0, self.keep_idx, self.W_kept)
        return W

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        h = F.linear(x, self.W_kept, self.bias_kept)  # (..., k)
        out = torch.zeros(
            *x.shape[:-1],
            self.original_out_features,
            dtype=h.dtype,
            device=h.device,
        )
        out.scatter_(-1, self.keep_idx.expand(*x.shape[:-1], -1), h)
        if self.lora_r > 0 and self.lora_A is not None and self.lora_B is not None:
            lora_out = F.linear(F.linear(x.float(), self.lora_A), self.lora_B)
            out = out + self.scaling * lora_out.to(out.dtype)
        return out

    def storage_bytes(self) -> int:
        total = self.W_kept.numel() * self.W_kept.element_size()
        total += self.keep_idx.numel() * self.keep_idx.element_size()
        if self.bias_kept is not None:

        if self.lora_A is not None:
            total += self.lora_A.numel() * self.lora_A.element_size()
        if self.lora_B is not None:
            total += self.lora_B.numel() * self.lora_B.element_size()
        return total

    Frozen pruned base weight + trainable LoRA adapters.

    NOTE: `weight` is the ALREADY-PRUNED weight matrix. Forward does not
    re-apply the mask. `mask` is kept only as a compact bool buffer for
    downstream analysis / merge operations.
    """

    def __init__(
        self,
        in_features: int,
        out_features: int,
        pruned_weight: torch.Tensor,
        mask: torch.Tensor,
        bias: torch.Tensor | None = None,
        lora_r: int = 0,
        lora_alpha: float = 1.0,
        device: torch.device | None = None,
        dtype: torch.dtype | None = None,
    ) -> None:
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.lora_r = int(lora_r)
        self.lora_alpha = float(lora_alpha)
        self.scaling = (self.lora_alpha / self.lora_r) if self.lora_r > 0 else 1.0

        target_device = device if device is not None else pruned_weight.device
        target_dtype = dtype if dtype is not None else pruned_weight.dtype

        # PS-FIX-02: weight is already masked; do not multiply again.
        self.weight = nn.Parameter(
            pruned_weight.to(device=target_device, dtype=target_dtype),
            requires_grad=False,
        )
        # PS-FIX-03: bool buffer (1 byte per element, non-persistent).
        self.register_buffer(
            "mask",
            mask.bool().to(device=target_device),
            persistent=False,
        )

        if bias is not None:
            self.bias = nn.Parameter(
                bias.to(device=target_device, dtype=target_dtype),
                requires_grad=False,
            )
        else:
            self.register_parameter("bias", None)

            self.lora_A = nn.Parameter(
                torch.empty(
                    (self.lora_r, in_features),
                    dtype=torch.float32,

                )

                    device=target_device,

        cls_name = base_mod.__class__.__name__.lower()

    def _soft_threshold(self, tensor: torch.Tensor, threshold: float) -> torch.Tensor:
        """prox_{a ||.||_1}(x) = sign(x) * max(|x| - a, 0)."""
        return torch.sign(tensor) * torch.clamp(torch.abs(tensor) - threshold, min=0.0)

    def _project_sparse(self, tensor: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        """Hard projection onto sparse constraint S. Returns (pruned, mask)."""
        if self.structured_24:
            pruned, mask = apply_24_structured_mask(tensor, saliency=torch.abs(tensor), dim=1)
            return pruned, (mask > 0)
        elif self.block_size != (1, 1):
            pruned, mask = apply_block_mask(
                tensor, block_size=self.block_size, sparsity=self.sparsity
            )
            return pruned, (mask > 0)
        else:
            flat = tensor.abs().reshape(-1)
            k = max(1, int(flat.numel() * (1.0 - self.sparsity)))
            thr = torch.topk(flat, k, largest=True).values[-1]
            mask = tensor.abs() >= thr
            return tensor * mask.float(), mask

    def compress(self, **kwargs: Any) -> tuple[nn.Module, dict[str, Any]]:
        self.compressed_layers.clear()
        self.visited_modules.clear()
        self.protected_layers.clear()
        self.skipped_layers.clear()
        self.fallback_layers.clear()
        self.failed_layers.clear()
        self.layer_errors.clear()
        self.layer_actual_sparsity.clear()

        for name, module in list(self.model.named_modules()):
            if not self._is_target_module(name, module):
                continue

            mod_id = id(module)
            if mod_id in self.visited_modules:
                self.skipped_layers.append(name)
                continue
            self.visited_modules.add(mod_id)

            try:
                base_mod = getattr(module, "base_layer", module)
                raw_w = base_mod.weight.data
                cls_name = base_mod.__class__.__name__
                is_conv1d = "conv1d" in cls_name.lower() or hasattr(base_mod, "nf")
                device = raw_w.device
                dtype = raw_w.dtype

                # Always operate in (out, in) orientation.
                W_0 = raw_w.t().contiguous().float() if is_conv1d else raw_w.contiguous().float()

                # Early NaN/Inf guard on source.
                if not torch.isfinite(W_0).all():
                    logger.warning("ProxSparse: %s has non-finite weights; skipping.", name)
                    self.failed_layers.append(name)
                    continue

                # ------------------------------------------------
                # PS-FIX-01: Correct proximal gradient iteration.
                # ------------------------------------------------
                eta = self.learning_rate
                W = W_0.clone()
                mask = None
                for _ in range(self.steps):
                    grad_W = W - W_0  # grad of 0.5*||W - W_0||^2
                    W_grad = W - eta * grad_W  # gradient descent step
                    W_prox = self._soft_threshold(W_grad, eta * self.lambda_prox)
                    W, mask = self._project_sparse(W_prox)

                # Final projection (mask from last iteration already applied).
                W_final = torch.nan_to_num(W, nan=0.0, posinf=0.0, neginf=0.0)

                w_norm = torch.norm(W_0, p="fro").clamp_min(1e-12).item()
                err = (torch.norm(W_0 - W_final, p="fro") / w_norm).item()
                self.layer_errors[name] = float(err)

                # Actual sparsity in the final weight.
                actual_zero = (W_final.abs() < 1e-12).float().mean().item()

    def _soft_threshold(self, tensor: torch.Tensor, threshold: float) -> torch.Tensor:
        """prox_{a ||.||_1}(x) = sign(x) * max(|x| - a, 0)."""
        return torch.sign(tensor) * torch.clamp(torch.abs(tensor) - threshold, min=0.0)

    def _project_sparse(self, tensor: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        """Hard projection onto sparse constraint S. Returns (pruned, mask)."""
        if self.structured_24:
            pruned, mask = apply_24_structured_mask(tensor, saliency=torch.abs(tensor), dim=1)
            return pruned, (mask > 0)
        elif self.block_size != (1, 1):
            pruned, mask = apply_block_mask(
                tensor, block_size=self.block_size, sparsity=self.sparsity
            )
            return pruned, (mask > 0)
        else:
            flat = tensor.abs().reshape(-1)
            k = max(1, int(flat.numel() * (1.0 - self.sparsity)))
            thr = torch.topk(flat, k, largest=True).values[-1]
            mask = tensor.abs() >= thr
            return tensor * mask.float(), mask

            logger.warning("ProxSparse forward sanity check failed: %s", ex_sanity)
            forward_check_success = False

        self._compressed_params = self._count_params(self.model)
        avg_err = (
            sum(self.layer_errors[k] for k in self.compressed_layers)
            / max(1, len(self.compressed_layers))
            if self.compressed_layers
            else 0.0
        )

        # PS-FIX-08: effective_parameter_ratio separate from compression_ratio.
        effective_ratio = 1.0
        if self.compressed_layers:
            ratios = []
            for n in self.compressed_layers:
                actual_sp = self.layer_actual_sparsity.get(n, 0.0)
                # Non-zeros in the pruned base + LoRA params (if any)
                ratio = 1.0 - actual_sp
                if self.use_lora and self.lora_r > 0:
                    # Very rough estimate; LoRA adds lora_r*(in+out) params.
                    ratio += 0.02  # approximate placeholder; real accounting per layer
                ratios.append(ratio)
            effective_ratio = sum(ratios) / len(ratios)

        stats = self.get_compression_stats()
        stats.update(
            {
                "effective_parameter_ratio": round(effective_ratio, 4),
                "pruned_layers": self.compressed_layers,
                "compressed_layers": self.compressed_layers,
                "compressed_layer_count": len(self.compressed_layers),

                "fallback_layer_count": len(self.fallback_layers),
                "failed_layers": self.failed_layers,
                "failed_layer_count": len(self.failed_layers),
                "layer_errors": self.layer_errors,
                "layer_actual_sparsity": self.layer_actual_sparsity,

                "forward_check_success": forward_check_success,

                "use_lora": self.use_lora,
                "lora_r": self.lora_r,
                "method": "proxsparse_regularized",
            }
        )
        self.stats = stats
        return self.model, stats

class SparseGPTCompressor(BaseCompressor):
    """
    In-memory PyTorch SparseGPT Compressor.

    Performs layer-by-layer second-order pruning with Optimal Brain Surgeon (OBS) weight
    compensation, supporting unstructured sparsity, block sparsity, and strict 2:4 structured sparsity.
    """

    def __init__(
        self,
        model: nn.Module,
        tokenizer: Any | None = None,
        sparsity: float = 0.5,
        prunen: int = 2,
        prunem: int = 4,
        block_size: int = 128,
        percdamp: float = 0.01,
        calibration_loader: Any | None = None,
        calibration_texts: list[str] | None = None,
        target_attention_only: bool = False,
        device: str | torch.device | None = None,
        structured_24: bool = False,
        max_error: float = 0.5,
        fallback_to_original: bool = True,
        target_modules: list[str] | None = None,
        **kwargs: Any,
    ) -> None:
        super().__init__(model, name="sparsegpt", **kwargs)
        cfg = kwargs.get("comp_config", kwargs.get("cfg"))
        if cfg is not None:
            sparsity = _cfg_get(cfg, "sparsegpt_sparsity", _cfg_get(cfg, "sparsity", sparsity))
            structured_24 = _cfg_get(
                cfg,
                "sparsegpt_structured_24",
                _cfg_get(cfg, "structured_24", structured_24),
            )
            max_error = _cfg_get(cfg, "max_error", max_error)
            fallback_to_original = _cfg_get(cfg, "fallback_to_original", fallback_to_original)
            target_attention_only = _cfg_get(cfg, "target_attention_only", target_attention_only)

        self.tokenizer = tokenizer or getattr(model, "tokenizer", None)
        self.sparsity = max(0.0, min(float(sparsity), 0.95))
        self.prunen = prunen
        self.prunem = prunem
        self.block_size = block_size
        self.percdamp = percdamp
        self.calibration_loader = calibration_loader
        self.calibration_texts = calibration_texts
        self.target_attention_only = bool(target_attention_only)
        self.structured_24 = structured_24 or (prunen == 2 and prunem == 4 and self.sparsity == 0.5)
        if self.structured_24:
            self.sparsity = 0.5
            self.prunen = 2
            self.prunem = 4
        self.target_device = (
            torch.device(device)
            if device is not None
            else next(model.parameters(), torch.tensor(0.0)).device
        )
        self.max_error = float(max_error)
        self.fallback_to_original = bool(fallback_to_original)
        self.target_modules = target_modules
        self.layer_errors: dict[str, float] = {}
        self.stats: dict[str, Any] = {}

    def _is_target_module(self, name: str, module: nn.Module) -> bool:
        name_l = name.lower()
        if any(k in name_l for k in ["classifier", "score", "lm_head", "pooler", "embedding"]):
            return False

        if self.target_attention_only:
            attn_keywords = [
                "attn",
                "attention",
                "q_proj",
                "k_proj",
                "v_proj",
                "o_proj",
                "c_attn",
                "c_proj",
                "query",
                "key",
                "value",
            ]
            if not any(k in name_l for k in attn_keywords):
                return False

        base_mod = getattr(module, "base_layer", module)
        cls_name = base_mod.__class__.__name__
        is_linear_like = (
            isinstance(base_mod, nn.Linear)
            or "conv1d" in cls_name.lower()
            or hasattr(base_mod, "nf")
        )
        if not is_linear_like:
            return False
        if not should_compress_linear(self.model, name):
            return False
        if self.target_modules is None:
            return True
        return any(t in name for t in self.target_modules)

    def _collect_layer_activations(self) -> dict[str, torch.Tensor]:
        """Capture layer inputs via forward hooks."""
        self.model.eval()
        activations: dict[str, list[torch.Tensor]] = {}
        hooks = []

        def make_hook(name: str):
            def hook(module: nn.Module, inputs: tuple[torch.Tensor, ...], output: Any):
                if inputs and isinstance(inputs[0], torch.Tensor):
                    inp = inputs[0].detach()
                    if name not in activations:
                        activations[name] = []
                    activations[name].append(inp.cpu())

            return hook

        for name, module in self.model.named_modules():
            if self._is_target_module(name, module):
                hooks.append(module.register_forward_hook(make_hook(name)))

        calib_inputs = get_calibration_tensors(
            self.model,
            tokenizer=self.tokenizer,
            calibration_loader=self.calibration_loader,
            calibration_texts=self.calibration_texts,
            device=self.target_device,
        )

        with torch.no_grad():
            try:
                if isinstance(calib_inputs, dict):
                    self.model(**calib_inputs)
                elif isinstance(calib_inputs, torch.Tensor):
                    self.model(calib_inputs)
                elif isinstance(calib_inputs, (list, tuple)):
                    self.model(*calib_inputs)
            except Exception as e:
                logger.debug("SparseGPT activation collection forward pass exception: %s", e)

        for h in hooks:
            h.remove()

        aggregated: dict[str, torch.Tensor] = {}
        for name, act_list in activations.items():
            if act_list:
                aggregated[name] = torch.cat(act_list, dim=0)

        return aggregated

    def _compress_linear_module(
        self,
        name: str,
        module: nn.Module,
        activations: torch.Tensor | None,
    ) -> bool:
        """Executes second-order SparseGPT pruning on a single Linear or Conv1D layer."""
        base_mod = getattr(module, "base_layer", module)
        raw_w = base_mod.weight.data
        cls_name = base_mod.__class__.__name__
        is_conv1d = (
            "conv1d" in cls_name.lower()
            or hasattr(base_mod, "nf")
            or (
                raw_w.ndim == 2
                and hasattr(base_mod, "in_features")
                and base_mod.in_features == raw_w.shape[0]
                and raw_w.shape[0] < raw_w.shape[1]
            )
        )

        W = raw_w.t().clone().float() if is_conv1d else raw_w.clone().float()
        d_out, d_in = W.shape
        device = W.device

        # Compute empirical Hessian and inverse
        if activations is not None and activations.numel() > 0:
            act_flat = activations.reshape(-1, activations.shape[-1]).float()
            if act_flat.shape[-1] == d_in:
                _, H_inv = compute_empirical_hessian(act_flat, damping=self.percdamp, device=device)
            else:
                2.0 * torch.eye(d_in, device=device)
                H_inv = 0.5 * torch.eye(d_in, device=device)
        else:
            2.0 * torch.eye(d_in, device=device)
            H_inv = 0.5 * torch.eye(d_in, device=device)

        H_inv = H_inv.to(device=device, dtype=torch.float32)

        # Iterate over columns in blocks
        W_pruned = W.clone()
        block_cols = min(self.block_size, d_in)

        for i1 in range(0, d_in, block_cols):
            i2 = min(i1 + block_cols, d_in)
            count = i2 - i1

            W_block_init = W_pruned[:, i1:i2].clone()
            W_block = W_block_init.clone()
            H_inv_block = H_inv[i1:i2, i1:i2].clone()

            for j in range(count):
                w_col = W_block[:, j].clone()
                diag_inv = max(H_inv_block[j, j].item(), 1e-8)
                saliency_col = (w_col**2) / (2.0 * diag_inv)

                # N:M structured or unstructured pruning with OBS error compensation (B10, H25)
                if self.structured_24 or (self.prunem > 0 and self.prunen > 0):
                    M = max(1, self.prunem)
                    N = max(1, min(self.prunen, M))
                    # At the end of every M-group or block boundary, apply N:M mask across the group
                    if (j + 1) % M == 0 or (j + 1 == count):
                        start_c = (j // M) * M
                        sub_w = W_block[:, start_c : j + 1]
                        sub_h_diag = torch.diagonal(
                            H_inv_block[start_c : j + 1, start_c : j + 1]
                        ).clamp(min=1e-8)
                        sub_saliency = (sub_w**2) / (2.0 * sub_h_diag.unsqueeze(0))
                        k_keep = min(N, sub_w.shape[1])
                        _, top_idx = torch.topk(sub_saliency, k=k_keep, dim=1, largest=True)
                        sub_mask = torch.zeros_like(sub_w)
                        sub_mask.scatter_(1, top_idx, 1.0)
                        pruned_sub = sub_w * sub_mask
                        err_sub = sub_w - pruned_sub
                        W_block[:, start_c : j + 1] = pruned_sub

                        # Optimal Brain Surgeon error compensation update
                        if j < count - 1:
                            err_col = err_sub[:, -1]
                            h_slice = H_inv_block[j, j + 1 : count] / diag_inv
                            W_block[:, j + 1 : count] -= torch.outer(err_col, h_slice)
                else:
                    k_prune = int(d_out * self.sparsity)
                    if k_prune > 0:
                        _, prune_idx = torch.topk(saliency_col, k=k_prune, largest=False)
                        mask_col = torch.ones(d_out, device=device)
                        mask_col[prune_idx] = 0.0
                        pruned_w_col = w_col * mask_col
                        err = w_col - pruned_w_col
                        W_block[:, j] = pruned_w_col

                        # Optimal Brain Surgeon error compensation update
                        if j < count - 1:
                            h_slice = H_inv_block[j, j + 1 : count] / diag_inv
                            W_block[:, j + 1 : count] -= torch.outer(err, h_slice)

            W_pruned[:, i1:i2] = W_block

            # Update remaining columns outside the current block
            if i2 < d_in:
                err_block = W_block_init - W_block
                h_cross = H_inv[i1:i2, i2:]  # [count, d_in - i2]
                inv_diag_block = torch.linalg.pinv(H_inv_block)
                update = torch.matmul(err_block, torch.matmul(inv_diag_block, h_cross))
                W_pruned[:, i2:] -= update

        # Relative Frobenius error check
        norm_orig = torch.norm(W, p="fro")
        rel_err = (torch.norm(W - W_pruned, p="fro") / (norm_orig + 1e-8)).item()
        self.layer_errors[name] = rel_err

        if self.fallback_to_original and rel_err > self.max_error:
            logger.warning(
                "SparseGPT reconstruction error %.4f on %s exceeded threshold %.4f. Skipping.",
                rel_err,
                name,
                self.max_error,
            )
            logger.warning(
                "Fallback triggered for %s. Layer is kept original and NOT wrapped in LoRA. It will not be fine-tuned.",
                name,
            )
            for p in module.parameters():
                p.requires_grad = False
            return False

        final_w = W_pruned.t() if is_conv1d else W_pruned
        base_mod.weight.data = final_w.to(
            dtype=base_mod.weight.dtype, device=base_mod.weight.device
        )
        compressed_layer = base_mod
        # Preserve existing LoRA adapter weights if the source module has them
        if (
            hasattr(module, "lora_A")
            and hasattr(module, "lora_B")
            and module.lora_A is not None
            and module.lora_B is not None
        ):
            src_lA = getattr(module.lora_A, "weight", module.lora_A)
            src_lB = getattr(module.lora_B, "weight", module.lora_B)
            if hasattr(compressed_layer, "lora_A") and compressed_layer.lora_A is not None:
                if hasattr(src_lA, "data") and src_lA.shape == compressed_layer.lora_A.shape:
                    compressed_layer.lora_A.data.copy_(src_lA.data)
            if hasattr(compressed_layer, "lora_B") and compressed_layer.lora_B is not None:
                if hasattr(src_lB, "data") and src_lB.shape == compressed_layer.lora_B.shape:
                    compressed_layer.lora_B.data.copy_(src_lB.data)

        # Determine if module is wrapped (LoRALinear) or standalone
        is_wrapped = hasattr(module, "base_layer")
        if is_wrapped:
            module.base_layer = compressed_layer
            new_module = module
        else:
            new_module = compressed_layer
            replace_module(self.model, name, new_module)
        return True

    def compress(self, **kwargs: Any) -> tuple[nn.Module, dict[str, Any]]:
        """Executes SparseGPT compression across all target linear layers."""
        self.layer_errors = {}
        activations_map = self._collect_layer_activations()
        pruned_layers_count = 0

        for name, module in self.model.named_modules():
            if self._is_target_module(name, module):

                "fallback_to_original": self.fallback_to_original,
                "max_error": self.max_error,
                "method": "sparsegpt_second_order",
            }
        )
        self.stats = stats
        return self.model, stats
