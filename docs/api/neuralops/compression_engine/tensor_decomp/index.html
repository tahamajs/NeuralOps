class BaseCompressor:
    """Base class for all compressors. Not a nn.Module to avoid parameter storage."""

    def __init__(self, model: nn.Module, name: str = "base", **kwargs) -> None:
        self.model = model
        self.name = name
        self.kwargs = kwargs
        self._original_params = self._count_params(model)

    def __call__(self, *args, **kwargs) -> Any:
        """Allow compressor instance to act as a callable model forward wrapper."""
        if not getattr(self, "_compressed_once", False):
            self.compress()
            self._compressed_once = True
        return self.model(*args, **kwargs)

    @abstractmethod
    def compress(self) -> tuple[nn.Module, dict[str, Any]]:
        """Return (compressed_model, statistics)."""
        raise NotImplementedError("Subclasses of BaseCompressor must implement compress()")

    def _count_params(self, model: nn.Module) -> int:
        """count params."""
        return sum(p.numel() for p in model.parameters())

    def get_compression_stats(self) -> dict[str, Any]:
        """Return the compression stats."""
        compressed = self._count_params(self.model)
        orig = self._original_params
        return {
            "compressor_name": self.name,
            "original_parameters": orig,
            "compressed_parameters": compressed,
            "saved_parameters": orig - compressed,
            "compression_ratio": compressed / orig if orig > 0 else 1.0,
            "reduction_percentage": ((orig - compressed) / orig * 100 if orig > 0 else 0.0),
        }

    def summary(self) -> str:
        """Summary."""
        s = self.get_compression_stats()
        return (
            f"\n========================================\n"
            f"Compression Summary: {self.name}\n"
            f"========================================\n"
            f"Original parameters:  {s['original_parameters']:,}\n"
            f"Compressed parameters: {s['compressed_parameters']:,}\n"
            f"Parameters saved:     {s['saved_parameters']:,}\n"
            f"Compression ratio:    {s['compression_ratio']:.3f}\n"
            f"Reduction:            {s['reduction_percentage']:.2f}%\n"
            f"========================================\n"
        )

    def _is_gqa_asymmetric(self, name: str, module: nn.Module) -> bool:
        if not self.skip_attention_qkv:
            return False
        cfg = getattr(self.model, "config", None)
        num_heads = getattr(cfg, "num_attention_heads", None)
        num_kv_heads = getattr(cfg, "num_key_value_heads", None)
        if num_heads is not None and num_kv_heads is not None and num_kv_heads < num_heads:
            if any(
                p in name.lower()
                for p in (
                    "q_proj",
                    "k_proj",
                    "v_proj",
                    "query",
                    "key",
                    "value",
                    "c_attn",
                )
            ):
                return True
        return False

    # KC-FIX-12: debug-enriched target check.
    def _is_target_module(self, name: str, module: nn.Module, _debug: bool = False) -> bool:
        name_lower = name.lower()
        if any(part in name.split(".") for part in ("lora_A", "lora_B", "base_layer")):
            if _debug:
                logger.info(f"  [skip] {name}: lora/base_layer part")
            return False
        if any(
            skip in name_lower
            for skip in (
                "classifier",
                "score",
                "lm_head",
                "pooler",
                "embedding",
                "router",
                "gate",
            )
        ):
            if _debug:
                logger.info(f"  [skip] {name}: skip keyword")
            return False

        if self.target_attention_only:
            if not any(k in name_lower for k in _ATTN_NAMES):
                if _debug:
                    logger.info(f"  [skip] {name}: not attention")
                return False

        if self._is_gqa_asymmetric(name, module):
            if _debug:
                logger.info(f"  [skip] {name}: GQA asymmetric")
            return False

        target_mod = getattr(module, "base_layer", module)
        cls_lower = target_mod.__class__.__name__.lower()
        is_linear = isinstance(target_mod, nn.Linear)
        is_conv1d = "conv1d" in cls_lower or hasattr(target_mod, "nf")
        if not (is_linear or is_conv1d):
            if _debug:
                logger.info(f"  [skip] {name}: not linear-like")
            return False
        if isinstance(target_mod, KroneckerLinear):
            if _debug:
                logger.info(f"  [skip] {name}: already KroneckerLinear")
            return False

        w = getattr(target_mod, "weight", None)
        if w is None:
            if _debug:
                logger.info(f"  [skip] {name}: weight None")
            return False

        # KC-FIX-11: filter tiny layers.
        if min(w.shape[0], w.shape[1]) < self.min_layer_dim:
            if _debug:
                logger.info(f"  [skip] {name}: dim too small {tuple(w.shape)}")
            return False

        # KC-FIX-14: guard against framework errors.
        if not self.skip_should_compress_filter:
            try:
                if not should_compress_linear(self.model, name):
                    if _debug:
                        logger.info(f"  [skip] {name}: should_compress_linear=False")
                    return False
            except Exception as e:
                if _debug:
                    logger.info(f"  [skip] {name}: should_compress_linear raised {e}")
                return False

        # KC-FIX-13: empty target_layers means "no targets".
        if self.target_layers is not None:
            ok = any(t in name for t in self.target_layers)
            if not ok and _debug:
                logger.info(f"  [skip] {name}: not in target_layers")
            return ok

        if _debug:
            logger.info(f"  [TARGET] {name}: shape={tuple(w.shape)}")
        return True

    # ------------------------------------------------------------------
    def compress(self, model: nn.Module | None = None) -> tuple[nn.Module, dict[str, Any]]:
        # KC-FIX-09: idempotence.
        if getattr(self.model, "_kronecker_compressed", False):
            logger.warning("KroneckerCompressor: model already compressed; skipping.")
            try:
                return self.model, self.get_compression_stats()
            except Exception:
                return self.model, {}

        if model is not None:
            self.model = model
            try:
                self._original_params = self._count_params(self.model)
            except Exception:
                self._original_params = sum(p.numel() for p in self.model.parameters())

        self.compressed_layers = []
        self.fallback_layers = []
        self.skipped_layers = []
        self.layer_errors = {}
        self.layer_ranks = {}
        self.layer_modes = {}
        seen_weights: set[int] = set()

        total_orig_params = 0
        total_comp_params = 0

        # Freeze non-targets when attention_only.
        if self.target_attention_only:
            for name, module in self.model.named_modules():
                target_mod = getattr(module, "base_layer", module)
                cls_lower = target_mod.__class__.__name__.lower()
                if (
                    isinstance(target_mod, nn.Linear)
                    or "conv1d" in cls_lower
                    or hasattr(target_mod, "nf")
                ):
                    if not self._is_target_module(name, module):
                        for p in module.parameters():
                            p.requires_grad = False
                        self.skipped_layers.append(name)

        targets = [
            (name, mod)
            for name, mod in list(self.model.named_modules())
            if self._is_target_module(name, mod)
        ]

        if not targets:
            # KC-FIX-12: diagnostic breakdown.
            logger.warning("KroneckerCompressor: no targets found. Diagnostic...")
            n_scanned = 0
            for name, module in self.model.named_modules():
                n_scanned += 1
                self._is_target_module(name, module, _debug=True)
            logger.warning(
                "KroneckerCompressor diagnostic: %d modules scanned, 0 targets.",
                n_scanned,
            )
            try:
                return self.model, self.get_compression_stats()
            except Exception:
                return self.model, {}

        logger.info("KroneckerCompressor: %d target layers identified.", len(targets))

        for name, module in targets:
            target_mod = getattr(module, "base_layer", module)
            if not hasattr(target_mod, "weight") or target_mod.weight is None:
                continue

            w_id = id(target_mod.weight)
            if w_id in seen_weights:
                self.skipped_layers.append(name)
                continue
            seen_weights.add(w_id)

            cls_lower = target_mod.__class__.__name__.lower()
            is_conv1d = "conv1d" in cls_lower or hasattr(target_mod, "nf")
            raw_w = target_mod.weight.data

            orig_layer_params = raw_w.numel()
            bias_mod = getattr(target_mod, "bias", None)
            if bias_mod is not None:
                orig_layer_params += bias_mod.numel()
            total_orig_params += orig_layer_params

            W_target = raw_w.t().float() if is_conv1d else raw_w.float()
            W_norm = torch.norm(W_target, p="fro").item()

            new_layer: KroneckerLinear | None = None
            err = 1.0
            try:
                new_layer = KroneckerLinear.from_linear(
                    module,
                    rank=self.rank,
                    max_error=self.max_error,
                    lora_r=self.lora_r,
                    lora_alpha=self.lora_alpha,
                )
                W_comp = new_layer.get_weight().float()
                err = float(torch.norm(W_target - W_comp, p="fro").item() / max(W_norm, 1e-12))
            except Exception as e:
                logger.warning("KroneckerLinear factorization failed on %s: %s", name, e)
                new_layer = None
                err = 1.0

            self.layer_errors[name] = err

            # KC-FIX-15: only set mode/rank after decision.
            if new_layer is None or err > self.max_error:
                logger.warning(
                    "Kronecker: error %.4f on %s > %.4f. %s",
                    err,
                    name,
                    self.max_error,
                    ("Falling back." if self.fallback_to_original else "Keeping original weight."),
                )
                if self.fallback_to_original:
                    self.fallback_layers.append(name)
                    self.layer_ranks[name] = None
                    self.layer_modes[name] = "fallback"
                    for p in module.parameters():
                        p.requires_grad = False
                    # KC-FIX-06: use orig_layer_params (no double-count).
                    total_comp_params += orig_layer_params
                    continue
                else:
                    self.fallback_layers.append(name)
                    self.layer_ranks[name] = None
                    self.layer_modes[name] = "fallback"
                    total_comp_params += orig_layer_params
                    continue

            # Accepted.
            mode = "kronecker" if getattr(new_layer, "can_decompose", False) else "svd"
            self.layer_modes[name] = mode
            self.layer_ranks[name] = getattr(new_layer, "rank", None)

            comp_layer_params = sum(p.numel() for p in new_layer.parameters())
            total_comp_params += comp_layer_params

            try:
                replace_module(self.model, name, new_layer)
                self.compressed_layers.append(name)
            except Exception as e:
                logger.warning("Kronecker: replace_module failed for %s: %s", name, e)
                self.fallback_layers.append(name)

        # KC-FIX-08: robust forward check.
        forward_check_success = True
        try:
            with torch.no_grad():
                dev = _get_device(self.model)
                dummy = torch.randint(0, 100, (1, 8), device=dev, dtype=torch.long)
                try:
                    self.model(input_ids=dummy, attention_mask=torch.ones_like(dummy))
                except Exception:
                    try:
                        self.model(dummy)
                    except Exception as e2:
                        logger.warning("Kronecker forward check failed: %s", e2)
                        forward_check_success = False
        except Exception as e:
            logger.warning("Kronecker forward check raised: %s", e)
            forward_check_success = False

        # KC-FIX-03: safe param count.
        try:
            self._compressed_params = self._count_params(self.model)
        except Exception:
            self._compressed_params = sum(p.numel() for p in self.model.parameters())

        # KC-FIX-05: avg_err only over compressed layers.
        compressed_errs = [
            self.layer_errors[n] for n in self.compressed_layers if n in self.layer_errors
        ]
        avg_err = sum(compressed_errs) / len(compressed_errs) if compressed_errs else 0.0
        param_ratio = (
            total_comp_params / max(total_orig_params, 1) if total_orig_params > 0 else 1.0
        )

        try:
            stats = self.get_compression_stats()
        except Exception:
            stats = {}

        stats.update(
            {
                "compressed_layers": self.compressed_layers,
                "fallback_layers": self.fallback_layers,
                "skipped_layers": self.skipped_layers,
                "num_compressed": len(self.compressed_layers),
                "num_fallback": len(self.fallback_layers),
                "num_skipped": len(self.skipped_layers),
                "layer_errors": self.layer_errors,
                "layer_ranks": self.layer_ranks,
                "layer_modes": self.layer_modes,
                "rank": self.rank,
                "max_error": self.max_error,
                "fallback_to_original": self.fallback_to_original,
                "lora_r": self.lora_r,
                "lora_alpha": self.lora_alpha,
                "target_attention_only": self.target_attention_only,
                "forward_check_success": forward_check_success,
                "avg_reconstruction_error": avg_err,
                "compressed_layer_count": len(self.compressed_layers),
                "fallback_layer_count": len(self.fallback_layers),
                "effective_parameter_ratio": round(param_ratio, 4),
                "method": "kronecker",
            }
        )
        self.stats = stats

        # KC-FIX-09: write stats onto model.
        try:
            if not hasattr(self.model, "compression_stats") or self.model.compression_stats is None:
                self.model.compression_stats = {}
            self.model.compression_stats.update(
                {
                    "num_compressed": len(self.compressed_layers),
                    "num_fallback": len(self.fallback_layers),
                    "compression_method": "kronecker",
                    "effective_parameter_ratio": param_ratio,
                    "avg_reconstruction_error": avg_err,
                }
            )
        except Exception:
            pass

        self.model._kronecker_compressed = True
        logger.info(
            "Kronecker: %d compressed, %d fallback. param_ratio=%.4f, avg_err=%.4f.",
            len(self.compressed_layers),
            len(self.fallback_layers),
            param_ratio,
            avg_err,
        )
        return self.model, stats

class KroneckerLinear(nn.Module):
    # KL-FIX-01: this is a linear layer, not an attention layer.
    is_custom_attention: bool = False
    is_custom_linear: bool = True

    def __init__(
        self,
        in_features: int,
        out_features: int,
        bias: bool = True,
        rank: int | None = None,
        m1: int | None = None,
        n1: int | None = None,
        m2: int | None = None,
        n2: int | None = None,
        dtype: torch.dtype | None = None,
        device: torch.device | None = None,
        lora_r: int = 0,
        lora_alpha: float = 1.0,
    ) -> None:
        super().__init__()
        self.in_features = int(in_features)
        self.out_features = int(out_features)
        self.nf = self.out_features  # Conv1D compatibility
        self.lora_r = int(lora_r)
        self.lora_alpha = float(lora_alpha)
        self.scaling = (self.lora_alpha / self.lora_r) if self.lora_r > 0 else 1.0

        # ---- Factor dimensions ----
        if m1 is not None and m2 is not None and n1 is not None and n2 is not None:
            if m1 * m2 != self.out_features or n1 * n2 != self.in_features:
                raise ValueError(
                    f"Factor dims mismatch: {m1}*{m2}!={self.out_features} "
                    f"or {n1}*{n2}!={self.in_features}"
                )
            self.m1, self.m2 = int(m1), int(m2)
            self.n1, self.n2 = int(n1), int(n2)
        else:
            self.n1, self.n2 = factorize_dim(self.in_features)
            self.m1, self.m2 = factorize_dim(self.out_features)

        # Decomposable when both factorizations are non-trivial OR dims are tiny.
        self.can_decompose = (
            (self.n1 * self.n2 == self.in_features)
            and (self.m1 * self.m2 == self.out_features)
            and (self.n1 > 1 or self.in_features <= 3)
            and (self.m1 > 1 or self.out_features <= 3)
        )

        # ---- Parameter allocation ----
        requires_grad = not (self.lora_r > 0)
        self._init_base_parameters(
            bias=bias,
            dtype=dtype,
            device=device,
            requires_grad=requires_grad,
            rank=rank,
        )

        # ---- LoRA parameters ----
        if self.lora_r > 0:
            # KL-FIX-12: keep LoRA in the same dtype as base parameters.
            param_dtype = (
                next(self.parameters()).dtype
                if any(True for _ in self.parameters())
                else torch.float32
            )
            param_device = (
                next(self.parameters()).device
                if any(True for _ in self.parameters())
                else torch.device("cpu")
            )
            self.lora_A = nn.Parameter(
                torch.empty(
                    (self.lora_r, self.in_features),
                    dtype=param_dtype,
                    device=param_device,
                )
            )
            self.lora_B = nn.Parameter(
                torch.zeros(
                    (self.out_features, self.lora_r),
                    dtype=param_dtype,
                    device=param_device,
                )
            )
            nn.init.kaiming_uniform_(self.lora_A, a=math.sqrt(5))
        else:
            self.register_parameter("lora_A", None)
            self.register_parameter("lora_B", None)

    # ------------------------------------------------------------------
    # KL-FIX-09: clean parameter initialization
    # ------------------------------------------------------------------
    def _init_base_parameters(
        self,
        bias: bool,
        dtype: torch.dtype | None,
        device: torch.device | None,
        requires_grad: bool,
        rank: int | None,
    ) -> None:
        if self.can_decompose:
            # Kronecker factors A (m1, n1), B (m2, n2)
            self.A = nn.Parameter(
                torch.empty(self.m1, self.n1, dtype=dtype, device=device),
                requires_grad=requires_grad,
            )
            self.B = nn.Parameter(
                torch.empty(self.m2, self.n2, dtype=dtype, device=device),
                requires_grad=requires_grad,
            )
            fan_in = max(self.in_features, 1)
            bound = 1.0 / math.sqrt(fan_in)
            nn.init.uniform_(self.A, -bound, bound)
            nn.init.uniform_(self.B, -bound, bound)

            # Ensure U / V don't exist.
            self.register_parameter("U", None)
            self.register_parameter("V", None)

            if bias:
                self.bias = nn.Parameter(
                    torch.empty(self.out_features, dtype=dtype, device=device),
                    requires_grad=requires_grad,
                )
                nn.init.uniform_(self.bias, -bound, bound)
            else:
                self.register_parameter("bias", None)
        else:
            effective_rank = rank or max(1, min(self.in_features, self.out_features) // 4)
            self.rank = min(effective_rank, self.in_features, self.out_features)
            self.U = nn.Parameter(
                torch.empty(self.out_features, self.rank, dtype=dtype, device=device),
                requires_grad=requires_grad,
            )
            self.V = nn.Parameter(
                torch.empty(self.rank, self.in_features, dtype=dtype, device=device),
                requires_grad=requires_grad,
            )
            nn.init.xavier_uniform_(self.U)
            nn.init.xavier_uniform_(self.V)

            # Ensure A / B don't exist.
            self.register_parameter("A", None)
            self.register_parameter("B", None)

            if bias:
                self.bias = nn.Parameter(
                    torch.zeros(self.out_features, dtype=dtype, device=device),
                    requires_grad=requires_grad,
                )
            else:
                self.register_parameter("bias", None)

    def _load_from_state_dict(self, *args, **kwargs):
        # KL-FIX-07: state_dict load invalidates weight cache.
        self._cached_weight = None

        self._cached_weight = None

    # Weight accessors (KL-FIX-10: no stale caching)

    @property
    def weight(self) -> torch.Tensor:
        """Reconstructed effective weight matrix (out_features, in_features)."""
        return self.get_weight()

    def get_weight(self) -> torch.Tensor:
        if self.can_decompose and self.A is not None and self.B is not None:
            return torch.kron(self.A, self.B)
        if self.U is not None and self.V is not None:
            return torch.matmul(self.U, self.V)
        # Fallback: should never happen for a properly initialized layer.
        raise RuntimeError("KroneckerLinear has no valid parameter set.")

    # ------------------------------------------------------------------
    # Forward
    # ------------------------------------------------------------------
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        if self.can_decompose and self.A is not None and self.B is not None:
            out = self._forward_kron(x)
        else:
            out = self._forward_svd(x)

        # KL-FIX-12: LoRA in native dtype (no explicit fp32 cast).
        if self.lora_r > 0 and self.lora_A is not None and self.lora_B is not None:
            lora_out = F.linear(F.linear(x, self.lora_A), self.lora_B)
            out = out + self.scaling * lora_out.to(out.dtype)
        return out

    def _forward_kron(self, x: torch.Tensor) -> torch.Tensor:
        """(A ⊗ B) x computed without materializing the full weight."""
        orig_shape = x.shape
        x_flat = x.reshape(-1, self.in_features)  # (K, n1 * n2)
        K = x_flat.shape[0]

        # Step 1: contract with B over n2  ->  (K * n1, m2)
        x_b = torch.matmul(
            x_flat.reshape(K * self.n1, self.n2),
            self.B.t().contiguous(),
        )
        # Step 2: (K, m2, n1) -> (K * m2, n1) @ A^T  ->  (K * m2, m1)
        x_b = (
            x_b.reshape(K, self.n1, self.m2)
            .transpose(1, 2)
            .contiguous()
            .reshape(K * self.m2, self.n1)
        )
        x_a = torch.matmul(x_b, self.A.t().contiguous())
        # Step 3: back to (K, m1 * m2)
        out = (
            x_a.reshape(K, self.m2, self.m1)
            .transpose(1, 2)
            .contiguous()
            .reshape(K, self.out_features)
        )

        if self.bias is not None:
            out = out + self.bias
        return out.reshape(*orig_shape[:-1], self.out_features)

    def _forward_svd(self, x: torch.Tensor) -> torch.Tensor:
        return F.linear(F.linear(x, self.V), self.U, self.bias)

    # ------------------------------------------------------------------
    # SVD fitting (KL-FIX-03, KL-FIX-05, KL-FIX-09, KL-FIX-11)
    # ------------------------------------------------------------------
    def _fit_svd(
        self,
        weight: torch.Tensor,
        bias: torch.Tensor | None = None,
        rank: int | None = None,
        max_error: float = 0.20,
    ) -> None:
        """Fit low-rank SVD factors ensuring parameter reduction."""
        in_f, out_f = self.in_features, self.out_features
        target_dtype = weight.dtype
        target_device = weight.device
        W_norm = torch.norm(weight, p="fro").item()

        # Parameter reduction constraint: r * (in + out) < in * out
        max_allowed_rank = max(
            1,
            min(in_f, out_f, int(0.75 * (in_f * out_f) / max(in_f + out_f, 1))),
        )
        base_r = rank if rank is not None and rank > 0 else max(1, min(in_f, out_f) // 4)
        base_r = min(base_r, max_allowed_rank)

        try:
            U, S, Vt = torch.linalg.svd(weight.float(), full_matrices=False)
            candidate_ranks = [base_r]
            if base_r * 2 <= max_allowed_rank:
                candidate_ranks.append(base_r * 2)
            if base_r * 4 <= max_allowed_rank:
                candidate_ranks.append(base_r * 4)

            best_r = base_r
            best_U_r = None
            best_V_r = None
            for cand_r in candidate_ranks:
                U_r = U[:, :cand_r] * torch.sqrt(S[:cand_r].unsqueeze(0))
                V_r = torch.sqrt(S[:cand_r].unsqueeze(1)) * Vt[:cand_r, :]
                W_cand = torch.matmul(U_r, V_r)
                err = float(
                    torch.norm(weight.float() - W_cand, p="fro").item() / max(W_norm, 1e-12)
                )
                best_r = cand_r
                best_U_r = U_r
                best_V_r = V_r
                if err <= max_error:
                    break
        except Exception:
            best_r = base_r
            best_U_r = torch.zeros(out_f, best_r, device=target_device, dtype=torch.float32)
            best_V_r = torch.zeros(best_r, in_f, device=target_device, dtype=torch.float32)

        self.can_decompose = False
        # KL-FIX-05: clean removal of Kronecker parameters.
        self.register_parameter("A", None)
        self.register_parameter("B", None)

        self.rank = best_r
        # KL-FIX-09: clean registration of SVD parameters.
        self.U = nn.Parameter(best_U_r.to(dtype=target_dtype, device=target_device))
        self.V = nn.Parameter(best_V_r.to(dtype=target_dtype, device=target_device))

        # KL-FIX-03 + KL-FIX-11: unconditional, dtype-matched bias.
        if bias is not None:
            self.bias = nn.Parameter(bias.clone().to(dtype=target_dtype, device=target_device))
        else:
            self.register_parameter("bias", None)

        # KL-FIX-10: invalidate cache.
        self._invalidate_cache()

    # Factory (KL-FIX-04, KL-FIX-08)
    # ------------------------------------------------------------------
    @classmethod
    def from_linear(
        cls,
        linear: nn.Module,
        rank: int | None = None,
        max_error: float = 0.20,
        m1: int | None = None,
        n1: int | None = None,
        m2: int | None = None,
        n2: int | None = None,
        lora_r: int = 0,
        lora_alpha: float = 1.0,
    ) -> KroneckerLinear:
        """Build a KroneckerLinear from a pretrained Linear/Conv1D/LoRA wrapper."""
        target_mod = getattr(linear, "base_layer", linear)
        cls_lower = target_mod.__class__.__name__.lower()
        is_conv1d = "conv1d" in cls_lower or hasattr(target_mod, "nf")
        raw_w = target_mod.weight.data

        if is_conv1d:
            in_features = raw_w.shape[0]
            out_features = raw_w.shape[1]
            weight = raw_w.t().contiguous().float()
        else:
            out_features = raw_w.shape[0]
            in_features = raw_w.shape[1]
            weight = raw_w.float()

        bias_holder = getattr(target_mod, "bias", None)
        bias = bias_holder.data.clone() if bias_holder is not None else None
        target_dtype = raw_w.dtype
        target_device = raw_w.device

        # KL-FIX-04: robust LoRA detection.
        existing_lora_A = getattr(linear, "lora_A", None)
        existing_lora_B = getattr(linear, "lora_B", None)
        effective_lora_r = lora_r
        effective_lora_alpha = lora_alpha
        if existing_lora_A is not None and existing_lora_B is not None:
            w_A = getattr(existing_lora_A, "weight", existing_lora_A)
            if isinstance(w_A, torch.Tensor) and w_A.dim() >= 1:
                effective_lora_r = int(w_A.shape[0])
            scaling = getattr(linear, "scaling", None)
            if isinstance(scaling, (int, float)) and scaling > 0:
                effective_lora_alpha = float(scaling) * effective_lora_r
            elif hasattr(linear, "lora_alpha"):
                effective_lora_alpha = float(linear.lora_alpha)

        kron_layer = cls(
            in_features=in_features,
            out_features=out_features,
            bias=bias is not None,
            rank=rank,
            m1=m1,
            n1=n1,
            m2=m2,
            n2=n2,
            dtype=target_dtype,
            device=target_device,
            lora_r=effective_lora_r,
            lora_alpha=effective_lora_alpha,
        )

        # Transfer existing LoRA if shapes match.
        if (
            existing_lora_A is not None
            and existing_lora_B is not None
            and kron_layer.lora_A is not None
            and kron_layer.lora_B is not None
        ):
            w_A = getattr(existing_lora_A, "weight", existing_lora_A)
            w_B = getattr(existing_lora_B, "weight", existing_lora_B)
            if isinstance(w_A, torch.Tensor) and w_A.shape == kron_layer.lora_A.shape:
                kron_layer.lora_A.data.copy_(
                    w_A.to(kron_layer.lora_A.device, dtype=kron_layer.lora_A.dtype)
                )
            if isinstance(w_B, torch.Tensor) and w_B.shape == kron_layer.lora_B.shape:
                kron_layer.lora_B.data.copy_(
                    w_B.to(kron_layer.lora_B.device, dtype=kron_layer.lora_B.dtype)
                )

        W_norm = torch.norm(weight, p="fro").item()
        err = 1.0

        if kron_layer.can_decompose:
            m1_, m2_ = kron_layer.m1, kron_layer.m2
            n1_, n2_ = kron_layer.n1, kron_layer.n2
            try:
                # Rearrangement for Nearest Kronecker Product.
                W_tensor = weight.view(m1_, m2_, n1_, n2_).permute(0, 2, 1, 3).contiguous()
                W_reordered = W_tensor.view(m1_ * n1_, m2_ * n2_).float()

                U, S, Vt = torch.linalg.svd(W_reordered, full_matrices=False)
                sigma1 = S[0].item() if S.numel() > 0 else 1.0
                scale = math.sqrt(max(sigma1, 1e-12))

                u1 = U[:, 0] * scale
                v1 = Vt[0, :] * scale

                kron_layer.A.data.copy_(
                    u1.view(m1_, n1_).to(dtype=target_dtype, device=target_device)
                )
                kron_layer.B.data.copy_(
                    v1.view(m2_, n2_).to(dtype=target_dtype, device=target_device)
                )

                W_recon = torch.kron(kron_layer.A, kron_layer.B).float()
                err = float(torch.norm(weight - W_recon, p="fro").item() / max(W_norm, 1e-12))
            except Exception as e:
                import logging

                logging.getLogger(__name__).debug(
                    "KroneckerLinear NKP failed: %s; falling back to SVD.", e
                )
                err = 1.0

            if err > max_error:
                kron_layer._fit_svd(weight, bias, rank=rank, max_error=max_error)
        else:
            kron_layer._fit_svd(weight, bias, rank=rank, max_error=max_error)

        if bias is not None and kron_layer.bias is not None:
            kron_layer.bias.data.copy_(
                bias.to(dtype=kron_layer.bias.dtype, device=kron_layer.bias.device)
            )

        # KL-FIX-08: forward sanity check.
        try:
            with torch.no_grad():
                dummy = torch.randn(2, in_features, device=target_device, dtype=target_dtype)
                _ = kron_layer(dummy)
        except Exception as e:
            import logging

            logging.getLogger(__name__).warning(
                "KroneckerLinear forward sanity check failed: %s", e
            )

        return kron_layer

            if not any(k in name_lower for k in _ATTN_NAMES):
                if _debug:
                    logger.info(f"  [skip] {name}: not attention")
                return False

        if self._is_gqa_asymmetric(name, module):
            if _debug:
                logger.info(f"  [skip] {name}: GQA asymmetric")
            return False

        target_mod = getattr(module, "base_layer", module)
        cls_lower = target_mod.__class__.__name__.lower()
        is_linear_like = (
            isinstance(target_mod, nn.Linear) or "conv1d" in cls_lower or hasattr(target_mod, "nf")
        )
        if not is_linear_like:
            if _debug:
                logger.info(f"  [skip] {name}: not linear-like")
            return False
        if isinstance(target_mod, KroneckerTTLoRALayer):
            if _debug:
                logger.info(f"  [skip] {name}: already KroneckerTTLoRALayer")
            return False

        # KTL-FIX-12
        if not self.skip_should_compress_filter:
            try:
                if not should_compress_linear(self.model, name):
                    if _debug:
                        logger.info(f"  [skip] {name}: should_compress_linear=False")
                    return False
            except Exception as e:
                if _debug:
                    logger.info(f"  [skip] {name}: should_compress_linear raised {e}")
                return False

        w = getattr(target_mod, "weight", None)
        if w is None:
            if _debug:
                logger.info(f"  [skip] {name}: weight None")
            return False
        if min(w.shape[0], w.shape[1]) < self.min_layer_dim:
            if _debug:
                logger.info(f"  [skip] {name}: dim too small {tuple(w.shape)}")
            return False

        if _debug:
            logger.info(f"  [TARGET] {name}: shape={tuple(w.shape)}")
        return True

    def compress(self, model: nn.Module | None = None) -> tuple[nn.Module, dict[str, Any]]:
        # KTL-FIX-08: idempotence.
        if getattr(self.model, "_kronecker_tt_lora_compressed", False):
            logger.warning("KroneckerTTLoRA: model already compressed; skipping.")
            try:
                return self.model, self.get_compression_stats()

                return self.model, {}

        if model is not None:
            self.model = model
            try:
                self._original_params = self._count_params(self.model)
            except Exception:
                self._original_params = sum(p.numel() for p in self.model.parameters())

        self.compressed_layers = []
        self.fallback_layers = []
        self.skipped_layers = []
        self.layer_errors = {}
        seen_weights: set[int] = set()

        total_orig_params = 0
        total_comp_params = 0

        target_list = [
            (name, module)
            for name, module in self.model.named_modules()
            if self._is_target_module(name, module)
        ]
        if not target_list:
            logger.warning("KroneckerTTLoRA: no target layers found. Diagnostic...")
            n = 0
            for name, module in self.model.named_modules():
                n += 1
                self._is_target_module(name, module, _debug=True)
            logger.warning("KroneckerTTLoRA diagnostic: %d modules scanned, 0 targets.", n)
            try:
                return self.model, self.get_compression_stats()
            except Exception:
                return self.model, {}

        logger.info("KroneckerTTLoRA: %d target layers identified.", len(target_list))

        # Freeze non-targets when attention_only.
        if self.target_attention_only:
            for name, module in self.model.named_modules():
                target_mod = getattr(module, "base_layer", module)
                if (
                    isinstance(target_mod, nn.Linear)
                    or "conv1d" in target_mod.__class__.__name__.lower()
                    or hasattr(target_mod, "nf")
                ):
                    if not self._is_target_module(name, module):
                        for p in module.parameters():
                            p.requires_grad = False
                        self.skipped_layers.append(name)

        for name, module in target_list:
            target_mod = getattr(module, "base_layer", module)
            if not hasattr(target_mod, "weight") or target_mod.weight is None:
                continue

            w_id = id(target_mod.weight)
            if w_id in seen_weights:
                self.skipped_layers.append(name)
                continue
            seen_weights.add(w_id)

            raw_w = target_mod.weight.data
            orig_params = raw_w.numel()
            if getattr(target_mod, "bias", None) is not None:
                orig_params += target_mod.bias.numel()
            total_orig_params += orig_params

            try:
                adapted = self.compress_linear(module)
            except Exception as e:
                logger.warning("KroneckerTTLoRA: compress_linear failed for %s: %s", name, e)
                self.fallback_layers.append(name)
                for p in module.parameters():
                    p.requires_grad = False
                total_comp_params += orig_params
                continue

            # Error check
            cls_lower = target_mod.__class__.__name__.lower()
            is_conv1d = "conv1d" in cls_lower or hasattr(target_mod, "nf")
            W_orig = raw_w.t().contiguous().float() if is_conv1d else raw_w.float()
            delta_w = adapted.get_delta_w().float()
            # Note: delta_w is the adapter delta, not a reconstruction of the base weight.
            # Compute the *relative adapter magnitude* as a stability metric.
            base_norm = torch.norm(W_orig, p="fro").item()
            adapter_norm = torch.norm(delta_w, p="fro").item()
            err = adapter_norm / max(base_norm, 1e-12)
            self.layer_errors[name] = float(err)

            if err > self.max_error:
                logger.warning(
                    "KroneckerTTLoRA: adapter delta norm ratio %.4f on %s > %.4f.",
                    err,
                    name,
                    self.max_error,

            adapter_params = sum(

                for p in (
                    adapted.A1,
                    adapted.A2,
                    adapted.G1,
                    adapted.G2,
                    adapted.G3,
                    adapted.lora_A,
                    adapted.lora_B,
                )
                if p is not None
            )
            # KTL-FIX-13: total stored params = base + adapter.
            total_comp_params += orig_params + adapter_params

            try:

                self.compressed_layers.append(name)
            except Exception as e:
                logger.warning("KroneckerTTLoRA: replace_module failed for %s: %s", name, e)
                self.fallback_layers.append(name)

        # KTL-FIX-02: actual forward sanity check.
        forward_check_success = True
        try:
            with torch.no_grad():
                first_p = next(iter(self.model.parameters()), None)
                dev = first_p.device if first_p is not None else torch.device("cpu")

                    except Exception as e2:

        except Exception as e:
            logger.warning("KroneckerTTLoRA: forward check raised: %s", e)
            forward_check_success = False

        # KTL-FIX-05: safe param count.
        try:
            self._compressed_params = self._count_params(self.model)
        except Exception:
            self._compressed_params = sum(p.numel() for p in self.model.parameters())

        # KTL-FIX-07: honest stats.
        param_ratio = (
            total_comp_params / max(total_orig_params, 1) if total_orig_params > 0 else 1.0
        )
        avg_err = (
            sum(self.layer_errors.values()) / max(1, len(self.layer_errors))
            if self.layer_errors
            else 0.0
        )

                "skipped_layers": self.skipped_layers,
                "num_compressed": len(self.compressed_layers),

                "num_skipped": len(self.skipped_layers),
                "tt_rank": self.rank,
                "gamma": self.gamma,

                "lora_alpha": self.lora_alpha,
                "max_error": self.max_error,
                "target_attention_only": self.target_attention_only,
                "avg_adapter_delta_ratio": avg_err,
                "effective_parameter_ratio": round(param_ratio, 4),
                "forward_check_success": forward_check_success,
                "method": "kronecker_tt_lora",

        self,

        n1: int,
        m2: int,

        self.base_layer = base_layer
        self.base_layer.weight.requires_grad = False

        self.out_features = base_layer.out_features

        self.padded_out_features = m1 * m2
        self.padded_in_features = n1 * n2

        self.m2, self.n2 = m2, n2
        self.tt_shapes = tuple(tt_shapes)
        self.tt_ranks = list(tt_ranks)
        self.gamma = float(gamma)
        self.lora_r = max(1, int(lora_r))

            _init_orthogonal_spectral((r1, d2, r2), (r1 * d2, r2), device, dtype)

        self.G3 = nn.Parameter(_init_orthogonal_spectral((r2, d3, 1), (r2, d3), device, dtype))

        )
        nn.init.kaiming_uniform_(self.lora_A, a=math.sqrt(5))
        nn.init.zeros_(self.lora_B)

    @classmethod
    def factorize_dimensions(cls, in_f: int, out_f: int) -> dict[str, Any]:
        """Auto-factorize dims and pick 3-factor TT shape."""

        def best_factor_pair(dim: int) -> tuple[int, int]:
            for d in range(int(math.isqrt(dim)), 0, -1):
                if dim % d == 0:
                    return d, dim // d
            return 1, dim

        m1, m2 = best_factor_pair(out_f)
        n1, n2 = best_factor_pair(in_f)

        total_padded = (m1 * m2) * (n1 * n2)
        c = int(round(total_padded ** (1.0 / 3.0)))
        best: tuple[int, int, int] | None = None
        best_spread = float("inf")
        for d in range(max(1, c - 64), c + 65):
            if total_padded % d != 0:
                continue
            rem = total_padded // d
            for d2 in range(1, int(math.isqrt(rem)) + 1):
                if rem % d2 == 0:
                    d3 = rem // d2
                    spread = max(d, d2, d3) - min(d, d2, d3)
                    if spread < best_spread:
                        best = (d, d2, d3)
                        best_spread = spread
                    break
        tt_shapes = best if best is not None else (m1, m2, n1 * n2)

        return {"m1": m1, "n1": n1, "m2": m2, "n2": n2, "tt_shapes": tt_shapes}

    def from_linear(
        cls,
        base_layer: nn.Module,

        cls_lower = target_mod.__class__.__name__.lower()

        raw_w = target_mod.weight.data

        has_bias = bias_holder is not None

        if is_conv1d:
            in_features = raw_w.shape[0]
            out_features = raw_w.shape[1]
            lin = nn.Linear(

        else:

            in_features = getattr(target_mod, "in_features", raw_w.shape[1])
            out_features = getattr(target_mod, "out_features", raw_w.shape[0])
            lin = nn.Linear(
                in_features,
                out_features,
                bias=has_bias,
                device=raw_w.device,
                dtype=raw_w.dtype,

            lin.weight.data.copy_(raw_w)

                lin.bias.data.copy_(bias_holder.data)

        existing_lora_A = getattr(base_layer, "lora_A", None)
        existing_lora_B = getattr(base_layer, "lora_B", None)

        if existing_lora_A is not None and existing_lora_B is not None:

            if isinstance(w_A, torch.Tensor) and w_A.dim() >= 1:
                eff_lora_r = int(w_A.shape[0])
            scaling = getattr(base_layer, "scaling", None)
            if isinstance(scaling, (int, float)) and scaling:
                eff_lora_alpha = float(scaling) * eff_lora_r
            elif hasattr(base_layer, "lora_alpha"):
                eff_lora_alpha = float(base_layer.lora_alpha)

        layer = cls(
            base_layer=effective_base,

            n1=n1 or dims["n1"],
            m2=m2 or dims["m2"],
            n2=n2 or dims["n2"],

    def get_tt_tensor(self) -> torch.Tensor:
        d1, d2, d3 = self.tt_shapes
        r1, r2 = self.tt_ranks[1], self.tt_ranks[2]
        g1 = self.G1.view(d1, r1)
        g2 = self.G2.view(r1, d2 * r2)
        core12 = torch.matmul(g1, g2).view(d1 * d2, r2)
        g3 = self.G3.view(r2, d3)
        tt = torch.matmul(core12, g3)
        return tt.view(self.padded_out_features, self.padded_in_features)

    def get_delta_w(self) -> torch.Tensor:
        kron = torch.kron(self.A1, self.A2)
        tt_tensor = self.get_tt_tensor()
        delta_padded = self.gamma * (kron * tt_tensor)
        return delta_padded[: self.out_features, : self.in_features]

    @property
    def weight(self) -> torch.Tensor:
        # KTL-FIX-15: consistent with forward (bias handled in forward, not here).
        W = self.base_layer.weight + self.get_delta_w()
        if self.lora_A is not None and self.lora_B is not None:
            W = W + self.lora_scaling * torch.matmul(self.lora_B, self.lora_A)
        return W

    @property
    def bias(self) -> torch.Tensor | None:
        return getattr(self.base_layer, "bias", None)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # KTL-FIX-03: compute delta_w once.
        base_out = F.linear(x, self.base_layer.weight, getattr(self.base_layer, "bias", None))
        delta_w = self.get_delta_w()
        kron_tt_out = F.linear(x, delta_w)
        lora_bypass = F.linear(F.linear(x, self.lora_A), self.lora_B) * self.lora_scaling
        return base_out + kron_tt_out + lora_bypass

    # Dynamic rank adaptation
    # ------------------------------------------------------------------
    def adapt_tt_ranks(self, delta_r: int = 1) -> None:
        with torch.no_grad():
            delta_w = self.get_delta_w().float()
            try:
                _, S, _ = torch.linalg.svd(delta_w, full_matrices=False)
            except Exception:
                return
            total = S.sum().item()
            if total > 1e-9:
                p = S / total
                p = p[p > 1e-9]
                h_sv = float(-torch.sum(p * torch.log(p)).item())
            else:
                h_sv = 0.0

            if h_sv > self.entropy_threshold:
                direction = 1
            elif h_sv < self.entropy_threshold * 0.5:
                direction = -1
            else:
                direction = 0

            if direction == 0:
                return
            new_r1 = max(self.r_min, min(self.r_max, self.tt_ranks[1] + direction * delta_r))
            new_r2 = max(self.r_min, min(self.r_max, self.tt_ranks[2] + direction * delta_r))
            if new_r1 != self.tt_ranks[1] or new_r2 != self.tt_ranks[2]:

    def _resize_cores(self, new_r1: int, new_r2: int) -> None:
        # KTL-FIX-06: validate inputs.
        if new_r1 < 1 or new_r2 < 1:
            return
        old_r1, old_r2 = self.tt_ranks[1], self.tt_ranks[2]
        d1, d2, d3 = self.tt_shapes

        new_G1 = _init_orthogonal_spectral(
            (1, d1, new_r1), (d1, new_r1), self.G1.device, self.G1.dtype
        )
        new_G2 = _init_orthogonal_spectral(
            (new_r1, d2, new_r2), (new_r1 * d2, new_r2), self.G2.device, self.G2.dtype
        )
        new_G3 = _init_orthogonal_spectral(
            (new_r2, d3, 1), (new_r2, d3), self.G3.device, self.G3.dtype
        )

        min_r1 = min(old_r1, new_r1)
        min_r2 = min(old_r2, new_r2)
        try:
            new_G1[:, :, :min_r1] = self.G1.data[:, :, :min_r1]
            new_G2[:min_r1, :, :min_r2] = self.G2.data[:min_r1, :, :min_r2]
            new_G3[:min_r2, :, :] = self.G3.data[:min_r2, :, :]
        except Exception as e:
            logger.warning("KTL: _resize_cores copy failed: %s", e)
            return

        self.G1 = nn.Parameter(new_G1)
        self.G2 = nn.Parameter(new_G2)
        self.G3 = nn.Parameter(new_G3)
        self.tt_ranks[1] = new_r1
        self.tt_ranks[2] = new_r2

class SVDLinearLayer(nn.Module):
    """
    Low-rank SVD fallback used when TT cannot achieve the error threshold.

    Stores A (out, r) and B (r, in) with bias. Forward: y = (x @ B.T) @ A.T + bias.
    Parameter count: r * (out + in) + out (bias).
    """

    is_custom_attention: bool = False

    def __init__(
        self,
        A: torch.Tensor,  # (out, r)
        B: torch.Tensor,  # (r, in)
        bias: torch.Tensor | None = None,
        is_conv1d: bool = False,
    ) -> None:
        super().__init__()
        self.register_buffer("A", A, persistent=True)
        self.register_buffer("B", B, persistent=True)
        self.rank = A.shape[1]
        self.in_features = B.shape[1]
        self.out_features = A.shape[0]
        self.is_conv1d = is_conv1d
        if bias is not None:
            self.bias = nn.Parameter(bias.clone(), requires_grad=False)
        else:
            self.register_parameter("bias", None)

    @property
    def weight(self) -> torch.Tensor:
        W = torch.matmul(self.A.float(), self.B.float())
        return W.to(self.A.dtype)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        orig_shape = x.shape
        x_flat = x.reshape(-1, self.in_features)
        out = F.linear(F.linear(x_flat, self.B), self.A)
        if self.bias is not None:
            out = out + self.bias
        return out.view(*orig_shape[:-1], self.out_features)

    def compress(self, model: nn.Module | None = None) -> tuple[nn.Module, dict[str, Any]]:
        if model is not None:
            self.model = model
            self._original_params = self._count_params(self.model)

        self.compressed_layers = []
        self.tt_layers = []
        self.svd_layers = []
        fallback_layers: list[str] = []
        total_error = 0.0
        orig_bytes = 0
        comp_bytes = 0

        for name, module in list(self.model.named_modules()):
            if not self._is_target_module(name, module):
                continue

            base_layer = getattr(module, "base_layer", module)
            raw_w = base_layer.weight.data
            cls_name = base_layer.__class__.__name__.lower()
            is_conv1d = "conv1d" in cls_name or hasattr(base_layer, "nf")

            if hasattr(base_layer, "nx") and hasattr(base_layer, "nf"):
                in_f, out_f = int(base_layer.nx), int(base_layer.nf)
            elif hasattr(base_layer, "in_features") and hasattr(base_layer, "out_features"):
                in_f, out_f = int(base_layer.in_features), int(base_layer.out_features)
            elif is_conv1d:
                in_f, out_f = raw_w.shape[0], raw_w.shape[1]
            else:
                out_f, in_f = raw_w.shape[0], raw_w.shape[1]

            # 1) Try TT
            tt_layer = TTLinearLayer.from_linear(
                module,
                tt_rank=self.tt_rank,
                num_factors=self.num_factors,
                max_error=self.max_error,
                energy_threshold=self.energy_threshold,
                strict_rank=self.strict_rank,
            )

            chosen_layer = None
            method = None
            rel_err = None

            if tt_layer is not None:
                rel_err = getattr(tt_layer, "reconstruction_error", None)
                chosen_layer = tt_layer
                method = "tt"

            # 2) If TT failed, try SVD fallback
            if chosen_layer is None and self.fallback_policy in ("svd", "original"):
                svd_layer = _try_svd_fallback(
                    module,
                    self.max_error,
                    target_rank=self.tt_rank * 4,
                )
                if svd_layer is not None:
                    chosen_layer = svd_layer
                    method = "svd"
                    rel_err = getattr(svd_layer, "reconstruction_error", None)

            # 3) If both failed, decide fallback
            if chosen_layer is None:
                if self.fallback_policy == "original":
                    fallback_layers.append(name)
                    for p in module.parameters():
                        p.requires_grad = False
                    continue
                else:
                    fallback_layers.append(name)
                    continue

            # Splice into model
            if hasattr(module, "base_layer"):
                module.base_layer = chosen_layer
            else:
                replace_module(self.model, name, chosen_layer)

            # Byte accounting (TT-FIX-08)
            o_bytes = raw_w.numel() * raw_w.element_size()
            if getattr(base_layer, "bias", None) is not None and base_layer.bias is not None:
                o_bytes += base_layer.bias.numel() * base_layer.bias.element_size()
            c_bytes = sum(p.numel() * p.element_size() for p in chosen_layer.parameters())
            c_bytes += sum(
                b.numel() * b.element_size() for b in chosen_layer.buffers() if b is not None
            )

            orig_bytes += o_bytes
            comp_bytes += c_bytes
            if method == "tt":
                self.tt_layers.append(name)
            elif method == "svd":
                self.svd_layers.append(name)

            self.compressed_layers.append(
                {
                    "name": name,
                    "shape": (in_f, out_f),
                    "method": method,
                    "rel_error": float(rel_err) if rel_err is not None else None,
                }
            )
            if rel_err is not None:
                total_error += float(rel_err)

        # Forward sanity check
        forward_ok = True
        try:
            self.model.eval()
            with torch.no_grad():
                device = next(self.model.parameters()).device
                cfg = getattr(self.model, "config", None)
                vocab = getattr(cfg, "vocab_size", None) or 100
                dummy = torch.randint(0, min(vocab, 100), (1, 8), device=device)
                self.model(dummy)
        except Exception as e:
            logger.warning("TT forward sanity failed: %s", e)
            forward_ok = False

        try:
            self._compressed_params = self._count_params(self.model)
        except Exception:
            self._compressed_params = sum(p.numel() for p in self.model.parameters())

        stats = self.get_compression_stats()
        avg_err = total_error / max(1, len(self.compressed_layers))
        storage_ratio = orig_bytes / comp_bytes if comp_bytes > 0 else 1.0

        stats.update(
            {
                "compressed_layers": self.compressed_layers,
                "num_compressed_layers": len(self.compressed_layers),
                "fallback_layers": fallback_layers,
                "num_fallback": len(fallback_layers),
                "tt_layers": self.tt_layers,
                "svd_layers": self.svd_layers,
                "num_tt_layers": len(self.tt_layers),
                "num_svd_layers": len(self.svd_layers),
                "tt_rank": self.tt_rank,
                "num_factors": self.num_factors,
                "avg_reconstruction_error": avg_err,
                "max_error": self.max_error,
                "energy_threshold": self.energy_threshold,
                "fallback_policy": self.fallback_policy,
                "forward_check_success": forward_ok,
                "orig_storage_bytes": orig_bytes,
                "comp_storage_bytes": comp_bytes,
                "storage_compression_ratio": round(storage_ratio, 3),
                "storage_reduction_pct": round(100.0 * (1.0 - comp_bytes / max(1, orig_bytes)), 2),

            len(self.svd_layers),
            len(fallback_layers),
            avg_err,
            storage_ratio,

    """
    Tensor-Train (TT) linear layer.

    Weight W ∈ R^{out × in} is reshaped to a multi-dim tensor of shape
    (q_0, q_1, ..., p_0, p_1, ...) with q_dims × p_dims = out × in, then
    factorized into TT cores G_1 ... G_d of shape (r_{k-1}, p_k, q_k, r_k),
    with boundary ranks r_0 = r_d = 1.

    Forward contracts TT cores with x directly, without materializing W
    for d == 2; for larger d, an optimized generic path is used.

    # TT-FIX-01
    is_custom_attention: bool = False

    @staticmethod
    def _max_rank_for_compression(
        out_features: int,
        in_features: int,
        p_dims: tuple[int, ...],
        q_dims: tuple[int, ...],
        safety_margin: float = 0.9,

        """
        Max TT-rank so TT parameter count stays below dense count.
        Param count ≈ r · Σ_i (p_i q_i) for uniform rank.
        """
        if not p_dims or not q_dims or len(p_dims) != len(q_dims):
            return 1
        sum_pq = sum(p * q for p, q in zip(p_dims, q_dims))
        if sum_pq <= 0:
            return 1
        breakeven = (out_features * in_features) / sum_pq
        return max(1, int(breakeven * safety_margin))

    def materialize_weight(self) -> torch.Tensor:
        """Reconstruct W ∈ R^{out × in} by contracting TT-cores."""
        if (
            not self.training
            and self._cached_weight is not None
            and self._cached_version == self._current_version()
        ):
            return self._cached_weight

        res = self.cores[0][0]  # (p0, q0, r1)
        for i in range(1, self.d):
            core = self.cores[i]  # (r_i, p_i, q_i, r_{i+1})
            res = torch.tensordot(res, core, dims=([-1], [0]))
        res = res[..., 0]  # drop trailing r=1

        q_idx = [2 * k + 1 for k in range(self.d)]
        p_idx = [2 * k for k in range(self.d)]
        res = res.permute(*(q_idx + p_idx)).contiguous()
        W = res.view(self.out_features, self.in_features)

        if not self.training:
            self._cached_weight = W
            self._cached_version = self._current_version()
        return W

    @classmethod
    def from_linear(
        cls,
        linear_layer: nn.Module,
        tt_rank: int = 16,
        num_factors: int = 2,
        max_error: float = 0.20,
        max_retries: int = 3,
        energy_threshold: float = 0.995,
        strict_rank: bool = False,
    ) -> TTLinearLayer | None:
        """
        Factorize a pretrained Linear/Conv1D into TT-cores with adaptive rank.

        Rank per core is chosen adaptively: the smallest r such that the
        retained energy ≥ energy_threshold, capped by tt_rank and by the
        max_rank_for_compression budget.

        Returns None if the layer cannot be compressed within the error
        threshold and parameter budget.
        """
        base_layer = getattr(linear_layer, "base_layer", linear_layer)
        raw_w = base_layer.weight.data
        orig_dtype = raw_w.dtype
        orig_device = raw_w.device

        cls_name = base_layer.__class__.__name__.lower()
        is_conv1d = "conv1d" in cls_name or hasattr(base_layer, "nf")

        if hasattr(base_layer, "nx") and hasattr(base_layer, "nf"):
            in_f = int(base_layer.nx)
            out_f = int(base_layer.nf)
            W = raw_w.t().contiguous().float()
        elif hasattr(base_layer, "in_features") and hasattr(base_layer, "out_features"):
            in_f = int(base_layer.in_features)
            out_f = int(base_layer.out_features)
            W = (
                raw_w.t().contiguous().float()
                if (raw_w.shape == (in_f, out_f))
                else raw_w.float().contiguous()
            )
        elif is_conv1d:
            in_f, out_f = raw_w.shape[0], raw_w.shape[1]
            W = raw_w.t().contiguous().float()
        else:
            out_f, in_f = raw_w.shape[0], raw_w.shape[1]
            W = raw_w.float().contiguous()

        norm_orig = W.norm().item() + 1e-12

        p_dims, q_dims = factorize_dimension_pair(in_f, out_f, num_factors=num_factors)
        if int(np.prod(p_dims)) != in_f or int(np.prod(q_dims)) != out_f:
            return None

        d = len(p_dims)
        # Reshape to (q_0, q_1, ..., p_0, p_1, ...)
        W_tensor = W.view(*(list(q_dims) + list(p_dims)))
        perm = []
        for k in range(d):
            perm.append(d + k)  # p_k
            perm.append(k)  # q_k
        W_interleaved = W_tensor.permute(*perm).contiguous()

        max_rank = cls._max_rank_for_compression(out_f, in_f, p_dims, q_dims, safety_margin=0.9)
        rank_cap = min(int(tt_rank), max_rank) if not strict_rank else min(int(tt_rank), max_rank)

        bias = base_layer.bias.data if getattr(base_layer, "bias", None) is not None else None

        # ---- Build the layer with adaptive TT-SVD ----
        layer = cls(
            in_features=in_f,
            out_features=out_f,
            tt_rank=rank_cap,
            bias=bias is not None,
            shape=(p_dims, q_dims),
            dtype=orig_dtype,
            device=orig_device,
        )

        try:
            curr_matrix = W_interleaved.view(p_dims[0] * q_dims[0], -1)
            r_left = 1
            for i in range(d - 1):
                p_i, q_i = p_dims[i], q_dims[i]
                rows = r_left * p_i * q_i
                curr_matrix = curr_matrix.view(rows, -1)

                U, S, Vt = torch.linalg.svd(curr_matrix, full_matrices=False)
                # Adaptive rank: smallest r with retained energy >= threshold
                S_sq = S**2
                total_energy = S_sq.sum().clamp_min(1e-12)
                cum = torch.cumsum(S_sq, dim=0) / total_energy
                r_energy = int((cum >= energy_threshold).nonzero()[0].item()) + 1
                # Cap by: configured tt_rank, max_rank budget, and the shape bound
                r_right = min(r_energy, rank_cap, U.shape[1], S.numel())
                r_right = max(1, r_right)

                U_r = U[:, :r_right]
                S_r = S[:r_right]
                Vt_r = Vt[:r_right, :]

                core_data = U_r.view(r_left, p_i, q_i, r_right)
                with torch.no_grad():
                    layer.cores[i].data.zero_()
                    layer.cores[i].data.copy_(core_data.to(dtype=orig_dtype, device=orig_device))

                curr_matrix = torch.diag(S_r) @ Vt_r
                r_left = r_right

            p_last, q_last = p_dims[-1], q_dims[-1]
            final_core = curr_matrix.view(r_left, p_last, q_last, 1)
            with torch.no_grad():
                layer.cores[-1].data.zero_()
                layer.cores[-1].data.copy_(final_core.to(dtype=orig_dtype, device=orig_device))

            if bias is not None and layer.bias is not None:
                with torch.no_grad():
                    layer.bias.data.copy_(bias.to(dtype=orig_dtype, device=orig_device))

            # ---- Error check ----
            W_rec = layer.materialize_weight().float()
            rel_err = (torch.norm(W - W_rec) / norm_orig).item()

            # TT-FIX-05: fall back to SVD if TT error is too high.
            if rel_err > max_error:
                logger.info(
                    "TT: rel_err=%.4f > max_error=%.4f, trying SVD fallback",
                    rel_err,
                    max_error,
                )
                return None

            # Check parameter count
            tt_params = sum(p.numel() for p in layer.parameters())
            if tt_params >= out_f * in_f:
                logger.info(
                    "TT: no parameter reduction (%d >= %d), aborting",
                    tt_params,
                    out_f * in_f,
                )
                return None

            layer.reconstruction_error = float(rel_err)
            layer.tt_rank_actual = r_left
            return layer

        except Exception as e:
            logger.warning("TT-SVD failed: %s", e)
            return None

            W = raw_w.float().contiguous()

        norm_orig = W.norm().item() + 1e-12

        p_dims, q_dims = factorize_dimension_pair(in_f, out_f, num_factors=num_factors)
        if int(np.prod(p_dims)) != in_f or int(np.prod(q_dims)) != out_f:
            return None

        d = len(p_dims)
        # Reshape to (q_0, q_1, ..., p_0, p_1, ...)
        W_tensor = W.view(*(list(q_dims) + list(p_dims)))
        perm = []
        for k in range(d):
            perm.append(d + k)  # p_k
            perm.append(k)  # q_k
        W_interleaved = W_tensor.permute(*perm).contiguous()

        max_rank = cls._max_rank_for_compression(out_f, in_f, p_dims, q_dims, safety_margin=0.9)
        rank_cap = min(int(tt_rank), max_rank) if not strict_rank else min(int(tt_rank), max_rank)

        bias = base_layer.bias.data if getattr(base_layer, "bias", None) is not None else None

        # ---- Build the layer with adaptive TT-SVD ----
        layer = cls(
            in_features=in_f,
            out_features=out_f,
            tt_rank=rank_cap,
            bias=bias is not None,
            shape=(p_dims, q_dims),
            dtype=orig_dtype,
            device=orig_device,
        )

        try:
            curr_matrix = W_interleaved.view(p_dims[0] * q_dims[0], -1)
            r_left = 1
            for i in range(d - 1):
                p_i, q_i = p_dims[i], q_dims[i]
                rows = r_left * p_i * q_i
                curr_matrix = curr_matrix.view(rows, -1)

                U, S, Vt = torch.linalg.svd(curr_matrix, full_matrices=False)
                # Adaptive rank: smallest r with retained energy >= threshold
                S_sq = S**2
                total_energy = S_sq.sum().clamp_min(1e-12)
                cum = torch.cumsum(S_sq, dim=0) / total_energy
                r_energy = int((cum >= energy_threshold).nonzero()[0].item()) + 1
                # Cap by: configured tt_rank, max_rank budget, and the shape bound
                r_right = min(r_energy, rank_cap, U.shape[1], S.numel())
                r_right = max(1, r_right)

                U_r = U[:, :r_right]
                S_r = S[:r_right]
                Vt_r = Vt[:r_right, :]

                core_data = U_r.view(r_left, p_i, q_i, r_right)
                with torch.no_grad():
                    layer.cores[i].data.zero_()
                    layer.cores[i].data.copy_(core_data.to(dtype=orig_dtype, device=orig_device))

                curr_matrix = torch.diag(S_r) @ Vt_r
                r_left = r_right

            p_last, q_last = p_dims[-1], q_dims[-1]
            final_core = curr_matrix.view(r_left, p_last, q_last, 1)
            with torch.no_grad():
                layer.cores[-1].data.zero_()
                layer.cores[-1].data.copy_(final_core.to(dtype=orig_dtype, device=orig_device))

            if bias is not None and layer.bias is not None:
                with torch.no_grad():
                    layer.bias.data.copy_(bias.to(dtype=orig_dtype, device=orig_device))

            # ---- Error check ----
            W_rec = layer.materialize_weight().float()
            rel_err = (torch.norm(W - W_rec) / norm_orig).item()

            # TT-FIX-05: fall back to SVD if TT error is too high.
            if rel_err > max_error:
                logger.info(
                    "TT: rel_err=%.4f > max_error=%.4f, trying SVD fallback",
                    rel_err,
                    max_error,
                )
                return None

            # Check parameter count
            tt_params = sum(p.numel() for p in layer.parameters())
            if tt_params >= out_f * in_f:
                logger.info(
                    "TT: no parameter reduction (%d >= %d), aborting",
                    tt_params,
                    out_f * in_f,
                )
                return None

            layer.reconstruction_error = float(rel_err)
            layer.tt_rank_actual = r_left
            return layer

        except Exception as e:
            logger.warning("TT-SVD failed: %s", e)
            return None

Tensor-Train (TT) linear layer.

Weight W ∈ R^{out × in} is reshaped to a multi-dim tensor of shape (q_0, q_1, …, p_0, p_1, …) with q_dims × p_dims = out × in, then factorized into TT cores G_1 … G_d of shape (r_{k-1}, p_k, q_k, r_k), with boundary ranks r_0 = r_d = 1.

Forward contracts TT cores with x directly, without materializing W for d == 2; for larger d, an optimized generic path is used.

Initialize internal Module state, shared by both nn.Module and ScriptModule.

def materialize_weight(self) -> torch.Tensor:
    """Reconstruct W ∈ R^{out × in} by contracting TT-cores."""
    if (
        not self.training
        and self._cached_weight is not None
        and self._cached_version == self._current_version()
    ):
        return self._cached_weight

    res = self.cores[0][0]  # (p0, q0, r1)
    for i in range(1, self.d):
        core = self.cores[i]  # (r_i, p_i, q_i, r_{i+1})
        res = torch.tensordot(res, core, dims=([-1], [0]))
    res = res[..., 0]  # drop trailing r=1

    q_idx = [2 * k + 1 for k in range(self.d)]
    p_idx = [2 * k for k in range(self.d)]
    res = res.permute(*(q_idx + p_idx)).contiguous()
    W = res.view(self.out_features, self.in_features)

    if not self.training:
        self._cached_weight = W
        self._cached_version = self._current_version()
    return W
