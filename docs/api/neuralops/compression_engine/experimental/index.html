class BaseCompressor:
    """Base class for all compressors. Not a nn.Module to avoid parameter storage."""

    def __init__(self, model: nn.Module, name: str = "base", **kwargs) -> None:
        self.model = model
        self.name = name
        self.kwargs = kwargs
        self._original_params = self._count_params(model)

    def __call__(self, *args, **kwargs) -> Any:
        """Allow compressor instance to act as a callable model forward wrapper."""
        if not getattr(self, "_compressed_once", False):
            self.compress()
            self._compressed_once = True
        return self.model(*args, **kwargs)

    @abstractmethod
    def compress(self) -> tuple[nn.Module, dict[str, Any]]:
        """Return (compressed_model, statistics)."""
        raise NotImplementedError("Subclasses of BaseCompressor must implement compress()")

    def _count_params(self, model: nn.Module) -> int:
        """count params."""
        return sum(p.numel() for p in model.parameters())

    def get_compression_stats(self) -> dict[str, Any]:
        """Return the compression stats."""
        compressed = self._count_params(self.model)
        orig = self._original_params
        return {
            "compressor_name": self.name,
            "original_parameters": orig,
            "compressed_parameters": compressed,
            "saved_parameters": orig - compressed,
            "compression_ratio": compressed / orig if orig > 0 else 1.0,
            "reduction_percentage": ((orig - compressed) / orig * 100 if orig > 0 else 0.0),
        }

    def summary(self) -> str:
        """Summary."""
        s = self.get_compression_stats()
        return (
            f"\n========================================\n"
            f"Compression Summary: {self.name}\n"
            f"========================================\n"
            f"Original parameters:  {s['original_parameters']:,}\n"
            f"Compressed parameters: {s['compressed_parameters']:,}\n"
            f"Parameters saved:     {s['saved_parameters']:,}\n"
            f"Compression ratio:    {s['compression_ratio']:.3f}\n"
            f"Reduction:            {s['reduction_percentage']:.2f}%\n"
            f"========================================\n"
        )

@COMPRESSOR_REGISTRY.register("flat_llm")
class FLATLLMCompressor(BaseCompressor):
    """
    FLAT-LLM compressor: Activation-aware output-channel pruning.

    Pipeline:
      1. Collect per-output-channel energy from calibration activations.
      2. Rank channels by energy.
      3. Keep top-k channels (k = round(keep_ratio * out_features)).
      4. Replace Linear with PrunedChannelLinear (factorized storage).

    Optionally applies 2:4 structured sparsity (compute-only; no storage savings
    unless a sparse kernel is used).
    """

    # FLT-FIX-09
    is_custom_attention: bool = False

    def __init__(
        self,
        model: nn.Module,
        tokenizer: Any | None = None,
        keep_ratio: float = 0.7,
        damping: float = 1e-4,
        max_error: float = 0.35,
        target_attention_only: bool = False,
        fallback_to_original: bool = True,
        calibration_loader: Any | None = None,
        calibration_texts: list[str] | None = None,
        device: str | torch.device | None = None,
        structured_24: bool = False,
        target_modules: list[str] | None = None,
        cfg: Any | None = None,
        num_calibration_batches: int = 8,
        **kwargs: Any,
    ) -> None:
        super().__init__(model, name="flat_llm", **kwargs)
        cfg = cfg or kwargs.get("comp_config")
        self.cfg = cfg

        self.tokenizer = tokenizer or getattr(model, "tokenizer", None)
        raw_kr = _cfg_get(cfg, "flat_llm_keep_ratio", _cfg_get(cfg, "keep_ratio", keep_ratio))
        self.keep_ratio = max(0.05, min(float(raw_kr), 1.0))
        # FLT-FIX-04
        self.damping = max(
            0.0,
            float(_cfg_get(cfg, "flat_llm_damping", _cfg_get(cfg, "damping", damping))),
        )
        self.structured_24 = bool(
            _cfg_get(
                cfg,
                "flat_llm_structured_24",
                _cfg_get(cfg, "structured_24", structured_24),
            )
        )
        c_max_error = _cfg_get(cfg, "flat_llm_max_error", _cfg_get(cfg, "max_error", max_error))
        if c_max_error is None or c_max_error == 0.35:
            c_max_error = adaptive_max_error(int(self.keep_ratio * 768), 768, 768, "distilbert")
            c_max_error = max(c_max_error, 0.60)

        self.max_error = float(c_max_error)
        self.target_attention_only = bool(
            _cfg_get(
                cfg,
                "flat_llm_target_attention_only",
                _cfg_get(cfg, "target_attention_only", target_attention_only),
            )
        )
        self.fallback_to_original = bool(
            _cfg_get(
                cfg,
                "flat_llm_fallback_to_original",
                _cfg_get(cfg, "fallback_to_original", fallback_to_original),
            )
        )
        self.calibration_loader = calibration_loader or _cfg_get(cfg, "calibration_loader", None)
        self.calibration_texts = calibration_texts
        self.target_modules = target_modules
        self.target_device = (
            torch.device(device)
            if device is not None
            else next(model.parameters(), torch.tensor(0.0)).device
        )
        self.num_calibration_batches = max(
            1, int(_cfg_get(cfg, "num_calibration_batches", num_calibration_batches))
        )

        self.compressed_layers: list[str] = []
        self.fallback_layers: list[str] = []
        self.skipped_layers: list[str] = []
        self.layer_errors: dict[str, float] = {}
        self.layer_keep_ratios: dict[str, float] = {}
        self._compressed_params: int = 0

    # ------------------------------------------------------------------
    def _is_target_module(self, name: str, module: nn.Module) -> bool:
        name_lower = name.lower()
        if any(part in name.split(".") for part in ["lora_A", "lora_B", "base_layer"]):
            return False
        if any(
            skip in name_lower
            for skip in (
                "classifier",
                "score",
                "lm_head",
                "pooler",
                "embedding",
                "router",
                "gate",
                "norm",
                "layernorm",
                "rmsnorm",
            )
        ):
            return False

        if isinstance(
            module,
            (
                nn.LayerNorm,
                nn.RMSNorm,
                nn.Embedding,
                nn.BatchNorm1d,
                nn.BatchNorm2d,
                PrunedChannelLinear,
            ),
        ):
            return False

        if self.target_attention_only:
            if not any(attn_kw in name_lower for attn_kw in _ATTN_NAMES):
                return False

        target_mod = getattr(module, "base_layer", module)
        cls_name = target_mod.__class__.__name__.lower()
        is_linear = isinstance(target_mod, nn.Linear)
        is_conv1d = "conv1d" in cls_name or hasattr(target_mod, "nf")
        if not (is_linear or is_conv1d):
            return False

        try:
            if not should_compress_linear(self.model, name):
                return False
        except Exception:
            pass

        if self.target_modules is not None:
            return any(t in name for t in self.target_modules)
        return True

    # ------------------------------------------------------------------
    # Calibration: per-output-channel energy (FLT-FIX-05)
    # ------------------------------------------------------------------
    def _normalize_calibration_batches(self, calib_inputs: Any) -> list[Any]:
        if calib_inputs is None:
            return []
        if isinstance(calib_inputs, (torch.Tensor, dict)):
            return [calib_inputs]
        if isinstance(calib_inputs, (list, tuple)):
            if len(calib_inputs) == 0:
                return []
            first = calib_inputs[0]
            if isinstance(first, (dict, torch.Tensor)):
                return list(calib_inputs)
            return [calib_inputs]
        return [calib_inputs]

    def _collect_layer_outputs(self) -> dict[str, torch.Tensor]:
        """
        Capture per-output-channel energy across calibration batches.

        Energy metric: E[x_out^2] per output channel, computed as the mean of
        per-batch variance of the output activations. For Conv1D, the axis is
        transposed accordingly.
        """
        self.model.eval()
        energy_sum: dict[str, torch.Tensor] = {}
        batch_count: dict[str, int] = {}
        hooks = []

        def make_hook(layer_name: str):
            def hook(module, inputs, output):
                if not isinstance(output, torch.Tensor):
                    return
                out = output.detach().float()
                target_mod = getattr(module, "base_layer", module)
                cls_name = target_mod.__class__.__name__.lower()
                is_conv1d = "conv1d" in cls_name or hasattr(target_mod, "nf")
                out_f = target_mod.weight.shape[1] if is_conv1d else target_mod.weight.shape[0]
                if out.shape[-1] != out_f:
                    return
                flat = out.reshape(-1, out_f).cpu()
                if flat.size(0) >= 2:
                    var = flat.var(dim=0)
                else:
                    var = flat.abs().mean(dim=0)
                if layer_name not in energy_sum:
                    energy_sum[layer_name] = var
                    batch_count[layer_name] = 1
                else:
                    energy_sum[layer_name] += var
                    batch_count[layer_name] += 1

            return hook

        for name, module in self.model.named_modules():
            if self._is_target_module(name, module):
                hooks.append(module.register_forward_hook(make_hook(name)))

        try:
            calib_inputs = get_calibration_tensors(
                self.model,
                tokenizer=self.tokenizer,
                calibration_loader=self.calibration_loader,
                calibration_texts=self.calibration_texts,
                device=self.target_device,
            )
            batches = self._normalize_calibration_batches(calib_inputs)
            n = min(len(batches), self.num_calibration_batches)

            with torch.no_grad():
                for i in range(n):
                    b = batches[i]
                    try:
                        if isinstance(b, dict):
                            self.model(**b)
                        elif isinstance(b, torch.Tensor):
                            self.model(b)
                        elif isinstance(b, (list, tuple)):
                            self.model(*b)
                    except Exception as e:
                        logger.debug("FLAT-LLM calibration batch %d failed: %s", i, e)
        finally:
            for h in hooks:
                h.remove()

        aggregated: dict[str, torch.Tensor] = {}
        for name, esum in energy_sum.items():
            aggregated[name] = (esum / max(1, batch_count[name])).to(self.target_device)
        return aggregated

    # ------------------------------------------------------------------
    def compress(self, **kwargs: Any) -> tuple[nn.Module, dict[str, Any]]:
        from neuralops.core.model_utils import replace_module

        outputs_map = self._collect_layer_outputs()
        self.compressed_layers = []
        self.fallback_layers = []
        self.skipped_layers = []
        self.layer_errors = {}
        self.layer_keep_ratios = {}

        # Honest accounting: storage params, not "effective" made-up numbers.
        total_orig_storage = 0
        total_comp_storage = 0

        targets = [
            (name, mod)
            for name, mod in self.model.named_modules()
            if self._is_target_module(name, mod)
        ]

        if not targets:
            logger.warning("FLAT-LLM: no compressible layers found.")
            return self.model, self.get_compression_stats()

        for name, module in targets:
            target_mod = getattr(module, "base_layer", module)
            cls_name = target_mod.__class__.__name__.lower()
            is_conv1d = "conv1d" in cls_name or hasattr(target_mod, "nf")
            raw_w = target_mod.weight.data
            device = raw_w.device
            dtype = raw_w.dtype

            W = raw_w.t().float().contiguous() if is_conv1d else raw_w.float()
            d_out, d_in = W.shape
            orig_layer_storage = d_out * d_in
            total_orig_storage += orig_layer_storage
            norm_w = torch.norm(W, p="fro").item()

            # Activation-aware saliency
            act = outputs_map.get(name, None)
            if act is not None and act.numel() > 0 and act.shape[0] == d_out:
                saliency = act.float().clamp(min=0.0).to(device)
            else:
                saliency = W.pow(2).sum(dim=1)

            # Apply damping to stabilize near-zero channels
            if self.damping > 0:
                saliency = saliency + self.damping * saliency.mean().clamp_min(1e-12)

            # ---- Decide kept channels ----
            if self.structured_24:
                # 2:4 is a compute-only structured sparsity. For storage, we still
                # keep all output channels; only the mask reduces the values.
                sal_mat = saliency.unsqueeze(1).expand_as(W)
                W_pruned, _mask = apply_24_structured_mask(W, saliency=sal_mat, dim=0)
                # Storage: unchanged (dense)
                kept_idx = torch.arange(d_out, device=device)
                W_kept = W_pruned.contiguous()
                keep_ratio_actual = 1.0
                bias_kept = (
                    target_mod.bias.data.float()
                    if getattr(target_mod, "bias", None) is not None and target_mod.bias is not None
                    else None
                )
                comp_layer_storage = orig_layer_storage
            else:
                keep_dims = max(1, min(int(round(d_out * self.keep_ratio)), d_out))
                _, keep_idx = torch.topk(saliency, k=keep_dims, largest=True)
                keep_idx, _ = torch.sort(keep_idx)  # stable order
                kept_idx = keep_idx.to(device)
                W_kept = W.index_select(0, kept_idx).contiguous()
                keep_ratio_actual = keep_dims / d_out

                # Compute error vs original (with pruned rows treated as zero)
                W_pruned = torch.zeros_like(W)
                W_pruned.index_copy_(0, kept_idx, W_kept)

                # Bias handling: keep only the corresponding entries
                if getattr(target_mod, "bias", None) is not None and target_mod.bias is not None:
                    bias_full = target_mod.bias.data.float()
                    bias_kept = bias_full.index_select(0, kept_idx).contiguous()
                else:
                    bias_kept = None

                # Storage: k * in instead of out * in
                comp_layer_storage = int(keep_dims) * d_in

            # ---- Error check ----
            if norm_w < 1e-12:
                final_err = 0.0
            else:
                if self.structured_24:
                    final_err = float(torch.norm(W - W_pruned, p="fro").item() / norm_w)
                else:
                    W_pruned_full = torch.zeros_like(W)
                    W_pruned_full.index_copy_(0, kept_idx, W_kept)
                    final_err = float(torch.norm(W - W_pruned_full, p="fro").item() / norm_w)

            self.layer_errors[name] = final_err
            self.layer_keep_ratios[name] = float(keep_ratio_actual)

            if final_err > self.max_error:
                logger.warning(
                    "FLAT-LLM error %.4f on %s exceeds %.4f.",
                    final_err,
                    name,
                    self.max_error,
                )
                if self.fallback_to_original:
                    self.fallback_layers.append(name)
                    total_comp_storage += orig_layer_storage
                    for p in module.parameters():
                        p.requires_grad = False
                    continue

            # ---- Build the new module ----
            if self.structured_24:
                # No storage reduction: keep original Linear, just overwrite
                # its weight with the masked values (compute-only sparsity).
                new_module = target_mod
                with torch.no_grad():
                    if is_conv1d:
                        target_mod.weight.data.copy_(W_kept.t().to(dtype=dtype, device=device))
                    else:
                        target_mod.weight.data.copy_(W_kept.to(dtype=dtype, device=device))
            else:
                new_module = PrunedChannelLinear(
                    in_features=d_in,
                    original_out_features=d_out,
                    kept_idx=kept_idx.cpu(),
                    W_kept=W_kept.to(dtype=dtype, device=device),
                    bias_kept=(
                        bias_kept.to(dtype=dtype, device=device) if bias_kept is not None else None
                    ),
                )

            # ---- Splice into the model (preserve LoRA wrappers) ----
            if hasattr(module, "base_layer"):
                module.base_layer = new_module
                # Keep LoRA params trainable
                for pn, p in module.named_parameters():
                    if "lora_" in pn.lower() and not pn.startswith("base_layer"):
                        p.requires_grad = True
            else:
                replace_module(self.model, name, new_module)

            self.compressed_layers.append(name)
            total_comp_storage += comp_layer_storage

        # ---- Forward sanity check (FLT-FIX-07) ----
        forward_check_success = True
        try:
            with torch.no_grad():
                first_p = next(self.model.parameters(), None)
                dev = first_p.device if first_p is not None else torch.device("cpu")
                dummy = torch.randint(0, 100, (1, 8), device=dev, dtype=torch.long)
                attn = torch.ones_like(dummy)
                try:
                    self.model(input_ids=dummy, attention_mask=attn)
                except Exception:
                    try:
                        self.model(dummy)
                    except Exception:
                        first_linear = next(
                            (
                                m
                                for m in self.model.modules()
                                if isinstance(m, (nn.Linear, PrunedChannelLinear))
                            ),
                            None,
                        )
                        in_dim = getattr(first_linear, "in_features", 8)
                        dt = first_p.dtype if first_p is not None else torch.float32
                        self.model(torch.randn(1, 8, in_dim, device=dev, dtype=dt))
        except Exception as e:
            logger.debug("FLAT-LLM forward sanity check failed: %s", e)
            forward_check_success = False

        # Honest storage ratio
        storage_ratio = total_orig_storage / total_comp_storage if total_comp_storage > 0 else 1.0
        effective_ratio = total_comp_storage / total_orig_storage if total_orig_storage > 0 else 1.0

        try:
            self._compressed_params = self._count_params(self.model)
        except Exception:
            self._compressed_params = sum(p.numel() for p in self.model.parameters())

        try:
            stats = self.get_compression_stats()
        except Exception:
            stats = {}

        avg_err = (
            sum(self.layer_errors[k] for k in self.compressed_layers)
            / max(1, len(self.compressed_layers))
            if self.compressed_layers
            else 0.0
        )

        stats.update(
            {
                "compressed_layers": self.compressed_layers,
                "fallback_layers": self.fallback_layers,
                "skipped_layers": self.skipped_layers,
                "layer_errors": self.layer_errors,
                "layer_keep_ratios": self.layer_keep_ratios,
                "keep_ratio": self.keep_ratio,
                "damping": self.damping,
                "structured_24": self.structured_24,

    """
    Linear layer with parameter sharing via a compact hash table.

      W[i, j] = sign[i, j] · weight_table[hash_idx[i, j]]
      y = x @ W^T + bias + (alpha/r) · (x @ lora_A^T) @ lora_B^T

    Storage:
      weight_table:   n_buckets · dtype_size        (the only trained/learned params)
      bias:           out_features · dtype_size
      lora_A, lora_B: (optional) r · (in + out) · dtype_size
      hash_idx:       out · in · 4 bytes  (int32, non-persistent)
      sign_val:       out · in · 1 byte   (int8, non-persistent)

    For compression to actually reduce storage, `out · in · 5` bytes must be
    less than `out · in · dtype_size`. With fp16 (2 bytes) this is almost
    never true unless a sparse storage format is used. Honest accounting is
    therefore critical (see HN-FIX-08).

        cls,

        lora_alpha: float = 1.0,
        fit_lora_residual: bool = True,
        **_ignored: Any,
    ) -> HashedLinear:
        """
        Build a HashedLinear by projecting W onto the hash basis.

        Reconstruction: W_hat[i,j] = sign[i,j] · mean_{bucket}(W · sign)
        which is the least-squares optimum under the HashedNet constraint.

        HN-FIX-07: no energy rescaling (it increases error).
        """
        target_mod = getattr(layer, "base_layer", layer)
        cls_name = target_mod.__class__.__name__.lower()
        is_conv1d = "conv1d" in cls_name or hasattr(target_mod, "nf")
        raw_w = target_mod.weight.data

        # Normalize to (out, in)
        if hasattr(target_mod, "nx") and hasattr(target_mod, "nf"):
            in_f = int(target_mod.nx)
            out_f = int(target_mod.nf)
            W = raw_w.t().contiguous().float()
        elif hasattr(target_mod, "in_features") and hasattr(target_mod, "out_features"):
            in_f = int(target_mod.in_features)
            out_f = int(target_mod.out_features)
            if raw_w.shape == (in_f, out_f):
                W = raw_w.t().contiguous().float()
            else:
                W = raw_w.view(out_f, in_f).float()
        elif is_conv1d:
            in_f, out_f = raw_w.shape[0], raw_w.shape[1]
            W = raw_w.t().contiguous().float()
        else:
            out_f, in_f = raw_w.shape[0], raw_w.shape[1]
            W = raw_w.float()

        # Clamp buckets to a sane range
        n_buckets = max(16, min(int(n_buckets), out_f * in_f))

        bias = (
            target_mod.bias.data
            if getattr(target_mod, "bias", None) is not None and target_mod.bias is not None
            else None
        )

        # Detect existing LoRA
        existing_lora_A = getattr(layer, "lora_A", None)
        existing_lora_B = getattr(layer, "lora_B", None)
        effective_lora_r = max(0, int(lora_rank))
        if existing_lora_A is not None and hasattr(existing_lora_A, "weight"):
            try:
                effective_lora_r = int(existing_lora_A.weight.shape[0])
            except Exception:
                pass

        hashed = cls(
            in_features=in_f,
            out_features=out_f,
            n_buckets=n_buckets,
            bias=bias is not None,
            use_sign=use_sign,
            lora_rank=effective_lora_r,
            lora_alpha=lora_alpha,
            dtype=raw_w.dtype,
            device=raw_w.device,
        )

        # ---- Least-squares fit for weight_table ----
        W_flat = W.flatten()
        h_flat = hashed.hash_idx.reshape(-1).long()
        s_flat = hashed.sign_val.reshape(-1).to(torch.float32)
        if not use_sign:
            s_flat = torch.ones_like(s_flat)

        # accum[b] = sum_{i,j: h(i,j)=b} W[i,j] · sign[i,j]
        accum = torch.zeros(n_buckets, dtype=torch.float32, device=W.device)
        weights_signed = W_flat * s_flat
        accum.scatter_add_(0, h_flat, weights_signed)

        counts = torch.bincount(h_flat, minlength=n_buckets).float()

        # HN-FIX-12: deterministic init for empty buckets
        non_zero = counts > 0
        mean_per_bucket = torch.zeros(n_buckets, dtype=torch.float32, device=W.device)
        mean_per_bucket[non_zero] = accum[non_zero] / counts[non_zero]

        if (~non_zero).any():
            bound = 1.0 / math.sqrt(max(1, in_f))
            gen = torch.Generator(device=W.device).manual_seed(0xCAFEBABE)
            n_empty = int((~non_zero).sum().item())
            mean_per_bucket[~non_zero] = torch.empty(
                n_empty, device=W.device, dtype=torch.float32
            ).uniform_(-bound, bound, generator=gen)

        with torch.no_grad():
            hashed.weight_table.data.copy_(
                mean_per_bucket.to(dtype=raw_w.dtype, device=raw_w.device)
            )
            if bias is not None and hashed.bias is not None:
                hashed.bias.data.copy_(bias.to(dtype=raw_w.dtype, device=raw_w.device))

        # ---- LoRA transfer / fit ----
        if (
            existing_lora_A is not None
            and existing_lora_B is not None
            and hashed.lora_A is not None
        ):
            w_A = getattr(existing_lora_A, "weight", existing_lora_A)
            w_B = getattr(existing_lora_B, "weight", existing_lora_B)
            try:
                if w_A.shape == hashed.lora_A.shape and w_B.shape == hashed.lora_B.shape:
                    hashed.lora_A.data.copy_(w_A.data)
                    hashed.lora_B.data.copy_(w_B.data)
            except Exception as e:
                logger.debug("HashedLinear existing LoRA copy failed: %s", e)

        elif fit_lora_residual and hashed.lora_A is not None and hashed.lora_B is not None:
            # Fit residual W − W_hat with low-rank SVD
            try:
                with torch.no_grad():
                    W_hat = hashed.get_weight().float()
                    R = W - W_hat
                    r_eff = min(hashed.lora_rank, R.shape[0], R.shape[1])
                    if r_eff > 0:
                        U, S, Vt = torch.linalg.svd(R, full_matrices=False)
                        # lora_B @ lora_A / scaling = R (approx)
                        sqrt_S = torch.sqrt(torch.clamp(S[:r_eff], min=1e-12))
                        B_init = U[:, :r_eff] * sqrt_S.unsqueeze(0)
                        A_init = (sqrt_S.unsqueeze(1) * Vt[:r_eff, :]) / max(1e-6, hashed.scaling)
                        hashed.lora_B.data.copy_(
                            B_init.to(dtype=hashed.lora_B.dtype, device=hashed.lora_B.device)
                        )
                        hashed.lora_A.data.copy_(
                            A_init.to(dtype=hashed.lora_A.dtype, device=hashed.lora_A.device)
                        )
            except Exception as e:
                logger.debug("HashedLinear residual SVD fit failed: %s", e)

        hashed._invalidate_cache()
        return hashed

            out = out + lora

    @classmethod
    def from_linear(
        cls,
        layer: nn.Module,
        n_buckets: int = 65536,
        use_sign: bool = True,
        lora_rank: int = 0,
        lora_alpha: float = 1.0,
        fit_lora_residual: bool = True,
        **_ignored: Any,
    ) -> HashedLinear:
        """
        Build a HashedLinear by projecting W onto the hash basis.

        Reconstruction: W_hat[i,j] = sign[i,j] · mean_{bucket}(W · sign)
        which is the least-squares optimum under the HashedNet constraint.

        HN-FIX-07: no energy rescaling (it increases error).
        """
        target_mod = getattr(layer, "base_layer", layer)
        cls_name = target_mod.__class__.__name__.lower()
        is_conv1d = "conv1d" in cls_name or hasattr(target_mod, "nf")
        raw_w = target_mod.weight.data

        # Normalize to (out, in)
        if hasattr(target_mod, "nx") and hasattr(target_mod, "nf"):
            in_f = int(target_mod.nx)
            out_f = int(target_mod.nf)
            W = raw_w.t().contiguous().float()
        elif hasattr(target_mod, "in_features") and hasattr(target_mod, "out_features"):
            in_f = int(target_mod.in_features)
            out_f = int(target_mod.out_features)
            if raw_w.shape == (in_f, out_f):
                W = raw_w.t().contiguous().float()
            else:
                W = raw_w.view(out_f, in_f).float()
        elif is_conv1d:
            in_f, out_f = raw_w.shape[0], raw_w.shape[1]
            W = raw_w.t().contiguous().float()
        else:
            out_f, in_f = raw_w.shape[0], raw_w.shape[1]
            W = raw_w.float()

        # Clamp buckets to a sane range
        n_buckets = max(16, min(int(n_buckets), out_f * in_f))

        bias = (
            target_mod.bias.data
            if getattr(target_mod, "bias", None) is not None and target_mod.bias is not None
            else None
        )

        # Detect existing LoRA
        existing_lora_A = getattr(layer, "lora_A", None)
        existing_lora_B = getattr(layer, "lora_B", None)
        effective_lora_r = max(0, int(lora_rank))
        if existing_lora_A is not None and hasattr(existing_lora_A, "weight"):
            try:
                effective_lora_r = int(existing_lora_A.weight.shape[0])
            except Exception:
                pass

        hashed = cls(
            in_features=in_f,
            out_features=out_f,
            n_buckets=n_buckets,
            bias=bias is not None,
            use_sign=use_sign,
            lora_rank=effective_lora_r,
            lora_alpha=lora_alpha,
            dtype=raw_w.dtype,
            device=raw_w.device,
        )

        # ---- Least-squares fit for weight_table ----
        W_flat = W.flatten()
        h_flat = hashed.hash_idx.reshape(-1).long()
        s_flat = hashed.sign_val.reshape(-1).to(torch.float32)
        if not use_sign:
            s_flat = torch.ones_like(s_flat)

        # accum[b] = sum_{i,j: h(i,j)=b} W[i,j] · sign[i,j]
        accum = torch.zeros(n_buckets, dtype=torch.float32, device=W.device)
        weights_signed = W_flat * s_flat
        accum.scatter_add_(0, h_flat, weights_signed)

        counts = torch.bincount(h_flat, minlength=n_buckets).float()

        # HN-FIX-12: deterministic init for empty buckets
        non_zero = counts > 0
        mean_per_bucket = torch.zeros(n_buckets, dtype=torch.float32, device=W.device)
        mean_per_bucket[non_zero] = accum[non_zero] / counts[non_zero]

        if (~non_zero).any():
            bound = 1.0 / math.sqrt(max(1, in_f))
            gen = torch.Generator(device=W.device).manual_seed(0xCAFEBABE)
            n_empty = int((~non_zero).sum().item())
            mean_per_bucket[~non_zero] = torch.empty(
                n_empty, device=W.device, dtype=torch.float32
            ).uniform_(-bound, bound, generator=gen)

        with torch.no_grad():
            hashed.weight_table.data.copy_(
                mean_per_bucket.to(dtype=raw_w.dtype, device=raw_w.device)
            )
            if bias is not None and hashed.bias is not None:
                hashed.bias.data.copy_(bias.to(dtype=raw_w.dtype, device=raw_w.device))

        # ---- LoRA transfer / fit ----
        if (
            existing_lora_A is not None
            and existing_lora_B is not None
            and hashed.lora_A is not None
        ):
            w_A = getattr(existing_lora_A, "weight", existing_lora_A)
            w_B = getattr(existing_lora_B, "weight", existing_lora_B)
            try:
                if w_A.shape == hashed.lora_A.shape and w_B.shape == hashed.lora_B.shape:
                    hashed.lora_A.data.copy_(w_A.data)
                    hashed.lora_B.data.copy_(w_B.data)
            except Exception as e:
                logger.debug("HashedLinear existing LoRA copy failed: %s", e)

        elif fit_lora_residual and hashed.lora_A is not None and hashed.lora_B is not None:
            # Fit residual W − W_hat with low-rank SVD
            try:
                with torch.no_grad():
                    W_hat = hashed.get_weight().float()
                    R = W - W_hat
                    r_eff = min(hashed.lora_rank, R.shape[0], R.shape[1])
                    if r_eff > 0:
                        U, S, Vt = torch.linalg.svd(R, full_matrices=False)
                        # lora_B @ lora_A / scaling = R (approx)
                        sqrt_S = torch.sqrt(torch.clamp(S[:r_eff], min=1e-12))
                        B_init = U[:, :r_eff] * sqrt_S.unsqueeze(0)
                        A_init = (sqrt_S.unsqueeze(1) * Vt[:r_eff, :]) / max(1e-6, hashed.scaling)
                        hashed.lora_B.data.copy_(
                            B_init.to(dtype=hashed.lora_B.dtype, device=hashed.lora_B.device)
                        )
                        hashed.lora_A.data.copy_(
                            A_init.to(dtype=hashed.lora_A.dtype, device=hashed.lora_A.device)
                        )
            except Exception as e:
                logger.debug("HashedLinear residual SVD fit failed: %s", e)

        hashed._invalidate_cache()
        return hashed

Linear layer with parameter sharing via a compact hash table.

W[i, j] = sign[i, j] · weight_table[hash_idx[i, j]] y = x @ W^T + bias + (alpha/r) · (x @ lora_A^T) @ lora_B^T

Storage -----= weight_table: n_buckets · dtype_size (the only trained/learned params) bias: out_features · dtype_size lora_A, lora_B: (optional) r · (in + out) · dtype_size hash_idx: out · in · 4 bytes (int32, non-persistent) sign_val: out · in · 1 byte (int8, non-persistent)

For compression to actually reduce storage, out · in · 5 bytes must be less than out · in · dtype_size. With fp16 (2 bytes) this is almost never true unless a sparse storage format is used. Honest accounting is therefore critical (see HN-FIX-08).

    def compress(self, model: nn.Module | None = None) -> tuple[nn.Module, dict[str, Any]]:
        if model is not None:
            self.model = model
            try:
                self._original_params = self._count_params(self.model)
            except Exception:
                self._original_params = sum(p.numel() for p in self.model.parameters())

        self.compressed_layers = []
        self.fallback_layers = []
        self.skipped_layers = []
        self.layer_errors = {}
        self.layer_buckets = {}
        self.layer_storage = {}

        seen_weights: set[int] = set()

        total_orig_bytes = 0
        total_comp_bytes = 0

        targets = [
            (name, mod)
            for name, mod in list(self.model.named_modules())
            if self._is_target_module(name, mod)
        ]

        if not targets:
            logger.warning("HashedNet: no compressible layers found.")
            return self.model, self.get_compression_stats()

        for name, module in targets:
            target_mod = getattr(module, "base_layer", module)
            if not hasattr(target_mod, "weight") or target_mod.weight is None:
                continue

            w_id = id(target_mod.weight)
            if w_id in seen_weights:
                self.skipped_layers.append(name)
                continue
            seen_weights.add(w_id)

            cls_name = target_mod.__class__.__name__.lower()
            is_conv1d = "conv1d" in cls_name or hasattr(target_mod, "nf")
            raw_w = target_mod.weight.data

            # Normalize W to (out, in)
            if hasattr(target_mod, "nx") and hasattr(target_mod, "nf"):
                in_f, out_f = int(target_mod.nx), int(target_mod.nf)
                W_target = raw_w.t().float()
            elif hasattr(target_mod, "in_features") and hasattr(target_mod, "out_features"):
                in_f, out_f = int(target_mod.in_features), int(target_mod.out_features)
                W_target = (
                    raw_w.t().float()
                    if raw_w.shape == (in_f, out_f)
                    else raw_w.view(out_f, in_f).float()
                )
            elif is_conv1d:
                in_f, out_f = raw_w.shape[0], raw_w.shape[1]
                W_target = raw_w.t().float()
            else:
                out_f, in_f = raw_w.shape[0], raw_w.shape[1]
                W_target = raw_w.float()

            # Original storage bytes
            orig_bytes = raw_w.numel() * raw_w.element_size()
            if getattr(target_mod, "bias", None) is not None and target_mod.bias is not None:
                orig_bytes += target_mod.bias.numel() * target_mod.bias.element_size()
            total_orig_bytes += orig_bytes

            W_norm = torch.norm(W_target, p="fro").item()

            init_buckets = self._determine_layer_buckets(in_f, out_f)
            current_buckets = init_buckets
            best_layer: HashedLinear | None = None
            best_err = float("inf")

            for attempt in range(self.max_retries + 1):
                try:
                    candidate = HashedLinear.from_linear(
                        module,
                        n_buckets=current_buckets,
                        use_sign=self.use_sign,
                        lora_rank=self.lora_rank,
                        lora_alpha=self.lora_alpha,
                    )
                except Exception as e:
                    logger.warning("HashedLinear creation failed on %s: %s", name, e)
                    break

                if (
                    torch.isnan(candidate.weight_table).any()
                    or torch.isinf(candidate.weight_table).any()
                ):
                    logger.warning("HashedLinear weight_table has NaN/Inf on %s", name)
                    break

                W_rec = candidate.get_weight().float()
                err = (
                    torch.norm(W_target - W_rec, p="fro").item() / W_norm if W_norm > 1e-12 else 0.0
                )

                if err < best_err:
                    best_err = err
                    best_layer = candidate

                if err <= self.max_error:
                    break

                if attempt < self.max_retries:
                    next_buckets = min(int(current_buckets * 2), in_f * out_f)
                    if next_buckets <= current_buckets:
                        break
                    current_buckets = next_buckets

            self.layer_errors[name] = best_err
            self.layer_buckets[name] = current_buckets

            if best_layer is None or best_err > self.max_error:
                logger.warning(
                    "HashedNet: err=%.4f on %s > max_error=%.4f; keeping original.",
                    best_err,
                    name,
                    self.max_error,
                )
                if self.fallback_to_original:
                    self.fallback_layers.append(name)
                    for p in module.parameters():
                        p.requires_grad = False
                    total_comp_bytes += orig_bytes
                    continue

            # Apply freeze policy
            if self.freeze_hash_table and best_layer is not None:

                    logger.warning("HashedLinear weight_table has NaN/Inf on %s", name)
                    break

                W_rec = candidate.get_weight().float()
                err = (
                    torch.norm(W_target - W_rec, p="fro").item() / W_norm if W_norm > 1e-12 else 0.0
                )

                if err < best_err:
                    best_err = err
                    best_layer = candidate

                if err <= self.max_error:
                    break

                if attempt < self.max_retries:
                    next_buckets = min(int(current_buckets * 2), in_f * out_f)
                    if next_buckets <= current_buckets:
                        break
                    current_buckets = next_buckets

            self.layer_errors[name] = best_err
            self.layer_buckets[name] = current_buckets

            if best_layer is None or best_err > self.max_error:
                logger.warning(
                    "HashedNet: err=%.4f on %s > max_error=%.4f; keeping original.",
                    best_err,
                    name,
                    self.max_error,
                )
                if self.fallback_to_original:
                    self.fallback_layers.append(name)
                    for p in module.parameters():
                        p.requires_grad = False
                    total_comp_bytes += orig_bytes
                    continue

            # Apply freeze policy
            if self.freeze_hash_table and best_layer is not None:
                best_layer.weight_table.requires_grad = False

            # Splice
            try:
                if hasattr(module, "base_layer"):
                    module.base_layer = best_layer
                    for pn, p in module.named_parameters():
                        if "lora_" in pn.lower() and not pn.startswith("base_layer"):
                            p.requires_grad = True
                else:
                    replace_module(self.model, name, best_layer)
            except Exception as e:
                logger.warning("HashedNet replacement failed on %s: %s", name, e)
                if self.fallback_to_original:
                    self.fallback_layers.append(name)
                    for p in module.parameters():
                        p.requires_grad = False
                    total_comp_bytes += orig_bytes
                    continue

            self.compressed_layers.append(name)
            comp_bytes = sum(best_layer.storage_bytes().values())
            self.layer_storage[name] = best_layer.storage_bytes()
            total_comp_bytes += comp_bytes

        forward_check_success = True

            with torch.no_grad():

                            (

                                if isinstance(m, (nn.Linear, HashedLinear))
                            ),

        except Exception as e:
            logger.debug("HashedNet forward sanity check failed: %s", e)
            forward_check_success = False

        storage_ratio = total_orig_bytes / total_comp_bytes if total_comp_bytes > 0 else 1.0

        try:
            self._compressed_params = self._count_params(self.model)
        except Exception:
            self._compressed_params = sum(p.numel() for p in self.model.parameters())

            stats = self.get_compression_stats()
        except Exception:
            stats = {}

            sum(self.layer_errors[k] for k in self.compressed_layers)
            / max(1, len(self.compressed_layers))
            if self.compressed_layers
            else 0.0
        )

            {
                "compressed_layers": self.compressed_layers,
                "fallback_layers": self.fallback_layers,
                "skipped_layers": self.skipped_layers,

                "num_skipped": len(self.skipped_layers),

    def _linearize_module(
        self,
        module: nn.Module,
        prefix: str = "",
        _depth: int = 0,
    ) -> None:
        """Recursively replaces activation functions and (optionally) attention blocks."""
        if _depth > 16:
            return

        for name, child in list(module.named_children()):
            full_name = f"{prefix}.{name}" if prefix else name

            if self._is_critical(full_name):
                continue

            if self._is_already_compressed(child):
                if hasattr(child, "lora_A") or child.__class__.__name__.lower() == "loralinear":
                    continue
                if isinstance(child, _LinearizedAttentionWrapper):
                    continue

            # Skip LoRA-wrapped modules.
            if hasattr(child, "lora_A") or child.__class__.__name__.lower() == "loralinear":
                continue

            is_attn_layer = any(k in full_name.lower() for k in _ATTN_NAMES)

            # --- Activation linearization ---
            if isinstance(child, nn.ReLU):
                if self.target_attention_only and not is_attn_layer:
                    self.skipped_layers.append(full_name)
                    continue
                setattr(module, name, nn.Identity())
                self.linearized_activations_count += 1
                self.replaced_modules.append(full_name)
                continue

            if isinstance(child, _SILU_CLASSES):
                if self.target_attention_only and not is_attn_layer:
                    self.skipped_layers.append(full_name)
                    continue
                setattr(module, name, _LinearizedGELU(alpha=self.gelu_coefficient))
                self.linearized_activations_count += 1
                self.replaced_modules.append(full_name)
                continue

            # --- Attention linearization ---
            if self.replace_attention and is_attn_layer:
                if "norm" in name.lower():
                    self._linearize_module(child, prefix=full_name, _depth=_depth + 1)
                    continue
                if isinstance(child, (nn.Linear, nn.LayerNorm, nn.RMSNorm, nn.Embedding)):
                    self._linearize_module(child, prefix=full_name, _depth=_depth + 1)
                    continue

                # BERT / RoBERTa nested structure: linearize inner self-attention.
                if hasattr(child, "self") and hasattr(child, "output"):
                    self._linearize_module(child, prefix=full_name, _depth=_depth + 1)
                    continue

                if isinstance(child, _LinearizedAttentionWrapper):
                    continue

                if not hasattr(child, "_liftr_skip") and not isinstance(
                    child, _LinearizedAttentionWrapper
                ):
                    try:
                        wrapped = _LinearizedAttentionWrapper(child, layer_name=full_name)
                        setattr(module, name, wrapped)

                        self.replaced_modules.append(full_name)
                    except Exception as e:
                        logger.warning("LIFTR: failed to wrap %s: %s", full_name, e)

            # --- Recurse ---
            self._linearize_module(child, prefix=full_name, _depth=_depth + 1)

            for name, module in self.model.named_modules()
            if self._is_target_module(name, module)
        ]
        if not target_list:
            return {}

        for name, module in target_list:
            hooks.append(module.register_forward_hook(make_hook(name)))

        try:
            calib_inputs = get_calibration_tensors(
                self.model,
                tokenizer=self.tokenizer,
                calibration_loader=self.calibration_loader,
                calibration_texts=self.calibration_texts,
                device=self.target_device,
            )
            batches = self._normalize_batches(calib_inputs)
            n_batches = min(len(batches), self.max_calibration_batches)
            with torch.no_grad():
                for i in range(n_batches):
                    batch = batches[i]
                    try:
                        if isinstance(batch, dict):
                            batch = {
                                k: (v.to(self.target_device) if isinstance(v, torch.Tensor) else v)
                                for k, v in batch.items()
                            }
                            self.model(**batch)
                        elif isinstance(batch, torch.Tensor):
                            self.model(batch.to(self.target_device))
                        elif isinstance(batch, (list, tuple)):
                            batch = [
                                (v.to(self.target_device) if isinstance(v, torch.Tensor) else v)
                                for v in batch
                            ]
                            self.model(*batch)
                    except Exception as e:
                        logger.debug("Calibration batch %d failed: %s", i, e)
        except Exception as e:
            logger.warning("LiLLaMA: calibration forward raised: %s", e)
        finally:
            for h in hooks:
                h.remove()

        aggregated: dict[str, tuple[torch.Tensor, torch.Tensor]] = {}
        MAX_ACTIVATION_SAMPLES = 512
        for name, data in activations.items():
            if data["in"] and data["out"]:
                try:
                    cat_in = torch.cat(data["in"], dim=0)
                    cat_out = torch.cat(data["out"], dim=0)
                    if cat_in.size(0) > MAX_ACTIVATION_SAMPLES:
                        cat_in = cat_in[:MAX_ACTIVATION_SAMPLES]
                        cat_out = cat_out[:MAX_ACTIVATION_SAMPLES]
                    aggregated[name] = (cat_in, cat_out)
                except Exception as e:
                    logger.debug("Activation concat failed for %s: %s", name, e)
        return aggregated

    @staticmethod
    def _normalize_batches(calib_inputs: Any) -> list[Any]:
        if calib_inputs is None:
            return []
        if isinstance(calib_inputs, torch.Tensor):
            return [calib_inputs]
        if isinstance(calib_inputs, dict):
            return [calib_inputs]
        if isinstance(calib_inputs, (list, tuple)):
            if len(calib_inputs) == 0:
                return []
            first = calib_inputs[0]
            if isinstance(first, (dict, torch.Tensor)):
                return list(calib_inputs)
            return [calib_inputs]
        return [calib_inputs]

    # ------------------------------------------------------------------
    # SVD factorization helper
    # ------------------------------------------------------------------
    def _factorize(
        self,
        W: torch.Tensor,
        rank: int,
    ) -> tuple[torch.Tensor, int]:
        """Return (W_recon, rank_used)."""
        try:
            U, S, Vh = torch.linalg.svd(W, full_matrices=False)
            total_energy = (S**2).sum().clamp(min=1e-12)
            cum_ratio = torch.cumsum(S**2, 0) / total_energy
            r = min(rank, len(S))
            cum_e = cum_ratio[r - 1].item() if r > 0 else 0.0
            if cum_e < 0.95:
                idx = torch.where(cum_ratio >= 0.95)[0]
                if len(idx) > 0:
                    r = min(int(idx[0].item()) + 1, len(S))
            W_recon = (U[:, :r] * S[:r].unsqueeze(0)) @ Vh[:r, :]
            return W_recon, r
        except Exception as e:
            logger.debug("LiLLaMA SVD factorization failed: %s", e)
            return W, 0

                return self.model, self.get_compression_stats()
            except Exception:
                return self.model, {}

        logger.info("LiLLaMA: %d target layers identified.", len(target_list))

        total_orig_params = 0
        total_comp_params = 0

        # Deduplicated parameter accounting.
        seen_param_ids: set = set()
        count_param_ids: set = set()

        for p in self.model.parameters():
            if id(p) not in seen_param_ids:
                seen_param_ids.add(id(p))
                total_orig_params += p.numel()

        def _count_params_of(mod: nn.Module):
            nonlocal total_comp_params
            for p in mod.parameters(recurse=False):
                if id(p) not in count_param_ids:
                    count_param_ids.add(id(p))
                    total_comp_params += p.numel()

        for name, module in target_list:
            target_layer = getattr(module, "base_layer", module)
            raw_w = target_layer.weight.data
            is_conv1d = _is_conv1d(target_layer)
            device = raw_w.device
            dtype = target_layer.weight.dtype

            W = raw_w.t().contiguous().clone().float() if is_conv1d else raw_w.clone().float()
            d_out, d_in = W.shape
            used_cross_cov = False
            r_used = self.rank

            # LL-FIX-04: initialize W_new to W (safe default).
            W_new = W

            if name in activations_map:
                inp, out = activations_map[name]

                    inp_flat = inp.reshape(-1, inp.shape[-1]).float()
                    out_flat = out.reshape(-1, out.shape[-1]).float()

                except Exception as e:

                    W_new, r_used = self._factorize(W, self.rank)
            else:
                W_new, r_used = self._factorize(W, self.rank)

            norm_orig = torch.norm(W, p="fro")
            rel_err = (torch.norm(W - W_new, p="fro") / (norm_orig + 1e-8)).item()
            self.layer_errors[name] = rel_err

            if rel_err > self.max_error:
                logger.warning(

                    rel_err,
                    name,
                    self.max_error,

                        if self.fallback_to_original

                    tuple(final_w.shape),

                _count_params_of(module)
                continue

            target_layer.weight.requires_grad = False
            if getattr(target_layer, "bias", None) is not None:
                target_layer.bias.requires_grad = False

            # Preserve LoRA adapter trainability.
            if hasattr(module, "lora_A") and hasattr(module, "lora_B"):
                for attr in ("lora_A", "lora_B"):
                    p = getattr(module, attr, None)
                    if p is None:
                        continue
                    if hasattr(p, "weight"):
                        p.weight.requires_grad = True
                    elif isinstance(p, nn.Parameter):
                        p.requires_grad = True

            self.compressed_layers.append(name)
            _count_params_of(module)

            self.layer_details[name] = {
                "rank": int(r_used),
                "rel_error": float(rel_err),
                "used_cross_cov": bool(used_cross_cov),
                "shape": tuple(final_w.shape),
            }

        # LL-FIX-07: forward sanity check.
        forward_check_success = True
        try:

            self.layer_errors[n] for n in self.compressed_layers if n in self.layer_errors

                "layer_errors": self.layer_errors,
                "layer_details": self.layer_details,
                "effective_parameter_ratio": eff_ratio,
                "forward_check_success": forward_check_success,
                "method": "lillama_cross_covariance_svd",
                # LL-FIX-12: honest note.
                "note": (

                    "The dense weight keeps its full (out, in) shape; framework-level "
                    "compression_rate reads 1.0. Use effective_parameter_ratio for the "

                    "effective_parameter_ratio": eff_ratio,
                    "avg_reconstruction_error": avg_err,
                }
            )
        except Exception:
            pass

            len(self.compressed_layers),
            len(self.fallback_layers),
            avg_err,
            eff_ratio,
        )
        return self.model, stats

        self.skip_should_compress_filter = bool(skip_should_compress_filter)
        self.max_calibration_batches = max(1, int(max_calibration_batches))

        # LL-FIX-03: safe device.
        if device is not None:
            self.target_device = torch.device(device)
        else:
            self.target_device = _get_device(model)

    def _is_target_module(self, name: str, module: nn.Module, _debug: bool = False) -> bool:
        name_lower = name.lower()
        if any(part in name.split(".") for part in ("lora_A", "lora_B", "base_layer")):
            if _debug:
                logger.info(f"  [skip] {name}: lora/base_layer part")
            return False

        # Skip already-compressed.
        cls_name = module.__class__.__name__
        if cls_name in (
            "LSCPrunedLoRALinear",
            "DynamicSVDLinear",
            "SVDLinear",
            "FactorisedLinear",
            "LowRankPlusSparseLinear",
        ):
            if _debug:
                logger.info(f"  [skip] {name}: already compressed ({cls_name})")
            return False

        if any(
            skip in name_lower
            for skip in [
                "classifier",
                "score",
                "lm_head",
                "pooler",
                "embedding",
                "router",
                "gate",
            ]
        ):
            if _debug:
                logger.info(f"  [skip] {name}: skip keyword")
            return False

        target_mod = getattr(module, "base_layer", module)
        cls_lower = target_mod.__class__.__name__.lower()
        is_linear_like = (
            isinstance(target_mod, nn.Linear) or "conv1d" in cls_lower or hasattr(target_mod, "nf")
        )
        if not is_linear_like:
            if _debug:
                logger.info(f"  [skip] {name}: not linear-like")
            return False

        # LL-FIX-09: allow bypassing framework filter.
        if not self.skip_should_compress_filter:
            try:
                if not should_compress_linear(self.model, name):
                    if _debug:
                        logger.info(f"  [skip] {name}: should_compress_linear=False")
                    return False
            except Exception as e:
                if _debug:
                    logger.info(f"  [skip] {name}: should_compress_linear raised {e}")
                return False

        # LL-FIX-10: filter tiny layers.
        w = getattr(target_mod, "weight", None)
        if w is None:
            if _debug:
                logger.info(f"  [skip] {name}: weight None")
            return False
        if min(w.shape[0], w.shape[1]) < self.min_layer_dim:
            if _debug:
                logger.info(f"  [skip] {name}: dim too small {tuple(w.shape)}")
            return False

        if self.target_attention_only:
            is_attn = any(k in name_lower for k in _ATTN_NAMES)
            if not is_attn:
                if _debug:
                    logger.info(f"  [skip] {name}: not attention")
                return False

        if self.skip_attention:
            attn_keywords = {
                "query",
                "key",
                "value",
                "q_lin",
                "k_lin",
                "v_lin",
                "q_proj",
                "k_proj",
                "v_proj",
                "out_lin",
                "out_proj",
                "dense",
                "o_proj",
                "c_attn",
            }
            last = name.split(".")[-1].lower()
            if last in attn_keywords or any(k in name_lower for k in ("self_attn", "attn.c_attn")):
                if _debug:
                    logger.info(f"  [skip] {name}: skip_attention")
                return False

        if self.target_modules is not None:
            ok = any(t in name for t in self.target_modules)
            if not ok and _debug:
                logger.info(f"  [skip] {name}: not in target_modules")
            return ok

        if _debug:
            logger.info(f"  [TARGET] {name}: shape={tuple(w.shape)}")
        return True

    # ------------------------------------------------------------------
    # IO activation collection
    # ------------------------------------------------------------------
    def _collect_io_activations(
        self,
    ) -> dict[str, tuple[torch.Tensor, torch.Tensor]]:
        """Collect input/output activations across target layers."""
        self.model.eval()
        activations: dict[str, dict[str, list[torch.Tensor]]] = {}
        hooks = []

        def make_hook(layer_name: str):
            def hook(module: nn.Module, inputs: tuple[Any, ...], output: Any):
                inp = inputs[0].detach() if inputs and isinstance(inputs[0], torch.Tensor) else None
                if isinstance(output, torch.Tensor):
                    out = output.detach()
                elif (
                    isinstance(output, (tuple, list))
                    and output
                    and isinstance(output[0], torch.Tensor)
                ):
                    out = output[0].detach()
                else:
                    out = None
                if inp is not None and out is not None:
                    if layer_name not in activations:
                        activations[layer_name] = {"in": [], "out": []}
                    activations[layer_name]["in"].append(inp.cpu())
                    activations[layer_name]["out"].append(out.cpu())

            return hook

        target_list = [
            (name, module)
            for name, module in self.model.named_modules()
            if self._is_target_module(name, module)
        ]
        if not target_list:
            return {}

        for name, module in target_list:
            hooks.append(module.register_forward_hook(make_hook(name)))

        try:
            calib_inputs = get_calibration_tensors(
                self.model,
                tokenizer=self.tokenizer,
                calibration_loader=self.calibration_loader,
                calibration_texts=self.calibration_texts,
                device=self.target_device,
            )
            batches = self._normalize_batches(calib_inputs)
            n_batches = min(len(batches), self.max_calibration_batches)
            with torch.no_grad():
                for i in range(n_batches):
                    batch = batches[i]
                    try:
                        if isinstance(batch, dict):
                            batch = {
                                k: (v.to(self.target_device) if isinstance(v, torch.Tensor) else v)
                                for k, v in batch.items()
                            }
                            self.model(**batch)
                        elif isinstance(batch, torch.Tensor):
                            self.model(batch.to(self.target_device))
                        elif isinstance(batch, (list, tuple)):
                            batch = [
                                (v.to(self.target_device) if isinstance(v, torch.Tensor) else v)
                                for v in batch
                            ]
                            self.model(*batch)
                    except Exception as e:
                        logger.debug("Calibration batch %d failed: %s", i, e)
        except Exception as e:
            logger.warning("LiLLaMA: calibration forward raised: %s", e)
        finally:
            for h in hooks:
                h.remove()

        aggregated: dict[str, tuple[torch.Tensor, torch.Tensor]] = {}
        MAX_ACTIVATION_SAMPLES = 512
        for name, data in activations.items():
            if data["in"] and data["out"]:
                try:
                    cat_in = torch.cat(data["in"], dim=0)
                    cat_out = torch.cat(data["out"], dim=0)
                    if cat_in.size(0) > MAX_ACTIVATION_SAMPLES:
                        cat_in = cat_in[:MAX_ACTIVATION_SAMPLES]
                        cat_out = cat_out[:MAX_ACTIVATION_SAMPLES]
                    aggregated[name] = (cat_in, cat_out)
                except Exception as e:
                    logger.debug("Activation concat failed for %s: %s", name, e)
        return aggregated

    @staticmethod
    def _normalize_batches(calib_inputs: Any) -> list[Any]:
        if calib_inputs is None:
            return []
        if isinstance(calib_inputs, torch.Tensor):
            return [calib_inputs]
        if isinstance(calib_inputs, dict):
            return [calib_inputs]
        if isinstance(calib_inputs, (list, tuple)):
            if len(calib_inputs) == 0:
                return []
            first = calib_inputs[0]
            if isinstance(first, (dict, torch.Tensor)):
                return list(calib_inputs)
            return [calib_inputs]
        return [calib_inputs]

    # ------------------------------------------------------------------
    # SVD factorization helper
    # ------------------------------------------------------------------
    def _factorize(
        self,
        W: torch.Tensor,
        rank: int,
    ) -> tuple[torch.Tensor, int]:
        """Return (W_recon, rank_used)."""
        try:
            U, S, Vh = torch.linalg.svd(W, full_matrices=False)
            total_energy = (S**2).sum().clamp(min=1e-12)
            cum_ratio = torch.cumsum(S**2, 0) / total_energy
            r = min(rank, len(S))
            cum_e = cum_ratio[r - 1].item() if r > 0 else 0.0
            if cum_e < 0.95:
                idx = torch.where(cum_ratio >= 0.95)[0]
                if len(idx) > 0:
                    r = min(int(idx[0].item()) + 1, len(S))
            W_recon = (U[:, :r] * S[:r].unsqueeze(0)) @ Vh[:r, :]
            return W_recon, r
        except Exception as e:
            logger.debug("LiLLaMA SVD factorization failed: %s", e)
            return W, 0

    # Main compress
    # ------------------------------------------------------------------
    def compress(self, **kwargs: Any) -> tuple[nn.Module, dict[str, Any]]:
        # LL-FIX-08: idempotence.
        if getattr(self.model, "_lillama_compressed", False):
            logger.warning("LiLLaMA: model already compressed; skipping.")
            try:
                return self.model, self.get_compression_stats()
            except Exception:
                return self.model, {}

        self.compressed_layers = []
        self.fallback_layers = []
        self.layer_errors = {}
        self.layer_details = {}

        activations_map = self._collect_io_activations()
        logger.info("LiLLaMA: %d target activation sets collected.", len(activations_map))

        # LL-FIX-06: diagnostic if no targets.
        target_list = [
            (name, module)
            for name, module in self.model.named_modules()
            if self._is_target_module(name, module)
        ]
        if not target_list:
            logger.warning("LiLLaMA: no compressible layers found. Running diagnostic...")
            n_scanned = 0
            for name, module in self.model.named_modules():
                n_scanned += 1
                self._is_target_module(name, module, _debug=True)
            logger.warning("LiLLaMA diagnostic: %d modules scanned, 0 targets.", n_scanned)
            try:
                return self.model, self.get_compression_stats()
            except Exception:
                return self.model, {}

        logger.info("LiLLaMA: %d target layers identified.", len(target_list))

        total_orig_params = 0
        total_comp_params = 0

        # Deduplicated parameter accounting.
        seen_param_ids: set = set()
        count_param_ids: set = set()

        for p in self.model.parameters():
            if id(p) not in seen_param_ids:
                seen_param_ids.add(id(p))
                total_orig_params += p.numel()

        def _count_params_of(mod: nn.Module):
            nonlocal total_comp_params
            for p in mod.parameters(recurse=False):
                if id(p) not in count_param_ids:
                    count_param_ids.add(id(p))
                    total_comp_params += p.numel()

        for name, module in target_list:
            target_layer = getattr(module, "base_layer", module)
            raw_w = target_layer.weight.data
            is_conv1d = _is_conv1d(target_layer)
            device = raw_w.device
            dtype = target_layer.weight.dtype

            W = raw_w.t().contiguous().clone().float() if is_conv1d else raw_w.clone().float()
            d_out, d_in = W.shape
            used_cross_cov = False
            r_used = self.rank

            # LL-FIX-04: initialize W_new to W (safe default).
            W_new = W

            if name in activations_map:
                inp, out = activations_map[name]
                try:
                    inp_flat = inp.reshape(-1, inp.shape[-1]).float()
                    out_flat = out.reshape(-1, out.shape[-1]).float()
                    if inp_flat.shape[-1] == d_in and out_flat.shape[-1] == d_out:
                        C = torch.matmul(out_flat.t().contiguous(), inp_flat) / max(
                            len(inp_flat), 1
                        )
                        W_new, r_used = self._factorize(C, self.rank)
                        used_cross_cov = True
                    else:
                        W_new, r_used = self._factorize(W, self.rank)
                except Exception as e:
                    logger.debug("Cross-covariance factorization failed for %s: %s", name, e)
                    W_new, r_used = self._factorize(W, self.rank)
            else:
                W_new, r_used = self._factorize(W, self.rank)

            norm_orig = torch.norm(W, p="fro")
            rel_err = (torch.norm(W - W_new, p="fro") / (norm_orig + 1e-8)).item()
            self.layer_errors[name] = rel_err

            if rel_err > self.max_error:
                logger.warning(
                    "LiLLaMA: err %.4f on %s > %.4f. %s",
                    rel_err,
                    name,
                    self.max_error,
                    (
                        "Keeping original."
                        if self.fallback_to_original
                        else "Proceeding with distilled weights."
                    ),
                )
                if self.fallback_to_original:
                    self.fallback_layers.append(name)
                    for p in module.parameters(recurse=False):
                        p.requires_grad = False
                    _count_params_of(module)
                    continue

            # Shape integrity check.
            final_w = W_new.t().contiguous() if is_conv1d else W_new
            if final_w.shape != target_layer.weight.shape:
                logger.warning(
                    "LiLLaMA: shape mismatch for %s: expected %s, got %s; skipping.",
                    name,
                    tuple(target_layer.weight.shape),
                    tuple(final_w.shape),
                )
                _count_params_of(module)
                continue

            target_layer.weight.data.copy_(final_w.to(dtype=dtype, device=device))
            target_layer.weight.requires_grad = False
            if getattr(target_layer, "bias", None) is not None:
                target_layer.bias.requires_grad = False

            # Preserve LoRA adapter trainability.
            if hasattr(module, "lora_A") and hasattr(module, "lora_B"):
                for attr in ("lora_A", "lora_B"):
                    p = getattr(module, attr, None)
                    if p is None:
                        continue
                    if hasattr(p, "weight"):
                        p.weight.requires_grad = True
                    elif isinstance(p, nn.Parameter):
                        p.requires_grad = True

            self.compressed_layers.append(name)
            _count_params_of(module)

            self.layer_details[name] = {
                "rank": int(r_used),
                "rel_error": float(rel_err),
                "used_cross_cov": bool(used_cross_cov),
                "shape": tuple(final_w.shape),
            }

        # LL-FIX-07: forward sanity check.
        forward_check_success = True
        try:
            with torch.no_grad():
                dummy = torch.randint(0, 100, (1, 8), device=self.target_device, dtype=torch.long)
                try:
                    self.model(input_ids=dummy, attention_mask=torch.ones_like(dummy))
                except Exception:
                    try:
                        self.model(dummy)
                    except Exception as e2:
                        logger.warning("LiLLaMA: forward check failed: %s", e2)
                        forward_check_success = False
        except Exception as e:
            logger.warning("LiLLaMA: forward check raised: %s", e)
            forward_check_success = False

        try:
            self._compressed_params = self._count_params(self.model)
        except Exception:
            self._compressed_params = sum(p.numel() for p in self.model.parameters())

        # LL-FIX-04: avg error only over compressed layers.
        compressed_errs = [
            self.layer_errors[n] for n in self.compressed_layers if n in self.layer_errors
        ]
        avg_err = sum(compressed_errs) / len(compressed_errs) if compressed_errs else 0.0
        eff_ratio = total_comp_params / max(1, total_orig_params)

        try:
            stats = self.get_compression_stats()
        except Exception:
            stats = {}

        stats.update(
            {
                "distilled_layers": len(self.compressed_layers),
                "compressed_layers": self.compressed_layers,
                "fallback_layers": self.fallback_layers,
                "num_compressed": len(self.compressed_layers),
                "num_fallback": len(self.fallback_layers),
                "rank": self.rank,
                "max_error": self.max_error,
                "fallback_to_original": self.fallback_to_original,
                "avg_reconstruction_error": avg_err,
                "layer_errors": self.layer_errors,
                "layer_details": self.layer_details,
                "effective_parameter_ratio": eff_ratio,
                "forward_check_success": forward_check_success,
                "method": "lillama_cross_covariance_svd",
                # LL-FIX-12: honest note.
                "note": (
                    "LiLLaMA rewrites weights in-place with the distilled reconstruction. "
                    "The dense weight keeps its full (out, in) shape; framework-level "
                    "compression_rate reads 1.0. Use effective_parameter_ratio for the "
                    "theoretical storage metric."
                ),
            }
        )
        self.stats = stats

        # LL-FIX-13: write stats onto model.
        try:
            if not hasattr(self.model, "compression_stats") or self.model.compression_stats is None:
                self.model.compression_stats = {}
            self.model.compression_stats.update(
                {
                    "num_compressed": len(self.compressed_layers),
                    "num_fallback": len(self.fallback_layers),
                    "compression_method": "lillama_cross_covariance_svd",
                    "effective_parameter_ratio": eff_ratio,
                    "avg_reconstruction_error": avg_err,
                }
            )
        except Exception:
            pass

        self.model._lillama_compressed = True
        logger.info(
            "LiLLaMA: %d distilled, %d fallback. Avg err %.4f, eff_param_ratio %.2fx.",
            len(self.compressed_layers),
            len(self.fallback_layers),
            avg_err,
            eff_ratio,
        )
        return self.model, stats

class PrunedChannelLinear(nn.Module):
    """
    Linear layer with pruned output channels.

    Stores only the kept rows of the effective weight matrix:
      - W_kept:      (k, in_features)  fp16/fp32   -- only kept rows
      - bias_kept:   (k,)              optional
      - kept_idx:    (k,)              int64  -- output positions of kept rows
      - original_out_features: int

    Forward:
      h = F.linear(x, W_kept, bias_kept)          # (..., k)
      out = zeros(..., original_out_features)
      out.scatter_(-1, kept_idx, h)               # (..., original_out)
    """

    is_custom_attention = False

    def __init__(
        self,
        in_features: int,
        original_out_features: int,
        kept_idx: torch.Tensor,  # (k,) int64
        W_kept: torch.Tensor,  # (k, in_features)
        bias_kept: torch.Tensor | None = None,
    ) -> None:
        super().__init__()
        self.in_features = int(in_features)
        self.original_out_features = int(original_out_features)
        self.out_features = self.original_out_features  # for API compatibility
        self.num_kept = int(W_kept.shape[0])

        self.register_buffer("kept_idx", kept_idx.long(), persistent=True)
        self.W_kept = nn.Parameter(W_kept, requires_grad=False)

        if bias_kept is not None:
            self.bias_kept = nn.Parameter(bias_kept, requires_grad=False)
        else:
            self.register_parameter("bias_kept", None)

        self._cached_weight: torch.Tensor | None = None

    @property
    def weight(self) -> torch.Tensor:
        """Reconstruct effective (original_out, in) weight with zeros in pruned rows."""
        if self._cached_weight is not None:
            return self._cached_weight
        W = self.W_kept.new_zeros(self.original_out_features, self.in_features)
        W[self.kept_idx] = self.W_kept
        self._cached_weight = W
        return W

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        h = F.linear(x, self.W_kept, self.bias_kept)  # (..., k)
        out = torch.zeros(
            *x.shape[:-1],
            self.original_out_features,
            dtype=h.dtype,
            device=h.device,
        )
        out.scatter_(-1, self.kept_idx.expand(*x.shape[:-1], -1), h)
        return out

    def extra_repr(self) -> str:
        return (
            f"in_features={self.in_features}, original_out_features={self.original_out_features}, "
            f"num_kept={self.num_kept}"
        )

            k in name_l
            for k in [
                "classifier",
                "score",
                "lm_head",
                "pooler",
                "embed",
                "embedding",
                "cls_head",
                "head",
                "logits_processor",
            ]

            self.protected_layers.append(name)
            return False

        if self.target_attention_only:
            attn_keywords = [
                "attn",
                "attention",
                "q_proj",
                "k_proj",
                "v_proj",
                "o_proj",
                "c_attn",
                "c_proj",
                "query",
                "key",
                "value",
                "dense",
            ]
            if not any(k in name_l for k in attn_keywords):
                self.skipped_layers.append(name)
                return False

        if self._is_gqa_asymmetric(name, module):
            self.skipped_layers.append(name)
            return False

        base_mod = getattr(module, "base_layer", module)
        cls_name = base_mod.__class__.__name__
        raw_w = getattr(base_mod, "weight", None)
        if raw_w is None or not isinstance(raw_w, torch.Tensor):
            return False

        is_conv1d = "conv1d" in cls_name.lower() or hasattr(base_mod, "nf")
        is_linear_like = isinstance(base_mod, nn.Linear) or is_conv1d
        if not is_linear_like or isinstance(module, ROASTLinear):
            return False

        if raw_w.numel() < self.min_numel:
            self.skipped_layers.append(name)
            return False

        if not should_compress_linear(self.model, name):
            self.protected_layers.append(name)
            return False

        if self.target_modules is not None:
            return any(t in name for t in self.target_modules)
        return True

        """Compress the target modules using ROAST tiling with LoRA support."""
        self.compressed_layers.clear()
        self.visited_modules.clear()
        self.protected_layers.clear()
        self.skipped_layers.clear()
        self.fallback_layers.clear()
        self.failed_layers.clear()
        self.layer_errors.clear()

            if self._is_target_module(name, module):
                mod_id = id(module)

                    continue

class ROASTLinear(nn.Module):
    """
    Linear layer with weight sharing based on tiles (blocks) rather than
    individual scalar weights. Each tile of size (tile_size, tile_size) shares
    a single weight from a small table, with optional trainable LoRA adapters.
    """

    def __init__(
        self,
        in_features: int,
        out_features: int,
        n_buckets: int = 1024,
        tile_size: int = 32,
        bias: bool = True,
        lora_r: int = 0,
        lora_alpha: float = 1.0,
        device: torch.device | None = None,
        dtype: torch.dtype | None = None,
    ) -> None:
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.tile_size = max(1, int(tile_size))
        self.n_buckets = max(1, int(n_buckets))
        self.lora_r = int(lora_r)
        self.lora_alpha = float(lora_alpha)
        self.scaling = (self.lora_alpha / self.lora_r) if self.lora_r > 0 else 1.0

        # Number of tiles along each dimension (padded)
        self.tiles_out = (out_features + self.tile_size - 1) // self.tile_size
        self.tiles_in = (in_features + self.tile_size - 1) // self.tile_size

        # Weight table: frozen if LoRA is active
        requires_grad = not (self.lora_r > 0)
        self.weight_table = nn.Parameter(
            torch.randn(self.n_buckets, device=device, dtype=dtype),
            requires_grad=requires_grad,
        )

        if bias:
            self.bias = nn.Parameter(
                torch.zeros(out_features, device=device, dtype=dtype),
                requires_grad=requires_grad,
            )
        else:
            self.register_parameter("bias", None)

        # Precompute tile indices with high-entropy bitwise mixing (B11, J1)
        tile_idx_i = (
            torch.arange(self.tiles_out, device=device).unsqueeze(1).expand(-1, self.tiles_in)
        )
        tile_idx_j = (
            torch.arange(self.tiles_in, device=device).unsqueeze(0).expand(self.tiles_out, -1)
        )
        h = (tile_idx_i.long() * 0x9E3779B1) ^ (tile_idx_j.long() * 0x85EBCA77)
        h = h ^ (h >> 16)
        h = h * 0x7FEB352D
        h = (h ^ (h >> 15)) & 0x7FFFFFFF
        self.register_buffer(
            "tile_hash_idx",
            (h % self.n_buckets).long(),
            persistent=False,
        )

        self._cached_weight: torch.Tensor | None = None
        self._cache_stale: bool = True

        if self.lora_r > 0:
            self.lora_A = nn.Parameter(
                torch.empty((self.lora_r, in_features), dtype=torch.float32, device=device)
            )
            self.lora_B = nn.Parameter(
                torch.zeros((out_features, self.lora_r), dtype=torch.float32, device=device)
            )
            nn.init.kaiming_uniform_(self.lora_A, a=math.sqrt(5))
        else:
            self.register_parameter("lora_A", None)
            self.register_parameter("lora_B", None)

    def _invalidate_cache(self) -> None:
        self._cached_weight = None
        self._cache_stale = True

    @property
    def weight(self) -> torch.Tensor:
        """Reconstructed effective weight matrix (out_features, in_features) with evaluation caching."""
        if self._cached_weight is not None and not self._cache_stale and not self.training:
            return self._cached_weight
        W_tiles = self.weight_table[self.tile_hash_idx]  # (tiles_out, tiles_in)
        W_full = torch.repeat_interleave(
            torch.repeat_interleave(W_tiles, self.tile_size, dim=0),
            self.tile_size,
            dim=1,
        )
        W_res = W_full[: self.out_features, : self.in_features]
        if not self.training:
            self._cached_weight = W_res
            self._cache_stale = False
        return W_res

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """Vectorized forward pass without per-element python loops."""
        W = self.weight.to(device=x.device, dtype=x.dtype)
        out = F.linear(x, W, self.bias.to(dtype=x.dtype) if self.bias is not None else None)
        if self.lora_r > 0 and self.lora_A is not None and self.lora_B is not None:
            lora_out = F.linear(F.linear(x.float(), self.lora_A), self.lora_B) * self.scaling
            out = out + lora_out.to(out.dtype)
        return out

    @classmethod
    def from_linear(
        cls,
        linear: nn.Module,
        n_buckets: int = 1024,
        tile_size: int = 32,
        lora_r: int = 0,
        lora_alpha: float = 1.0,
    ) -> ROASTLinear:
        """Initialize from a pretrained nn.Linear, Conv1D, or LoRALinear base_layer."""
        base_mod = getattr(linear, "base_layer", linear)
        raw_w = base_mod.weight.data
        cls_name = base_mod.__class__.__name__
        is_conv1d = "conv1d" in cls_name.lower() or hasattr(base_mod, "nf")

        if is_conv1d:
            W = raw_w.t().contiguous().float()
            out_f, in_f = W.shape
        else:
            W = raw_w.contiguous().float()
            out_f, in_f = W.shape

        bias = getattr(base_mod, "bias", None) is not None
        roast = cls(
            in_features=in_f,
            out_features=out_f,
            n_buckets=n_buckets,
            tile_size=tile_size,
            bias=bias,
            lora_r=lora_r,
            lora_alpha=lora_alpha,
            device=raw_w.device,
            dtype=raw_w.dtype,
        )
        if bias and base_mod.bias is not None:
            roast.bias.data.copy_(base_mod.bias.data)

        # Preserve existing source LoRA weights if present
        if (
            hasattr(linear, "lora_A")
            and hasattr(linear, "lora_B")
            and linear.lora_A is not None
            and linear.lora_B is not None
        ):
            src_A = getattr(linear.lora_A, "weight", linear.lora_A)
            src_B = getattr(linear.lora_B, "weight", linear.lora_B)
            if roast.lora_A is not None and roast.lora_B is not None:
                if src_A.shape == roast.lora_A.shape:
                    roast.lora_A.data.copy_(src_A.data)
                if src_B.shape == roast.lora_B.shape:
                    roast.lora_B.data.copy_(src_B.data)

        ts = max(1, tile_size)
        tiles_out = (out_f + ts - 1) // ts
        tiles_in = (in_f + ts - 1) // ts

        # Vectorized computation of per-tile means and scatter into weight_table
        with torch.no_grad():
            pad_out = tiles_out * ts - out_f
            pad_in = tiles_in * ts - in_f
            if pad_out > 0 or pad_in > 0:
                W_padded = F.pad(W, (0, pad_in, 0, pad_out), mode="constant", value=0.0)
                mask = F.pad(
                    torch.ones_like(W),
                    (0, pad_in, 0, pad_out),
                    mode="constant",
                    value=0.0,
                )
                W_blocks = (
                    W_padded.view(tiles_out, ts, tiles_in, ts)
                    .permute(0, 2, 1, 3)
                    .contiguous()
                    .view(tiles_out, tiles_in, -1)
                )
                mask_blocks = (
                    mask.view(tiles_out, ts, tiles_in, ts)
                    .permute(0, 2, 1, 3)
                    .contiguous()
                    .view(tiles_out, tiles_in, -1)
                )
                tile_means = W_blocks.sum(dim=-1) / mask_blocks.sum(dim=-1).clamp(min=1.0)
            else:
                tile_means = (
                    W.view(tiles_out, ts, tiles_in, ts)
                    .permute(0, 2, 1, 3)
                    .contiguous()
                    .view(tiles_out, tiles_in, -1)
                    .mean(dim=-1)
                )

            flat_idx = roast.tile_hash_idx.view(-1).to(device=roast.weight_table.device)
            flat_means = tile_means.view(-1).to(
                device=roast.weight_table.device, dtype=roast.weight_table.dtype
            )
            roast.weight_table.data.scatter_(0, flat_idx, flat_means)

        return roast

    out = F.linear(x, W, self.bias.to(dtype=x.dtype) if self.bias is not None else None)
    if self.lora_r > 0 and self.lora_A is not None and self.lora_B is not None:
        lora_out = F.linear(F.linear(x.float(), self.lora_A), self.lora_B) * self.scaling
        out = out + lora_out.to(out.dtype)
    return out
