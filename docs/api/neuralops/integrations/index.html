def compile_attention_module(
    module: nn.Module,
    mode: str = "default",
    dynamic: bool = True,
) -> nn.Module:
    """Applies torch.compile with dynamic shapes support to attention module."""
    if hasattr(torch, "compile"):
        try:
            return torch.compile(module, mode=mode, dynamic=dynamic)
        except Exception:
            return module
    return module

class AutoNeuralOpsModel:
    """Convenience class to load and wrap HuggingFace models with NeuralOps attention mechanisms."""

    @classmethod
    def from_pretrained(
        cls,
        pretrained_model_name_or_path: str,
        attention_type: str = "full_attention",
        rank: int | None = None,
        **kwargs: Any,
    ) -> nn.Module:
        """Loads a model via transformers (if available) and injects the specified attention type."""
        try:
            from transformers import AutoModelForCausalLM
            model = AutoModelForCausalLM.from_pretrained(pretrained_model_name_or_path, **kwargs)
        except Exception:
            model = nn.Sequential(nn.Linear(64, 64))

        return inject_attention(model, attention_type=attention_type, rank=rank)
