class ClassificationAdapter:
    """Task adapter for sequence classification benchmarks (e.g. GLUE tasks)."""

    def __init__(
        self,
        module_name: str,
        module_type: str,
        model_name: str,
        dataset_name: str,
        rank: Optional[int] = None,
        device: Optional[str] = None,
        dtype: Optional[str] = None,
        batch_size: int = 32,
        max_length: int = 128,
        epochs: int = 3,
        lr: float = 2e-5,
        seed: int = 42,
        eval_only: bool = False,
        **kwargs: Any,
    ):
        self.module_name = module_name
        self.module_type = module_type
        self.model_name = model_name
        self.dataset_name = dataset_name
        self.rank = rank
        self.device = device or ("cuda" if torch.cuda.is_available() else "cpu")
        self.dtype = dtype or "float32"
        self.batch_size = batch_size
        self.max_length = max_length
        self.epochs = epochs
        self.lr = lr
        self.seed = seed
        self.eval_only = eval_only
        self.extra_kwargs = kwargs

        self.model = None
        self.tokenizer = None
        self.train_loader = None
        self.eval_loader = None
        self.model_meta: Optional[ModelMetadata] = None
        self.dataset_meta: Optional[DatasetMetadata] = None

    def setup(self) -> None:
        """Load datasets, tokenizers, and base model with classification head."""
        torch.manual_seed(self.seed)

        # 1. Determine number of labels for dataset
        num_labels = get_num_labels(self.dataset_name)

        # 2. Load model and tokenizer with sequence classification head
        self.model, self.tokenizer, model_cfg = load_model_and_tokenizer(
            self.model_name,
            device=self.device,
            dtype=self.dtype,
            task="classification",
            num_labels=num_labels,
        )
        self.model_meta = get_model_metadata(self.model, model_cfg)

        # 3. Tokenize datasets
        tokenized_train = load_dataset_by_name(
            self.dataset_name,
            tokenizer=self.tokenizer,
            max_length=self.max_length,
            split="train",
        )
        # Determine eval split
        eval_split = "validation"
        try:
            tokenized_eval = load_dataset_by_name(
                self.dataset_name,
                tokenizer=self.tokenizer,
                max_length=self.max_length,
                split="validation",
            )
        except Exception:
            eval_split = "test"
            tokenized_eval = load_dataset_by_name(
                self.dataset_name,
                tokenizer=self.tokenizer,
                max_length=self.max_length,
                split="test",
            )

        self.dataset_meta = get_dataset_metadata(
            tokenized_train,
            name=self.dataset_name,
            split=f"train+{eval_split}",
        )

        self.train_loader = make_dataloader(tokenized_train, batch_size=self.batch_size, shuffle=True)
        self.eval_loader = make_dataloader(tokenized_eval, batch_size=self.batch_size, shuffle=False)

    def apply_module(self) -> None:
        """Apply attention module replacement or compression algorithm."""
        if self.model is None:
            raise RuntimeError("Setup must be run before applying module.")

        if self.module_type in ("baseline", "none", "full", "dense"):
            logger.info("Baseline run: skipping module injection.")
            return


        if self.module_type == "attention":
            self.model = apply_attention_module(
                self.model,
                module_name=self.module_name,
                rank=self.rank,
                **self.extra_kwargs,
            )
        elif self.module_type == "compressor":
            self.model = apply_compressor(
                self.model,
                module_name=self.module_name,
                rank=self.rank,
                calibration_loader=self.train_loader,
                **self.extra_kwargs,
            )
        else:
            raise ValueError(f"Unknown module type: {self.module_type}")


        # Update params in metadata
        if self.model_meta is not None:
            self.model_meta.num_params = count_parameters(self.model)

    def run(self) -> Dict[str, Any]:
        """Execute setup, module injection, training (if applicable), and evaluation."""
        self.setup()
        self.apply_module()

        train_metrics = {}
        if not self.eval_only and self.epochs > 0:
            train_metrics = finetune_classification(
                model=self.model,
                train_loader=self.train_loader,
                eval_loader=self.eval_loader,
                epochs=self.epochs,
                lr=self.lr,
                device=self.device,
            )

        # Baseline evaluation
        eval_metrics = evaluate_classification(
            model=self.model,
            eval_loader=self.eval_loader,
            device=self.device,
        )

        # Latency and throughput
        latency_stats = measure_latency_and_throughput(
            model=self.model,
            dataloader=self.eval_loader,
            device=self.device,
        )

        # Memory measurement
        peak_mem = measure_peak_memory_mb(self.device)

        return {
            "model_metadata": self.model_meta,
            "dataset_metadata": self.dataset_meta,
            "train_metrics": train_metrics,
            "eval_metrics": eval_metrics,
            "latency": latency_stats,
            "peak_memory_mb": peak_mem,
        }

    def cleanup(self) -> None:
        """Clean up memory and resources."""
        if self.model is not None:
            cleanup_model(self.model)
            self.model = None
        self.train_loader = None
        self.eval_loader = None
        self.tokenizer = None

class RulerAdapter:
    """Evaluates long-context effective capacity using Needle-in-a-Haystack synthetic tasks."""

    def __init__(
        self,
        module_name: str,
        module_type: str,
        model_name: str,
        dataset_name: str = "ruler",
        rank: int | None = 64,
        device: str | None = None,
        dtype: str = "float32",
        context_length: int = 4096,
        num_samples: int = 10,
        seed: int = 42,
        **kwargs: Any,
    ) -> None:
        self.module_name = module_name
        self.module_type = module_type
        self.model_name = model_name
        self.dataset_name = dataset_name
        self.rank = rank
        self.device = device or ("cuda" if torch.cuda.is_available() else "cpu")
        self.dtype = dtype
        self.context_length = context_length
        self.num_samples = num_samples
        self.seed = seed
        self.extra_kwargs = kwargs

        self.model: nn.Module | None = None
        self.tokenizer: Any = None
        self.model_meta: ModelMetadata | None = None
        self.dataset_meta: DatasetMetadata | None = None

    def setup(self) -> None:
        """Load target model and tokenizer with extended position context."""
        torch.manual_seed(self.seed)
        random.seed(self.seed)

        self.model, self.tokenizer, model_cfg = load_model_and_tokenizer(
            self.model_name,
            device=self.device,
            dtype=self.dtype,
            task="lm",
        )
        self.model_meta = get_model_metadata(self.model, model_cfg)

        self.dataset_meta = DatasetMetadata(
            name=f"ruler_{self.context_length}",
            task="ruler",
            num_labels=1,
            num_train=0,
            num_eval=self.num_samples,
        )

    def apply_module(self) -> None:
        """Inject candidate attention replacement or compression algorithm."""
        if self.model is None:
            raise RuntimeError("Setup must be called before applying module.")

        if self.module_type in ("baseline", "none", "full", "dense"):
            LOG.info("Baseline RULER run: skipping module injection.")
            return

        if self.module_type == "attention":
            self.model = apply_attention_module(
                self.model,
                module_name=self.module_name,
                rank=self.rank or 64,
                **self.extra_kwargs,
            )
        elif self.module_type == "compressor":
            self.model = apply_compressor(
                self.model,
                module_name=self.module_name,
                rank=self.rank or 64,
                **self.extra_kwargs,
            )

        if self.model_meta is not None:
            self.model_meta.num_params = count_parameters(self.model)

    def _generate_synthetic_niah(self, depth: float) -> tuple[str, str, str]:
        """Generate needle, haystack, and query question at a relative context depth."""
        secret_number = str(random.randint(100000, 999999))
        needle = f" The special magic passcode is {secret_number}. Remember this passcode."
        haystack_filler = " The grass is green. The sky is blue. Numbers and facts surround us every day."

        # Estimate repetitions to reach desired context token length
        tokens_per_filler = max(len(self.tokenizer(haystack_filler).input_ids), 1) if self.tokenizer else 15
        needed_fillers = max(self.context_length // tokens_per_filler, 20)

        insertion_index = int(needed_fillers * depth)
        parts = [haystack_filler] * needed_fillers
        parts.insert(insertion_index, needle)

        prompt = "".join(parts) + "\nQuestion: What is the special magic passcode?\nAnswer: The special magic passcode is"
        return prompt, secret_number, secret_number

    def run(self) -> dict[str, Any]:
        """Execute Needle-in-a-Haystack sweeps across various depths."""
        self.setup()
        self.apply_module()

        depths = [0.1, 0.25, 0.5, 0.75, 0.9]
        found_count = 0
        total_tests = 0
        total_latency_s = 0.0

        self.model.eval()
        for i in range(self.num_samples):
            depth = depths[i % len(depths)]
            prompt, expected, _ = self._generate_synthetic_niah(depth)

            inputs = self.tokenizer(prompt, return_tensors="pt", truncation=True, max_length=self.context_length)
            inputs = {k: v.to(self.device) for k, v in inputs.items()}

            start_t = time.perf_counter()
            with torch.no_grad():
                gen_out = self.model.generate(
                    **inputs,
                    max_new_tokens=10,
                    do_sample=False,
                ) if hasattr(self.model, "generate") else None

            duration = time.perf_counter() - start_t
            total_latency_s += duration

            if gen_out is not None:
                decoded = self.tokenizer.decode(gen_out[0], skip_special_tokens=True)
                if expected in decoded:
                    found_count += 1
            total_tests += 1

        accuracy = found_count / max(total_tests, 1)
        peak_mem = measure_peak_memory_mb(self.device)
        avg_latency_ms = (total_latency_s / max(total_tests, 1)) * 1000.0

        return {
            "model_metadata": self.model_meta,
            "dataset_metadata": self.dataset_meta,
            "eval_metrics": {
                "accuracy": float(accuracy),
                "needle_found_rate": float(accuracy),
                "context_length": self.context_length,
                "num_samples": total_tests,
            },
            "latency": {
                "latency_ms_per_pass": float(avg_latency_ms),
                "throughput_tok_per_sec": float((self.context_length * total_tests) / max(total_latency_s, 1e-6)),
            },
            "peak_memory_mb": peak_mem,
        }

    def cleanup(self) -> None:
        """Clean up model and memory."""
        if self.model is not None:
            cleanup_model(self.model)
            self.model = None
        self.tokenizer = None

def run(self) -> dict[str, Any]:
    """Execute Needle-in-a-Haystack sweeps across various depths."""
    self.setup()
    self.apply_module()

    depths = [0.1, 0.25, 0.5, 0.75, 0.9]
    found_count = 0
    total_tests = 0
    total_latency_s = 0.0

    self.model.eval()
    for i in range(self.num_samples):
        depth = depths[i % len(depths)]
        prompt, expected, _ = self._generate_synthetic_niah(depth)

        inputs = self.tokenizer(prompt, return_tensors="pt", truncation=True, max_length=self.context_length)
        inputs = {k: v.to(self.device) for k, v in inputs.items()}

        start_t = time.perf_counter()
        with torch.no_grad():
            gen_out = self.model.generate(
                **inputs,
                max_new_tokens=10,
                do_sample=False,
            ) if hasattr(self.model, "generate") else None

        duration = time.perf_counter() - start_t
        total_latency_s += duration

        if gen_out is not None:
            decoded = self.tokenizer.decode(gen_out[0], skip_special_tokens=True)
            if expected in decoded:
                found_count += 1
        total_tests += 1

    accuracy = found_count / max(total_tests, 1)
    peak_mem = measure_peak_memory_mb(self.device)
    avg_latency_ms = (total_latency_s / max(total_tests, 1)) * 1000.0

    return {
        "model_metadata": self.model_meta,
        "dataset_metadata": self.dataset_meta,
        "eval_metrics": {
            "accuracy": float(accuracy),
            "needle_found_rate": float(accuracy),
            "context_length": self.context_length,
            "num_samples": total_tests,
        },
        "latency": {
            "latency_ms_per_pass": float(avg_latency_ms),
            "throughput_tok_per_sec": float((self.context_length * total_tests) / max(total_latency_s, 1e-6)),
        },
        "peak_memory_mb": peak_mem,
    }

class ZeroShotAdapter:
    """Zero-shot evaluation adapter using logprob-based choice ranking without fine-tuning."""

    def __init__(
        self,
        module_name: str,
        module_type: str,
        model_name: str,
        dataset_name: str = "arc_challenge",
        rank: int | None = 64,
        device: str | None = None,
        dtype: str = "float32",
        batch_size: int = 16,
        max_length: int = 512,
        seed: int = 42,
        num_samples: int | None = 100,
        **kwargs: Any,
    ) -> None:
        self.module_name = module_name
        self.module_type = module_type
        self.model_name = model_name
        self.dataset_name = dataset_name.lower()
        self.rank = rank
        self.device = device or ("cuda" if torch.cuda.is_available() else "cpu")
        self.dtype = dtype
        self.batch_size = batch_size
        self.max_length = max_length
        self.seed = seed
        self.num_samples = num_samples
        self.extra_kwargs = kwargs

        self.model: nn.Module | None = None
        self.tokenizer: Any = None
        self.model_meta: ModelMetadata | None = None
        self.dataset_meta: DatasetMetadata | None = None
        self.samples: list[dict[str, Any]] = []

    def setup(self) -> None:
        """Load causal LM model, tokenizer, and zero-shot prompts."""
        torch.manual_seed(self.seed)

        self.model, self.tokenizer, model_cfg = load_model_and_tokenizer(
            self.model_name,
            device=self.device,
            dtype=self.dtype,
            task="lm",
        )
        self.model_meta = get_model_metadata(self.model, model_cfg)

        self.dataset_meta = DatasetMetadata(
            name=self.dataset_name,
            task="zeroshot",
            num_labels=4,
            num_train=0,
            num_eval=self.num_samples or 100,
        )

        # Build prompt samples (cached from datasets or generated for evaluation)
        self.samples = self._load_samples()

    def _load_samples(self) -> list[dict[str, Any]]:
        """Load evaluation prompts and choice completions."""
        try:
            from datasets import load_dataset
            if "arc" in self.dataset_name:
                subset = "ARC-Challenge" if "challenge" in self.dataset_name else "ARC-Easy"
                ds = load_dataset("ai2_arc", subset, split="test")
                items = []
                limit = self.num_samples or len(ds)
                for i, row in enumerate(ds):
                    if i >= limit:
                        break
                    choices = row["choices"]["text"]
                    labels = row["choices"]["label"]
                    answer_key = row["answerKey"]
                    correct_idx = labels.index(answer_key) if answer_key in labels else 0
                    items.append({
                        "prompt": f"Question: {row['question']}\nAnswer:",
                        "choices": choices,
                        "answer_idx": correct_idx,
                    })
                return items
            elif "hellaswag" in self.dataset_name:
                ds = load_dataset("hellaswag", split="validation")
                items = []
                limit = self.num_samples or len(ds)
                for i, row in enumerate(ds):
                    if i >= limit:
                        break
                    items.append({
                        "prompt": f"Context: {row['ctx']}\nCompletion:",
                        "choices": row["endings"],
                        "answer_idx": int(row["label"]),
                    })
                return items
        except Exception as e:
            LOG.warning("Falling back to local zero-shot sample format: %s", e)

        # Fallback dummy sample structure when offline
        return [
            {
                "prompt": f"Evaluation prompt {i}: Which is the best answer?",
                "choices": ["Option A", "Option B", "Option C", "Option D"],
                "answer_idx": 0,
            }
            for i in range(min(self.num_samples or 10, 10))
        ]

    def apply_module(self) -> None:
        """Apply attention module replacement or compressor."""
        if self.model is None:
            raise RuntimeError("Setup must be run before applying module.")

        if self.module_type in ("baseline", "none", "full", "dense"):
            LOG.info("Baseline zero-shot run: skipping module injection.")
            return

        if self.module_type == "attention":
            self.model = apply_attention_module(
                self.model,
                module_name=self.module_name,
                rank=self.rank or 64,
                **self.extra_kwargs,
            )
        elif self.module_type == "compressor":
            self.model = apply_compressor(
                self.model,
                module_name=self.module_name,
                rank=self.rank or 64,
                **self.extra_kwargs,
            )

        if self.model_meta is not None:
            self.model_meta.num_params = count_parameters(self.model)

    def _compute_logprobs(self, prompt: str, choice: str) -> float:
        """Compute sum of log probabilities for a choice given prompt."""
        if self.model is None or self.tokenizer is None:
            return 0.0

        full_text = prompt + " " + choice
        enc_full = self.tokenizer(full_text, return_tensors="pt").to(self.device)
        enc_prompt = self.tokenizer(prompt, return_tensors="pt")

        prompt_len = enc_prompt.input_ids.shape[1]
        input_ids = enc_full.input_ids

        with torch.no_grad():
            outputs = self.model(**enc_full)
            logits = outputs.logits

        # Log-softmax over vocabulary
        log_probs = torch.log_softmax(logits, dim=-1)

        # Sum logprobs of tokens belonging to choice
        total_logprob = 0.0
        for i in range(prompt_len, input_ids.shape[1]):
            token_id = input_ids[0, i]
            total_logprob += float(log_probs[0, i - 1, token_id].item())

        return total_logprob

    def run(self) -> dict[str, Any]:
        """Execute zero-shot evaluation across prompts."""
        self.setup()
        self.apply_module()

        correct = 0
        total = 0

        self.model.eval()
        for sample in self.samples:
            prompt = sample["prompt"]
            choices = sample["choices"]
            target = sample["answer_idx"]

            choice_logprobs = [self._compute_logprobs(prompt, c) for c in choices]
            pred_idx = int(torch.argmax(torch.tensor(choice_logprobs)).item())

            if pred_idx == target:
                correct += 1
            total += 1

        accuracy = correct / max(total, 1)
        peak_mem = measure_peak_memory_mb(self.device)

        return {
            "model_metadata": self.model_meta,
            "dataset_metadata": self.dataset_meta,
            "eval_metrics": {
                "accuracy": float(accuracy),
                "num_samples": total,
                "correct": correct,
            },
            "latency": {
                "latency_ms_per_token": 0.0,
                "throughput_tok_per_sec": 0.0,
            },
            "peak_memory_mb": peak_mem,
        }

    def cleanup(self) -> None:
        """Clean up model and memory."""
        if self.model is not None:
            cleanup_model(self.model)
            self.model = None
        self.tokenizer = None
        self.samples.clear()
