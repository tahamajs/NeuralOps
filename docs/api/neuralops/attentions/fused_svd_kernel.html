Module neuralops.attentions.fused_svd_kernel

Fused SVD-Linear Layer with SRAM-Resident RMSNorm & Dual-Projection.

RESEARCH FIX: Eliminates GPU High-Bandwidth Memory (HBM) roundtrip latency by fusing: 1. Down-projection: Z = X @ V^T (hidden_dim -> rank) 2. In-SRAM RMSNorm: Z_norm = Z * rsqrt(mean(Z^2) + eps) * weight 3. Up-projection: Y = Z_norm @ U^T + bias (rank -> out_dim)

The intermediate activation matrix Z \in R^{B x L x r} is kept strictly inside GPU Shared Memory (SRAM) registers and never materialized to DRAM/HBM. Includes Triton JIT accelerated kernel with CUDA tensor-core blocking and PyTorch autograd function with fallback.

    layer: Any = None,

    target = layer if layer is not None else model
    if target is None:
        return None

    cfg = model_config or getattr(target, "config", None)

    # 1. Whole model traversal support
    if (
        old_attn is None
        and target is not None
        and (
            hasattr(target, "transformer")
            or hasattr(target, "distilbert")
            or hasattr(target, "roberta")
            or hasattr(target, "bert")
            or (hasattr(target, "model") and hasattr(target.model, "layers"))
            or (hasattr(target, "encoder") and hasattr(target.encoder, "layer"))
        )

            hasattr(target, "q_lin")
            or hasattr(target, "q_proj")
            or hasattr(target, "c_attn")
            or hasattr(target, "query")
        )
    ):
        try:
            layer_list = None
            if hasattr(target, "distilbert") and hasattr(target.distilbert, "transformer"):
                layer_list = target.distilbert.transformer.layer
            elif hasattr(target, "roberta") and hasattr(target.roberta, "encoder"):
                layer_list = target.roberta.encoder.layer
            elif hasattr(target, "bert") and hasattr(target.bert, "encoder"):
                layer_list = target.bert.encoder.layer
            elif hasattr(target, "transformer") and hasattr(target.transformer, "h"):
                layer_list = target.transformer.h
            elif hasattr(target, "transformer") and hasattr(target.transformer, "layer"):
                layer_list = target.transformer.layer
            elif hasattr(target, "model") and hasattr(target.model, "layers"):
                layer_list = target.model.layers
            elif hasattr(target, "encoder") and hasattr(target.encoder, "layer"):
                layer_list = target.encoder.layer

            if layer_list is not None:
                tot = len(layer_list)
                for idx, lyr in enumerate(layer_list):
                    inject_fused_svd_kernel(
                        layer=lyr,
                        model_config=cfg,
                        comp_config=comp_config,
                        layer_idx=idx,
                        total_layers=tot,
                        **kwargs,
                    )
                return target
        except Exception as e:
            logger.error("inject_fused_svd_kernel whole-model traversal failed: %s", e)
            return None

    # 2. Named modules whole-model fallback
    if (
        old_attn is None
        and target is not None
        and hasattr(target, "named_modules")
        and not (
            hasattr(target, "attention")
            or hasattr(target, "self")
            or hasattr(target, "attn")
            or hasattr(target, "q_proj")
            or hasattr(target, "q_lin")
            or hasattr(target, "query")
            or hasattr(target, "c_attn")
        )
    ):
        try:
            replaced = 0
            for name, module in list(target.named_modules()):
                if (
                    hasattr(module, "q_lin")
                    or hasattr(module, "query")
                    or hasattr(module, "q_proj")
                    or hasattr(module, "c_attn")
                ):
                    parent_name = name.rsplit(".", 1)[0] if "." in name else ""
                    child_name = name.rsplit(".", 1)[-1] if "." in name else name
                    parent = target.get_submodule(parent_name) if parent_name else target
                    new_attn = inject_fused_svd_kernel(
                        layer=None,
                        model_config=cfg,
                        comp_config=comp_config,
                        old_attn=module,
                        **kwargs,
                    )
                    if new_attn is not None:
                        setattr(parent, child_name, new_attn)
                        replaced += 1
            if replaced > 0:
                return target
        except Exception as e:
            logger.error("inject_fused_svd_kernel named_modules replacement failed: %s", e)
            return None

        src_attn = (
            old_attn
            if old_attn is not None
            else (
                getattr(
                    layer,
                    "attention",
                    getattr(layer, "self", getattr(layer, "attn", None)),
                )
                if layer is not None
                else None
            )
        )
        new_attn = FusedSVDKernelAttention(
            config=cfg,
            old_attn=src_attn,
            **kwargs,
        )
        if src_attn is not None:
            new_attn.copy_weights(src_attn)
        if layer is not None:
            # Transfer new_attn to the target layer's device and dtype
            target_param = next(layer.parameters(), None)
            if target_param is not None:
                new_attn.to(device=target_param.device, dtype=target_param.dtype)

                if hasattr(layer.attention, "self"):
                    layer.attention.self = new_attn
                    new_attn.is_nested_encoder = True
                else:
                    layer.attention = new_attn

                layer.self = new_attn
                new_attn.is_nested_encoder = True
            elif hasattr(layer, "attn"):
                layer.attn = new_attn
        return new_attn
    except Exception as e:
        logger.error("inject_fused_svd_kernel failed: %s", e)
        return None

        else:

class FusedSVDLinear(nn.Module):
    """
    RESEARCH SOTA: Fused SVD-Linear Module.
    Combines low-rank projection $Z = x V^T$, in-SRAM RMSNorm, and up-projection $Y = Z U^T + b$
    into a single kernel invocation, eliminating HBM roundtrips.
    """

    def __init__(
        self,
        in_features: int,
        out_features: int,
        rank: int = 32,
        bias: bool = True,
        use_rms_norm: bool = False,
        eps: float = 1e-6,
        dtype: torch.dtype = torch.float32,
        device: torch.device | None = None,
    ) -> None:
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.rank = rank
        self.eps = eps
        self.use_rms_norm = use_rms_norm

        # Low-rank factorized parameters
        self.V = nn.Parameter(torch.empty(rank, in_features, dtype=dtype, device=device))
        self.U = nn.Parameter(torch.empty(out_features, rank, dtype=dtype, device=device))

        if use_rms_norm:
            self.norm_weight = nn.Parameter(torch.ones(rank, dtype=dtype, device=device))
        else:
            self.register_parameter("norm_weight", None)

        if bias:
            self.bias = nn.Parameter(torch.empty(out_features, dtype=dtype, device=device))
        else:
            self.register_parameter("bias", None)

        self.reset_parameters()

    def reset_parameters(self) -> None:
        # Symmetrical square-root initialization
        nn.init.orthogonal_(self.U)
        nn.init.orthogonal_(self.V)
        if self.bias is not None:
            nn.init.zeros_(self.bias)
        if self.norm_weight is not None:
            nn.init.ones_(self.norm_weight)

    def weight(self) -> torch.Tensor:
        """Reconstructed effective weight matrix for compatibility."""
        return torch.matmul(self.U, self.V)

    def decompose_from_weight(self, weight: torch.Tensor, bias: torch.Tensor | None = None) -> None:
        """Decompose dense weight matrix into symmetric SVD factors."""
        with torch.no_grad():
            W = weight.float()
            U, S, Vt = torch.linalg.svd(W, full_matrices=False)
            r = min(self.rank, len(S))
            sqrt_s = torch.sqrt(S[:r])
            self.U.data.zero_()
            self.V.data.zero_()
            self.U.data[:, :r] = (U[:, :r] * sqrt_s).to(self.U.dtype)
            self.V.data[:r, :] = (sqrt_s.unsqueeze(1) * Vt[:r, :]).to(self.V.dtype)
            if bias is not None and self.bias is not None:
                self.bias.data.copy_(bias)

    def get_orthogonality_loss(self) -> torch.Tensor:
        """
        # RESEARCH FIX: Computes Stiefel manifold penalty to prevent singular subspace collapse.
        """
        u_gram = torch.matmul(self.U.T, self.U)
        v_gram = torch.matmul(self.V, self.V.T)
        eye_r = torch.eye(self.rank, device=self.U.device, dtype=self.U.dtype)
        loss_u = torch.norm(u_gram - eye_r, p="fro") ** 2
        loss_v = torch.norm(v_gram - eye_r, p="fro") ** 2
        return (loss_u + loss_v) / (self.rank * self.rank)

            f"rank={self.rank}, rms_norm={self.use_rms_norm}, bias={self.bias is not None}"

Define the forward of the custom autograd Function.

This function is to be overridden by all subclasses. There are two ways to define forward:

Usage 1 (Combined forward and ctx)::

@staticmethod
def forward(ctx: Any, *args: Any, **kwargs: Any) -> Any:
    pass
  • It must accept a context ctx as the first argument, followed by any number of arguments (tensors or other types).
  • See :ref:combining-forward-context for more details

Usage 2 (Separate forward and ctx)::

@staticmethod
def forward(*args: Any, **kwargs: Any) -> Any:
    pass


@staticmethod
def setup_context(ctx: Any, inputs: Tuple[Any, ...], output: Any) -> None:
    pass
  • The forward no longer accepts a ctx argument.
  • Instead, you must also override the :meth:torch.autograd.Function.setup_context staticmethod to handle setting up the ctx object. output is the output of the forward, inputs are a Tuple of inputs to the forward.
  • See :ref:extending-autograd for more details

The context can be used to store arbitrary data that can be then retrieved during the backward pass. Tensors should not be stored directly on ctx (though this is not currently enforced for backward compatibility). Instead, tensors should be saved either with :func:ctx.save_for_backward if they are intended to be used in backward (equivalently, vjp) or :func:ctx.save_for_forward if they are intended to be used for in jvp.
