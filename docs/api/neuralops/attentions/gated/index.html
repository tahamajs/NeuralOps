class BaseAttention(nn.Module):
    """Every custom attention module must inherit from this."""

    def __init__(self, config: Any | None = None, rank: int | None = None, **kwargs: Any) -> None:
        super().__init__()
        self.config = config
        self.rank = rank
        self.num_heads = (
            kwargs.get("num_heads")
            or kwargs.get("num_attention_heads")
            or (
                getattr(
                    config,
                    "num_attention_heads",
                    getattr(config, "num_heads", getattr(config, "n_heads", 12)),
                )
                if config is not None
                else 12
            )
        )
        self.hidden_size = (
            kwargs.get("hidden_size")
            or kwargs.get("dim")
            or kwargs.get("d_model")
            or (
                getattr(
                    config,
                    "hidden_size",
                    getattr(config, "d_model", getattr(config, "dim", 768)),
                )
                if config is not None
                else 768
            )
        )
        self.head_dim = getattr(
            config,
            "d_kv",
            getattr(
                config,
                "head_dim",
                self.hidden_size // self.num_heads if self.num_heads > 0 else 64,
            ),
        )

    @abstractmethod
    def forward(
        self,
        hidden_states,
        attention_mask=None,
        head_mask=None,
        output_attentions=False,
        **kwargs,
    ) -> tuple[Any, Any | None]:
        """Must return (output, attention_weights_or_None)."""
        raise NotImplementedError("Subclasses of BaseAttention must implement forward()")

    def copy_weights(self, old_attn: Any) -> None:
        """Universal fallback: copy pretrained weights from old attention layer."""
        if old_attn is None:
            return
        import torch

        from neuralops.core.adapters import UniversalAdapter

        adapter = UniversalAdapter()

        try:
            w_q, b_q = adapter.get_weight_bias(old_attn, "q")
            w_k, b_k = adapter.get_weight_bias(old_attn, "k")
            w_v, b_v = adapter.get_weight_bias(old_attn, "v")
            w_o, b_o = adapter.get_weight_bias(old_attn, "o")

            with torch.no_grad():
                dim = self.hidden_size
                for q_name in ["W_q", "q_proj", "q_lin", "Wq", "query", "q"]:
                    if hasattr(self, q_name):
                        mod = getattr(self, q_name)
                        if hasattr(mod, "weight") and mod.weight.shape == w_q.shape:
                            mod.weight.copy_(w_q)
                            if b_q is not None and getattr(mod, "bias", None) is not None:
                                mod.bias.copy_(b_q)

                for k_name in ["W_k", "k_proj", "k_lin", "Wk", "key", "k"]:
                    if hasattr(self, k_name):
                        mod = getattr(self, k_name)
                        if hasattr(mod, "weight") and mod.weight.shape == w_k.shape:
                            mod.weight.copy_(w_k)
                            if b_k is not None and getattr(mod, "bias", None) is not None:
                                mod.bias.copy_(b_k)

                for v_name in ["W_v", "v_proj", "v_lin", "Wv", "value", "v"]:
                    if hasattr(self, v_name):
                        mod = getattr(self, v_name)
                        if hasattr(mod, "weight") and mod.weight.shape == w_v.shape:
                            mod.weight.copy_(w_v)
                            if b_v is not None and getattr(mod, "bias", None) is not None:
                                mod.bias.copy_(b_v)

                for o_name in ["out", "out_proj", "o_proj", "out_lin", "proj", "dense"]:
                    if hasattr(self, o_name):
                        mod = getattr(self, o_name)
                        if hasattr(mod, "weight") and mod.weight.shape == w_o.shape:
                            mod.weight.copy_(w_o)
                            if b_o is not None and getattr(mod, "bias", None) is not None:
                                mod.bias.copy_(b_o)

                # Handle fused in_proj
                if hasattr(self, "in_proj") and hasattr(self.in_proj, "weight"):
                    if self.in_proj.weight.shape == torch.Size([dim * 2, dim]):
                        self.in_proj.weight[:dim].copy_(w_v[:dim])
                        self.in_proj.weight[dim:].copy_(w_q[:dim])
                    elif self.in_proj.weight.shape == torch.Size([dim * 3, dim]):
                        self.in_proj.weight[:dim].copy_(w_q[:dim])
                        self.in_proj.weight[dim : 2 * dim].copy_(w_k[:dim])
                        self.in_proj.weight[2 * dim :].copy_(w_v[:dim])
        except (AttributeError, RuntimeError, ValueError) as e:
            logger.debug("BaseAttention.copy_weights failed: %s", e)
        except Exception as e:
            logger.warning("Unexpected error in BaseAttention.copy_weights: %s", e)

    def copy_weights(self, old_attn: Any) -> None:
        """Transfer pretrained projection weights with strict Frobenius norm verification."""
        if old_attn is None:
            return

        cls_name = getattr(old_attn, "__class__", type(old_attn)).__name__
        has_out_lin = (
            hasattr(old_attn, "out_lin")
            or hasattr(old_attn, "c_proj")
            or hasattr(old_attn, "o_proj")
            or hasattr(old_attn, "dense")
        )
        if (
            (
                hasattr(old_attn, "self")
                and hasattr(old_attn.self, "query")
                and hasattr(old_attn, "output")
                and hasattr(old_attn.output, "dense")
            )
            or (
                cls_name
                in [
                    "BertSelfAttention",
                    "RobertaSelfAttention",
                    "XLMRobertaSelfAttention",
                    "ElectraSelfAttention",
                    "DebertaSelfAttention",
                ]
                or (cls_name.endswith("SelfAttention") and "distil" not in cls_name.lower())
            )
            and not has_out_lin
        ):
            self.is_nested_encoder = True

        if hasattr(old_attn, "rotary_emb"):
            self.rotary_emb = old_attn.rotary_emb

        extracted: dict[str, tuple[torch.Tensor | None, torch.Tensor | None]] = {
            "q": (None, None),
            "k": (None, None),
            "v": (None, None),
            "o": (None, None),
        }

        # 1. GPT-2: Fused c_attn + c_proj
        if hasattr(old_attn, "c_attn") and old_attn.c_attn is not None:
            c_attn_mod = old_attn.c_attn
            if hasattr(c_attn_mod, "merge") and not getattr(c_attn_mod, "merged", False):
                try:
                    c_attn_mod.merge()
                except Exception as e:
                    logger.debug(f"c_attn_mod.merge failed: {e}")
            base_c = getattr(c_attn_mod, "base_layer", c_attn_mod)
            c_w = getattr(base_c, "weight", getattr(c_attn_mod, "weight", None))
            if c_w is not None:
                c_w = c_w.data.clone()
                if (
                    ("Conv1D" in c_attn_mod.__class__.__name__)
                    or ("Conv1D" in base_c.__class__.__name__)
                    or hasattr(c_attn_mod, "nf")
                    or hasattr(base_c, "nf")
                    or (c_w.ndim == 2 and c_w.shape[0] < c_w.shape[1])
                ):
                    c_w = c_w.t().contiguous()
                split_size = c_w.size(0) // 3
                bias_mod = getattr(base_c, "bias", getattr(c_attn_mod, "bias", None))
                c_b = bias_mod.data.clone() if bias_mod is not None else None
                extracted["q"] = (
                    c_w[:split_size, :],
                    c_b[:split_size] if c_b is not None else None,
                )
                extracted["k"] = (
                    c_w[split_size : 2 * split_size, :],
                    c_b[split_size : 2 * split_size] if c_b is not None else None,
                )
                extracted["v"] = (
                    c_w[2 * split_size :, :],
                    c_b[2 * split_size :] if c_b is not None else None,
                )

            if hasattr(old_attn, "c_proj") and old_attn.c_proj is not None:
                c_proj_mod = old_attn.c_proj
                base_o = getattr(c_proj_mod, "base_layer", c_proj_mod)
                o_w = getattr(base_o, "weight", getattr(c_proj_mod, "weight", None))
                if o_w is not None:
                    o_w = o_w.data.clone()
                    if (
                        ("Conv1D" in c_proj_mod.__class__.__name__)
                        or ("Conv1D" in base_o.__class__.__name__)
                        or hasattr(c_proj_mod, "nf")
                        or hasattr(base_o, "nf")
                        or (o_w.ndim == 2 and o_w.shape[0] < o_w.shape[1])
                    ):
                        o_w = o_w.t().contiguous()
                    bias_o = getattr(base_o, "bias", getattr(c_proj_mod, "bias", None))
                    extracted["o"] = (
                        o_w,
                        bias_o.data.clone() if bias_o is not None else None,
                    )

        # 2. RoBERTa / BERT nested self-attention
        elif hasattr(old_attn, "self") and hasattr(old_attn.self, "query"):
            s = old_attn.self
            for proj_key, sub_name in [("q", "query"), ("k", "key"), ("v", "value")]:
                mod = getattr(s, sub_name, None)
                base_mod = getattr(mod, "base_layer", mod)
                if base_mod is not None and hasattr(base_mod, "weight"):
                    extracted[proj_key] = (
                        base_mod.weight.data.clone(),
                        (
                            base_mod.bias.data.clone()
                            if getattr(base_mod, "bias", None) is not None
                            else None
                        ),
                    )
            if hasattr(old_attn, "output") and hasattr(old_attn.output, "dense"):
                out_mod = old_attn.output.dense
                base_out = getattr(out_mod, "base_layer", out_mod)
                extracted["o"] = (
                    base_out.weight.data.clone(),
                    (
                        base_out.bias.data.clone()
                        if getattr(base_out, "bias", None) is not None
                        else None
                    ),
                )

        # 3. Standard linear projections (DistilBERT, Qwen, LLaMA)
        else:
            proj_candidates = {
                "q": ["q_lin", "q_proj", "query", "W_q"],
                "k": ["k_lin", "k_proj", "key", "W_k"],
                "v": ["v_lin", "v_proj", "value", "W_v"],
                "o": ["out_lin", "out_proj", "dense", "o_proj", "c_proj", "wo"],
            }
            for proj_key, names in proj_candidates.items():
                for name in names:
                    if hasattr(old_attn, name):
                        mod = getattr(old_attn, name)
                        base_mod = getattr(mod, "base_layer", mod)
                        if hasattr(base_mod, "weight"):
                            w_data = base_mod.weight.data.clone()
                            if (
                                (base_mod.__class__.__name__ == "Conv1D")
                                or hasattr(base_mod, "nf")
                                or (
                                    w_data.ndim == 2
                                    and proj_key == "o"
                                    and w_data.shape[0] < w_data.shape[1]
                                )
                            ):
                                w_data = w_data.t().contiguous()
                            b_data = getattr(base_mod, "bias", None)
                            extracted[proj_key] = (
                                w_data,
                                b_data.data.clone() if b_data is not None else None,
                            )
                            break

        # Fallback via UniversalAdapter
        missing = [
            k
            for k, (w, _) in extracted.items()
            if w is None and (k != "o" or not self.is_nested_encoder)
        ]
        if missing:
            try:
                from neuralops.core.adapters import UniversalAdapter

                adapter = UniversalAdapter()
                for k in missing:
                    w, b = adapter.get_weight_bias(old_attn, k)
                    if w is not None:
                        extracted[k] = (w.clone(), b.clone() if b is not None else None)
            except Exception as e:
                logger.debug(f"UniversalAdapter fallback: {e}")

        missing_after = [
            k
            for k, (w, _) in extracted.items()
            if w is None and (k != "o" or not self.is_nested_encoder)
        ]
        if missing_after:
            raise RuntimeError(
                f"DecayLinearAttention copy_weights: missing mandatory projections {missing_after} "
                f"from {old_attn.__class__.__name__}."
            )

        target_map = {
            "q": self.q_proj,
            "k": self.k_proj,
            "v": self.v_proj,
            "o": self.out_proj,
        }

        with torch.no_grad():
            for proj_key, target_mod in target_map.items():
                w, b = extracted[proj_key]
                if w is None:
                    if proj_key == "o" and self.is_nested_encoder:
                        nn.init.eye_(target_mod.weight)
                        if target_mod.bias is not None:
                            nn.init.zeros_(target_mod.bias)
                    continue

                if w.ndim != 2:
                    w = w.reshape(w.size(0), -1)

                if (
                    w.shape != target_mod.weight.shape
                    and target_mod.weight.shape == w.t().contiguous().shape
                ):
                    w = w.t().contiguous()
                out_d = min(target_mod.weight.size(0), w.size(0))
                in_d = min(target_mod.weight.size(1), w.size(1))

                target_mod.weight.data[:out_d, :in_d].copy_(
                    w[:out_d, :in_d].to(
                        device=target_mod.weight.device, dtype=target_mod.weight.dtype
                    )
                )

                if b is not None and target_mod.bias is not None:
                    min_b = min(target_mod.bias.numel(), b.numel())
                    target_mod.bias.data[:min_b].copy_(
                        b[:min_b].to(device=target_mod.bias.device, dtype=target_mod.bias.dtype)
                    )
                elif target_mod.bias is not None:
                    nn.init.zeros_(target_mod.bias)

                # Frobenius norm verification (>= 1e-4)
                w_norm = float(torch.linalg.norm(target_mod.weight.data.float()).item())
                if w_norm < 1e-4:
                    logger.warning(
                        f"DecayLinearAttention weight copy verification failed for '{proj_key}': norm too small ({w_norm:.2e})"
                    )

                if self.freeze_base:
                    target_mod.freeze_base_weights()

    def _decay_scan_linear(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
        lambdas: torch.Tensor,
        mask: torch.Tensor | None = None,
        reverse: bool = False,
        chunk_size: int = 32,
    ) -> torch.Tensor:
        r"""
        Numerically stable parallel decay scan with O(1) state per chunk (chunked recurrence).
        Eliminates quadratic O(L^2) memory footprint and prevents GPU out-of-memory.
        """
        B, H, L, d = q.shape

        # Path 1: Hardware-fused Triton kernel on CUDA
        if (
            HAS_TRITON_DECAY
            and q.is_cuda
            and d <= MAX_REGISTER_DIM
            and mask is None
            and not reverse
        ):
            try:
                return gla_scan_triton(q.float(), k.float(), v.float(), lambdas.float()).to(q.dtype)
            except Exception as e:
                logger.debug("Triton decay scan failed, falling back to chunked PyTorch: %s", e)

        q_f = q.float()
        k_f = k.float()
        v_f = v.float()
        lambdas_f = lambdas.float().clamp(min=1e-4, max=1.0 - 1e-4)

        if mask is not None:
            # mask is [B, L] boolean tensor where True = valid token
            m_f = mask.float().unsqueeze(1).unsqueeze(-1)  # [B, 1, L, 1]
            k_f = k_f * m_f
            v_f = v_f * m_f

        if reverse:
            q_f = q_f.flip(dims=[2])
            k_f = k_f.flip(dims=[2])
            v_f = v_f.flip(dims=[2])
            lambdas_f = lambdas_f.flip(dims=[-1])

        log_lambdas = torch.log(lambdas_f)
        cum_log = torch.cumsum(log_lambdas, dim=-1)

        if chunk_size >= L:
            log_decay = cum_log.unsqueeze(-1) - cum_log.unsqueeze(-2)
            decay = torch.exp(log_decay.clamp(max=0.0))
            causal_mask = torch.tril(torch.ones(L, L, device=q.device, dtype=torch.bool))
            decay = decay.masked_fill(~causal_mask, 0.0)
            scores = torch.matmul(q_f, k_f.transpose(-1, -2).contiguous()) * decay
            num = torch.matmul(scores, v_f)
            denom = scores.sum(dim=-1, keepdim=True)
        else:
            pad_len = (chunk_size - (L % chunk_size)) % chunk_size
            if pad_len > 0:
                q_pad = F.pad(q_f, (0, 0, 0, pad_len))
                k_pad = F.pad(k_f, (0, 0, 0, pad_len))
                v_pad = F.pad(v_f, (0, 0, 0, pad_len))
                last_cum = cum_log[:, :, -1:]
                cum_pad = torch.cat([cum_log, last_cum.expand(-1, -1, pad_len)], dim=-1)
            else:
                q_pad, k_pad, v_pad, cum_pad = q_f, k_f, v_f, cum_log

            total_L = L + pad_len
            num_chunks = total_L // chunk_size

            qc = q_pad.view(B, H, num_chunks, chunk_size, d)
            kc = k_pad.view(B, H, num_chunks, chunk_size, d)
            vc = v_pad.view(B, H, num_chunks, chunk_size, d)
            cum_c = cum_pad.view(B, H, num_chunks, chunk_size)

            intra_log_decay = cum_c.unsqueeze(-1) - cum_c.unsqueeze(-2)
            intra_decay = torch.exp(intra_log_decay.clamp(max=0.0))
            causal_mask = torch.tril(
                torch.ones(chunk_size, chunk_size, device=q.device, dtype=torch.bool)
            )
            intra_decay = intra_decay.masked_fill(~causal_mask, 0.0)

            if pad_len > 0:
                valid_in_last = chunk_size - pad_len
                intra_decay[:, :, -1, :, valid_in_last:] = 0.0

            intra_scores = torch.matmul(qc, kc.transpose(-1, -2).contiguous()) * intra_decay
            num_intra = torch.matmul(intra_scores, vc)
            denom_intra = intra_scores.sum(dim=-1, keepdim=True)

            S = torch.zeros(B, H, d, d, device=q.device, dtype=torch.float32)
            z = torch.zeros(B, H, d, 1, device=q.device, dtype=torch.float32)

            num_list = []
            denom_list = []

            prev_cum_end = None
            for c in range(num_chunks):
                curr_cum = cum_c[:, :, c]
                if c == 0:
                    inter_decay = torch.zeros(
                        B, H, chunk_size, 1, device=q.device, dtype=torch.float32
                    )
                else:
                    inter_decay = torch.exp((curr_cum - prev_cum_end).clamp(max=0.0)).unsqueeze(-1)

                num_inter = torch.matmul(qc[:, :, c], S) * inter_decay
                denom_inter = torch.matmul(qc[:, :, c], z) * inter_decay

                num_list.append(num_intra[:, :, c] + num_inter)
                denom_list.append(denom_intra[:, :, c] + denom_inter)

                cum_end = curr_cum[:, :, -1:]
                decay_to_end = torch.exp((cum_end - curr_cum).clamp(max=0.0)).unsqueeze(-1)

                k_curr = kc[:, :, c]
                v_curr = vc[:, :, c]
                if c == num_chunks - 1 and pad_len > 0:
                    valid_in_last = chunk_size - pad_len
                    k_curr = k_curr.clone()
                    k_curr[:, :, valid_in_last:] = 0.0

                k_to_end = k_curr * decay_to_end

                if c == 0:
                    decay_prev_to_end = 0.0
                else:
                    decay_prev_to_end = torch.exp(
                        (cum_end - prev_cum_end).clamp(max=0.0)
                    ).unsqueeze(-1)

                S = S * decay_prev_to_end + torch.matmul(
                    k_to_end.transpose(-1, -2).contiguous(), v_curr
                )
                z = (
                    z * decay_prev_to_end
                    + k_to_end.sum(dim=-2, keepdim=True).transpose(-1, -2).contiguous()
                )
                prev_cum_end = cum_end

            num = torch.cat(num_list, dim=2)[:, :, :L, :]
            denom = torch.cat(denom_list, dim=2)[:, :, :L, :]

        y = num / denom.clamp(min=1e-6)

        if reverse:
            y = y.flip(dims=[2])

        return y.to(q.dtype)

    def forward(
        self,
        hidden_states: torch.Tensor,
        attention_mask: torch.Tensor | None = None,
        head_mask: torch.Tensor | None = None,
        encoder_hidden_states: torch.Tensor | None = None,
        encoder_attention_mask: torch.Tensor | None = None,
        past_key_value: Any | None = None,
        output_attentions: bool | None = False,
        output_hidden_states: bool | None = False,
        return_dict: bool | None = True,
        position_ids: torch.Tensor | None = None,
        position_embeddings: tuple[torch.Tensor, torch.Tensor] | None = None,
        *args: Any,
        **kwargs: Any,
    ) -> tuple[torch.Tensor, torch.Tensor | None]:
        if not hidden_states.is_contiguous():
            hidden_states = hidden_states.contiguous()

        B, L, D = hidden_states.shape
        H, d = self.num_heads, self.head_dim

        is_cross_attention = encoder_hidden_states is not None
        kv_input = encoder_hidden_states if is_cross_attention else hidden_states
        kv_len = kv_input.shape[1]

        # 1. Projections
        q = self.q_proj(hidden_states).view(B, L, H, d).transpose(1, 2).contiguous()
        k = self.k_proj(kv_input).view(B, kv_len, self.num_kv_heads, d).transpose(1, 2).contiguous()
        v = self.v_proj(kv_input).view(B, kv_len, self.num_kv_heads, d).transpose(1, 2).contiguous()

        # 2. Rotary Position Embeddings (RoPE)
        if position_embeddings is not None and not is_cross_attention:
            if isinstance(position_embeddings, tuple):
                cos, sin = position_embeddings
            else:
                cos, sin = position_embeddings, None
            if sin is not None:
                q, k = apply_rotary_pos_emb(q, k, cos, sin, position_ids)
        elif self.rotary_emb is not None and not is_cross_attention:
            if position_ids is not None:
                cos, sin = self.rotary_emb(v, seq_len=kv_len)
                q, k = apply_rotary_pos_emb(q, k, cos, sin, position_ids)

        # 3. GQA expansion
        if self.kv_repeat > 1:
            k = repeat_kv(k, self.kv_repeat)
            v = repeat_kv(v, self.kv_repeat)

        # 4. Kernel feature mapping
        q = F.elu(q) + 1.0
        k = F.elu(k) + 1.0

        # Data-dependent decay lambdas in (0, 1) with bounded negative log decay bias
        decay_logits = self.decay_gate(hidden_states).transpose(
            1, 2
        ).contiguous() + self.log_decay.view(1, -1, 1)
        lambdas = torch.sigmoid(decay_logits).clamp(min=0.85, max=0.999)

        eff_mask = encoder_attention_mask if is_cross_attention else attention_mask
        mask_bool = None
        if eff_mask is not None:
            mask_bool = get_token_valid_mask(eff_mask, B, kv_len, hidden_states.device)

        # 5. True O(L) Linear Decay Scan
        y_fwd = self._decay_scan_linear(q, k, v, lambdas, mask=mask_bool, reverse=False)

        if self.is_decoder or is_cross_attention:
            y = y_fwd
        else:
            y_bwd = self._decay_scan_linear(q, k, v, lambdas, mask=mask_bool, reverse=True)
            y = 0.5 * (y_fwd + y_bwd)

        # 6. Head mask handling
        if head_mask is not None:
            if head_mask.dim() == 1:
                head_mask = head_mask.view(1, -1, 1, 1)
            elif head_mask.dim() == 2:
                head_mask = head_mask.view(head_mask.shape[0], head_mask.shape[1], 1, 1)
            elif head_mask.dim() == 3:
                head_mask = head_mask.unsqueeze(-1)
            y = y * head_mask

        y = torch.nan_to_num(y, nan=0.0)
        out = y.transpose(1, 2).contiguous().view(B, L, self.all_head_size)

        # 7. Output projection with nested encoder bypass (no output_scale)
        if self.is_nested_encoder:
            final_out = out
        else:
            final_out = self.out_proj(out)

        if mask_bool is not None and not is_cross_attention:
            final_out = final_out * mask_bool.unsqueeze(-1).to(final_out.dtype)

        return final_out, None

                in_d = min(target.weight.size(1), w.size(1))

                    w[:out_d, :in_d].to(target.weight.device, target.weight.dtype)
                )

                    mb = min(target.bias.numel(), b.numel())
                    target.bias.data[:mb].copy_(b[:mb].to(target.bias.device, target.bias.dtype))
                elif target.bias is not None:
                    nn.init.zeros_(target.bias)

                if n < 1e-4:
                    logger.warning("GatedDeltaNet: '%s' near-zero norm (%.2e)", key, n)

                    target.freeze_base_weights()

            w_q = extracted["q"][0]
            if w_q is not None:

                in_z = min(self.z_proj.weight.size(1), w_q.size(1))

    def forward(
        self,
        hidden_states: torch.Tensor,
        attention_mask: torch.Tensor | None = None,
        head_mask: torch.Tensor | None = None,
        encoder_hidden_states: torch.Tensor | None = None,
        encoder_attention_mask: torch.Tensor | None = None,
        past_key_value: Any | None = None,
        output_attentions: bool | None = False,
        output_hidden_states: bool | None = False,
        return_dict: bool | None = True,
        position_ids: torch.Tensor | None = None,
        position_embeddings: tuple[torch.Tensor, torch.Tensor] | None = None,
        *args: Any,
        **kwargs: Any,
    ):

            hidden_states = hidden_states.contiguous()

        B, L, D = hidden_states.shape
        H, d = self.num_heads, self.head_dim

        is_cross = encoder_hidden_states is not None
        kv_input = encoder_hidden_states if is_cross else hidden_states
        kv_len = kv_input.shape[1]

        q = self.q_proj(hidden_states).view(B, L, H, d).transpose(1, 2).contiguous()
        k = self.k_proj(kv_input).view(B, kv_len, self.num_kv_heads, d).transpose(1, 2).contiguous()
        v = self.v_proj(kv_input).view(B, kv_len, self.num_kv_heads, d).transpose(1, 2).contiguous()
        z = self.z_proj(hidden_states)

        # ---- RoPE ----
        if position_embeddings is not None and not is_cross:
            if isinstance(position_embeddings, tuple):
                cos, sin = position_embeddings
                q, k = apply_rotary_pos_emb(q, k, cos, sin, position_ids)
        elif self.rotary_emb is not None and not is_cross:
            if position_ids is None:
                position_ids = (
                    torch.arange(kv_len, device=hidden_states.device).unsqueeze(0).expand(B, -1)
                )
            try:
                cos, sin = self.rotary_emb(k, seq_len=kv_len)
            except TypeError:
                try:
                    cos, sin = self.rotary_emb(k, position_ids)
                except Exception:
                    cos, sin = None, None
            if cos is not None and sin is not None:
                q, k = apply_rotary_pos_emb(q, k, cos, sin, position_ids)

        # ---- GQA ----
        if self.num_kv_heads != self.num_heads:
            k = repeat_kv(k, self.kv_repeat)
            v = repeat_kv(v, self.kv_repeat)

        # ---- Gating ----
        hs_beta = hidden_states.to(self.beta_proj.weight.dtype)
        beta = torch.sigmoid(self.beta_proj(hs_beta)).transpose(1, 2).contiguous()  # (B,H,L)
        hs_alpha = hidden_states.to(self.alpha_proj.weight.dtype)
        alpha = torch.sigmoid(self.alpha_proj(hs_alpha)).transpose(1, 2).contiguous()

        # ---- Mask ----
        eff_mask = encoder_attention_mask if is_cross else attention_mask
        mask_bool = self._normalize_valid_mask(eff_mask, B, kv_len, hidden_states.device)

        # ---- is_causal (GDN-FIX-08) ----
        bool(self.is_decoder and not is_cross and L > 1 and past_key_value is None)

        # ---- Delta scan ----
        y_fwd = chunked_delta_rule_scan(
            q,
            k,
            v,
            beta,
            alpha,
            chunk_size=self.chunk_size,
        )
        if not self.is_decoder:
            q_b = torch.flip(q, dims=[2])
            k_b = torch.flip(k, dims=[2])
            v_b = torch.flip(v, dims=[2])
            beta_b = torch.flip(beta, dims=[2])
            alpha_b = torch.flip(alpha, dims=[2])
            y_bwd_f = chunked_delta_rule_scan(
                q_b, k_b, v_b, beta_b, alpha_b, chunk_size=self.chunk_size
            )
            y_bwd = torch.flip(y_bwd_f, dims=[2])
            y = 0.5 * (y_fwd + y_bwd)
        else:
            y = y_fwd

        # ---- head_mask (GDN-FIX-11) ----
        if head_mask is not None:
            hm = head_mask.to(y.dtype)
            if hm.dim() == 1:
                hm = hm.view(1, -1, 1, 1)
            elif hm.dim() == 2:
                hm = hm.view(hm.shape[0], hm.shape[1], 1, 1)
            elif hm.dim() == 3:
                hm = hm.unsqueeze(1)
            y = y * hm

        # ---- Recombine + SiLU gating ----
        y = torch.nan_to_num(y, nan=0.0, posinf=1e4, neginf=-1e4)
        y = y.transpose(1, 2).contiguous().view(B, L, self.all_head_size)
        gated_y = y.to(z.dtype) * F.silu(z)

        # ---- Output projection ----
        if self.is_nested_encoder:
            out = gated_y
        else:
            out = self.out_proj(gated_y)

        out = out * self.output_scale

        # ---- mask output (only once; GDN-FIX-05) ----
        if mask_bool is not None:
            out = out * mask_bool.unsqueeze(-1).to(out.dtype)

        return out, None

    def _gla_scan_chunked(self, q, k, v, lambdas, mask=None, reverse=False):
            B, H, L, d = q.shape
            device = q.device

            # اگر کرنل تریتون در دسترس باشد و مسک نداشته باشیم
            if HAS_TRITON_GLA and q.is_cuda and d <= MAX_REGISTER_DIM and mask is None and not reverse:
                try:
                    return gla_scan_triton(q.float(), k.float(), v.float(), lambdas.float()).to(q.dtype)
                except Exception as e:
                    logger.debug("GLA Triton scan failed, falling back: %s", e)

            if reverse:
                q = torch.flip(q, dims=[2])
                k = torch.flip(k, dims=[2])
                v = torch.flip(v, dims=[2])
                lambdas = torch.flip(lambdas, dims=[2])
                if mask is not None:
                    mask = torch.flip(mask, dims=[1])

            if mask is not None:
                m = mask.float()
                m_kv = m.unsqueeze(1).unsqueeze(-1)
                k = k * m_kv
                v = v * m_kv

            q_f = q.float()
            k_f = k.float()
            v_f = v.float()
            lam = lambdas.float().clamp(min=1e-4, max=1.0 - 1e-4)

            chunk = self.chunk_size
            n_chunks = (L + chunk - 1) // chunk

            # فقط S (استیت اصلی) را نگه می‌داریم و Z (مخرج) کاملا حذف می‌شود
            S = self.state_bias.unsqueeze(0).expand(B, -1, -1, -1).contiguous().to(torch.float32)
            outs = []

            for c in range(n_chunks):
                s = c * chunk
                e = min(s + chunk, L)
                q_c = q_f[:, :, s:e]
                k_c = k_f[:, :, s:e]
                v_c = v_f[:, :, s:e]
                lam_c = lam[:, :, s:e]

                log_lam = torch.log(lam_c)
                C = torch.cumsum(log_lam, dim=-1)
                C_i = C.unsqueeze(-1)
                C_j = C.unsqueeze(-2)
                
                # دقت: برای جلوگیری از سرریزِ اعداد، clamp روی تفاضل می‌زنیم
                D = torch.exp((C_i - C_j).clamp(min=-20.0, max=0.0)) * torch.tril(
                    torch.ones(e - s, e - s, device=device))

                lam_prefix = torch.exp(C)
                
                # محاسبه Y بر اساس State گذشته (بدون مخرج)
                y_prev = torch.einsum("bhl,bhld->bhld", lam_prefix,
                                    torch.einsum("bhlc,bhcd->bhld", q_c, S))

                # محاسبه Y محلی درون-چانکی
                qk = torch.matmul(q_c, k_c.transpose(-1, -2))
                y_cur = torch.matmul(qk * D, v_c)

                y_chunk = y_prev + y_cur
                outs.append(y_chunk)

                # آپدیت S برای چانک بعدی
                lam_prod_chunk = torch.exp(C[:, :, -1]).view(B, H, 1, 1)
                C_end = C[:, :, -1:].unsqueeze(-1)
                C_t = C.unsqueeze(-1)
                weight_t = torch.exp((C_end - C_t).clamp(min=-20.0, max=0.0))
                k_w = k_c * weight_t
                
                S = lam_prod_chunk * S + torch.einsum("bhld,bhle->bhde", k_w, v_c)

            y = torch.cat(outs, dim=2)
            if reverse:
                y = torch.flip(y, dims=[2])
            return y.to(q.dtype)

            q, k = apply_rotary_pos_emb(q, k, cos, sin)

        if self.num_kv_heads < self.num_heads:

            v = repeat_kv(v, self.kv_repeat)

        k = F.elu(k) + 1.0

        if decay_out.shape[-1] != self.num_heads:

                f"GLA: decay_gate output {decay_out.shape[-1]} != num_heads {self.num_heads}"
            )
        decay_logits = decay_out.transpose(1, 2) + self.log_decay.to(decay_out.dtype).view(1, -1, 1)
        lambdas = torch.sigmoid(decay_logits)

        eff_mask = encoder_attention_mask if is_cross else attention_mask
        mask_bool = self._normalize_valid_mask(eff_mask, B, L_kv, hidden_states.device)

        is_causal = bool(self.is_decoder and not is_cross)
        if is_causal:
            y = self._gla_scan_chunked(q, k, v, lambdas, mask=mask_bool, reverse=False)
        else:
            y_fwd = self._gla_scan_chunked(q, k, v, lambdas, mask=mask_bool, reverse=False)
            y_bwd = self._gla_scan_chunked(q, k, v, lambdas, mask=mask_bool, reverse=True)
            y = 0.5 * (y_fwd + y_bwd)

        y = self.state_norm(y)

        if head_mask is not None:
            hm = head_mask.to(y.dtype)
            if hm.dim() == 1:
                hm = hm.view(1, H, 1, 1)
            elif hm.dim() == 2:
                hm = hm.view(hm.shape[0], H, 1, 1)
            elif hm.dim() == 3:
                hm = hm.unsqueeze(1)
            if hm.shape[1] != H:
                raise RuntimeError(f"GLA: head_mask has {hm.shape[1]} heads, expected {H}")

        context = torch.nan_to_num(y * F.silu(z), nan=0.0, posinf=1e4, neginf=-1e4)

        else:

        out = out * self.output_scale

        if mask_bool is not None and not is_cross:
            out = out * mask_bool.unsqueeze(-1).to(out.dtype)

    def _chunked_scan(self, q, k, v, f, chunk_size=16):
        """
        Chunked scan for pure HGRN2 recurrence.
        
        FIX 12: Removed denominator Z (Katharopoulos-style normalization).
                Modern RNNs (HGRN2, GLA, Mamba) rely on f-gate for scaling.
                Dividing by Z fights the forget-gate dynamics.
        """
        B, H, L, d = q.shape
        num_chunks = (L + chunk_size - 1) // chunk_size

        # Only S state (no Z)
        S_prev = torch.zeros(B, H, d, d, device=q.device, dtype=q.dtype)
        outputs = []

        for c in range(num_chunks):
            start = c * chunk_size
            end = min(start + chunk_size, L)
            C = end - start

            q_c = q[:, :, start:end]
            k_c = k[:, :, start:end]
            v_c = v[:, :, start:end]
            f_c = f[:, :, start:end]

            log_f = torch.log(f_c.clamp_min(1e-6))
            log_F_c = torch.cumsum(log_f, dim=-1)

            log_F_prev = torch.cat([
                torch.zeros(B, H, 1, device=q.device, dtype=log_F_c.dtype),
                log_F_c,
            ], dim=-1)

            log_decay = log_F_c.unsqueeze(-2) - log_F_prev.unsqueeze(-1)

            i_idx = torch.arange(C + 1, device=q.device).view(C + 1, 1)
            j_idx = torch.arange(C, device=q.device).view(1, C)
            valid = (i_idx == 0) | (i_idx <= j_idx)
            fill_val = -1e4 if log_decay.dtype == torch.float16 else -1e9
            log_decay = log_decay.masked_fill(~valid, fill_val)
            decay = torch.exp(log_decay).to(q.dtype)

            one_minus_f = (1.0 - f_c).to(q.dtype)
            W_kv = one_minus_f[..., None, None] * (k_c.unsqueeze(-1) * v_c.unsqueeze(-2))

            combined_kv = torch.cat([S_prev.unsqueeze(2), W_kv], dim=2).to(q.dtype)

            B_, H_, Nc, _, _ = combined_kv.shape
            combined_kv_flat = combined_kv.view(B_, H_, Nc, -1)
            S_all_flat = torch.matmul(decay, combined_kv_flat)
            S_all = S_all_flat.view(B_, H_, C, d, d)

            # FIX 12: output is num only — no denominator!
            y_c = torch.einsum("bhjd,bhjde->bhje", q_c, S_all)

            outputs.append(y_c)
            S_prev = S_all[:, :, -1]

        return torch.cat(outputs, dim=2).to(q.dtype)

                past_key_value=None, output_attentions=False,

        k = self.k_proj(hidden_states).view(B, L, self.num_kv_heads, d).transpose(1, 2).contiguous()
        v = self.v_proj(hidden_states).view(B, L, self.num_kv_heads, d).transpose(1, 2).contiguous()

        # FIX 11: REMOVED RoPE completely
        # HGRN2 uses f-gate for positional tracking; RoPE is invalid for
        # outer-product linear attention (phases mix into noise).

        if self.num_kv_heads != self.num_heads:
            k = repeat_kv(k, self.num_key_value_groups)
            v = repeat_kv(v, self.num_key_value_groups)

        # FIX 13: REMOVED ELU+1 — only scale q by 1/sqrt(d)
        # ELU+1 was destroying negative geometry of pretrained GPT-2 weights.
        # Modern HGRN2/GLA/Mamba preserve signed values.
        q = q / math.sqrt(d)

        f = torch.exp((-F.softplus(self.W_f(hidden_states))).clamp(min=-3.0, max=-1e-4)).transpose(1, 2)

        mask_bool = None
        if attention_mask is not None:
            if attention_mask.dim() == 2:
                if attention_mask.dtype == torch.bool:
                    mask_bool = attention_mask
                elif attention_mask.dtype.is_floating_point:
                    mask_bool = attention_mask > -1e4
                else:
                    mask_bool = attention_mask > 0
            elif attention_mask.dim() == 4:
                if attention_mask.dtype == torch.bool:
                    m = attention_mask
                elif attention_mask.dtype.is_floating_point:
                    m = attention_mask > -1e4
                else:
                    m = attention_mask > 0
                mask_bool = m.any(dim=-2).squeeze(1)
            if mask_bool is not None and mask_bool.shape[-1] != L:
                mask_bool = None

        if mask_bool is not None:
            m_exp = mask_bool.unsqueeze(1).unsqueeze(-1).to(k.dtype)
            k = k * m_exp
            v = v * m_exp

        y_fwd = self._chunked_scan(q, k, v, f, chunk_size=self.chunk_size)

        if self.is_decoder:
            y = y_fwd
        else:
            q_r = torch.flip(q, dims=[2])
            k_r = torch.flip(k, dims=[2])
            v_r = torch.flip(v, dims=[2])
            f_r = torch.flip(f, dims=[2])
            y_bwd_r = self._chunked_scan(q_r, k_r, v_r, f_r, chunk_size=self.chunk_size)
            y_bwd = torch.flip(y_bwd_r, dims=[2])
            y = 0.5 * (y_fwd + y_bwd)
            y = y.to(hidden_states.dtype)

        sn_dtype = (self.state_norm.weight.dtype
                    if hasattr(self.state_norm, "weight") and self.state_norm.weight is not None
                    else hidden_states.dtype)
        y = self.state_norm(y.to(sn_dtype)).to(hidden_states.dtype)
        y = torch.nan_to_num(y, nan=0.0, posinf=1e4, neginf=-1e4)

        if head_mask is not None:
            hm = head_mask.to(y.dtype)
            if hm.dim() == 1:
                hm = hm.view(1, -1, 1, 1)
            elif hm.dim() == 2:
                hm = hm.view(hm.shape[0], hm.shape[1], 1, 1)
            elif hm.dim() == 3:
                hm = hm.unsqueeze(-1)
            y = y * hm

        y = y.transpose(1, 2).contiguous().view(B, L, H * d)

        if self.is_nested_encoder:
            out = y
        else:
            out = self.out_proj(y)

        out = out * self.output_scale

        if mask_bool is not None:
            out = out * mask_bool.unsqueeze(-1).to(out.dtype)

        return out, None

    def _chunked_scan(self, q, k, v, f, chunk_size=16):
        """
        Chunked scan for pure HGRN2 recurrence.
        
        FIX 12: Removed denominator Z (Katharopoulos-style normalization).
                Modern RNNs (HGRN2, GLA, Mamba) rely on f-gate for scaling.
                Dividing by Z fights the forget-gate dynamics.
        """
        B, H, L, d = q.shape
        num_chunks = (L + chunk_size - 1) // chunk_size

        # Only S state (no Z)
        S_prev = torch.zeros(B, H, d, d, device=q.device, dtype=q.dtype)
        outputs = []

        for c in range(num_chunks):
            start = c * chunk_size
            end = min(start + chunk_size, L)
            C = end - start

            q_c = q[:, :, start:end]
            k_c = k[:, :, start:end]
            v_c = v[:, :, start:end]
            f_c = f[:, :, start:end]

            log_f = torch.log(f_c.clamp_min(1e-6))
            log_F_c = torch.cumsum(log_f, dim=-1)

            log_F_prev = torch.cat([
                torch.zeros(B, H, 1, device=q.device, dtype=log_F_c.dtype),
                log_F_c,
            ], dim=-1)

            log_decay = log_F_c.unsqueeze(-2) - log_F_prev.unsqueeze(-1)

            i_idx = torch.arange(C + 1, device=q.device).view(C + 1, 1)
            j_idx = torch.arange(C, device=q.device).view(1, C)
            valid = (i_idx == 0) | (i_idx <= j_idx)
            fill_val = -1e4 if log_decay.dtype == torch.float16 else -1e9
            log_decay = log_decay.masked_fill(~valid, fill_val)
            decay = torch.exp(log_decay).to(q.dtype)

            one_minus_f = (1.0 - f_c).to(q.dtype)
            W_kv = one_minus_f[..., None, None] * (k_c.unsqueeze(-1) * v_c.unsqueeze(-2))

            combined_kv = torch.cat([S_prev.unsqueeze(2), W_kv], dim=2).to(q.dtype)

            B_, H_, Nc, _, _ = combined_kv.shape
            combined_kv_flat = combined_kv.view(B_, H_, Nc, -1)
            S_all_flat = torch.matmul(decay, combined_kv_flat)
            S_all = S_all_flat.view(B_, H_, C, d, d)

            # FIX 12: output is num only — no denominator!
            y_c = torch.einsum("bhjd,bhjde->bhje", q_c, S_all)

            outputs.append(y_c)
            S_prev = S_all[:, :, -1]

        return torch.cat(outputs, dim=2).to(q.dtype)

            y_bwd = torch.flip(y_bwd_r, dims=[2])

                    if hasattr(self.state_norm, "weight") and self.state_norm.weight is not None

            if hm.dim() == 1:
                hm = hm.view(1, -1, 1, 1)
            elif hm.dim() == 2:
                hm = hm.view(hm.shape[0], hm.shape[1], 1, 1)
            elif hm.dim() == 3:
                hm = hm.unsqueeze(-1)
            y = y * hm

        y = y.transpose(1, 2).contiguous().view(B, L, H * d)

        if self.is_nested_encoder:
            out = y
        else:
            out = self.out_proj(y)

        out = out * self.output_scale

        if mask_bool is not None:
            out = out * mask_bool.unsqueeze(-1).to(out.dtype)

        return out, None

    def _scan_direction(self, f, iv, chunk_size=32):
        if HAS_TRITON_HGRN2 and hgrn2_scan_triton is not None and f.is_cuda:
            try:
                return hgrn2_scan_triton(f.contiguous(), iv.contiguous(), res_clip=self.res_clip)
            except Exception as e:
                logger.debug("HGRN2 Triton scan failed, falling back: %s", e)

        B, L, D = iv.shape
        f_fp32 = f.float()
        iv_fp32 = iv.float()
        log_f = torch.log(f_fp32.clamp(min=1e-6))

        h_chunks = []
        h_prev = torch.zeros(B, D, device=iv.device, dtype=torch.float32)

        for start in range(0, L, chunk_size):
            end = min(start + chunk_size, L)
            C = end - start
            f_c = log_f[:, start:end]
            iv_c = iv_fp32[:, start:end]
            A = torch.cumsum(f_c, dim=1)
            A_diff = A.unsqueeze(2) - A.unsqueeze(1)
            mask = torch.tril(torch.ones(C, C, device=iv.device, dtype=torch.bool))
            A_diff = torch.where(mask.unsqueeze(0).unsqueeze(-1), A_diff, torch.full_like(A_diff, -1e4))
            decay_weights = torch.exp(A_diff) * mask.unsqueeze(0).unsqueeze(-1)
            intra = torch.sum(decay_weights * iv_c.unsqueeze(1), dim=2)
            inter = h_prev.unsqueeze(1) * torch.exp(A)
            h_c = inter + intra
            if self.res_clip is not None and self.res_clip > 0:
                h_c = torch.clamp(h_c, -self.res_clip, self.res_clip)
            h_chunks.append(h_c)
            h_prev = h_c[:, -1]

        return torch.cat(h_chunks, dim=1).to(iv.dtype)

    def _recurrent_scan_parallel(self, f, iv, mask=None):
        if mask is not None:
            if mask.dim() == 2:
                mask_float = mask.float()
            elif mask.dim() == 4:
                v = (mask > -1e4) if mask.dtype.is_floating_point else mask.bool()
                v = v.any(dim=1)
                mask_float = v.any(dim=1).float() if v.dim() == 3 else v.float()
            else:
                mask_float = mask.float()

            iv = iv * mask_float.unsqueeze(-1)
            f = torch.where(
                mask_float.unsqueeze(-1).bool(),
                f,
                torch.ones_like(f),
            )

        if self.is_decoder:
            return h_fwd, None
        f_rev = torch.flip(f, dims=[1])
        iv_rev = torch.flip(iv, dims=[1])
        h_rev = self._scan_direction(f_rev, iv_rev)
        h_bwd = torch.flip(h_rev, dims=[1])
        return h_fwd, h_bwd

    def _local_attention(self, hidden_states):
        """FIX 8: local sliding-window attention."""
        B, L, D = hidden_states.shape
        H, d = self.num_heads, self.head_dim

        q = self.q_proj(hidden_states).view(B, L, H, d).transpose(1, 2)
        k = self.k_proj(hidden_states).view(B, L, H, d).transpose(1, 2)
        v = self.v_proj_attn(hidden_states).view(B, L, H, d).transpose(1, 2)

        scale = 1.0 / math.sqrt(d)
        scores = torch.matmul(q, k.transpose(-1, -2)) * scale

        idx = torch.arange(L, device=hidden_states.device)
        window_mask = (idx.unsqueeze(1) - idx.unsqueeze(0)).abs() <= self.window_size
        if self.is_decoder:
            causal = idx.unsqueeze(1) >= idx.unsqueeze(0)
            window_mask = window_mask & causal
        scores = scores.masked_fill(~window_mask.unsqueeze(0).unsqueeze(0), -1e4)

        attn = F.softmax(scores, dim=-1)
        out = torch.matmul(attn, v)
        out = out.transpose(1, 2).contiguous().view(B, L, D)
        return out

    def forward(self, hidden_states, attention_mask=None, head_mask=None,
                encoder_hidden_states=None, encoder_attention_mask=None,
                past_key_value=None, output_attentions=False,
                output_hidden_states=False, return_dict=True, *args, **kwargs):
        if not hidden_states.is_contiguous():
            hidden_states = hidden_states.contiguous()

        if encoder_hidden_states is not None:
            raise NotImplementedError("Cross-attention not supported in HGRN2Layer")

        B, L, D = hidden_states.shape
        target_dtype = hidden_states.dtype

        # ---- HGRN-2 branch ----
        raw_log_decay = -F.softplus(self.forget_gate(hidden_states))
        bounded_log_decay = raw_log_decay.clamp(min=-3.0, max=-1e-4)
        f = torch.exp(bounded_log_decay).to(target_dtype)

        i = torch.sigmoid(self.input_gate(hidden_states)).to(target_dtype)
        input_scale = ((1.0 - f) * i).to(target_dtype)
        v = self.v_proj(hidden_states).to(target_dtype)
        if self.lora_rank > 0 and self.v_lora_A is not None and self.v_lora_B is not None:
            v_lora = (torch.matmul(
                torch.matmul(hidden_states, self.v_lora_A.t().contiguous().to(target_dtype)),
                self.v_lora_B.t().contiguous().to(target_dtype)) * self.lora_scaling)
            v = v + v_lora

        z = self.g_proj(hidden_states).to(target_dtype)
        iv = (input_scale * v).to(target_dtype)

        # ---- FIX 12: 4D mask handling ----
        mask_bool = None
        if attention_mask is not None:
            if attention_mask.dim() == 2:
                mask_bool = attention_mask.bool()
            elif attention_mask.dim() == 3:
                m = (attention_mask > -1e4) if attention_mask.dtype.is_floating_point else attention_mask.bool()
                if m.dim() == 3:
                    mask_bool = m.any(dim=1) if m.shape[1] > 1 else m[:, 0, :]
            elif attention_mask.dim() == 4:
                m = (attention_mask > -1e4) if attention_mask.dtype.is_floating_point else attention_mask.bool()
                m = m.any(dim=-2)
                if m.dim() == 3:
                    mask_bool = m.any(dim=1)
                else:
                    mask_bool = m

        # ---- Scan ----
        h_fwd, h_bwd = self._recurrent_scan_parallel(f, iv, mask_bool)
        if self.is_decoder or h_bwd is None:
            h_ctx = h_fwd.to(target_dtype)
        else:
            h_ctx = (h_fwd + h_bwd).to(target_dtype)

        sn_dtype = (self.state_norm.weight.dtype
                    if hasattr(self.state_norm, "weight") and self.state_norm.weight is not None
                    else target_dtype)
        h_norm = self.state_norm(h_ctx.to(sn_dtype)).to(target_dtype)
        hgrn_out = (h_norm * F.silu(z)).to(target_dtype)

        # ---- Local attention branch ----
        local_out = self._local_attention(hidden_states)

        # ---- Fusion ----
        fusion_logits = self.fusion_gate(hidden_states).to(target_dtype)
        gate = torch.sigmoid(fusion_logits)
        fused = gate * hgrn_out + (1 - gate) * local_out

        # ---- Output ----
        out = self.out_proj(fused).to(target_dtype)
        if self.lora_rank > 0 and self.out_lora_A is not None and self.out_lora_B is not None:
            out_lora = (torch.matmul(
                torch.matmul(fused, self.out_lora_A.t().contiguous().to(target_dtype)),
                self.out_lora_B.t().contiguous().to(target_dtype)) * self.lora_scaling)
            out = out + out_lora

        out = out * self.output_scale

        if out.shape[-1] != hidden_states.shape[-1]:
            if out.shape[-1] < hidden_states.shape[-1]:
                out = F.pad(out, (0, hidden_states.shape[-1] - out.shape[-1]))
            else:
                out = out[..., :hidden_states.shape[-1]]

        if attention_mask is not None and mask_bool is not None:
            out = out * mask_bool.unsqueeze(-1).to(out.dtype)

        return out.to(target_dtype), None

                w_k = src.k_lin.weight.detach()
                b_k = src.k_lin.bias.detach() if src.k_lin.bias is not None else None

                    b_o = src.out_lin.bias.detach() if src.out_lin.bias is not None else None

                    adapter = UniversalAdapter()
                    w_q, b_q = adapter.get_weight_bias(src, "q")

    def _hla_scan(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
        decay: torch.Tensor,
        mask: torch.Tensor | None = None,
        reverse: bool = False,
    ) -> torch.Tensor:
        """Vectorized parallel linear associative scan via tensor contraction."""
        B, H, L, d = q.shape
        q_f = torch.nn.functional.elu(q.float()) + 1.0
        k_f = torch.nn.functional.elu(k.float()) + 1.0
        v_f = v.float()

        if mask is not None:
            m_f = mask.float().view(B, 1, L, 1)
            k_f = k_f * m_f
            v_f = v_f * m_f

        t_idx = torch.arange(L, device=q.device)
        diff = (
            (t_idx.unsqueeze(0) - t_idx.unsqueeze(1))
            if reverse
            else (t_idx.unsqueeze(1) - t_idx.unsqueeze(0))
        )
        tri_mask = (
            torch.triu(torch.ones(L, L, device=q.device, dtype=torch.bool))
            if reverse
            else torch.tril(torch.ones(L, L, device=q.device, dtype=torch.bool))
        )

        log_decay = torch.log(
            decay.squeeze(-1).squeeze(-1).clamp(min=1e-5, max=1.0 - 1e-5)
        )  # [1, H]
        exp_arg = torch.where(
            tri_mask.unsqueeze(0).unsqueeze(0),
            diff.unsqueeze(0).unsqueeze(0).float() * log_decay.unsqueeze(-1).unsqueeze(-1),
            float("-inf"),
        )
        decay_mat = torch.exp(exp_arg)  # [1, H, L, L], exactly 0.0 outside causal triangle

        # Attention weights with exponential decay: scores = (q @ k^T) * decay_mat
        qk = torch.matmul(q_f, k_f.transpose(-1, -2).contiguous())  # [B, H, L, L]
        scores = qk * decay_mat

        # Numerator: y_num = scores @ v
        num = torch.matmul(scores, v_f)  # [B, H, L, d]

        # Denominator: y_denom = sum_j scores
        denom = scores.sum(dim=-1, keepdim=True) + 1e-5  # [B, H, L, 1]

        y = num / denom.clamp(min=1e-5)
        return y.to(q.dtype)

    def forward(
        self,
        hidden_states: torch.Tensor,
        attention_mask: torch.Tensor | None = None,
        head_mask: torch.Tensor | None = None,
        encoder_hidden_states: torch.Tensor | None = None,
        encoder_attention_mask: torch.Tensor | None = None,
        past_key_value: Any | None = None,
        output_attentions: bool | None = False,
        output_hidden_states: bool | None = False,
        return_dict: bool | None = True,
        *args: Any,
        **kwargs: Any,
    ) -> tuple[torch.Tensor, torch.Tensor | None] | torch.Tensor:
        """Bidirectional / Causal linear attention forward pass."""
        if not hidden_states.is_contiguous():
            hidden_states = hidden_states.contiguous()

        B, L, D = hidden_states.shape
        H, Dh = self.num_heads, self.head_dim

        q = self.q_proj(hidden_states).view(B, L, H, Dh).transpose(1, 2).contiguous()
        k = (
            self.k_proj(hidden_states)
            .view(B, L, self.num_kv_heads, Dh)
            .transpose(1, 2)
            .contiguous()
        )
        v = (
            self.v_proj(hidden_states)
            .view(B, L, self.num_kv_heads, Dh)
            .transpose(1, 2)
            .contiguous()
        )

        # Partial Rotary Position Embeddings (RoPE)
        pos_emb = kwargs.get("position_embeddings")
        if pos_emb is not None and isinstance(pos_emb, tuple) and len(pos_emb) == 2:
            cos, sin = pos_emb
            q, k = apply_rotary_pos_emb(q, k, cos, sin)

        # Grouped-Query Attention expansion
        if self.num_kv_heads != self.num_heads:
            k = repeat_kv(k, self.num_key_value_groups)
            v = repeat_kv(v, self.num_key_value_groups)

        # ELU + 1 kernelization
        q = F.elu(q) + 1.0
        k = F.elu(k) + 1.0

        decay = torch.sigmoid(self.beta).unsqueeze(0)  # [1, H, 1, 1]

        # Mask processing
        mask_bool = None
        if attention_mask is not None:
            if attention_mask.dim() == 2:
                mask_bool = attention_mask.bool()
            elif attention_mask.dim() == 4:
                mask_bool = (attention_mask[:, 0, 0, :] > -10000.0).bool()

        y_fwd = self._hla_scan(q, k, v, decay, mask=mask_bool, reverse=False)

        if self.is_decoder:
            y = y_fwd
        else:
            y_bwd = self._hla_scan(q, k, v, decay, mask=mask_bool, reverse=True)
            y = 0.5 * (y_fwd + y_bwd)

        if head_mask is not None:
            if head_mask.dim() == 1:
                head_mask = head_mask.view(1, -1, 1, 1)
            elif head_mask.dim() == 2:
                head_mask = head_mask.view(head_mask.size(0), head_mask.size(1), 1, 1)
            y = y * head_mask

        context = y.transpose(1, 2).contiguous().view(B, L, H * Dh)

        if self.is_nested_encoder:
            out = context * self.output_scale
        else:
            out = self.out_proj(context) * self.output_scale

        if attention_mask is not None and mask_bool is not None:
            out = out * mask_bool.unsqueeze(-1).to(out.dtype)

        return (out, None)

    )

        v = repeat_kv(v, self.num_key_value_groups)

            kc = k[:, :, start:end, :].float()
            vc = v[:, :, start:end, :].float()

            y_fwd_list.append(yc)

            decay_to_end = torch.exp(
                ((Lc - 1 - pos).unsqueeze(0).unsqueeze(0).unsqueeze(-1) * log_gamma).clamp(
                    min=-40.0, max=0.0
                )
            )
            k_decayed = kc * decay_to_end
            chunk_S = k_decayed.transpose(-1, -2) @ vc
            chunk_Z = k_decayed.transpose(-1, -2) @ torch.ones_like(vc[..., :1])

            gamma_chunk = torch.exp((Lc * log_gamma).clamp(min=-40.0, max=0.0))
            S = S * gamma_chunk + chunk_S
            Z = Z * gamma_chunk + chunk_Z

        y_fwd = torch.cat(y_fwd_list, dim=2)

        if (not self.is_decoder) and _allow_bidirectional:
            # FIX: Added `.contiguous()` to prevent CuBLAS kernel crash on negative strides
            q_rev = q.flip(dims=[2]).contiguous()
            k_rev = k.flip(dims=[2]).contiguous()
            v_rev = v.flip(dims=[2]).contiguous()
            mask_rev = mask_float.flip(dims=[-1]).contiguous() if mask_float is not None else None
            
            y_bwd_rev, _ = self._kda_scan_chunked(
                q_rev, k_rev, v_rev, decay, mask_rev,
                chunk_size=chunk_size, _allow_bidirectional=False,
            )
            y_bwd = y_bwd_rev.flip(dims=[2]).contiguous()
            return y_fwd, y_bwd

        return y_fwd, None

    # ------------------------------------------------------------------
    # Parallel scan (short sequences)
    # ------------------------------------------------------------------
    def _kda_scan_parallel(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
        decay: torch.Tensor,
        mask: torch.Tensor | None = None,
    ) -> tuple[torch.Tensor, torch.Tensor | None]:
        B, H, L, d = q.shape

        if L > 512:
            return self._kda_scan_chunked(q, k, v, decay, mask, chunk_size=256)

        log_decay = torch.log(decay.clamp(min=1e-4, max=1.0 - 1e-4))
        pos = torch.arange(L, device=q.device, dtype=torch.float32)
        pos_diff = pos.unsqueeze(-1) - pos.unsqueeze(-2)

        tril_mask = torch.tril(torch.ones(L, L, device=q.device, dtype=torch.bool))
        D_causal = (
            torch.exp((pos_diff.unsqueeze(0).unsqueeze(0) * log_decay).clamp(min=-40.0, max=0.0))
            * tril_mask.float()
        )

        attn_scores = q.float() @ k.float().transpose(-1, -2)

        mask_2d = None
        if mask is not None:
            valid_m = get_token_valid_mask(mask, B, L, q.device)
            mask_float = valid_m.float()
            mask_2d = mask_float.unsqueeze(1).unsqueeze(-1) * mask_float.unsqueeze(1).unsqueeze(-2)
            D_causal = D_causal * mask_2d

        num_fwd = (attn_scores * D_causal) @ v.float()
        denom_fwd = (attn_scores * D_causal).sum(dim=-1, keepdim=True)
        y_fwd = torch.clamp(
            num_fwd / denom_fwd.clamp(min=1e-4),
            min=-1e4,
            max=1e4,
        ).to(q.dtype)

        if not self.is_decoder:
            triu_mask = torch.triu(torch.ones(L, L, device=q.device, dtype=torch.bool))
            D_anti = (
                torch.exp(
                    (-pos_diff.unsqueeze(0).unsqueeze(0) * log_decay).clamp(min=-40.0, max=0.0)
                )
                * triu_mask.float()
            )
            if mask_2d is not None:
                D_anti = D_anti * mask_2d
            num_bwd = (attn_scores * D_anti) @ v.float()
            denom_bwd = (attn_scores * D_anti).sum(dim=-1, keepdim=True)
            y_bwd = torch.clamp(
                num_bwd / denom_bwd.clamp(min=1e-4),
                min=-1e4,
                max=1e4,
            ).to(q.dtype)
            return y_fwd, y_bwd

        return y_fwd, None

    # ------------------------------------------------------------------
    # Forward
    # ------------------------------------------------------------------
    def forward(
        self,
        hidden_states: torch.Tensor,
        attention_mask: torch.Tensor | None = None,
        head_mask: torch.Tensor | None = None,
        encoder_hidden_states: torch.Tensor | None = None,
        encoder_attention_mask: torch.Tensor | None = None,
        past_key_value: Any | None = None,
        output_attentions: bool | None = False,
        output_hidden_states: bool | None = False,
        return_dict: bool | None = True,
        position_embeddings: tuple[torch.Tensor, torch.Tensor] | None = None,
        *args: Any,
        **kwargs: Any,
    ) -> tuple[torch.Tensor, torch.Tensor | None]:
        if not hidden_states.is_contiguous():
            hidden_states = hidden_states.contiguous()

        B, L, D = hidden_states.shape
        H, d = self.num_heads, self.head_dim

        q_raw = self.q_proj(hidden_states).view(B, L, H, d).transpose(1, 2).contiguous()
        k_raw = (
            self.k_proj(hidden_states).view(B, L, self.num_kv_heads, d).transpose(1, 2).contiguous()
        )
        v = self.v_proj(hidden_states).view(B, L, self.num_kv_heads, d).transpose(1, 2).contiguous()

        # RoPE
        if position_embeddings is not None:
            cos, sin = position_embeddings
            q_raw, k_raw = apply_rotary_pos_emb(q_raw, k_raw, cos, sin)
        elif self.rotary_emb is not None:
            rotary_module = self.rotary_emb
            pos_ids = kwargs.get("position_ids")
            if pos_ids is None:
                pos_ids = torch.arange(L, device=hidden_states.device).unsqueeze(0).expand(B, -1)
            try:
                cos, sin = rotary_module(q_raw, pos_ids)
                q_raw, k_raw = apply_rotary_pos_emb(q_raw, k_raw, cos, sin, position_ids=pos_ids)
            except Exception:
                try:
                    cos, sin = rotary_module(q_raw, seq_len=L)
                    q_raw, k_raw = apply_rotary_pos_emb(
                        q_raw, k_raw, cos, sin, position_ids=pos_ids
                    )
                except Exception as e:
                    logger.debug("KDA rotary_emb failed: %s", e)

        if self.kv_repeat > 1:
            k_raw = repeat_kv(k_raw, self.kv_repeat)
            v = repeat_kv(v, self.kv_repeat)

        q = F.elu(q_raw) + 1.0
        k = (F.elu(k_raw) + 1.0) / (d**0.5)

        decay = torch.sigmoid(self.lora_beta).view(1, H, 1, 1)

        mask_bool = None
        if attention_mask is not None:
            mask_bool = get_token_valid_mask(attention_mask, B, L, hidden_states.device)

        y_fwd, y_bwd = self._kda_scan_parallel(q, k, v, decay, mask_bool)

        if self.is_decoder or y_bwd is None:
            y = y_fwd
        else:
            y = 0.5 * (y_fwd + y_bwd)

        y = torch.nan_to_num(y, nan=0.0)

        if head_mask is not None:
            if head_mask.dim() == 1:
                hm = head_mask.view(1, -1, 1, 1)
            elif head_mask.dim() == 2:
                hm = head_mask.view(head_mask.size(0), -1, 1, 1)
            else:
                hm = head_mask
            y = y * hm

        out = y.transpose(1, 2).contiguous().view(B, L, self.all_head_size)

        out = out * self.lora_output_scale
        if self.is_nested_encoder:
            final_out = out
        else:
            final_out = self.out_proj(out)

        return final_out, None

class LoRALinear(nn.Module):
    def __init__(
        self,
        in_features: int,
        out_features: int,
        bias: bool = True,
        lora_rank: int = 16,
        lora_alpha: float = 32.0,
        freeze_base: bool = False,
        device: torch.device | None = None,
        dtype: torch.dtype | None = None,
    ) -> None:
        super().__init__()
        factory_kwargs = {"device": device, "dtype": dtype}
        self.in_features = in_features
        self.out_features = out_features
        self.lora_rank = lora_rank
        self.lora_alpha = lora_alpha
        self.lora_scaling = (lora_alpha / lora_rank) if lora_rank > 0 else 1.0
        self.freeze_base = freeze_base

        self.weight = nn.Parameter(torch.empty((out_features, in_features), **factory_kwargs))
        if bias:
            self.bias = nn.Parameter(torch.empty(out_features, **factory_kwargs))
        else:
            self.register_parameter("bias", None)

        if self.lora_rank > 0:
            self.lora_A = nn.Parameter(torch.empty((lora_rank, in_features), **factory_kwargs))
            self.lora_B = nn.Parameter(torch.empty((out_features, lora_rank), **factory_kwargs))
        else:
            self.register_parameter("lora_A", None)
            self.register_parameter("lora_B", None)

        self.reset_parameters()
        if freeze_base:
            self.freeze_base_weights()

    def reset_parameters(self) -> None:
        nn.init.kaiming_uniform_(self.weight, a=math.sqrt(5))
        if self.bias is not None:
            fan_in, _ = nn.init._calculate_fan_in_and_fan_out(self.weight)
            bound = 1 / math.sqrt(fan_in) if fan_in > 0 else 0
            nn.init.uniform_(self.bias, -bound, bound)

        if self.lora_rank > 0 and self.lora_A is not None:
            nn.init.kaiming_uniform_(self.lora_A, a=math.sqrt(5))
            self.lora_A.requires_grad = True
        if self.lora_rank > 0 and self.lora_B is not None:
            nn.init.zeros_(self.lora_B)
            self.lora_B.requires_grad = True

    def freeze_base_weights(self) -> None:
        self.weight.requires_grad = False
        if self.bias is not None:
            self.bias.requires_grad = False
        if self.lora_rank > 0 and self.lora_A is not None:
            self.lora_A.requires_grad = True
        if self.lora_rank > 0 and self.lora_B is not None:
            self.lora_B.requires_grad = True

    def unfreeze_base_weights(self) -> None:
        self.weight.requires_grad = True
        if self.bias is not None:
            self.bias.requires_grad = True

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        out = F.linear(x, self.weight, self.bias)
        if self.lora_rank > 0 and self.lora_A is not None and self.lora_B is not None:
            lora_out = F.linear(F.linear(x, self.lora_A), self.lora_B) * self.lora_scaling
            out = out + lora_out
        return out

Modules can also contain other Modules, allowing them to be nested in a tree structure. You can assign the submodules as regular attributes::

import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    def __init__(self) -> None:
        super().__init__()
        self.conv1 = nn.Conv2d(1, 20, 5)
        self.conv2 = nn.Conv2d(20, 20, 5)

    def forward(self, x):
        x = F.relu(self.conv1(x))
        return F.relu(self.conv2(x))

class UniversalAdapter:
    """Universal adapter supporting BERT, DistilBERT, RoBERTa, LLaMA, GPT2, DeBERTa, etc."""

    @staticmethod
    def get_weight_bias(old_attn: Any, proj: str) -> tuple:
        """Extract (weight, bias) for projection 'q', 'k', 'v', or 'o'.

        Handles:
        - RoBERTa/BERT: query/key/value attributes on .self sub-module
        - LLaMA/Qwen: q_proj/k_proj/v_proj on attention module directly
        - GPT-2: c_attn (fused, already handled by fused-QKV path); c_proj
        - DistilBERT: q_lin/k_lin/v_lin/out_lin
        - LoRALinear: unwraps .base_layer to get the frozen base weight
        - Conv1D: transposes weight from [in, out] to [out, in]
        """
        target = getattr(old_attn, "self", old_attn)
        candidate_names = CANONICAL_PROJECTION_ALIASES.get(proj, [])

        def _unwrap(mod: Any) -> Any:
            """Unwrap LoRALinear / merged adapter layers to their base_layer."""
            if mod is None:
                return mod
            # LoRALinear stores the original frozen layer as base_layer
            if hasattr(mod, "base_layer") and hasattr(mod.base_layer, "weight"):
                return mod.base_layer
            return mod

        def _extract(mod: Any):
            """Return (weight_data, bias) from a module, or None if not a valid weight-holder."""
            mod = _unwrap(mod)
            if mod is None:
                return None
            if not hasattr(mod, "weight") or mod.weight is None:
                return None
            import torch
            import torch.nn as nn

            w = mod.weight
            # Must be either an nn.Parameter or a plain Tensor (Conv1D)
            if not (isinstance(w, nn.Parameter) or isinstance(w, torch.Tensor)):
                return None
            w = w.data if hasattr(w, "data") else w
            # Projections must strictly be 2D weight matrices (excludes 1D norms, 3D convolutions, etc.)
            if w.ndim != 2:
                return None
            # Conv1D stores weights transposed: [in_features, out_features]
            is_conv1d = (mod.__class__.__name__ == "Conv1D") or hasattr(mod, "nf")
            if is_conv1d or (w.shape[0] < w.shape[1]):
                w = w.t().contiguous()
            bias = getattr(mod, "bias", None)
            bias = bias.data if (bias is not None and hasattr(bias, "data")) else bias
            # Qwen3.5 Gated Q: query is fused with gate (dim 0 has 2 * num_heads * head_dim)
            if proj == "q" and (
                hasattr(old_attn, "q_norm")
                or old_attn.__class__.__name__ == "Qwen3_5Attention"
                or hasattr(old_attn, "k_norm")
            ):
                cfg = getattr(old_attn, "config", None)
                tc = getattr(cfg, "text_config", cfg)
                h_dim = getattr(tc, "head_dim", 256)
                n_heads = getattr(tc, "num_attention_heads", getattr(old_attn, "num_heads", 8))
                if w.shape[0] == n_heads * h_dim * 2:
                    w = (
                        w.view(n_heads, 2, h_dim, -1)[:, 0, :, :]
                        .reshape(n_heads * h_dim, -1)
                        .contiguous()
                    )
                    if bias is not None:
                        bias = (
                            bias.view(n_heads, 2, h_dim)[:, 0, :]
                            .reshape(n_heads * h_dim)
                            .contiguous()
                        )
            return w, bias

        # 1. Search in target (old_attn.self or old_attn)
        for name in candidate_names:
            if hasattr(target, name):
                result = _extract(getattr(target, name))
                if result is not None:
                    return result

        # 2. Search in old_attn.output (BERT / RoBERTa output projection 'o')
        if proj == "o" and hasattr(old_attn, "output"):
            out_mod = old_attn.output
            if hasattr(out_mod, "dense"):
                result = _extract(out_mod.dense)
                if result is not None:
                    return result
            result = _extract(out_mod)
            if result is not None:
                return result

        # 3. Search directly in old_attn (catches cases where there's no .self sub-module)
        for name in candidate_names:
            if hasattr(old_attn, name):
                result = _extract(getattr(old_attn, name))
                if result is not None:
                    return result

        # 4. Fallback for Fused GPT-2 c_attn (must precede positional fallback)
        if proj in ("q", "k", "v") and hasattr(old_attn, "c_attn"):
            fused_result = _extract(old_attn.c_attn)
            if fused_result is not None:
                w, b = fused_result
                split_w = w.size(0) // 3
                split_b = b.size(0) // 3 if b is not None else 0
                if proj == "q":
                    return w[:split_w, :], (b[:split_b] if b is not None else None)
                elif proj == "k":
                    return w[split_w : 2 * split_w, :], (
                        b[split_b : 2 * split_b] if b is not None else None
                    )
                elif proj == "v":
                    return w[2 * split_w :, :], (b[2 * split_b :] if b is not None else None)

        # 4b. Fallback for Qwen3.5 / fused in_proj_qkv
        if proj in ("q", "k", "v") and hasattr(old_attn, "in_proj_qkv"):
            fused_result = _extract(old_attn.in_proj_qkv)
            if fused_result is not None:
                w, b = fused_result
                k_dim = getattr(old_attn, "key_dim", getattr(old_attn, "k_dim", w.size(0) // 3))
                v_dim = getattr(old_attn, "value_dim", getattr(old_attn, "v_dim", w.size(0) // 3))
                q_dim = w.size(0) - k_dim - v_dim if (k_dim + v_dim < w.size(0)) else k_dim
                if proj == "q":
                    return w[:q_dim, :], (b[:q_dim] if b is not None else None)
                elif proj == "k":
                    return w[q_dim : q_dim + k_dim, :], (
                        b[q_dim : q_dim + k_dim] if b is not None else None
                    )
                elif proj == "v":
                    return w[q_dim + k_dim : q_dim + k_dim + v_dim, :], (
                        b[q_dim + k_dim : q_dim + k_dim + v_dim] if b is not None else None
                    )

        # 4c. PyTorch nn.MultiheadAttention in_proj_weight / separate proj_weight
        for obj in (target, old_attn):
            if (
                proj in ("q", "k", "v")
                and hasattr(obj, "in_proj_weight")
                and obj.in_proj_weight is not None
            ):
                w = (
                    obj.in_proj_weight.data
                    if hasattr(obj.in_proj_weight, "data")
                    else obj.in_proj_weight
                )
                b = getattr(obj, "in_proj_bias", None)
                b = b.data if (b is not None and hasattr(b, "data")) else b
                split_w = w.size(0) // 3
                split_b = b.size(0) // 3 if b is not None else 0
                if proj == "q":
                    return w[:split_w, :].clone(), (b[:split_b].clone() if b is not None else None)
                elif proj == "k":
                    return w[split_w : 2 * split_w, :].clone(), (
                        b[split_b : 2 * split_b].clone() if b is not None else None
                    )
                elif proj == "v":
                    return w[2 * split_w :, :].clone(), (
                        b[2 * split_b :].clone() if b is not None else None
                    )

            if proj in ("q", "k", "v"):
                attr_w = f"{proj}_proj_weight"
                if hasattr(obj, attr_w) and getattr(obj, attr_w) is not None:
                    w = getattr(obj, attr_w)
                    w = w.data if hasattr(w, "data") else w
                    b = getattr(obj, f"{proj}_proj_bias", None)
                    b = b.data if (b is not None and hasattr(b, "data")) else b
                    return w.clone(), (b.clone() if b is not None else None)

        # 5. Positional fallback: scan all Linear children of target in declaration order.
        #    Only valid if at least 3 children exist (the typical pattern for BERT/RoBERTa variants).
        ordered_proj_map = {"q": 0, "k": 1, "v": 2}
        if proj in ordered_proj_map:
            children_with_weight = [
                (n, m) for n, m in target.named_children() if _extract(m) is not None
            ]
            if len(children_with_weight) >= 3:
                idx = ordered_proj_map[proj]
                if idx < len(children_with_weight):
                    return _extract(children_with_weight[idx][1])

        # 6. Fallback for 'o' projection when replacing leaf self-attention without local out_proj
        # (e.g. RoBERTa/BERT RobertaSelfAttention where output.dense resides in the parent container).
        # Returning an Identity matrix ensures:
        #   (a) custom modules don't crash during weight copying, and
        #   (b) context passes through transparently to parent.output.dense without double projection!
        if proj == "o":
            for mod in target.modules():
                if (
                    hasattr(mod, "weight")
                    and isinstance(getattr(mod, "weight", None), torch.Tensor)
                    and mod.weight.ndim == 2
                ):
                    d = mod.weight.shape[0]
                    eye_w = torch.eye(d, dtype=mod.weight.dtype, device=mod.weight.device)
                    zero_b = torch.zeros(d, dtype=mod.weight.dtype, device=mod.weight.device)
                    return eye_w, zero_b

        raise ValueError(
            f"Could not extract '{proj}' projection weights from {type(old_attn).__name__}"
        )

@staticmethod
def get_weight_bias(old_attn: Any, proj: str) -> tuple:
    """Extract (weight, bias) for projection 'q', 'k', 'v', or 'o'.

    Handles:
    - RoBERTa/BERT: query/key/value attributes on .self sub-module
    - LLaMA/Qwen: q_proj/k_proj/v_proj on attention module directly
    - GPT-2: c_attn (fused, already handled by fused-QKV path); c_proj
    - DistilBERT: q_lin/k_lin/v_lin/out_lin
    - LoRALinear: unwraps .base_layer to get the frozen base weight
    - Conv1D: transposes weight from [in, out] to [out, in]
    """
    target = getattr(old_attn, "self", old_attn)
    candidate_names = CANONICAL_PROJECTION_ALIASES.get(proj, [])

    def _unwrap(mod: Any) -> Any:
        """Unwrap LoRALinear / merged adapter layers to their base_layer."""
        if mod is None:
            return mod
        # LoRALinear stores the original frozen layer as base_layer
        if hasattr(mod, "base_layer") and hasattr(mod.base_layer, "weight"):
            return mod.base_layer
        return mod

    def _extract(mod: Any):
        """Return (weight_data, bias) from a module, or None if not a valid weight-holder."""
        mod = _unwrap(mod)
        if mod is None:
            return None
        if not hasattr(mod, "weight") or mod.weight is None:
            return None
        import torch
        import torch.nn as nn

        w = mod.weight
        # Must be either an nn.Parameter or a plain Tensor (Conv1D)
        if not (isinstance(w, nn.Parameter) or isinstance(w, torch.Tensor)):
            return None
        w = w.data if hasattr(w, "data") else w
        # Projections must strictly be 2D weight matrices (excludes 1D norms, 3D convolutions, etc.)
        if w.ndim != 2:
            return None
        # Conv1D stores weights transposed: [in_features, out_features]
        is_conv1d = (mod.__class__.__name__ == "Conv1D") or hasattr(mod, "nf")
        if is_conv1d or (w.shape[0] < w.shape[1]):
            w = w.t().contiguous()
        bias = getattr(mod, "bias", None)
        bias = bias.data if (bias is not None and hasattr(bias, "data")) else bias
        # Qwen3.5 Gated Q: query is fused with gate (dim 0 has 2 * num_heads * head_dim)
        if proj == "q" and (
            hasattr(old_attn, "q_norm")
            or old_attn.__class__.__name__ == "Qwen3_5Attention"
            or hasattr(old_attn, "k_norm")
        ):
            cfg = getattr(old_attn, "config", None)
            tc = getattr(cfg, "text_config", cfg)
            h_dim = getattr(tc, "head_dim", 256)
            n_heads = getattr(tc, "num_attention_heads", getattr(old_attn, "num_heads", 8))
            if w.shape[0] == n_heads * h_dim * 2:
                w = (
                    w.view(n_heads, 2, h_dim, -1)[:, 0, :, :]
                    .reshape(n_heads * h_dim, -1)
                    .contiguous()
                )
                if bias is not None:
                    bias = (
                        bias.view(n_heads, 2, h_dim)[:, 0, :]
                        .reshape(n_heads * h_dim)
                        .contiguous()
                    )
        return w, bias

    # 1. Search in target (old_attn.self or old_attn)
    for name in candidate_names:
        if hasattr(target, name):
            result = _extract(getattr(target, name))
            if result is not None:
                return result

    # 2. Search in old_attn.output (BERT / RoBERTa output projection 'o')
    if proj == "o" and hasattr(old_attn, "output"):
        out_mod = old_attn.output
        if hasattr(out_mod, "dense"):
            result = _extract(out_mod.dense)
            if result is not None:
                return result
        result = _extract(out_mod)
        if result is not None:
            return result

    # 3. Search directly in old_attn (catches cases where there's no .self sub-module)
    for name in candidate_names:
        if hasattr(old_attn, name):
            result = _extract(getattr(old_attn, name))
            if result is not None:
                return result

    # 4. Fallback for Fused GPT-2 c_attn (must precede positional fallback)
    if proj in ("q", "k", "v") and hasattr(old_attn, "c_attn"):
        fused_result = _extract(old_attn.c_attn)
        if fused_result is not None:
            w, b = fused_result
            split_w = w.size(0) // 3
            split_b = b.size(0) // 3 if b is not None else 0
            if proj == "q":
                return w[:split_w, :], (b[:split_b] if b is not None else None)
            elif proj == "k":
                return w[split_w : 2 * split_w, :], (
                    b[split_b : 2 * split_b] if b is not None else None
                )
            elif proj == "v":
                return w[2 * split_w :, :], (b[2 * split_b :] if b is not None else None)

    # 4b. Fallback for Qwen3.5 / fused in_proj_qkv
    if proj in ("q", "k", "v") and hasattr(old_attn, "in_proj_qkv"):
        fused_result = _extract(old_attn.in_proj_qkv)
        if fused_result is not None:
            w, b = fused_result
            k_dim = getattr(old_attn, "key_dim", getattr(old_attn, "k_dim", w.size(0) // 3))
            v_dim = getattr(old_attn, "value_dim", getattr(old_attn, "v_dim", w.size(0) // 3))
            q_dim = w.size(0) - k_dim - v_dim if (k_dim + v_dim < w.size(0)) else k_dim
            if proj == "q":
                return w[:q_dim, :], (b[:q_dim] if b is not None else None)
            elif proj == "k":
                return w[q_dim : q_dim + k_dim, :], (
                    b[q_dim : q_dim + k_dim] if b is not None else None
                )
            elif proj == "v":
                return w[q_dim + k_dim : q_dim + k_dim + v_dim, :], (
                    b[q_dim + k_dim : q_dim + k_dim + v_dim] if b is not None else None
                )

    # 4c. PyTorch nn.MultiheadAttention in_proj_weight / separate proj_weight
    for obj in (target, old_attn):
        if (
            proj in ("q", "k", "v")
            and hasattr(obj, "in_proj_weight")
            and obj.in_proj_weight is not None
        ):
            w = (
                obj.in_proj_weight.data
                if hasattr(obj.in_proj_weight, "data")
                else obj.in_proj_weight
            )
            b = getattr(obj, "in_proj_bias", None)
            b = b.data if (b is not None and hasattr(b, "data")) else b
            split_w = w.size(0) // 3
            split_b = b.size(0) // 3 if b is not None else 0
            if proj == "q":
                return w[:split_w, :].clone(), (b[:split_b].clone() if b is not None else None)
            elif proj == "k":
                return w[split_w : 2 * split_w, :].clone(), (
                    b[split_b : 2 * split_b].clone() if b is not None else None
                )
            elif proj == "v":
                return w[2 * split_w :, :].clone(), (
                    b[2 * split_b :].clone() if b is not None else None
                )

        if proj in ("q", "k", "v"):
            attr_w = f"{proj}_proj_weight"
            if hasattr(obj, attr_w) and getattr(obj, attr_w) is not None:
                w = getattr(obj, attr_w)
                w = w.data if hasattr(w, "data") else w
                b = getattr(obj, f"{proj}_proj_bias", None)
                b = b.data if (b is not None and hasattr(b, "data")) else b
                return w.clone(), (b.clone() if b is not None else None)

    # 5. Positional fallback: scan all Linear children of target in declaration order.
    #    Only valid if at least 3 children exist (the typical pattern for BERT/RoBERTa variants).
    ordered_proj_map = {"q": 0, "k": 1, "v": 2}
    if proj in ordered_proj_map:
        children_with_weight = [
            (n, m) for n, m in target.named_children() if _extract(m) is not None
        ]
        if len(children_with_weight) >= 3:
            idx = ordered_proj_map[proj]
            if idx < len(children_with_weight):
                return _extract(children_with_weight[idx][1])

    # 6. Fallback for 'o' projection when replacing leaf self-attention without local out_proj
    # (e.g. RoBERTa/BERT RobertaSelfAttention where output.dense resides in the parent container).
    # Returning an Identity matrix ensures:
    #   (a) custom modules don't crash during weight copying, and
    #   (b) context passes through transparently to parent.output.dense without double projection!
    if proj == "o":
        for mod in target.modules():
            if (
                hasattr(mod, "weight")
                and isinstance(getattr(mod, "weight", None), torch.Tensor)
                and mod.weight.ndim == 2
            ):
                d = mod.weight.shape[0]
                eye_w = torch.eye(d, dtype=mod.weight.dtype, device=mod.weight.device)
                zero_b = torch.zeros(d, dtype=mod.weight.dtype, device=mod.weight.device)
                return eye_w, zero_b

    raise ValueError(
        f"Could not extract '{proj}' projection weights from {type(old_attn).__name__}"
    )
