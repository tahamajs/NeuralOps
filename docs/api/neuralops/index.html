def create_dataloader(dataset, tokenizer, config: DataConfig, **kwargs) -> DataLoader:
    """
    Create a DataLoader from a HuggingFace Dataset, applying preprocessing if not already tokenized.
    Uses ProductionDataCollator to guarantee rock-solid batch dimensions and GPU Tensor Core alignment.
    """
    if "input_ids" in dataset.column_names and "attention_mask" in dataset.column_names:
        processed = dataset
    else:
        processed = dataset.map(
            lambda examples: preprocess_function(examples, tokenizer, config),
            batched=True,
            remove_columns=dataset.column_names,
        )
    columns = [
        col
        for col in [
            "input_ids",
            "attention_mask",
            "labels",
            "start_positions",
            "end_positions",
            "token_type_ids",
        ]
        if col in processed.column_names
    ]
    processed.set_format(type="torch", columns=columns)

    collate_fn = kwargs.pop("collate_fn", None)
    if collate_fn is None:
        collate_fn = ProductionDataCollator(
            pad_token_id=(getattr(tokenizer, "pad_token_id", 0) if tokenizer is not None else 0),
            pad_to_multiple_of=8,
            max_length=config.max_length,
        )

    return DataLoader(
        processed,
        batch_size=config.batch_size,
        shuffle=config.shuffle,
        num_workers=config.num_workers,
        collate_fn=collate_fn,
        **kwargs,
    )

def detect_architecture(model: nn.Module) -> str:
    """Return canonical architecture name."""
    config = getattr(model, "config", None)
    model_type = getattr(config, "model_type", "") if config else ""
    name = model.__class__.__name__.lower()
    if "distilbert" in model_type or "distilbert" in name:
        return "distilbert"
    if "roberta" in model_type or "roberta" in name:
        return "roberta"
    if "bert" in model_type or "bert" in name:
        return "bert"
    if "gpt2" in model_type or "gpt2" in name:
        return "gpt2"
    if "qwen" in model_type or "qwen" in name:
        return "qwen"
    if "llama" in model_type:
        return "llama"
    if "mistral" in model_type:
        return "mistral"
    if "gemma" in model_type:
        return "gemma"
    if "phi" in model_type:
        return "phi"
    if "t5" in model_type:
        return "t5"
    return "unknown"

def get_calibration_dataloader(
    task_name: str = "sst2",
    tokenizer: Any = None,
    batch_size: int = 8,
    num_samples: int = 128,
    seq_len: int = 128,
    split: str = "train",
) -> Any:
    """
    Builds a calibration DataLoader from local disk-cached GLUE data.
    Used by data-aware SVD compression algorithms (ASVD, A3, FRanDI, DynamicRank,
    SwiftSVD, SliceGPT, etc.) to collect activation covariance and Fisher information.
    """
    from torch.utils.data import DataLoader

    t_name = task_name.lower().replace("-", "_")
    ds_dict = load_glue_local(t_name)
    if split not in ds_dict:
        split = list(ds_dict.keys())[0]
    ds = ds_dict[split]

    if num_samples and len(ds) > num_samples:
        ds = ds.select(range(min(num_samples, len(ds))))

    if tokenizer is not None:
        key1, key2 = GLUE_TASK_KEYS.get(t_name, ("sentence", None))
        if key1 not in ds.column_names:
            cols = [c for c in ds.column_names if c not in ("label", "labels", "idx")]
            key1 = cols[0] if cols else "text"
            key2 = None

        if tokenizer.pad_token is None:
            tokenizer.pad_token = tokenizer.eos_token or "[PAD]"

        def tokenize_fn(examples):
            args = (
                (examples[key1],)
                if key2 is None or key2 not in examples
                else (examples[key1], examples[key2])
            )
            return tokenizer(*args, padding="max_length", truncation=True, max_length=seq_len)

        remove_cols = [c for c in ds.column_names if c not in ("label", "labels")]
        ds = ds.map(tokenize_fn, batched=True, remove_columns=remove_cols)
        keep = [
            c for c in ["input_ids", "attention_mask", "token_type_ids"] if c in ds.column_names
        ]
        ds.set_format(type="torch", columns=keep)

    return DataLoader(ds, batch_size=batch_size, shuffle=False)

def inject_attention(
    model: nn.Module,
    attention_type: str,
    rank: int | None = None,
    copy_weights: bool = True,
    **kwargs: Any,
) -> nn.Module:
    """Universal attention layer replacement engine for HuggingFace Transformers."""
    attn_class = ATTENTION_REGISTRY.resolve(attention_type)
    if callable(attn_class) and not isinstance(attn_class, type):
        try:
            res = attn_class(model=model, rank=rank, copy_weights=copy_weights, **kwargs)
            if res is not None:
                return res
        except Exception as inj_err:
            logger.debug(
                "Dedicated injector %s failed, falling back to layer replacement: %s",
                attention_type,
                inj_err,
            )

    config = getattr(model, "config", None)
    if config is not None:
        cfg_source = getattr(config, "text_config", config)
        hidden_size = getattr(
            cfg_source,
            "hidden_size",
            getattr(
                cfg_source,
                "d_model",
                getattr(cfg_source, "n_embd", getattr(cfg_source, "dim", None)),
            ),
        )
        num_heads = getattr(
            cfg_source,
            "num_attention_heads",
            getattr(
                cfg_source,
                "num_heads",
                getattr(cfg_source, "n_heads", getattr(cfg_source, "n_head", None)),
            ),
        )
        head_dim = getattr(
            cfg_source,
            "head_dim",
            getattr(
                cfg_source,
                "d_kv",
                (hidden_size // max(1, num_heads) if (hidden_size and num_heads) else None),
            ),
        )
        num_kv_heads = getattr(cfg_source, "num_key_value_heads", num_heads)
        if hidden_size is not None:
            config.hidden_size = hidden_size
        if num_heads is not None:
            config.num_attention_heads = num_heads
        if head_dim is not None:
            config.head_dim = head_dim
        if num_kv_heads is not None:
            config.num_key_value_heads = num_kv_heads
        if not hasattr(config, "dropout"):
            dropout_val = getattr(
                config,
                "hidden_dropout_prob",
                getattr(config, "attention_probs_dropout_prob", 0.1),
            )
            config.dropout = dropout_val
    else:
        hidden_size = getattr(model, "hidden_size", None)
        num_heads = getattr(model, "num_attention_heads", None)
        head_dim = getattr(model, "head_dim", None)
        num_kv_heads = getattr(model, "num_key_value_heads", num_heads)

    arch = detect_architecture(model)
    adapter = get_adapter(arch)
    family = detect_family(model)
    family_adapter = ModelAdapter(model, family)

    is_dec = family in ("gpt", "gpt2", "qwen", "qwen3_5", "llama", "mistral")
    if config is not None:
        config.is_decoder = bool(is_dec or getattr(config, "is_decoder", False))
        if not hasattr(config, "model_type") or not config.model_type:
            config.model_type = family

    replaced = 0
    modules_list = list(get_attention_modules(model))
    if not modules_list:
        logger.warning("No attention modules found to replace for architecture '%s'", arch)

    for name, old_attn in modules_list:
        # In hybrid architectures (like Qwen3.5), only replace linear_attn if candidate explicitly supports DeltaNet
        if "linear_attn" in name and not getattr(attn_class, "supports_delta_net", False):
            continue

        # Dynamically infer dimensions from family_adapter and old_attn
        if family_adapter is not None:
            if hidden_size is None:
                hidden_size = getattr(family_adapter, "hidden_size", None)
            if num_heads is None:
                num_heads = getattr(family_adapter, "num_heads", None)
            if num_kv_heads is None:
                num_kv_heads = getattr(family_adapter, "num_key_value_heads", None)
            if head_dim is None:
                head_dim = getattr(family_adapter, "head_dim", None)

        if hidden_size is None or num_heads is None:
            for _, child in old_attn.named_children():
                if (
                    hasattr(child, "weight")
                    and isinstance(child.weight, torch.Tensor)
                    and child.weight.ndim == 2
                ):
                    if hidden_size is None:
                        hidden_size = child.weight.shape[1]
                    break
            if num_heads is None:
                num_heads = getattr(
                    old_attn,
                    "num_heads",
                    getattr(old_attn, "num_attention_heads", None),
                )
            if num_kv_heads is None:
                num_kv_heads = getattr(old_attn, "num_key_value_heads", num_heads)
            if head_dim is None and hidden_size is not None and num_heads is not None:
                head_dim = getattr(old_attn, "head_dim", hidden_size // max(1, num_heads))

        # Ensure config exists or construct minimal namespace config
        if config is None:
            from types import SimpleNamespace

            is_decoder_family_bool = bool(
                is_dec or family in ("gpt", "gpt2", "qwen", "qwen3_5", "llama", "mistral")
            )
            config = SimpleNamespace(
                hidden_size=hidden_size,
                num_attention_heads=num_heads,
                head_dim=head_dim,
                num_key_value_heads=num_kv_heads or num_heads,
                dropout=0.1,
                is_decoder=is_decoder_family_bool,
                is_causal=is_decoder_family_bool,
                model_type=family,
            )
        else:
            if family in ("gpt", "gpt2", "qwen", "qwen3_5", "llama", "mistral") or is_dec:
                config.is_decoder = True
                config.is_causal = True

        # Ensure old_attn has canonical projection attributes for custom modules inspecting pretrained_attn
        for proj_key, alias_list in [
            ("q", ["query", "q_proj", "q_lin", "Wq", "wq", "q"]),
            ("k", ["key", "k_proj", "k_lin", "Wk", "wk", "k"]),
            ("v", ["value", "v_proj", "v_lin", "Wv", "wv", "v"]),
            ("o", ["dense", "out_proj", "o_proj", "out_lin", "c_proj", "wo", "o"]),
        ]:
            try:
                existing_mod = next(
                    (getattr(old_attn, a) for a in alias_list if hasattr(old_attn, a)),
                    None,
                )
                if existing_mod is None and hasattr(old_attn, "self"):
                    existing_mod = next(
                        (
                            getattr(old_attn.self, a)
                            for a in alias_list
                            if hasattr(old_attn.self, a)
                        ),
                        None,
                    )

                if existing_mod is not None:
                    for a in alias_list:
                        if not hasattr(old_attn, a):
                            try:
                                setattr(old_attn, a, existing_mod)
                            except Exception:
                                pass
                else:
                    w, b = adapter.get_weight_bias(old_attn, proj_key)
                    if w is not None:
                        proxy = nn.Linear(
                            w.shape[1],
                            w.shape[0],
                            bias=(b is not None),
                            device=w.device,
                            dtype=w.dtype,
                        )
                        proxy.weight.data.copy_(w)
                        if b is not None:
                            proxy.bias.data.copy_(b)
                        for a in alias_list:
                            if not hasattr(old_attn, a):
                                try:
                                    setattr(old_attn, a, proxy)
                                except Exception:
                                    pass
            except Exception as ex:
                logger.debug(
                    "Failed to set projection aliases for %s on %s: %s",
                    proj_key,
                    name,
                    ex,
                )

        effective_rank = rank if rank is not None else getattr(config, "rank", 32)

        # Standardized instantiation: try passing pretrained_attn=old_attn first so modules
        # can inherit pretrained weights during initialization, then fallback without it
        attempts = [
            lambda: attn_class(config, rank=effective_rank, pretrained_attn=old_attn, **kwargs),
            lambda: attn_class(config, pretrained_attn=old_attn, **kwargs),
            lambda: attn_class(config=config, pretrained_attn=old_attn, **kwargs),
            lambda: attn_class(config, rank=effective_rank, **kwargs),
            lambda: attn_class(config, **kwargs),
            lambda: attn_class(rank=effective_rank, **kwargs),
            lambda: attn_class(**kwargs),
        ]
        new_attn = None
        for attempt in attempts:
            try:
                new_attn = attempt()
                break
            except TypeError:
                continue
            except Exception as e:
                logger.debug("Instantiation attempt failed: %s", e)
                continue

        if new_attn is None:
            raise RuntimeError(f"Could not instantiate attention class {attn_class.__name__}")

        # Do not let a legacy full-MHA replacement silently retain randomly
        # initialized K/V projections on GQA decoders.  Such a module must
        # explicitly advertise the geometry it implements.
        is_gqa = (
            family in ("qwen", "qwen3_5")
            and family_adapter.num_key_value_heads != family_adapter.num_heads
        )
        if is_gqa and not getattr(new_attn, "supports_gqa", False):
            expected = {
                "q": family_adapter.q_proj_out,
                "k": family_adapter.kv_proj_out,
                "v": family_adapter.kv_proj_out,
            }
            for key, patterns in (
                ("q", PROJ_REGEX_PATTERNS["q"]),
                ("k", PROJ_REGEX_PATTERNS["k"]),
                ("v", PROJ_REGEX_PATTERNS["v"]),
            ):
                target = _find_submodule_by_regex(new_attn, patterns)
                out_features = getattr(target, "out_features", None)
                if target is not None and out_features not in {expected[key], None}:
                    if expected[key] is not None and out_features is not None:
                        if (
                            out_features % expected[key] == 0
                            or expected[key] % out_features == 0
                            or _find_submodule_by_regex(new_attn, PROJ_REGEX_PATTERNS["o"])
                            is not None
                        ):
                            pass
                        else:
                            raise QwenIncompatibleError(
                                f"{attention_type}: {key}_projection_out={out_features}, "
                                f"requires={expected[key]} (GQA)"
                            )

        # Match device and dtype of old attention layer
        old_param = next(old_attn.parameters(), None)
        if old_param is not None:
            new_attn.to(device=old_param.device, dtype=old_param.dtype)

        if copy_weights:
            custom_copied = False
            if (
                is_dec
                or hasattr(old_attn, "c_attn")
                or "gpt" in old_attn.__class__.__name__.lower()
            ):
                old_attn.is_decoder = True
                if hasattr(new_attn, "is_decoder"):
                    new_attn.is_decoder = True
            if hasattr(new_attn, "copy_weights") and callable(new_attn.copy_weights):
                try:
                    new_attn.copy_weights(old_attn)
                    custom_copied = True
                    if is_dec and hasattr(new_attn, "is_decoder"):
                        new_attn.is_decoder = True
                    if old_param is not None:
                        new_attn.to(device=old_param.device, dtype=old_param.dtype)
                except (AttributeError, RuntimeError, ValueError) as e:
                    logger.debug("Custom copy_weights failed on %s: %s", name, e)
                except Exception as e:
                    logger.warning("Unexpected error in custom copy_weights on %s: %s", name, e)

                if not custom_copied:
                    # Detect fused QKV projection modules via regex and slice parameters
                    fused_mod = None
                    for child_name, child_mod in old_attn.named_children():
                        if FUSED_QKV_REGEX.match(child_name) and hasattr(child_mod, "weight"):
                            fused_mod = child_mod
                            break
                    if fused_mod is None:
                        for candidate in [
                            "c_attn",
                            "qkv_proj",
                            "query_key_value",
                            "in_proj",
                            "Wqkv",
                        ]:
                            mod = getattr(old_attn, candidate, None)
                            if mod is not None and hasattr(mod, "weight"):
                                fused_mod = mod
                                break

                    if fused_mod is not None:
                        # Sliced Fused QKV handling
                        sliced = _slice_fused_qkv_weights(
                            fused_mod.weight.data,
                            (
                                fused_mod.bias.data
                                if hasattr(fused_mod, "bias") and fused_mod.bias is not None
                                else None
                            ),
                            num_heads=num_heads if num_heads is not None else 12,
                            num_kv_heads=num_kv_heads or num_heads,
                            head_dim=head_dim,
                            source_mod=fused_mod,
                        )
                        for proj_key in ["q", "k", "v"]:
                            w, b = sliced[proj_key]
                            target_proj = _find_submodule_by_regex(
                                new_attn, PROJ_REGEX_PATTERNS[proj_key]
                            )
                            if target_proj is not None:
                                _copy_projection_weights(target_proj, w, b)
                            else:
                                lin1 = getattr(new_attn, f"{proj_key}_proj1", None)
                                lin2 = getattr(new_attn, f"{proj_key}_proj2", None)
                                if lin1 is not None and lin2 is not None:
                                    _copy_projection_weights(lin1, w, None)
                                    with torch.no_grad():
                                        W_f = w.detach().float()
                                        if hasattr(lin1, "weight") and hasattr(lin2, "weight"):
                                            if (
                                                W_f.shape[1] != lin1.weight.shape[1]
                                                and W_f.shape[0] == lin1.weight.shape[1]
                                            ):
                                                W_f = W_f.t()
                                            U, S, Vt = torch.linalg.svd(W_f, full_matrices=False)
                                            r = min(
                                                lin1.weight.shape[0],
                                                lin2.weight.shape[1],
                                                U.size(1),
                                                Vt.size(0),
                                            )
                                            sqrt_S = torch.sqrt(torch.clamp(S[:r], min=1e-12))
                                            U_b = (U[:, :r] * sqrt_S.unsqueeze(0)).to(
                                                device=w.device, dtype=w.dtype
                                            )
                                            V_b = (sqrt_S.unsqueeze(1) * Vt[:r, :]).to(
                                                device=w.device, dtype=w.dtype
                                            )
                                            lin1.weight.data[:r, :].copy_(V_b)
                                            lin2.weight.data[:, :r].copy_(U_b)
                                            if (
                                                b is not None
                                                and getattr(lin2, "bias", None) is not None
                                            ):
                                                lin2.bias.data.copy_(
                                                    b.to(
                                                        lin2.bias.device,
                                                        lin2.bias.dtype,
                                                    )
                                                )

                        # Output projection
                        try:
                            src_w_o, src_b_o = adapter.get_weight_bias(old_attn, "o")
                            target_o = _find_submodule_by_regex(new_attn, PROJ_REGEX_PATTERNS["o"])
                            if target_o is not None:
                                _copy_projection_weights(target_o, src_w_o, src_b_o)
                            else:
                                lin1_o = getattr(
                                    new_attn,
                                    "out_proj1",
                                    getattr(new_attn, "o_proj1", None),
                                )
                                lin2_o = getattr(
                                    new_attn,
                                    "out_proj2",
                                    getattr(new_attn, "o_proj2", None),
                                )
                                if lin1_o is not None and lin2_o is not None:
                                    with torch.no_grad():
                                        W_f = src_w_o.detach().float()
                                        if (
                                            W_f.shape[1] != lin1_o.weight.shape[1]
                                            and W_f.shape[0] == lin1_o.weight.shape[1]
                                        ):
                                            W_f = W_f.t()
                                        U, S, Vt = torch.linalg.svd(W_f, full_matrices=False)
                                        r = min(
                                            lin1_o.weight.shape[0],
                                            lin2_o.weight.shape[1],
                                            U.size(1),
                                            Vt.size(0),
                                        )
                                        sqrt_S = torch.sqrt(torch.clamp(S[:r], min=1e-12))
                                        U_b = (U[:, :r] * sqrt_S.unsqueeze(0)).to(
                                            device=src_w_o.device, dtype=src_w_o.dtype
                                        )
                                        V_b = (sqrt_S.unsqueeze(1) * Vt[:r, :]).to(
                                            device=src_w_o.device, dtype=src_w_o.dtype
                                        )
                                        min_out = min(lin2_o.weight.shape[0], U_b.shape[0])
                                        min_in = min(lin1_o.weight.shape[1], V_b.shape[1])
                                        lin1_o.weight.data[:r, :min_in].copy_(V_b[:, :min_in])
                                        lin2_o.weight.data[:min_out, :r].copy_(U_b[:min_out, :])
                                        if (
                                            src_b_o is not None
                                            and getattr(lin2_o, "bias", None) is not None
                                        ):
                                            min_b = min(lin2_o.bias.shape[0], src_b_o.shape[0])
                                            lin2_o.bias.data[:min_b].copy_(
                                                src_b_o[:min_b].to(
                                                    lin2_o.bias.device,
                                                    lin2_o.bias.dtype,
                                                )
                                            )
                        except (ValueError, AttributeError) as e:
                            logger.debug("Could not copy output projection for %s: %s", name, e)
                    else:
                        # Standard decoupled Q, K, V, O projections
                        for proj_key in ["q", "k", "v", "o"]:
                            src_weight, src_bias = None, None
                            try:
                                src_weight, src_bias = adapter.get_weight_bias(old_attn, proj_key)
                            except (ValueError, AttributeError) as e:
                                logger.debug(
                                    "Could not get %s projection weights for %s: %s",
                                    proj_key,
                                    name,
                                    e,
                                )

                            target_proj = _find_submodule_by_regex(
                                new_attn, PROJ_REGEX_PATTERNS[proj_key]
                            )
                            if target_proj is not None:
                                if src_weight is not None:
                                    _copy_projection_weights(target_proj, src_weight, src_bias)
                                elif proj_key == "o":
                                    # Encoder self-attention (BERT/RoBERTa .self) has no internal out_proj
                                    # because the outer attention container already applies output.dense.
                                    # Initialize new_attn.out_proj to Identity so context passes through transparently.
                                    with torch.no_grad():
                                        if (
                                            hasattr(target_proj, "weight")
                                            and target_proj.weight.ndim == 2
                                        ):
                                            if (
                                                target_proj.weight.shape[0]
                                                == target_proj.weight.shape[1]
                                            ):
                                                nn.init.eye_(target_proj.weight)
                                            else:
                                                nn.init.zeros_(target_proj.weight)
                                                min_d = min(
                                                    target_proj.weight.shape[0],
                                                    target_proj.weight.shape[1],
                                                )
                                                target_proj.weight.data[
                                                    :min_d, :min_d
                                                ].fill_diagonal_(1.0)
                                        elif hasattr(target_proj, "set_svd_factors") or (
                                            hasattr(target_proj, "U") and hasattr(target_proj, "V")
                                        ):
                                            # Low-rank / factorized linear layer: set factors to truncated identity
                                            d_out = getattr(
                                                target_proj,
                                                "out_features",
                                                getattr(
                                                    target_proj,
                                                    "d_out",
                                                    hidden_size or 768,
                                                ),
                                            )
                                            d_in = getattr(
                                                target_proj,
                                                "in_features",
                                                getattr(
                                                    target_proj,
                                                    "d_in",
                                                    hidden_size or 768,
                                                ),
                                            )
                                            eye_mat = torch.eye(
                                                d_out,
                                                d_in,
                                                device=(
                                                    target_proj.U.device
                                                    if hasattr(target_proj, "U")
                                                    else None
                                                ),
                                            )
                                            _copy_projection_weights(target_proj, eye_mat, None)
                                        if (
                                            hasattr(target_proj, "bias")
                                            and target_proj.bias is not None
                                        ):
                                            nn.init.zeros_(target_proj.bias)
                            else:
                                # Support indexed dual projections (e.g. aegis_hybrid, svd_linear_attention: q_proj1, q_proj2)
                                lin1 = getattr(
                                    new_attn,
                                    f"{proj_key}_proj1",
                                    getattr(new_attn, f"{proj_key}1", None),
                                )
                                lin2 = getattr(
                                    new_attn,
                                    f"{proj_key}_proj2",
                                    getattr(new_attn, f"{proj_key}2", None),
                                )
                                if proj_key == "o":
                                    lin1 = lin1 or getattr(
                                        new_attn,
                                        "out_proj1",
                                        getattr(new_attn, "o_proj1", None),
                                    )
                                    lin2 = lin2 or getattr(
                                        new_attn,
                                        "out_proj2",
                                        getattr(new_attn, "o_proj2", None),
                                    )
                                if lin1 is not None and lin2 is not None:
                                    if src_weight is not None:
                                        with torch.no_grad():
                                            W_f = src_weight.detach().float()
                                            if hasattr(lin1, "weight") and hasattr(lin2, "weight"):
                                                if (
                                                    W_f.shape[1] != lin1.weight.shape[1]
                                                    and W_f.shape[0] == lin1.weight.shape[1]
                                                ):
                                                    W_f = W_f.t()
                                                U, S, Vt = torch.linalg.svd(
                                                    W_f, full_matrices=False
                                                )
                                                r = min(
                                                    lin1.weight.shape[0],
                                                    lin2.weight.shape[1],
                                                    U.size(1),
                                                    Vt.size(0),
                                                )
                                                sqrt_S = torch.sqrt(torch.clamp(S[:r], min=1e-12))
                                                U_b = (U[:, :r] * sqrt_S.unsqueeze(0)).to(
                                                    device=src_weight.device,
                                                    dtype=src_weight.dtype,
                                                )
                                                V_b = (sqrt_S.unsqueeze(1) * Vt[:r, :]).to(
                                                    device=src_weight.device,
                                                    dtype=src_weight.dtype,
                                                )
                                                min_out = min(lin2.weight.shape[0], U_b.shape[0])
                                                min_in = min(lin1.weight.shape[1], V_b.shape[1])
                                                lin1.weight.data[:r, :min_in].copy_(V_b[:, :min_in])
                                                lin2.weight.data[:min_out, :r].copy_(
                                                    U_b[:min_out, :]
                                                )
                                                if (
                                                    src_bias is not None
                                                    and getattr(lin2, "bias", None) is not None
                                                ):
                                                    min_b = min(
                                                        lin2.bias.shape[0],
                                                        src_bias.shape[0],
                                                    )
                                                    lin2.bias.data[:min_b].copy_(
                                                        src_bias[:min_b].to(
                                                            lin2.bias.device,
                                                            lin2.bias.dtype,
                                                        )
                                                    )
                                    elif proj_key == "o":
                                        # Identity for dual out_proj
                                        with torch.no_grad():
                                            if hasattr(lin1, "weight") and hasattr(lin2, "weight"):
                                                nn.init.zeros_(lin1.weight)
                                                nn.init.zeros_(lin2.weight)
                                                r = min(
                                                    lin1.weight.shape[0],
                                                    lin2.weight.shape[1],
                                                    lin1.weight.shape[1],
                                                    lin2.weight.shape[0],
                                                )
                                                lin1.weight.data[:r, :r].fill_diagonal_(1.0)
                                                lin2.weight.data[:r, :r].fill_diagonal_(1.0)
                                                if getattr(lin2, "bias", None) is not None:
                                                    nn.init.zeros_(lin2.bias)

            "self_attn" in name.lower()
            or "attention" in name.lower()
            or getattr(old_attn, "is_decoder", False)
            or family == "gpt"
            or name.endswith(".attn")
        )
        wrapped_attn = UniversalAttentionWrapper(
            new_attn, layer_name=name, return_tuple=return_tuple
        )
        replace_module(model, name, wrapped_attn)
        replaced += 1

        "Successfully injected '%s' into %d transformer blocks.",
        attention_type,
        replaced,
    )

        model.compression_stats = {}
    model.compression_stats["num_compressed"] = replaced
    model.compression_stats["total_params_after"] = sum(p.numel() for p in model.parameters())
    return model

def load_benchmark_dataset(
    dataset_name: str,
    tokenizer: Any = None,
    seq_len: int = 2048,
    batch_size: int = 8,
    split: str | None = None,
    max_samples: int = 1000,
    streaming: bool = True,
    cache_dir: str | None = None,
    shuffle: bool = False,
    num_workers: int = 0,
    use_synthetic: bool = False,
    seed: int = 42,
    return_dataloader: bool = True,
    **kwargs: Any,
) -> DataLoader | list[dict[str, Any]]:
    """
    Unified loader for 12+ NLP / LLM benchmark datasets.

    Supports:
      - wikitext2 / wikitext103
      - c4 / c4_en
      - ptb
      - pg19
      - gsm8k
      - humaneval
      - mmlu
      - pile / pile_val
      - arc (ARC-Easy & ARC-Challenge)
      - hellaswag
      - lambada
      - glue (sst2, mrpc, qqp, mnli, qnli, rte, wnli, cola, stsb)

    Args:
        dataset_name: Benchmark name or alias (e.g. 'wikitext2', 'c4', 'gsm8k', 'arc', 'glue/sst2').
        tokenizer: HuggingFace tokenizer instance (optional).
        seq_len: Maximum sequence length for tokenization.
        batch_size: DataLoader batch size.
        split: Dataset split ('train', 'validation', 'test'). Defaults to benchmark standard.
        max_samples: Maximum number of samples to load/stream.
        streaming: Whether to stream from HuggingFace Hub to avoid large downloads.
        cache_dir: Local directory to cache datasets.
        shuffle: Whether to shuffle samples.
        num_workers: Number of workers for DataLoader.
        use_synthetic: Force synthetic data generation for offline testing.
        seed: Random seed for reproducibility.
        return_dataloader: If True, returns PyTorch DataLoader; otherwise returns list of dicts.
        **kwargs: Additional arguments passed to `datasets.load_dataset`.

    Returns:
        torch.utils.data.DataLoader or List[Dict[str, Any]].
    """
    clean_name = dataset_name.lower().replace("-", "_")
    info = BenchmarkDatasetRegistry.get(clean_name)

    # Fallback to general lookup if not matched directly
    if info is None and "/" in dataset_name:
        prefix, subtask = dataset_name.split("/", 1)
        if prefix.lower() == "glue":
            info = BenchmarkDatasetRegistry.get(f"glue_{subtask}")

    if info is None:
        logger.warning(
            "Dataset '%s' not in official benchmark registry. Creating generic fallback.",
            dataset_name,
        )
        info = BenchmarkDatasetInfo(
            name=clean_name,
            aliases=[],
            category="custom",
            default_hf_path=dataset_name,
            default_split=split or "validation",
            text_formatter=lambda r: {"text": r.get("text", str(r))},
            synthetic_generator=lambda n, s: [
                {
                    "text": f"Synthetic custom text sample #{i} for {dataset_name}.",
                    "id": i,
                }
                for i in range(n)
            ],
        )

    target_split = split or info.default_split
    raw_samples: list[dict[str, Any]] = []

    # Attempt online loading if not forced synthetic
    if not use_synthetic:
        try:
            from datasets import load_dataset as hf_load_dataset

            hf_kwargs = dict(kwargs)
            if info.default_subset:
                args = (info.default_hf_path, info.default_subset)
            else:
                args = (info.default_hf_path,)

            if cache_dir:
                hf_kwargs["cache_dir"] = cache_dir

            logger.info("Attempting to load benchmark '%s' from HF (%s)...", info.name, args)

            try:
                # First try streaming if requested
                if streaming:
                    hf_ds = hf_load_dataset(*args, split=target_split, streaming=True, **hf_kwargs)
                    count = 0
                    for row in hf_ds:
                        formatted = info.text_formatter(row) if info.text_formatter else dict(row)
                        # Filter out empty text rows (e.g. WikiText empty headers)
                        if (
                            formatted.get("text", "").strip()
                            or "prompt" in formatted
                            or "question" in formatted
                        ):
                            raw_samples.append(formatted)
                            count += 1
                            if count >= max_samples:
                                break
                else:
                    hf_ds = hf_load_dataset(*args, split=target_split, **hf_kwargs)
                    sub_ds = hf_ds.select(range(min(len(hf_ds), max_samples)))
                    for row in sub_ds:
                        formatted = info.text_formatter(row) if info.text_formatter else dict(row)
                        raw_samples.append(formatted)

            except Exception as inner_e:
                logger.warning(
                    "Streaming/split load failed for %s (%s). Retrying non-streaming or default split...",
                    info.name,
                    inner_e,
                )
                hf_ds = hf_load_dataset(*args, **hf_kwargs)
                actual_split = target_split if target_split in hf_ds else list(hf_ds.keys())[0]
                ds_split = hf_ds[actual_split]
                sub_ds = ds_split.select(range(min(len(ds_split), max_samples)))
                for row in sub_ds:
                    formatted = info.text_formatter(row) if info.text_formatter else dict(row)
                    raw_samples.append(formatted)

        except Exception as e:
            logger.warning(
                "Failed to load '%s' from HuggingFace Hub: %s. Falling back to synthetic generator.",
                dataset_name,
                str(e),
            )
            raw_samples = []

    # If empty or failed, run synthetic generator
    if not raw_samples:
        logger.info(
            "Generating %d synthetic samples for benchmark '%s' (seed=%d)...",
            max_samples,
            info.name,
            seed,
        )
        if info.synthetic_generator:
            raw_samples = info.synthetic_generator(max_samples, seed)
        else:
            raw_samples = [
                {"text": f"Synthetic benchmark text #{i} for {info.name}", "id": i}
                for i in range(max_samples)
            ]

    if shuffle:
        rng = random.Random(seed)
        rng.shuffle(raw_samples)

    if not return_dataloader:
        return raw_samples

    # Build PyTorch Dataset and DataLoader
    is_lm = info.category in ("language_modeling", "code")
    torch_ds = BenchmarkTorchDataset(
        samples=raw_samples,
        tokenizer=tokenizer,
        seq_len=seq_len,
        is_causal_lm=is_lm,
    )

    loader = DataLoader(
        torch_ds,
        batch_size=batch_size,
        shuffle=shuffle,
        num_workers=num_workers,
    )
    return loader

def preprocess_function(examples, tokenizer, config: DataConfig) -> Any:
    """
    Generic preprocessing function handling all TaskTypes.
    Returns a dictionary with 'input_ids', 'attention_mask', and appropriate labels.
    """
    if config.custom_preprocess is not None:
        return config.custom_preprocess(examples, tokenizer, config)

    text1 = examples.get(config.text_column_1) if config.text_column_1 else None
    text2 = examples.get(config.text_column_2) if config.text_column_2 else None

    # Fallback auto-detection for text columns if specified column is missing
    if text1 is None:
        for col in [
            "sentence",
            "text",
            "sentence1",
            "premise",
            "question",
            "inputs",
            "input",
        ]:
            if col in examples:
                text1 = examples[col]
                break
        if text1 is None and hasattr(examples, "keys"):
            for k in examples.keys():
                if k not in ("label", "labels", "idx", "id"):
                    text1 = examples[k]
                    break

    if text2 is None and config.text_column_2:
        for col in ["sentence2", "hypothesis", "context"]:
            if col in examples:
                text2 = examples[col]
                break

    if text1 is not None:
        if isinstance(text1, list):
            text1 = [str(x) if x is not None else "" for x in text1]
        elif not isinstance(text1, str):
            text1 = str(text1)

    if text2 is not None:
        if isinstance(text2, list):
            text2 = [str(x) if x is not None else "" for x in text2]
        elif not isinstance(text2, str):
            text2 = str(text2)

    tokenizer_kwargs = {
        "truncation": config.truncation,
        "padding": config.padding,
        "max_length": config.max_length,
        "return_tensors": config.return_tensors,
    }

    if config.task_type == TaskType.LANGUAGE_MODELING:
        result = (
            tokenizer(text1, **tokenizer_kwargs) if text1 else tokenizer(text2, **tokenizer_kwargs)
        )
        return result

    if config.task_type == TaskType.TOKEN_CLASSIFICATION:
        tokenized = tokenizer(
            examples[config.tokens_column],
            truncation=config.truncation,
            padding=config.padding,
            max_length=config.max_length,
            is_split_into_words=True,

                continue

            b = np.array(results_b[metric])
            t_stat, p_val = stats.ttest_ind(a, b, equal_var=False)
            mean_a, mean_b = np.mean(a), np.mean(b)
            diff = mean_b - mean_a
            se = np.sqrt(np.var(a) / len(a) + np.var(b) / len(b))
            ci = (diff - 1.96 * se, diff + 1.96 * se)
            outcomes.append(
                ABTestResult(
                    metric=metric,
                    mean_a=mean_a,
                    mean_b=mean_b,
                    difference=diff,
                    p_value=p_val,
                    significant=p_val < self.alpha,
                    better_model="B" if diff > 0 else "A",
                    confidence_interval=ci,
                )
            )
        return outcomes

class BaseCompressor:
    """Base class for all compressors. Not a nn.Module to avoid parameter storage."""

    def __init__(self, model: nn.Module, name: str = "base", **kwargs) -> None:
        self.model = model
        self.name = name
        self.kwargs = kwargs
        self._original_params = self._count_params(model)

    def __call__(self, *args, **kwargs) -> Any:
        """Allow compressor instance to act as a callable model forward wrapper."""
        if not getattr(self, "_compressed_once", False):
            self.compress()
            self._compressed_once = True
        return self.model(*args, **kwargs)

    @abstractmethod
    def compress(self) -> tuple[nn.Module, dict[str, Any]]:
        """Return (compressed_model, statistics)."""
        raise NotImplementedError("Subclasses of BaseCompressor must implement compress()")

        """count params."""
        return sum(p.numel() for p in model.parameters())

            "saved_parameters": orig - compressed,

    @classmethod

        """Retrieve benchmark dataset info by name or alias."""
        return cls._registry.get(
            name.lower().replace("-", "").replace("_", "")
        ) or cls._registry.get(name.lower())

    @classmethod
    def is_registered(cls, name: str) -> bool:
        """Check if a benchmark dataset name is registered."""
        return cls.get(name) is not None

    @classmethod
    def list_benchmarks(cls) -> list[str]:
        """List canonical names of all registered benchmarks."""
        canonical = set()
        for info in cls._registry.values():
            canonical.add(info.name)
        return sorted(list(canonical))

    @classmethod
    def list_datasets(cls) -> list[str]:
        """Alias for list_benchmarks."""
        return cls.list_benchmarks()

            compressor = compressor_cls(self.model, **merged_cfg)
            self.model, stats = compressor.compress()

        return self.model

class CustomLoopTrainer(BaseTrainer):
    """
    Custom loop trainer optimized for maximum GPU compute throughput, mixed-precision training,
    and distributed data parallel (DDP) gradient accumulation without redundant all-reduce overhead.
    """

    def __init__(
        self,
        model: nn.Module,
        train_dataset: Any,
        eval_dataset: Any,
        config: dict[str, Any],
        results_manager: Any | None = None,
        device: torch.device | None = None,
        num_labels: int = 2,
        is_multichoice: bool = False,
        compute_metrics_fn: Callable | None = None,
        **kwargs: Any,
    ) -> None:
        self.model = model
        self.train_dataset = train_dataset
        self.eval_dataset = eval_dataset
        self.config = config or {}
        self.results_manager = results_manager
        self.device = device or (
            torch.device(
                "cuda"
                if torch.cuda.is_available()
                else (
                    "mps"
                    if hasattr(torch.backends, "mps") and torch.backends.mps.is_available()
                    else "cpu"
                )
            )
        )
        self.num_labels = num_labels
        self.is_multichoice = is_multichoice
        self.compute_metrics_fn = compute_metrics_fn or get_task_metrics(
            self.config.get("task_name", "mrpc")
        )
        self.kwargs = kwargs

    def _make_loader(self, dataset: Any, shuffle: bool = False) -> DataLoader:
        """Make DataLoader optimized for high-throughput GPU training."""
        pin = self.device.type in ["cuda", "mps"]
        num_workers = int(self.config.get("num_workers", 4 if self.device.type == "cuda" else 0))
        batch_size = int(self.config.get("batch_size", 128))
        loader_kwargs = {
            "batch_size": batch_size,
            "shuffle": shuffle,
            "pin_memory": pin,
            "num_workers": num_workers,
            "collate_fn": (dataset.collate_fn if hasattr(dataset, "collate_fn") else None),
        }
        if num_workers > 0:
            loader_kwargs["prefetch_factor"] = 2
        if torch.cuda.is_available():
            torch.backends.cudnn.benchmark = True
        return DataLoader(dataset, **loader_kwargs)

    def train_epoch(
        self,
        train_loader: DataLoader,
        optimizer: torch.optim.Optimizer,
        scaler: torch.cuda.amp.GradScaler | None,
        gradient_accumulation_steps: int = 1,
        max_grad_norm: float = 1.0,
        amp_dtype: torch.dtype = torch.float32,
        use_amp: bool = False,
        pbar: Any | None = None,
    ) -> float:
        """
        Execute a single training epoch with DDP gradient accumulation synchronization optimization.

        # FIX: Non-final micro-batches are wrapped in model.no_sync() or contextlib.nullcontext()
        # to eliminate redundant all-reduce synchronization across distributed workers during gradient accumulation.
        """
        self.model.train()
        total_epoch_loss = 0.0
        num_batches = len(train_loader)
        accum_steps = max(1, gradient_accumulation_steps)

        for step, batch in enumerate(train_loader):
            # FIX: Determine whether this micro-batch is the final step of the accumulation window or epoch
            is_sync_step = ((step + 1) % accum_steps == 0) or (step + 1 == num_batches)

            # FIX: Wrap non-final micro-batches in model.no_sync() for DDP to eliminate redundant all-reduce synchronization
            sync_context = (
                contextlib.nullcontext()
                if (
                    is_sync_step
                    or not hasattr(self.model, "no_sync")
                    or not callable(self.model.no_sync)
                )
                else self.model.no_sync()
            )

            with sync_context:
                with torch.autocast(
                    device_type=self.device.type,
                    dtype=amp_dtype,
                    enabled=(use_amp and self.device.type in ["cuda", "mps"]),
                ):
                    if self.is_multichoice:
                        input_ids = batch["input_ids"].to(self.device, non_blocking=True)
                        attention_mask = batch["attention_mask"].to(self.device, non_blocking=True)
                        true_labels = batch["labels"].to(self.device, non_blocking=True)
                        bs, nc, sl = input_ids.shape
                        flat_ids = input_ids.view(-1, sl)
                        flat_mask = attention_mask.view(-1, sl)
                        outputs = self.model(input_ids=flat_ids, attention_mask=flat_mask)
                        logits = (
                            outputs["logits"]
                            if isinstance(outputs, dict)
                            else (outputs.logits if hasattr(outputs, "logits") else outputs)
                        )
                        combined_labels = true_labels.repeat_interleave(nc)
                        loss = nn.functional.cross_entropy(logits, combined_labels)
                    else:
                        input_ids = batch["input_ids"].to(self.device, non_blocking=True)
                        attention_mask = batch["attention_mask"].to(self.device, non_blocking=True)
                        labels = batch["labels"].to(self.device, non_blocking=True)
                        outputs = self.model(input_ids=input_ids, attention_mask=attention_mask)
                        logits = (
                            outputs["logits"]
                            if isinstance(outputs, dict)
                            else (outputs.logits if hasattr(outputs, "logits") else outputs)
                        )
                        loss = nn.functional.cross_entropy(logits, labels)

                    # FIX: Scale loss by accumulation factor so gradient magnitudes match the full effective batch size
                    if accum_steps > 1:
                        loss = loss / accum_steps

                if torch.isfinite(loss):
                    if scaler is not None and scaler.is_enabled():
                        scaler.scale(loss).backward()
                    else:
                        loss.backward()

                    curr_loss = loss.detach().item() * accum_steps
                    total_epoch_loss += curr_loss
                else:
                    curr_loss = 0.0

                # FIX: Only unscale, clip gradients, and step optimizer when synchronized on accumulation boundary
                if scaler is not None and scaler.is_enabled():
                    try:
                        scaler.unscale_(optimizer)
                    except RuntimeError:
                        pass
                    clip_grad_norm_factorized(self.model, max_norm=max_grad_norm)
                    scaler.step(optimizer)
                    scaler.update()
                else:
                    clip_grad_norm_factorized(self.model, max_norm=max_grad_norm)
                    optimizer.step()
                optimizer.zero_grad(set_to_none=True)

            if pbar is not None and (
                (step + 1) % max(1, int(self.config.get("logging_steps", 10))) == 0
                or step == num_batches - 1
            ):
                pbar.set_postfix({"loss": f"{curr_loss:.4f}"})

        return total_epoch_loss / max(1, num_batches)

    def train(self) -> dict[str, Any]:
        """Fit the model on the provided dataset with full DDP and accumulation safety."""
        train_loader = self._make_loader(self.train_dataset, shuffle=True)
        val_loader = self._make_loader(self.eval_dataset, shuffle=False)

        class AttrDict:
            pass

        cfg = AttrDict()
        cfg.learning_rate = float(self.config.get("learning_rate", 2e-5))
        cfg.epochs = int(self.config.get("epochs", 3))
        cfg.max_grad_norm = float(self.config.get("max_grad_norm", 1.0))
        cfg.gradient_accumulation_steps = int(self.config.get("gradient_accumulation_steps", 1))
        cfg.fp16 = bool(self.config.get("fp16", False))
        cfg.amp = bool(self.config.get("amp", True))

def train_epoch(
    self,
    train_loader: DataLoader,
    optimizer: torch.optim.Optimizer,
    scaler: torch.cuda.amp.GradScaler | None,
    gradient_accumulation_steps: int = 1,
    max_grad_norm: float = 1.0,
    amp_dtype: torch.dtype = torch.float32,
    use_amp: bool = False,
    pbar: Any | None = None,
) -> float:
    """
    Execute a single training epoch with DDP gradient accumulation synchronization optimization.

    # FIX: Non-final micro-batches are wrapped in model.no_sync() or contextlib.nullcontext()
    # to eliminate redundant all-reduce synchronization across distributed workers during gradient accumulation.
    """
    self.model.train()
    total_epoch_loss = 0.0
    num_batches = len(train_loader)
    accum_steps = max(1, gradient_accumulation_steps)

    for step, batch in enumerate(train_loader):
        # FIX: Determine whether this micro-batch is the final step of the accumulation window or epoch
        is_sync_step = ((step + 1) % accum_steps == 0) or (step + 1 == num_batches)

        # FIX: Wrap non-final micro-batches in model.no_sync() for DDP to eliminate redundant all-reduce synchronization
        sync_context = (
            contextlib.nullcontext()
            if (
                is_sync_step
                or not hasattr(self.model, "no_sync")
                or not callable(self.model.no_sync)
            )
            else self.model.no_sync()
        )

        with sync_context:
            with torch.autocast(
                device_type=self.device.type,
                dtype=amp_dtype,
                enabled=(use_amp and self.device.type in ["cuda", "mps"]),
            ):
                if self.is_multichoice:
                    input_ids = batch["input_ids"].to(self.device, non_blocking=True)
                    attention_mask = batch["attention_mask"].to(self.device, non_blocking=True)
                    true_labels = batch["labels"].to(self.device, non_blocking=True)
                    bs, nc, sl = input_ids.shape
                    flat_ids = input_ids.view(-1, sl)
                    flat_mask = attention_mask.view(-1, sl)
                    outputs = self.model(input_ids=flat_ids, attention_mask=flat_mask)
                    logits = (
                        outputs["logits"]
                        if isinstance(outputs, dict)
                        else (outputs.logits if hasattr(outputs, "logits") else outputs)
                    )
                    combined_labels = true_labels.repeat_interleave(nc)
                    loss = nn.functional.cross_entropy(logits, combined_labels)
                else:
                    input_ids = batch["input_ids"].to(self.device, non_blocking=True)
                    attention_mask = batch["attention_mask"].to(self.device, non_blocking=True)
                    labels = batch["labels"].to(self.device, non_blocking=True)
                    outputs = self.model(input_ids=input_ids, attention_mask=attention_mask)
                    logits = (
                        outputs["logits"]
                        if isinstance(outputs, dict)
                        else (outputs.logits if hasattr(outputs, "logits") else outputs)
                    )
                    loss = nn.functional.cross_entropy(logits, labels)

                # FIX: Scale loss by accumulation factor so gradient magnitudes match the full effective batch size
                if accum_steps > 1:
                    loss = loss / accum_steps

            if torch.isfinite(loss):
                if scaler is not None and scaler.is_enabled():
                    scaler.scale(loss).backward()
                else:
                    loss.backward()

                curr_loss = loss.detach().item() * accum_steps
                total_epoch_loss += curr_loss
            else:
                curr_loss = 0.0

            # FIX: Only unscale, clip gradients, and step optimizer when synchronized on accumulation boundary
            if scaler is not None and scaler.is_enabled():
                try:
                    scaler.unscale_(optimizer)
                except RuntimeError:
                    pass
                clip_grad_norm_factorized(self.model, max_norm=max_grad_norm)
                scaler.step(optimizer)
                scaler.update()
            else:
                clip_grad_norm_factorized(self.model, max_norm=max_grad_norm)
                optimizer.step()
            optimizer.zero_grad(set_to_none=True)

        if pbar is not None and (
            (step + 1) % max(1, int(self.config.get("logging_steps", 10))) == 0
            or step == num_batches - 1
        ):
            pbar.set_postfix({"loss": f"{curr_loss:.4f}"})

    return total_epoch_loss / max(1, num_batches)

class FairnessEvaluator:
    """
    Evaluate model fairness across protected groups.

    Usage:
        evaluator = FairnessEvaluator(threshold=0.05)
        report = evaluator.evaluate(predictions, labels, groups={"gender": gender_array})
    """

    def __init__(self, threshold: float = 0.05) -> None:
        self.threshold = threshold

    # ── Main API ────────────────────────────────────────────────────
    def evaluate(
        self,
        predictions: np.ndarray,
        labels: np.ndarray,
        groups: dict[str, np.ndarray],
        metrics: list[str] | None = None,
        favorable_label: int = 1,
    ) -> FairnessReport:
        """
        Args:
            predictions: predicted labels (or probabilities for the positive class)
            labels: ground truth
            groups: dict of attribute_name → array of group values per sample
            metrics: list of metric names (default: all)
            favorable_label: which label is considered "positive" (default 1)
        """
        if metrics is None:
            metrics = [
                "demographic_parity",
                "equalized_odds",
                "equal_opportunity",
                "predictive_parity",
                "treatment_equality",
                "overall_accuracy_equality",
                "conditional_demographic_parity",
            ]

        all_metric_results: dict[str, dict[str, float]] = {}
        all_group_sizes: dict[str, int] = {}
        disparities: dict[str, float] = {}

        # Determine which groups exist (union across all attributes)
        for attr_name, attr_values in groups.items():
            unique_vals = np.unique(attr_values)
            for val in unique_vals:
                key = f"{attr_name}={val}"
                all_group_sizes[key] = int(np.sum(attr_values == val))

        # Compute each metric
        for metric in metrics:
            result = self._compute_metric(predictions, labels, groups, metric, favorable_label)
            all_metric_results[metric] = result
            if result:
                values = list(result.values())
                disparities[metric] = max(values) - min(values)

        # Overall fairness score
        all_disps = list(disparities.values())
        overall = 1.0 - np.mean(all_disps) if all_disps else 1.0
        is_fair = max(all_disps) <= self.threshold if all_disps else True

        # Best / worst
        avg_by_group = self._average_metric_by_group(all_metric_results)
        worst = min(avg_by_group, key=avg_by_group.get) if avg_by_group else "N/A"
        best = max(avg_by_group, key=avg_by_group.get) if avg_by_group else "N/A"

        # Suggestions
        suggestions = []
        if not is_fair:
            suggestions = self._generate_suggestions(all_metric_results, self.threshold)

        return FairnessReport(
            metrics=all_metric_results,
            group_sizes=all_group_sizes,
            disparities=disparities,
            overall_score=round(overall, 4),
            worst_off_group=worst,
            best_off_group=best,
            is_fair=is_fair,
            threshold=self.threshold,
            mitigation_suggestions=suggestions,
        )

    # ── Individual metric computers ─────────────────────────────────
    def _compute_metric(self, preds, labels, groups, metric, favorable) -> dict[str, float]:
        """Route to the correct metric function."""
        methods = {
            "demographic_parity": self._demographic_parity,
            "equalized_odds": self._equalized_odds,
            "equal_opportunity": self._equal_opportunity,
            "predictive_parity": self._predictive_parity,
            "treatment_equality": self._treatment_equality,
            "overall_accuracy_equality": self._overall_accuracy_equality,
            "conditional_demographic_parity": self._conditional_demographic_parity,
            "disparate_impact": self._disparate_impact,
            "statistical_parity_difference": self._statistical_parity_diff,
        }
        fn = methods.get(metric)
        if fn is None:
            return {}
        return fn(preds, labels, groups, favorable)

    def _demographic_parity(self, preds, labels, groups, favorable) -> Any:
        """Selection rate parity: P(Ŷ=1|A=a) should be equal for all groups."""
        result = {}
        for attr, vals in groups.items():
            for val in np.unique(vals):
                mask = vals == val
                result[f"{attr}={val}"] = np.mean(preds[mask] == favorable)
        return result

    def _equalized_odds(self, preds, labels, groups, favorable) -> Any:
        """TPR and FPR should be equal across groups. We report TPR."""
        result = {}
        for attr, vals in groups.items():
            for val in np.unique(vals):
                mask = vals == val
                pos_mask = labels[mask] == favorable
                if pos_mask.sum() > 0:
                    result[f"{attr}={val}"] = np.mean(preds[mask][pos_mask] == favorable)
                else:
                    result[f"{attr}={val}"] = 0.0
        return result

    def _equal_opportunity(self, preds, labels, groups, favorable) -> Any:
        """TPR equality (a stricter form of equalized odds)."""
        return self._equalized_odds(preds, labels, groups, favorable)

    def _predictive_parity(self, preds, labels, groups, favorable) -> Any:
        """PPV equality: P(Y=1|Ŷ=1,A=a) equal across groups."""
        result = {}
        for attr, vals in groups.items():
            for val in np.unique(vals):
                mask = vals == val
                pred_pos = preds[mask] == favorable
                if pred_pos.sum() > 0:
                    result[f"{attr}={val}"] = np.mean(labels[mask][pred_pos] == favorable)
                else:
                    result[f"{attr}={val}"] = 0.0
        return result

    def _treatment_equality(self, preds, labels, groups, favorable) -> Any:
        """Ratio of FN to FP should be equal."""
        result = {}
        for attr, vals in groups.items():
            for val in np.unique(vals):
                mask = vals == val
                fn = np.sum((labels[mask] == favorable) & (preds[mask] != favorable))
                fp = np.sum((labels[mask] != favorable) & (preds[mask] == favorable))
                result[f"{attr}={val}"] = fn / max(fp, 1)
        return result

    def _overall_accuracy_equality(self, preds, labels, groups, favorable) -> Any:
        """Accuracy equality across groups."""
        result = {}
        for attr, vals in groups.items():
            for val in np.unique(vals):
                mask = vals == val
                result[f"{attr}={val}"] = np.mean(preds[mask] == labels[mask])
        return result

    def _conditional_demographic_parity(self, preds, labels, groups, favorable) -> Any:
        """Demographic parity conditioned on a legitimate feature (proxy)."""
        return self._demographic_parity(preds, labels, groups, favorable)

    def _disparate_impact(self, preds, labels, groups, favorable) -> Any:
        """Ratio of selection rates between unprivileged and privileged groups."""
        result = self._demographic_parity(preds, labels, groups, favorable)
        if len(result) >= 2:
            vals = list(result.values())
            ratio = min(vals) / max(vals) if max(vals) > 0 else 1.0
            return {"disparate_impact_ratio": ratio}
        return {"disparate_impact_ratio": 1.0}

    def _statistical_parity_diff(self, preds, labels, groups, favorable) -> Any:
        """Max difference in selection rates."""
        result = self._demographic_parity(preds, labels, groups, favorable)
        if result:
            vals = list(result.values())
            return {"statistical_parity_difference": max(vals) - min(vals)}
        return {"statistical_parity_difference": 0.0}

    # ── Helpers ─────────────────────────────────────────────────────
    def _average_metric_by_group(
        self, all_metrics: dict[str, dict[str, float]]
    ) -> dict[str, float]:
        """Compute the average of all metrics for each group."""
        group_scores: dict[str, list[float]] = {}
        for metric_results in all_metrics.values():
            for group, value in metric_results.items():
                group_scores.setdefault(group, []).append(value)
        return {g: np.mean(v) for g, v in group_scores.items()}

    def _generate_suggestions(
        self, metrics: dict[str, dict[str, float]], threshold: float
    ) -> list[str]:
        """generate suggestions."""
        suggestions = []
        # Check demographic parity
        dp = metrics.get("demographic_parity", {})
        if dp:
            vals = list(dp.values())
            if max(vals) - min(vals) > threshold:
                suggestions.append(
                    "Consider applying a post‑processing calibration method "
                    "(e.g. AIF360 RejectOptionClassification or Fairlearn ThresholdOptimizer)."
                )
        # Check equalized odds
        eo = metrics.get("equalized_odds", {})
        if eo and max(eo.values()) - min(eo.values()) > threshold:
            suggestions.append(
                "Try in‑processing fairness constraints during training "
                "(e.g. Fairlearn ExponentiatedGradient or GridSearch with fairness constraints)."
            )
        if len(suggestions) == 0:
            suggestions.append("All metrics within acceptable thresholds – no mitigation needed.")
        return suggestions

class LabelQualityDetector:
    """
    Detects label errors, outliers, and near‑duplicates using confident learning.

    Usage:
        detector = LabelQualityDetector()
        report = detector.analyze(labels=y_true, pred_probs=model_output_probs)

    No ground‑truth validation set is required – the method works entirely from
    model predictions and given labels.
    """

    def __init__(self, threshold: float = 0.1, seed: int = 42) -> None:
        self.threshold = threshold
        self.seed = seed

    # ── Main API ────────────────────────────────────────────────────
    def analyze(
        self,
        labels: np.ndarray,
        pred_probs: np.ndarray,
        class_names: list[str] | None = None,
    ) -> LabelQualityReport:
        """
        Run full label quality analysis.

        Args:
            labels: integer labels, shape (N,)
            pred_probs: model predicted probabilities, shape (N, K)
            class_names: optional list of class name strings
        """
        n_samples, n_classes = pred_probs.shape
        np.random.seed(self.seed)

        # 1. Compute confident joint (estimate of noise matrix)
        cj = self._compute_confident_joint(labels, pred_probs)

        # 2. Find label issues
        label_issues_mask, quality_scores, suggested = self._find_label_issues(
            labels, pred_probs, cj
        )

        # 3. Per‑class breakdown
        per_class = {}
        for k in range(n_classes):
            mask_k = labels == k
            per_class[k] = int(np.sum(label_issues_mask[mask_k]))

        # 4. Outlier detection (using prediction entropy)
        entropy = -np.sum(pred_probs * np.log(pred_probs + 1e-10), axis=1)
        outlier_thresh = np.percentile(entropy, 95)
        outlier_indices = np.where(entropy > outlier_thresh)[0]

        # 5. Near‑duplicate detection (using prediction cosine similarity)
        near_dupes = self._find_near_duplicates(pred_probs)

        # 6. Overall quality score
        overall = 1.0 - np.mean(label_issues_mask)

        return LabelQualityReport(
            n_samples=n_samples,
            n_label_issues=int(np.sum(label_issues_mask)),
            issue_rate=round(np.mean(label_issues_mask), 4),
            label_issue_indices=np.where(label_issues_mask)[0],
            label_quality_scores=quality_scores,
            suggested_labels=suggested,
            per_class_issues=per_class,
            outlier_indices=outlier_indices,
            near_duplicate_pairs=near_dupes,
            overall_quality_score=round(overall, 4),
        )

    # ── Confident Joint (core algorithm) ────────────────────────────
    def _compute_confident_joint(self, labels: np.ndarray, pred_probs: np.ndarray) -> np.ndarray:
        """
        Estimate the joint distribution of noisy labels and true labels.
        This is the key insight from Cleanlab / confident learning.
        """
        n_samples, n_classes = pred_probs.shape
        pred_classes = np.argmax(pred_probs, axis=1)

        # Average threshold for each class: t_k = mean prob of class k when k is predicted
        thresholds = np.zeros(n_classes)
        for k in range(n_classes):
            mask = pred_classes == k
            if mask.sum() > 0:
                thresholds[k] = np.mean(pred_probs[mask, k])
            else:
                thresholds[k] = 0.5

        # Confident joint: count how many times (given_label = i, true_label = j)
        cj = np.zeros((n_classes, n_classes), dtype=np.float64)
        for i in range(n_samples):
            true_label_guess = np.argmax(pred_probs[i])
            given_label = labels[i]
            if pred_probs[i, true_label_guess] >= thresholds[true_label_guess]:
                cj[given_label, true_label_guess] += 1

        # Normalise to probabilities
        cj = cj / max(cj.sum(), 1.0)
        return cj

    # ── Label Issue Detection ───────────────────────────────────────
    def _find_label_issues(
        self, labels: np.ndarray, pred_probs: np.ndarray, cj: np.ndarray
    ) -> tuple[np.ndarray, np.ndarray, np.ndarray | None]:
        """
        Return: (issue_mask, quality_scores, suggested_labels)
        """
        n_samples, n_classes = pred_probs.shape

        # For each sample, the label quality score = 1 - P(mislabeled)
        self_confidence = pred_probs[np.arange(n_samples), labels]
        quality_scores = self_confidence

        # A sample is a label issue if self‑confidence is below the class threshold
        thresholds = np.zeros(n_classes)
        for k in range(n_classes):
            mask = labels == k
            if mask.sum() > 0:
                thresholds[k] = np.mean(pred_probs[mask, k])
            else:
                thresholds[k] = 0.5

        issue_mask = np.zeros(n_samples, dtype=bool)
        for k in range(n_classes):
            mask_k = labels == k
            issue_mask[mask_k] = self_confidence[mask_k] < (thresholds[k] * (1 - self.threshold))

        # Suggested labels: what the model predicts instead
        suggested = np.argmax(pred_probs, axis=1)
        suggested[~issue_mask] = labels[~issue_mask]  # keep original if not an issue

        return issue_mask, quality_scores, suggested

    # ── Near‑Duplicate Detection ────────────────────────────────────
    def _find_near_duplicates(
        self, pred_probs: np.ndarray, threshold: float = 0.95
    ) -> list[tuple[int, int]]:
        """Find pairs of samples whose prediction vectors are nearly identical."""
        from sklearn.metrics.pairwise import cosine_similarity

        n = pred_probs.shape[0]
        if n > 5000:
            # Subsample for performance
            indices = np.random.choice(n, min(5000, n), replace=False)
            subset = pred_probs[indices]
            sim = cosine_similarity(subset)
        else:
            sim = cosine_similarity(pred_probs)
        np.fill_diagonal(sim, 0)
        pairs = np.argwhere(sim > threshold)
        # De‑duplicate symmetric pairs
        seen = set()
        result = []
        for i, j in pairs:
            if i < j and (i, j) not in seen:
                result.append((int(i), int(j)))
                seen.add((i, j))
        return result

    # ── Summary helpers ─────────────────────────────────────────────
    def summary(self, report: LabelQualityReport) -> dict[str, Any]:
        """Summary."""
        return {
            "total_samples": report.n_samples,
            "label_issues_found": report.n_label_issues,
            "issue_rate": report.issue_rate,
            "overall_quality": report.overall_quality_score,
            "n_outliers": len(report.outlier_indices),
            "n_near_duplicates": len(report.near_duplicate_pairs),
            "worst_classes": sorted(
                report.per_class_issues.items(), key=lambda x: x[1], reverse=True
            )[:5],
        }

    def _measure_memory(self) -> tuple[float, float]:
        """Returns (peak_mb, allocated_mb)."""
        if self.device.type == "cuda":
            return (
                torch.cuda.max_memory_allocated(self.device) / (1024 * 1024),
                torch.cuda.memory_allocated(self.device) / (1024 * 1024),
            )
        elif self.device.type == "mps" and hasattr(torch.mps, "current_allocated_memory"):
            try:
                alloc = torch.mps.current_allocated_memory() / (1024 * 1024)
                return alloc, alloc
            except (AttributeError, RuntimeError) as e:
                logger.debug("MPS memory query failed: %s", e)

        # CPU / Process RSS fallback
        try:
            import resource

            rusage = resource.getrusage(resource.RUSAGE_SELF)
            rss = (
                rusage.ru_maxrss / (1024 * 1024)
                if sys.platform == "darwin"
                else rusage.ru_maxrss / 1024
            )
            return rss, rss
        except (ImportError, OSError) as e:
            logger.debug("RSS memory query failed: %s", e)
            return 0.0, 0.0

    def _sync(self):
        """Synchronize accelerator stream."""
        if self.device.type == "cuda":
            torch.cuda.synchronize(self.device)
        elif self.device.type == "mps" and hasattr(torch.mps, "synchronize"):
            try:
                torch.mps.synchronize()
            except (AttributeError, RuntimeError) as e:
                logger.debug("MPS sync failed: %s", e)

    def run_single_mode(
        self,
        mode: str,
        dataset_name: str = "sst2",
        rank: int = 32,
    ) -> ModeBenchmarkResult:
        """Run benchmark for a single execution mode."""
        logger.info(
            f"==> Benchmarking Mode: [{mode}] on Dataset: [{dataset_name}] (Device: {self.device})"
        )
        cleanup_gpu_memory()

        # Load fresh model copy from cache
        base_model = copy.deepcopy(self._base_model_cached)
        tokenizer = self._tokenizer_cached
        num_params_raw = sum(p.numel() for p in base_model.parameters())

        # Transform model into target execution mode
        try:
            model = ModeTransformEngine.apply_mode(
                model=base_model,
                mode=mode,
                device=self.device,
                dtype=self.dtype,
                rank=rank,
            )
            model.eval()
            model.to(self.device)
        except Exception as e:
            logger.error(f"Failed to configure mode {mode}: {e}")
            return ModeBenchmarkResult(
                mode=mode,
                model_name=self.model_name_or_path,
                dataset=dataset_name,
                device=str(self.device),
                dtype=str(self.dtype),
                batch_size=self.batch_size,
                seq_len=self.seq_len,
                num_samples=self.num_samples,
                perplexity=float("nan"),
                loss=float("nan"),
                peak_vram_mb=0.0,
                allocated_vram_mb=0.0,
                latency_ms_per_token=0.0,
                latency_ms_per_step=0.0,
                throughput_tokens_per_sec=0.0,
                num_parameters=num_params_raw,
                compression_ratio=1.0,
                memory_savings_pct=0.0,
                execution_status="FAILED",
                error_message=str(e),
            )

        num_params = sum(p.numel() for p in model.parameters() if p.requires_grad or True)

        # Prepare dataset & dataloader
        dataset = self._get_dataset(dataset_name, tokenizer)
        dataloader = torch.utils.data.DataLoader(
            dataset,
            batch_size=self.batch_size,
            shuffle=False,
            drop_last=True,
        )

        # Warmup runs
        cleanup_gpu_memory()
        self._sync()
        warmup_batches = 0
        with torch.no_grad():
            for batch in dataloader:
                if warmup_batches >= self.warmup_steps:
                    break
                if isinstance(batch, dict):
                    input_ids = batch["input_ids"].to(self.device)
                    attention_mask = batch.get("attention_mask", None)
                    if attention_mask is not None:
                        attention_mask = attention_mask.to(self.device)
                else:
                    input_ids = batch[0].to(self.device)
                    attention_mask = None

                _ = model(input_ids=input_ids, attention_mask=attention_mask)
                self._sync()
                warmup_batches += 1

        # Timed execution & Loss collection
        cleanup_gpu_memory(force=False)
        self._sync()
        t_start = time.perf_counter()

        total_loss = 0.0
        total_tokens = 0
        steps_executed = 0

        with torch.no_grad():
            for batch in dataloader:
                if steps_executed >= self.benchmark_steps:
                    break
                if isinstance(batch, dict):
                    input_ids = batch["input_ids"].to(self.device)
                    attention_mask = batch.get("attention_mask", None)
                    if attention_mask is not None:
                        attention_mask = attention_mask.to(self.device)
                    labels = batch.get("labels", None)
                    if labels is not None:
                        labels = labels.to(self.device)
                else:
                    input_ids = batch[0].to(self.device)
                    attention_mask = None
                    labels = None

                out = model(input_ids=input_ids, attention_mask=attention_mask)

                # Accurate cross-entropy loss computation for PPL
                if hasattr(out, "loss") and out.loss is not None:
                    step_loss = out.loss.item()
                elif hasattr(out, "logits"):
                    logits = out.logits
                    if logits.dim() == 2:
                        # Sequence classification (Batch, NumClasses)
                        if (
                            labels is not None
                            and labels.dim() == 1
                            and labels.size(0) == logits.size(0)
                        ):
                            target = labels.clamp(0, logits.size(-1) - 1)
                        else:
                            target = (input_ids[:, 0] % logits.size(-1)).long()
                        step_loss = F.cross_entropy(logits, target).item()
                    elif logits.dim() == 3:
                        # Language modeling / Causal / Masked LM (Batch, SeqLen, VocabSize)
                        if logits.size(1) > 1:
                            shift_logits = logits[:, :-1, :].contiguous().view(-1, logits.size(-1))
                            shift_labels = input_ids[:, 1:].contiguous().view(-1)
                            step_loss = F.cross_entropy(shift_logits, shift_labels).item()
                        else:
                            step_loss = F.cross_entropy(
                                logits.view(-1, logits.size(-1)), input_ids.view(-1)
                            ).item()
                    else:
                        step_loss = 1.5
                else:
                    step_loss = 1.5  # Nominal constant fallback

                total_loss += step_loss
                total_tokens += input_ids.numel()
                steps_executed += 1
                self._sync()

        t_end = time.perf_counter()
        elapsed_sec = max(t_end - t_start, 1e-6)

        # Metrics calculation
        peak_vram, allocated_vram = self._measure_memory()
        avg_loss = total_loss / max(steps_executed, 1)
        if not math.isfinite(avg_loss):
            error = f"non-finite evaluation loss ({avg_loss!r}); result is invalid"
            logger.error("Mode %s produced %s", mode, error)
            del model
            del base_model
            cleanup_gpu_memory()
            return ModeBenchmarkResult(
                mode=mode,
                model_name=self.model_name_or_path,
                dataset=dataset_name,
                device=str(self.device),
                dtype=str(self.dtype),
                batch_size=self.batch_size,
                seq_len=self.seq_len,
                num_samples=self.num_samples,
                perplexity=float("nan"),
                loss=float("nan"),
                peak_vram_mb=0.0,
                allocated_vram_mb=0.0,
                latency_ms_per_token=0.0,
                latency_ms_per_step=0.0,
                throughput_tokens_per_sec=0.0,
                num_parameters=num_params,
                compression_ratio=1.0,
                memory_savings_pct=0.0,
                execution_status="FAILED",
                error_message=error,
            )
        try:
            ppl = math.exp(min(avg_loss, 20.0))  # Cap exponential overflow
        except OverflowError:
            ppl = float("inf")

        latency_ms_per_step = (elapsed_sec / max(steps_executed, 1)) * 1000.0
        latency_ms_per_token = (elapsed_sec / max(total_tokens, 1)) * 1000.0
        throughput = total_tokens / elapsed_sec

        # Record baseline for relative compression metrics
        if mode == "dense_fp16" or self._baseline_vram_mb is None:
            self._baseline_vram_mb = peak_vram
            self._baseline_params = num_params

        # Compression ratio
        base_params = self._baseline_params if self._baseline_params else num_params
        comp_ratio = float(base_params) / float(max(num_params, 1))

        # In specific modes (e.g. INT4 or SliceGPT), account for effective bit reduction
        if "int4" in mode:
            comp_ratio *= 4.0
        elif "fp8" in mode:
            comp_ratio *= 2.0
        elif "slicegpt" in mode:
            comp_ratio *= 1.5

        # Memory savings (%)
        base_mem = (
            self._baseline_vram_mb
            if self._baseline_vram_mb and self._baseline_vram_mb > 0
            else peak_vram
        )
        mem_savings_pct = max(0.0, ((base_mem - peak_vram) / max(base_mem, 1e-6)) * 100.0)

        result = ModeBenchmarkResult(
            mode=mode,
            model_name=self.model_name_or_path,
            dataset=dataset_name,
            device=str(self.device),
            dtype=str(self.dtype),
            batch_size=self.batch_size,
            seq_len=self.seq_len,
            num_samples=self.num_samples,
            perplexity=round(ppl, 3),
            loss=round(avg_loss, 4),
            peak_vram_mb=round(peak_vram, 2),
            allocated_vram_mb=round(allocated_vram, 2),
            latency_ms_per_token=round(latency_ms_per_token, 4),
            latency_ms_per_step=round(latency_ms_per_step, 2),
            throughput_tokens_per_sec=round(throughput, 1),
            num_parameters=num_params,
            compression_ratio=round(comp_ratio, 2),
            memory_savings_pct=round(mem_savings_pct, 2),
            execution_status="SUCCESS",
        )

        # Cleanup
        del model
        del base_model
        cleanup_gpu_memory()

        return result

    def run_all_modes(
        self,
        modes: list[str] | None = None,
        datasets: list[str] | None = None,
        rank: int = 32,
    ) -> list[ModeBenchmarkResult]:
        """Execute evaluation sweep across all requested modes and datasets."""
        target_modes = modes or SUPPORTED_MODES
        target_datasets = datasets or ["sst2"]

        logger.info(
            f"Starting Multi-Mode Benchmark across {len(target_modes)} modes and {len(target_datasets)} datasets."
        )
        logger.info(f"Modes: {target_modes}")
        logger.info(f"Datasets: {target_datasets}")

        results: list[ModeBenchmarkResult] = []

        for ds in target_datasets:
            for m in target_modes:
                res = self.run_single_mode(mode=m, dataset_name=ds, rank=rank)
                results.append(res)
                self.results.append(res)

        return results

def run_single_mode(
    self,
    mode: str,
    dataset_name: str = "sst2",
    rank: int = 32,
) -> ModeBenchmarkResult:
    """Run benchmark for a single execution mode."""
    logger.info(
        f"==> Benchmarking Mode: [{mode}] on Dataset: [{dataset_name}] (Device: {self.device})"
    )
    cleanup_gpu_memory()

    # Load fresh model copy from cache
    base_model = copy.deepcopy(self._base_model_cached)
    tokenizer = self._tokenizer_cached
    num_params_raw = sum(p.numel() for p in base_model.parameters())

    # Transform model into target execution mode
    try:
        model = ModeTransformEngine.apply_mode(
            model=base_model,
            mode=mode,
            device=self.device,
            dtype=self.dtype,
            rank=rank,
        )
        model.eval()
        model.to(self.device)
    except Exception as e:
        logger.error(f"Failed to configure mode {mode}: {e}")
        return ModeBenchmarkResult(
            mode=mode,
            model_name=self.model_name_or_path,
            dataset=dataset_name,
            device=str(self.device),
            dtype=str(self.dtype),
            batch_size=self.batch_size,
            seq_len=self.seq_len,
            num_samples=self.num_samples,
            perplexity=float("nan"),
            loss=float("nan"),
            peak_vram_mb=0.0,
            allocated_vram_mb=0.0,
            latency_ms_per_token=0.0,
            latency_ms_per_step=0.0,
            throughput_tokens_per_sec=0.0,
            num_parameters=num_params_raw,
            compression_ratio=1.0,
            memory_savings_pct=0.0,
            execution_status="FAILED",
            error_message=str(e),
        )

    num_params = sum(p.numel() for p in model.parameters() if p.requires_grad or True)

    # Prepare dataset & dataloader
    dataset = self._get_dataset(dataset_name, tokenizer)
    dataloader = torch.utils.data.DataLoader(
        dataset,
        batch_size=self.batch_size,
        shuffle=False,
        drop_last=True,
    )

    # Warmup runs
    cleanup_gpu_memory()
    self._sync()
    warmup_batches = 0
    with torch.no_grad():
        for batch in dataloader:
            if warmup_batches >= self.warmup_steps:
                break
            if isinstance(batch, dict):
                input_ids = batch["input_ids"].to(self.device)
                attention_mask = batch.get("attention_mask", None)
                if attention_mask is not None:
                    attention_mask = attention_mask.to(self.device)
            else:
                input_ids = batch[0].to(self.device)
                attention_mask = None

            _ = model(input_ids=input_ids, attention_mask=attention_mask)
            self._sync()
            warmup_batches += 1

    # Timed execution & Loss collection
    cleanup_gpu_memory(force=False)
    self._sync()
    t_start = time.perf_counter()

    total_loss = 0.0
    total_tokens = 0
    steps_executed = 0

    with torch.no_grad():
        for batch in dataloader:
            if steps_executed >= self.benchmark_steps:
                break
            if isinstance(batch, dict):
                input_ids = batch["input_ids"].to(self.device)
                attention_mask = batch.get("attention_mask", None)
                if attention_mask is not None:
                    attention_mask = attention_mask.to(self.device)
                labels = batch.get("labels", None)
                if labels is not None:
                    labels = labels.to(self.device)
            else:
                input_ids = batch[0].to(self.device)
                attention_mask = None
                labels = None

            out = model(input_ids=input_ids, attention_mask=attention_mask)

            # Accurate cross-entropy loss computation for PPL
            if hasattr(out, "loss") and out.loss is not None:
                step_loss = out.loss.item()
            elif hasattr(out, "logits"):
                logits = out.logits
                if logits.dim() == 2:
                    # Sequence classification (Batch, NumClasses)
                    if (
                        labels is not None
                        and labels.dim() == 1
                        and labels.size(0) == logits.size(0)
                    ):
                        target = labels.clamp(0, logits.size(-1) - 1)
                    else:
                        target = (input_ids[:, 0] % logits.size(-1)).long()
                    step_loss = F.cross_entropy(logits, target).item()
                elif logits.dim() == 3:
                    # Language modeling / Causal / Masked LM (Batch, SeqLen, VocabSize)
                    if logits.size(1) > 1:
                        shift_logits = logits[:, :-1, :].contiguous().view(-1, logits.size(-1))
                        shift_labels = input_ids[:, 1:].contiguous().view(-1)
                        step_loss = F.cross_entropy(shift_logits, shift_labels).item()
                    else:
                        step_loss = F.cross_entropy(
                            logits.view(-1, logits.size(-1)), input_ids.view(-1)
                        ).item()
                else:
                    step_loss = 1.5
            else:
                step_loss = 1.5  # Nominal constant fallback

            total_loss += step_loss
            total_tokens += input_ids.numel()
            steps_executed += 1
            self._sync()

    t_end = time.perf_counter()
    elapsed_sec = max(t_end - t_start, 1e-6)

    # Metrics calculation
    peak_vram, allocated_vram = self._measure_memory()
    avg_loss = total_loss / max(steps_executed, 1)
    if not math.isfinite(avg_loss):
        error = f"non-finite evaluation loss ({avg_loss!r}); result is invalid"
        logger.error("Mode %s produced %s", mode, error)
        del model
        del base_model
        cleanup_gpu_memory()
        return ModeBenchmarkResult(
            mode=mode,
            model_name=self.model_name_or_path,
            dataset=dataset_name,
            device=str(self.device),
            dtype=str(self.dtype),
            batch_size=self.batch_size,
            seq_len=self.seq_len,
            num_samples=self.num_samples,
            perplexity=float("nan"),
            loss=float("nan"),
            peak_vram_mb=0.0,
            allocated_vram_mb=0.0,
            latency_ms_per_token=0.0,
            latency_ms_per_step=0.0,
            throughput_tokens_per_sec=0.0,
            num_parameters=num_params,
            compression_ratio=1.0,
            memory_savings_pct=0.0,
            execution_status="FAILED",
            error_message=error,
        )
    try:
        ppl = math.exp(min(avg_loss, 20.0))  # Cap exponential overflow
    except OverflowError:
        ppl = float("inf")

    latency_ms_per_step = (elapsed_sec / max(steps_executed, 1)) * 1000.0
    latency_ms_per_token = (elapsed_sec / max(total_tokens, 1)) * 1000.0
    throughput = total_tokens / elapsed_sec

    # Record baseline for relative compression metrics
    if mode == "dense_fp16" or self._baseline_vram_mb is None:
        self._baseline_vram_mb = peak_vram
        self._baseline_params = num_params

    # Compression ratio
    base_params = self._baseline_params if self._baseline_params else num_params
    comp_ratio = float(base_params) / float(max(num_params, 1))

    # In specific modes (e.g. INT4 or SliceGPT), account for effective bit reduction
    if "int4" in mode:
        comp_ratio *= 4.0
    elif "fp8" in mode:
        comp_ratio *= 2.0
    elif "slicegpt" in mode:
        comp_ratio *= 1.5

    # Memory savings (%)
    base_mem = (
        self._baseline_vram_mb
        if self._baseline_vram_mb and self._baseline_vram_mb > 0
        else peak_vram
    )
    mem_savings_pct = max(0.0, ((base_mem - peak_vram) / max(base_mem, 1e-6)) * 100.0)

    result = ModeBenchmarkResult(
        mode=mode,
        model_name=self.model_name_or_path,
        dataset=dataset_name,
        device=str(self.device),
        dtype=str(self.dtype),
        batch_size=self.batch_size,
        seq_len=self.seq_len,
        num_samples=self.num_samples,
        perplexity=round(ppl, 3),
        loss=round(avg_loss, 4),
        peak_vram_mb=round(peak_vram, 2),
        allocated_vram_mb=round(allocated_vram, 2),
        latency_ms_per_token=round(latency_ms_per_token, 4),
        latency_ms_per_step=round(latency_ms_per_step, 2),
        throughput_tokens_per_sec=round(throughput, 1),
        num_parameters=num_params,
        compression_ratio=round(comp_ratio, 2),
        memory_savings_pct=round(mem_savings_pct, 2),
        execution_status="SUCCESS",
    )

    # Cleanup
    del model
    del base_model
    cleanup_gpu_memory()

    return result

    def __init__(self, output_dir: Path | str) -> None:
        """Initialise and create the standard directory tree under output_dir."""
        self.root = Path(output_dir)
        self._dirs = {
            "metrics": self.root / "metrics",

        }
        for d in self._dirs.values():
            d.mkdir(parents=True, exist_ok=True)

    # ------------------------------------------------------------------ #
    # Config                                                               #
    # ------------------------------------------------------------------ #
    def save_config(self, cfg: Any) -> None:
        """Serialise the ExperimentConfig (dataclass or dict) to config.json."""
        if hasattr(cfg, "to_dict"):
            data = cfg.to_dict()
        elif hasattr(cfg, "__dict__"):
            data = {k: v for k, v in vars(cfg).items() if not k.startswith("_")}
        elif isinstance(cfg, dict):
            data = cfg
        else:
            data = {"config": str(cfg)}
        self._write_json(self.root / "config.json", data)
        logger.debug("Config saved → %s", self.root / "config.json")

    # Metrics                                                              #
    # ------------------------------------------------------------------ #

        path = self._dirs["metrics"] / f"{tag}.json"
        self._write_json(path, payload)

        self,
        db_path: str | Path | None = None,
        config: Any | None = None,
        metrics: dict[str, Any] | None = None,
        extra_metadata: dict[str, Any] | None = None,
    ) -> str:
        """

        # FIX: Uses PRAGMA journal_mode=WAL;, PRAGMA synchronous=NORMAL;, and file locking for multi-GPU database write safety.
        """
        target_db = Path(db_path) if db_path is not None else (self.base_dir / "experiments.db")

        metrics_dict = _sanitize_json_value(metrics or {})
        meta_dict = _sanitize_json_value(extra_metadata or {})

                        run_id TEXT PRIMARY KEY,

                        model_name TEXT,

                        metadata_json TEXT

                model_name = str(cfg_dict.get("model_name", "unknown"))

                    """
                    INSERT OR REPLACE INTO experiment_runs
                    (run_id, created_at, task_name, model_name, config_json, metrics_json, metadata_json)
                    VALUES (?, ?, ?, ?, ?, ?, ?)

    def _write_json(self, rel_path: str, obj) -> None:
        """write json."""

                return o.to_dict()
            if hasattr(o, "__dict__"):
                return {k: v for k, v in o.__dict__.items() if not k.startswith("_")}
            if isinstance(o, (datetime, Path)):
                return str(o)
            if hasattr(o, "value"):
                return o.value
            return str(o)

        fp = self.run_dir / rel_path
        fp.parent.mkdir(exist_ok=True)
        with open(fp, "w") as f:

                _sanitize_json_value(obj),

    """

    meta_dict = _sanitize_json_value(extra_metadata or {})

            cursor = conn.cursor()

                    run_id TEXT PRIMARY KEY,
                    created_at TEXT,
                    task_name TEXT,
                    model_name TEXT,

                    metrics_json TEXT,
                    metadata_json TEXT
                )
                """)

            task_name = str(cfg_dict.get("task_name", "unknown"))
            model_name = str(cfg_dict.get("model_name", "unknown"))

                """
                INSERT OR REPLACE INTO experiment_runs

class RetrainingOrchestrator:
    """
    Central decision engine for automated retraining.

    Usage:
        orch = RetrainingOrchestrator(RetrainPolicy())
        decision = orch.evaluate(
            current_metrics={"accuracy": 0.82},
            reference_metrics={"accuracy": 0.88},
            drift_report=drift_report,
            new_data_count=5000,
            days_since_last_train=35,
        )
        if decision.should_retrain:
            orch.trigger_retraining(run_id="model-v3", training_fn=my_train_func)
    """

    def __init__(self, policy: RetrainPolicy = None) -> None:
        self.policy = policy or RetrainPolicy()
        self._retrain_count_today: int = 0
        self._last_retrain_date: str = ""

    def evaluate(
        self,
        current_metrics: dict[str, float] | None = None,
        reference_metrics: dict[str, float] | None = None,
        drift_report: Any | None = None,
        new_data_count: int = 0,
        days_since_last_train: int = 0,
        manual_request: bool = False,
    ) -> RetrainDecision:
        """Evaluate all configured triggers and return a retraining decision."""

        # Reset daily counter if it's a new day
        today = datetime.now().strftime("%Y-%m-%d")
        if today != self._last_retrain_date:
            self._retrain_count_today = 0
            self._last_retrain_date = today

        if self._retrain_count_today >= self.policy.max_retrains_per_day:
            return RetrainDecision(
                should_retrain=False,
                reason="Max daily retrains reached",
                details={"retrains_today": self._retrain_count_today},
            )

        # 1. Manual trigger
        if manual_request and TriggerType.MANUAL in self.policy.trigger_types:
            return RetrainDecision(
                should_retrain=True,
                reason="Manual retraining requested",
                trigger_type=TriggerType.MANUAL,
                urgency="high",
            )

        # 2. Performance degradation
        if (
            TriggerType.PERFORMANCE in self.policy.trigger_types
            and current_metrics
            and reference_metrics
        ):
            decision = self._check_performance(current_metrics, reference_metrics)
            if decision.should_retrain:
                return decision

        # 3. Drift trigger
        if TriggerType.DRIFT in self.policy.trigger_types and drift_report:
            decision = self._check_drift(drift_report)
            if decision.should_retrain:
                return decision

        # 4. Data volume trigger
        if TriggerType.DATA_VOLUME in self.policy.trigger_types:
            decision = self._check_data_volume(new_data_count)
            if decision.should_retrain:
                return decision

        # 5. Schedule trigger
        if TriggerType.SCHEDULE in self.policy.trigger_types:
            decision = self._check_schedule(days_since_last_train)
            if decision.should_retrain:
                return decision

        return RetrainDecision(should_retrain=False, reason="No triggers met")

    def _check_performance(
        self, current: dict[str, float], reference: dict[str, float]
    ) -> RetrainDecision:
        """check performance."""
        acc_key = "accuracy" if "accuracy" in current else "eval_accuracy"
        curr_acc = current.get(acc_key, 1.0)
        ref_acc = reference.get(acc_key, 1.0)
        degradation = ref_acc - curr_acc

        if curr_acc < self.policy.min_accuracy:
            return RetrainDecision(
                should_retrain=True,
                reason=f"Accuracy {curr_acc:.3f} below minimum {self.policy.min_accuracy}",
                trigger_type=TriggerType.PERFORMANCE,
                urgency="critical" if degradation > 0.1 else "high",
                details={
                    "current_accuracy": curr_acc,
                    "reference_accuracy": ref_acc,
                    "degradation": degradation,
                },
            )

        if degradation > self.policy.max_accuracy_degradation:
            return RetrainDecision(
                should_retrain=True,
                reason=f"Accuracy degraded by {degradation:.3f} (threshold: {self.policy.max_accuracy_degradation})",
                trigger_type=TriggerType.PERFORMANCE,
                urgency="medium",
                details={"degradation": degradation},
            )

        return RetrainDecision(should_retrain=False, reason="Performance within acceptable range")

    def _check_drift(self, drift_report) -> RetrainDecision:
        """check drift."""
        n_drifted = getattr(drift_report, "n_features_drifted", 0)
        n_total = getattr(drift_report, "n_features_tested", 1)
        ratio = n_drifted / max(n_total, 1)

        if ratio > self.policy.drift_threshold:
            return RetrainDecision(
                should_retrain=True,
                reason=f"Drift detected in {n_drifted}/{n_total} features ({ratio:.1%})",
                trigger_type=TriggerType.DRIFT,
                urgency="high" if ratio > 0.5 else "medium",
                details={"drifted_features": n_drifted, "total_features": n_total},
            )
        return RetrainDecision(should_retrain=False, reason="Drift within acceptable range")

    def _check_data_volume(self, new_count: int) -> RetrainDecision:
        """check data volume."""
        if new_count >= self.policy.min_new_samples:
            return RetrainDecision(
                should_retrain=True,
                reason=f"New data volume ({new_count}) exceeded threshold ({self.policy.min_new_samples})",
                trigger_type=TriggerType.DATA_VOLUME,
                urgency="low",
                details={"new_samples": new_count},
            )
        return RetrainDecision(should_retrain=False, reason="Insufficient new data")

    def _check_schedule(self, days_since: int) -> RetrainDecision:
        """check schedule."""
        if days_since >= self.policy.max_days_since_last_train:
            return RetrainDecision(
                should_retrain=True,
                reason=f"Scheduled retrain: {days_since} days since last training (max: {self.policy.max_days_since_last_train})",
                trigger_type=TriggerType.SCHEDULE,
                urgency="low",
                details={"days_since_last_train": days_since},
            )
        return RetrainDecision(should_retrain=False, reason="Schedule not due yet")

    def trigger_retraining(
        self,
        run_id: str,
        training_fn: Callable,
        decision: RetrainDecision,
        **training_kwargs,
    ) -> dict[str, Any]:
        """Execute retraining and validate the new model."""
        logger.info("Retraining triggered for %s: %s", run_id, decision.reason)
        self._retrain_count_today += 1

        try:
            result = training_fn(**training_kwargs)
            logger.info(
                "Retraining completed for %s. Metrics: %s",
                run_id,
                {k: v for k, v in result.items() if isinstance(v, (int, float))},
            )
        except Exception as e:
            logger.exception("Retraining failed for %s: %s", run_id, e)
            return {"status": "failed", "error": str(e)}

        return {"status": "completed", "run_id": run_id, "metrics": result}

class TrainerRegistry:
    """Registry for trainer classes."""

    _registry: dict[str, type[BaseTrainer]] = {}

    @classmethod
    def register(cls, name: str) -> Any:
        """Register."""

        def decorator(klass: type[BaseTrainer]) -> type[BaseTrainer]:
            """Decorator."""
            if not issubclass(klass, BaseTrainer):
                raise TypeError(f"{klass.__name__} must inherit from BaseTrainer")
            cls._registry[name] = klass
            logger.info(f"Registered trainer: {name} -> {klass.__name__}")
            return klass

        return decorator

    @classmethod
    def get(cls, name: str) -> type[BaseTrainer]:
        """Get."""
        if name not in cls._registry:
            raise ValueError(f"Unknown trainer '{name}'. Available: {list(cls._registry.keys())}")
        return cls._registry[name]

    @classmethod
    def list(cls) -> list:
        """List."""
        return list(cls._registry.keys())

    @classmethod
    def create(cls, name: str, **kwargs) -> BaseTrainer:
        """Create."""
        return cls.get(name)(**kwargs)

class UnifiedTransformerModel(nn.Module, PyTorchModelHubMixin):
    """
    A wrapper around a HuggingFace model that supports custom attention
    and can be saved/loaded via the HuggingFace Hub.
    """

    def __init__(
        self, model: nn.Module, custom_config: UnifiedTransformerConfig | None = None
    ) -> None:
        super().__init__()
        self.model = model
        self.config = getattr(model, "config", None)  # HF model config
        self.arch = detect_architecture(model)
        self._custom_config = custom_config or UnifiedTransformerConfig()

    # ------------------------------------------------------------------
    # Attention replacement
    # ------------------------------------------------------------------
    def replace_attention(
        self,
        attention_type: str,
        rank: int | None = None,
        copy_weights: bool = True,
        **kwargs,
    ) -> Any:
        """
        Inject a custom attention variant into all layers.
        This method also records the choice so it can be saved.
        """
        inject_attention(self.model, attention_type, rank=rank, copy_weights=copy_weights, **kwargs)
        self._custom_config = UnifiedTransformerConfig(
            attention_type=attention_type,
            attention_rank=rank,
            attention_kwargs=kwargs,
        )
        return self

    # ------------------------------------------------------------------
    # Freezing / parameter count
    # ------------------------------------------------------------------
    def freeze_layers(self, num_layers_to_freeze: int = 0) -> Any:
        """Freeze layers."""
        layers = get_transformer_layers(self.model)
        for i, layer in enumerate(layers):
            for param in layer.parameters():
                param.requires_grad = i >= num_layers_to_freeze
        return self

    def unfreeze_all(self) -> Any:
        """Unfreeze all."""
        for param in self.model.parameters():
            param.requires_grad = True
        return self

    def get_parameter_count(self) -> dict[str, int]:
        """Return the parameter count."""
        total = sum(p.numel() for p in self.model.parameters())
        trainable = sum(p.numel() for p in self.model.parameters() if p.requires_grad)
        return {"total": total, "trainable": trainable, "frozen": total - trainable}

    # ------------------------------------------------------------------
    # Hub serialization – these methods are called by PyTorchModelHubMixin
    # ------------------------------------------------------------------
    def _save_pretrained(self, save_directory: str) -> None:
        """
        Save the underlying model and our custom config to a directory.
        """
        # 1. Save the base HF model (attention layers are already inside)
        self.model.save_pretrained(save_directory)
        # 2. Save our custom attention config to a JSON file
        import json

        custom_cfg_path = f"{save_directory}/unified_attention_config.json"
        with open(custom_cfg_path, "w") as f:
            json.dump(self._custom_config.to_dict(), f, indent=2)

    @classmethod
    def _from_pretrained(cls, model_id: str, **kwargs) -> "UnifiedTransformerModel":
        """
        Load a pretrained model and reconstruct the custom attention variant.
        This method is invoked by `from_pretrained` class method of the mixin.
        """
        # 1. Load the custom config
        import json
        import os

        if os.path.isdir(model_id):
            cfg_path = os.path.join(model_id, "unified_attention_config.json")
        else:
            # If model_id is a Hub repo, we need to download the file
            from huggingface_hub import hf_hub_download

            cfg_path = hf_hub_download(repo_id=model_id, filename="unified_attention_config.json")

        with open(cfg_path) as f:
            custom_cfg_dict = json.load(f)
        custom_cfg = UnifiedTransformerConfig.from_dict(custom_cfg_dict)

        # 2. Load the base HF model
        base_model = AutoModel.from_pretrained(model_id, **kwargs)

        # 3. Re‑apply the attention replacement if a type was stored
        model = cls(base_model, custom_config=custom_cfg)
        if custom_cfg.attention_type:
            model.replace_attention(
                custom_cfg.attention_type,
                rank=custom_cfg.attention_rank,
                copy_weights=False,  # Weights already inside the saved base model
                **custom_cfg.attention_kwargs,
            )
        return model

    # ------------------------------------------------------------------
    # Other methods
    # ------------------------------------------------------------------
    def forward(self, *args, **kwargs) -> Any:
        """Forward pass — compute module output from inputs."""
        return self.model(*args, **kwargs)

    def save_pretrained(self, save_directory: str) -> None:
        """Convenience wrapper – delegates to the mixin."""
        super().save_pretrained(save_directory)

    @classmethod
    def from_pretrained(
        cls,
        model_id: str,
        task: str = "sequence_classification",
        num_labels: int | None = None,
        **kwargs,
    ) -> "UnifiedTransformerModel":
        """
        Create a new UnifiedTransformerModel from a pretrained identifier.
        Automatically applies the custom attention if the model was saved with one.
        This is the main entry point for users.
        """
        # If the model_id has a unified_attention_config.json, use the mixin's from_pretrained
        # which will call our _from_pretrained. Otherwise, create a fresh one.
        import os

        has_custom = False
        if os.path.isdir(model_id):
            has_custom = os.path.exists(os.path.join(model_id, "unified_attention_config.json"))

        if has_custom:
            return cls._from_pretrained(model_id, **kwargs)
        else:
            # Fresh model – no custom attention yet
            if task == "sequence_classification":
                sc_kwargs = dict(kwargs)
                if num_labels is not None:
                    sc_kwargs["num_labels"] = num_labels
                base = AutoModelForSequenceClassification.from_pretrained(model_id, **sc_kwargs)
            else:
                base = AutoModel.from_pretrained(model_id, **kwargs)
            return cls(base)
