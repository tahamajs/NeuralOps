```python
def prune_mlp_intermediate_l1(linear_layer: torch.nn.Linear, prune_ratio: float = 0.5) -> torch.nn.Linear:
    """Prune linear layer intermediate channels using L1-norm magnitude ranking."""
    weight = linear_layer.weight.data
    l1_norms = torch.norm(weight, p=1, dim=1)
    keep_count = int(weight.size(0) * (1.0 - prune_ratio))
    _, keep_indices = torch.topk(l1_norms, k=keep_count, largest=True)
    keep_indices, _ = torch.sort(keep_indices)
    
    new_layer = torch.nn.Linear(linear_layer.in_features, keep_count, bias=linear_layer.bias is not None)
    new_layer.weight.data = weight[keep_indices, :]
    if linear_layer.bias is not None:
        new_layer.bias.data = linear_layer.bias.data[keep_indices]
    return new_layer
```
