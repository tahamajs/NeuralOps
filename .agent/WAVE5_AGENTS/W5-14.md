## Recipe 2: Upgrading Benchmark Configurations

### Before (v2.3) — untyped dict
```python
config = yaml.safe_load(open("benchmark.yaml"))
run_benchmark(config["suite"], config["kernels"])
```

### After (v2.4) — Pydantic v2 validated
```python
from neuralops.benchmarks.config_v2.loader import load_benchmark_config
config = load_benchmark_config("benchmark.yaml")
# config is now a frozen, type-safe KernelSuiteConfig object
run_benchmark(config.suite, config.kernels)
```

## Automated Migration Script

```python
# scripts/migrate_v23_to_v24.py
import re, sys

def migrate_file(path: str) -> None:
    with open(path) as f:
        content = f.read()
    content = content.replace(
        "continuous_batching_generate",
        "generate_sequential"
    )
    content = re.sub(
        r'yaml\.safe_load\(open\(["\']([^"\']+)["\']\)\)',
        r'load_benchmark_config("\1")',
        content
    )
    with open(path, "w") as f:
        f.write(content)
    print(f"Migrated: {path}")

if __name__ == "__main__":
    for path in sys.argv[1:]:
        migrate_file(path)
```

import glob, re, ast

m_files = glob.glob('docs/migrations/*.md')
results = []

errors = 0

for fpath in m_files:
    with open(fpath) as f:

    # Extract \`\`\`python ... \`\`\` blocks
    snippets = re.findall(r'\`\`\`python\n(.*?)\`\`\`', content, re.DOTALL)
