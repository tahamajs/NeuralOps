### Step 4: Inspect Every Wave 4 Branch — Commit Count vs Main

Using the actual agent list discovered above, run a **robust** (no shell=True) Python script:

```python
# Save as: /tmp/w5_2_branch_audit.py
import subprocess
import os

# Read actual agent file list discovered in Step 3
agents_file = 'audits/wave5/w5_2_evidence/wave4_agents_actual.txt'
with open(agents_file) as f:
    raw_lines = [l.strip() for l in f.readlines() if l.strip() and not l.startswith('WAVE4_AGENTS')]

# Strip .md extension and derive agent IDs
agents = []
for line in raw_lines:
    name = line.replace('.md', '').strip()
    if name:
        agents.append(name)

if not agents:
    # Fallback to known W4 naming convention
    agents = [f'W4-{x}' for x in ['L1','L2','L3','L4','O1','O2','C1','T1','T2','T3','A1','C2','E1','U1','U2','B1','B2','B3','I1','R1']]

report = []
for a in agents:
    branch = f'wave4/{a}'
    try:
        result = subprocess.run(
            ['git', 'rev-list', '--count', f'main..{branch}'],
            capture_output=True, text=True, timeout=10
        )
        count = result.stdout.strip() if result.returncode == 0 else 'NO_BRANCH'
    except Exception as exc:
        count = f'ERROR: {exc}'
    report.append(f'{a} ({branch}): {count} commits ahead of main')

with open('audits/wave5/w5_2_evidence/unmerged_remediation_analysis.txt', 'w') as f:
    f.write('\n'.join(report) + '\n')
print('Wave 4 branch audit completed.')
print('\n'.join(report))
```
