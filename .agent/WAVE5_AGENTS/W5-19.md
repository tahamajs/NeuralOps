Full profiling script:
```bash
python -c "
import json, time, torch, datetime, subprocess

# Get git commit
try:
    commit = subprocess.check_output(['git', 'rev-parse', 'HEAD']).decode().strip()
except Exception:
    commit = 'unknown'

device = torch.device('mps' if torch.backends.mps.is_available() else 'cpu')
targets = ['cosformer', 'decay_linear_attention', 'gla_attention']
configs = [
    {'seq_len': 128, 'batch_size': 2, 'dim': 64, 'heads': 4},
    {'seq_len': 512, 'batch_size': 2, 'dim': 64, 'heads': 4},
]

timestamp = datetime.datetime.utcnow().strftime('%Y-%m-%dT%H:%M:%SZ')
raw_timings = {}
baseline_summary = {}

try:
    from neuralops.core.registry import create_attention
    registry_available = True
except Exception as e:
    print(f'Registry not available: {e}')
    registry_available = False

for name in targets:
    raw_timings[name] = {}
    baseline_summary[name] = {}
    for cfg in configs:
        cfg_key = f's{cfg[\"seq_len\"]}_b{cfg[\"batch_size\"]}'
        try:
            if not registry_available:
                raise ImportError('Registry not available')
            model = create_attention(name, dim=cfg['dim'], num_heads=cfg['heads']).to(device)
            model.eval()
            x = torch.randn(cfg['batch_size'], cfg['seq_len'], cfg['dim'], device=device)

            # Rule 9.1: Warmup 10 iterations
            for _ in range(10):
                with torch.no_grad():
                    _ = model(x)

            # Timed iterations (25 runs)
            runs = []
            for _ in range(25):
                t0 = time.perf_counter()
                with torch.no_grad():
                    _ = model(x)
                runs.append((time.perf_counter() - t0) * 1000)

            # Rule 9.2: Sort and calculate percentiles, discard top/bottom 5%
            runs.sort()
            filtered = runs[1:-1]  # discard extreme outliers
            median = filtered[len(filtered) // 2]
            p90 = filtered[int(len(filtered) * 0.9)]
            p99 = filtered[-1]

            raw_timings[name][cfg_key] = runs
            baseline_summary[name][cfg_key] = {
                'benchmark_name': name,
                'device': str(device),
                'seq_len': cfg['seq_len'],
                'batch_size': cfg['batch_size'],
                'dim': cfg['dim'],
                'heads': cfg['heads'],
                'median_ms': round(median, 4),
                'p90_ms': round(p90, 4),
                'p99_ms': round(p99, 4),
                'git_commit': commit,
                'timestamp': timestamp,
            }
        except Exception as e:
            baseline_summary[name][cfg_key] = {
                'benchmark_name': name,
                'device': str(device),
                'seq_len': cfg['seq_len'],
                'batch_size': cfg['batch_size'],
                'dim': cfg['dim'],
                'heads': cfg['heads'],
                'error': str(e),
                'git_commit': commit,
                'timestamp': timestamp,
            }

with open('audits/wave5/w5_19_evidence/profiling_raw_timings.json', 'w') as f:
    json.dump(raw_timings, f, indent=2)

with open('audits/wave5/w5_19_baseline.json', 'w') as f:
    json.dump(baseline_summary, f, indent=2)

print('Baseline profiling completed.')
"
```

### Step 4: GPU Profiling (if CUDA available — Rule 2.2)
If `torch.cuda.is_available()` returns True, additionally time with `torch.cuda.Event`:
```bash
python -c "
import torch, json, time, datetime, subprocess

if not torch.cuda.is_available():
    print('CUDA not available — skipping GPU profiling')
    exit(0)

commit = subprocess.check_output(['git', 'rev-parse', 'HEAD']).decode().strip()
device = torch.device('cuda')
timestamp = datetime.datetime.utcnow().strftime('%Y-%m-%dT%H:%M:%SZ')

try:
    from neuralops.core.registry import create_attention
    # Rule 2.3: Flush L2 cache before timing
    def flush_l2(dev):
        try:
            l2_bytes = torch.cuda.get_device_properties(dev).L2_cache_size
            buf = torch.empty(l2_bytes // 4, dtype=torch.int32, device=dev)
            buf.zero_()
            torch.cuda.synchronize(dev)
        except Exception:
            pass

    gpu_baselines = {}
    for name in ['cosformer', 'gla_attention']:
        x = torch.randn(2, 128, 64, device=device)
        model = create_attention(name, dim=64, num_heads=4).to(device)
        model.eval()

        # Warmup (Rule 9.1)
        for _ in range(10):
            with torch.no_grad():
                _ = model(x)
        torch.cuda.synchronize()

        runs = []
        for _ in range(25):
            flush_l2(device)
            # Rule 2.2: Use torch.cuda.Event
            start = torch.cuda.Event(enable_timing=True)
            end = torch.cuda.Event(enable_timing=True)
            start.record()
            with torch.no_grad():
                _ = model(x)
            end.record()
            torch.cuda.synchronize()
            runs.append(start.elapsed_time(end))

        runs.sort()
        filtered = runs[1:-1]
        gpu_baselines[name] = {
            'benchmark_name': name,
            'device': 'cuda',
            'median_ms': round(filtered[len(filtered)//2], 4),
            'p90_ms': round(filtered[int(len(filtered)*0.9)], 4),
            'p99_ms': round(filtered[-1], 4),
            'git_commit': commit,
            'timestamp': timestamp,
        }

    # Merge into existing baseline
    try:
        with open('audits/wave5/w5_19_baseline.json') as f:
            baseline = json.load(f)
    except Exception:
        baseline = {}

    for name, data in gpu_baselines.items():
        if name not in baseline:
            baseline[name] = {}
        baseline[name]['s128_b2_cuda'] = data

    with open('audits/wave5/w5_19_baseline.json', 'w') as f:
        json.dump(baseline, f, indent=2)

    print('GPU profiling completed.')
except Exception as e:
    print(f'GPU profiling error: {e}')
"
```
