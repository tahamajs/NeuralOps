### Step 2: Create/Update `.github/workflows/ci.yml`

This agent MUST CREATE this file with real content. Write the following YAML exactly:

```yaml
# .github/workflows/ci.yml
name: CI

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
      - run: pip install ruff mypy
      - run: ruff check neuralops/ tests/
      - run: ruff format --check neuralops/ tests/
      - run: mypy --strict neuralops/benchmarks/config_v2/ neuralops/benchmarks/results_v2/ neuralops/benchmarks/cli_v2/ neuralops/benchmarks/suites_v2/

  test:
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        python-version: ["3.10", "3.11", "3.12"]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: ${{ matrix.python-version }}
      - run: pip install -e ".[test]"
      - run: pytest tests/unit tests/contract -v
      - run: pytest tests/integration -v -m "not hub and not gpu"
```

### Step 4: Create/Update `.pre-commit-config.yaml`

This agent MUST CREATE this file showing exact hooks for ruff and mypy:

```yaml
# .pre-commit-config.yaml
repos:
  - repo: https://github.com/astral-sh/ruff-pre-commit
    rev: v0.3.0
    hooks:
      - id: ruff
        args: [--fix]
      - id: ruff-format

  - repo: https://github.com/pre-commit/mirrors-mypy
    rev: v1.8.0
    hooks:
      - id: mypy
        args:
          - --strict
          - --ignore-missing-imports
        files: ^neuralops/benchmarks/(config_v2|results_v2|cli_v2|suites_v2)/
        additional_dependencies:
          - pydantic>=2.0
          - torch

  - repo: https://github.com/pre-commit/pre-commit-hooks
    rev: v4.5.0
    hooks:
      - id: trailing-whitespace
      - id: end-of-file-fixer
      - id: check-yaml
      - id: check-added-large-files
        args: [--maxkb=500]
```
